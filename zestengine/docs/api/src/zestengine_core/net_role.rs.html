1//! Content roles for dedicated servers (NET-3): which side of a networked game
2//! an object or component belongs to.
3//!
4//! A scene object and each of its components carries a [`NetRole`]. `Both` (the
5//! default, and what every old scene means) keeps the thing everywhere; `Client`
6//! content is stripped from a dedicated server (cameras, audio, UI, particles ...);
7//! `Server` content is stripped from a client (spawners, score keepers, anything
8//! the players must not receive). A component without an explicit role takes its
9//! type's default ([`TypedComponent::NET_ROLE`](crate::TypedComponent::NET_ROLE),
10//! with a table for the built-ins in [`builtin_default_role`]).
11//!
12//! The decision rules live in [`NetRole::kept_on`]; the strip itself runs on scene
13//! entries in `zestengine_assets::net_strip` (load time and server cook).
14
15use std::sync::atomic::{AtomicU8, Ordering};
16
17use serde::{Deserialize, Serialize};

19/// Where an object or component exists.
20#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash, PartialOrd, Ord, Serialize, Deserialize)]
21pub enum NetRole {
22    /// Only on players' machines.
23    Client,
24    /// Only on the dedicated server.
25    Server,
26    /// Everywhere (listen hosts, offline play, editor included).
27    #[default]
28    Both,
29}
30
31impl NetRole {
32    /// Every role, in declaration order.
33    pub const ALL: [NetRole; 3] = [NetRole::Both, NetRole::Client, NetRole::Server];
34
35    /// True for `Both` (the role kept everywhere).
36    pub fn is_both(&self) -> bool {
37        *self == NetRole::Both
38    }
39
40    /// Whether content with this role exists in a process playing `side`.
41    /// `side` is the process role: `Server` = dedicated server, `Client` = a
42    /// joined player, `Both` = listen host / offline / editor (keeps everything).
43    pub fn kept_on(self, side: NetRole) -> bool {
44        self == NetRole::Both || side == NetRole::Both || self == side
45    }
46
47    /// Short name for lists and logs.
48    pub fn label(self) -> &'static str {
49        match self {
50            NetRole::Both => "Both",
51            NetRole::Client => "Client only",
52            NetRole::Server => "Server only",
53        }
54    }
55
56    /// One-letter hierarchy badge; `None` for `Both`.
57    pub fn badge(self) -> Option<char> {
58        match self {
59            NetRole::Both => None,
60            NetRole::Client => Some('C'),
61            NetRole::Server => Some('S'),
62        }
63    }
64}

128/// The default role of a built-in component type by registry name.
129pub fn builtin_default_role(name: &str) -> NetRole {
130    if CLIENT_ONLY.contains(&name) { NetRole::Client } else { NetRole::Both }
131}
132
133/// Every built-in type with a non-`Both` default (documentation and tests).
134pub fn builtin_client_only() -> &'static [&'static str] {
135    CLIENT_ONLY
136}

138static PROCESS_ROLE: AtomicU8 = AtomicU8::new(2);
139
140/// The role this process plays for content filtering at load time: `Both`
141/// (keeps everything) unless the player runs as a pure joined client. A
142/// dedicated server passes `Server` explicitly to its host instead of using this.
143pub fn process_role() -> NetRole {
144    match PROCESS_ROLE.load(Ordering::Relaxed) {
145        0 => NetRole::Client,
146        1 => NetRole::Server,
147        _ => NetRole::Both,
148    }
149}
150
151/// Set [`process_role`] (a player that only ever joins servers sets `Client`).
152pub fn set_process_role(role: NetRole) {
153    PROCESS_ROLE.store(
154        match role {
155            NetRole::Client => 0,
156            NetRole::Server => 1,
157            NetRole::Both => 2,
158        },
159        Ordering::Relaxed,
160    );
161}

163#[cfg(test)]
164mod tests {
165    use super::*;
166
167    #[test]
168    fn keep_rules() {
169        use NetRole::*;
170        for r in [Client, Server, Both] {
171            assert!(r.kept_on(Both), "{r:?} stays in a listen host");
172        }
173        assert!(Both.kept_on(Server) && Both.kept_on(Client));
174        assert!(Server.kept_on(Server) && !Server.kept_on(Client));
175        assert!(Client.kept_on(Client) && !Client.kept_on(Server));
176    }
177
178    #[test]
179    fn default_is_both_and_serde_round_trips() {
180        assert_eq!(NetRole::default(), NetRole::Both);
181        for r in NetRole::ALL {
182            let text = ron::to_string(&r).unwrap();
183            assert_eq!(ron::from_str::<NetRole>(&text).unwrap(), r);
184        }
185    }
186
187    #[derive(serde::Serialize, serde::Deserialize, Default)]
188    struct ServerOnlyThing;
189    impl crate::TypedComponent for ServerOnlyThing {
190        const NAME: &'static str = "Test Server Only Thing";
191        const NET_ROLE: NetRole = NetRole::Server;
192    }
193
194    #[test]
195    fn a_game_component_declares_its_own_default() {
196        let mut reg = crate::ComponentRegistry::with_builtins();
197        reg.register::<ServerOnlyThing>();
198        assert_eq!(reg.default_role("Test Server Only Thing"), NetRole::Server);
199        assert_eq!(reg.default_role("No Such Plugin Component"), NetRole::Both, "unknown types stay on both sides");
200    }
201
202    #[test]
203    fn builtin_table_names_registered_types() {
204        let reg = crate::ComponentRegistry::with_builtins();
205        let names: Vec<_> = reg.names().collect();
206        for n in builtin_client_only() {
207            assert!(names.contains(n), "table names unknown component {n:?}");
208            assert_eq!(reg.default_role(n), NetRole::Client);
209        }
210        for n in ["Box Collider", "Mesh Collider", "Rigid Body", "Network Object", "Network Manager", "Animator", "Spawn Point", "Health", "Voice Chat"] {
211            assert_eq!(reg.default_role(n), NetRole::Both, "{n}");
212        }
213    }
