1//! Shared spatial-region selection, independent of rendering or audio.
2use glam::{Mat4, Vec3};
3
4#[derive(Clone, Copy, Debug)]
5pub struct VolumeRegion {
6    pub world: Mat4,
7    pub half_extents: Vec3,
8    pub global: bool,
9    pub priority: f32,
10}
11
12impl VolumeRegion {
13    pub fn contains(&self, point: Vec3) -> bool {
14        if self.global { return true; }
15        if !self.world.is_finite() || self.world.determinant().abs() < 1e-8 { return false; }
16        let local = self.world.inverse().transform_point3(point);
17        local.is_finite() && local.abs().cmple(self.half_extents.abs()).all()
18    }
19}
20
21/// Highest priority wins; equal priority retains the first region in scene order.
22pub fn select_volume(regions: impl IntoIterator<Item = VolumeRegion>, point: Vec3) -> Option<usize> {
23    let mut selected: Option<(usize, f32)> = None;
24    for (i, region) in regions.into_iter().enumerate() {
25        if region.priority.is_finite() && region.contains(point)
26            && selected.is_none_or(|(_, p)| region.priority > p)
27        { selected = Some((i, region.priority)); }
28    }
29    selected.map(|(i, _)| i)
30}

32#[cfg(test)]
33mod tests {
34    use super::*;
35    #[test]
36    fn transformed_bounds_priority_and_stable_ties() {
37        let global = VolumeRegion { world: Mat4::IDENTITY, half_extents: Vec3::ONE, global: true, priority: -1.0 };
38        let local = VolumeRegion { world: Mat4::from_scale_rotation_translation(Vec3::new(2.0, 1.0, 1.0), glam::Quat::from_rotation_y(1.0), Vec3::splat(10.0)), global: false, priority: 1.0, ..global };
39        assert_eq!(select_volume([global, local, local], Vec3::splat(10.0)), Some(1));
40        assert_eq!(select_volume([global, local], Vec3::ZERO), Some(0));
41        assert!(local.contains(local.world.transform_point3(Vec3::new(0.9, 0.0, 0.0))));
42        assert!(!local.contains(local.world.transform_point3(Vec3::new(1.1, 0.0, 0.0))));
43        assert!(!VolumeRegion { world: Mat4::ZERO, ..local }.contains(Vec3::ZERO));
44    }
