1//! Cloth simulation resolution, independent of the visible mesh topology.
2use serde::{Deserialize,Serialize};
3
4/// A distance tier. The near tier uses the Mesh Cloth's ordinary settings.
5#[derive(Clone,Debug,Serialize,Deserialize,PartialEq)]
6#[serde(default)]
7pub struct ClothLodLevel {
8    /// Distance from viewer to object origin in world metres.
9    pub distance:f32,
10    /// Minimum particle spacing in object metres (never finer than the near tier).
11    pub particle_spacing:f32,
12    pub frequency:f32,
13    pub iterations:u32,
14}
15impl Default for ClothLodLevel {
16    fn default()->Self{Self{distance:10.0,particle_spacing:0.04,frequency:60.0,iterations:3}}
17}
18
19#[derive(Clone,Debug,Serialize,Deserialize,PartialEq)]
20#[serde(default)]
21pub struct ClothLod {
22    pub enabled:bool,
23    pub levels:Vec<ClothLodLevel>,
24    /// Metres on each side of a threshold, to avoid switching as the viewer jitters.
25    pub hysteresis:f32,
26    /// Seconds to fade the render detail that cannot fit the new particle layout.
27    pub transition:f32,
28}
29impl Default for ClothLod {
30    fn default()->Self{Self{enabled:false,levels:vec![ClothLodLevel::default(),ClothLodLevel{distance:25.0,particle_spacing:0.1,frequency:30.0,iterations:2}],hysteresis:1.0,transition:0.25}}
31}
32impl ClothLod {
33    fn valid(&self,index:usize)->Option<&ClothLodLevel>{
34        index.checked_sub(1).and_then(|i|self.levels.get(i)).filter(|l|l.distance.is_finite()&&l.distance>0.0&&l.particle_spacing.is_finite()&&l.particle_spacing>=0.0)
35    }
36    /// Zero is the near tier; other values index the authored array plus one.
37    /// Without a viewer, headless simulation uses the near tier deterministically.
38    pub fn select(&self,distance:Option<f32>,current:usize)->usize{
39        let Some(distance)=distance.filter(|d|d.is_finite())else{return 0};
40        if !self.enabled{return 0;}
41        let current_distance=self.valid(current).map_or(0.0,|l|l.distance);
42        let h=if self.hysteresis.is_finite(){self.hysteresis.max(0.0)}else{0.0};
43        let mut result=0;let mut threshold=0.0;
44        for i in 1..=self.levels.len(){
45            let Some(l)=self.valid(i)else{continue};
46            let boundary=if l.distance<=current_distance{l.distance-h}else{l.distance+h};
47            if distance>=boundary && l.distance>threshold{result=i;threshold=l.distance;}
48        }
49        result
50    }
51    pub fn settings(&self,level:usize,spacing:f32,frequency:f32,iterations:u32)->(f32,f32,u32){
52        match self.valid(level).filter(|_|self.enabled){
53            Some(l)=>(l.particle_spacing.max(spacing),if l.frequency.is_finite(){l.frequency.clamp(10.0,480.0)}else{frequency},l.iterations.clamp(1,32)),
54            None=>(spacing,frequency,iterations),
55        }
56    }
57}

59#[cfg(test)]
60mod tests{
61    use super::*;
62    #[test]
63    fn cloth_lod_threshold_hysteresis_unsorted_and_headless(){
64        let mut lod=ClothLod{enabled:true,..Default::default()};
65        assert_eq!(lod.select(Some(10.5),0),0);assert_eq!(lod.select(Some(11.0),0),1);
66        assert_eq!(lod.select(Some(9.5),1),1);assert_eq!(lod.select(Some(8.9),1),0);
67        assert_eq!(lod.select(Some(100.0),0),2);assert_eq!(lod.select(Some(0.0),2),0);
68        assert_eq!(lod.select(None,2),0);
69        lod.levels.swap(0,1);assert_eq!(lod.select(Some(15.0),0),2);assert_eq!(lod.select(Some(30.0),0),1);
70        lod.levels[0].distance=f32::NAN;assert_eq!(lod.select(Some(30.0),1),2);
71        lod.enabled=false;assert_eq!(lod.select(Some(100.0),2),0);
72    }
