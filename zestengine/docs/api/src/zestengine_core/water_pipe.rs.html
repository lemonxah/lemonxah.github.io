1//! Shallow-water simulation of a Water body's interactive layer: a virtual
2//! pipe model (O'Brien & Hodgins; Mei, Decaudin, Hu 2007) on a square window
3//! of cells. Each cell holds a water column, four outflow pipes and the
4//! water's horizontal velocity; gravity drives the pipes from the head
5//! difference, so water has mass and momentum, piles up, sloshes, flows
6//! round solid cells and reflects off them.
7//!
8//! The GPU runs this (`water_sim.comp` in zestengine-render); [`PipeSim`] is the
9//! same scheme on the CPU, pass for pass, as its reference and for tests.
10//! What the engine feeds both lives here: the window ([`choose_window`]),
11//! the per-cell field ([`SimField`]: rest depth, base flow, solid) and the
12//! bodies' footprints ([`WaterDisplacer`]).
13//!
14//! The simulated state is a perturbation of the rest water: a river's or
15//! current's base flow is not in the pipes, it advects the perturbation, and
16//! obstacles in it push back (the bow and the lee). Bodies occupy volume in
17//! the columns under them, so the water they displace is pushed out by the
18//! solver and mass is kept. Waves travel at the speed for `min(depth,
19//! wave depth)`; the column itself keeps the real depth.

30const GRAVITY: f32 = 9.81;
31/// Most substeps one frame takes; a longer frame runs the water slower.
32pub const MAX_SIM_SUBSTEPS: u32 = 16;
33/// Courant number the substeps keep under.
34const COURANT: f32 = 0.5;
35/// Water-dragging rate (1/s) at `sim_momentum` 1.
36const MOMENTUM_RATE: f32 = 1.5;
37/// Eddy viscosity rate (1/s) at Chop smoothing 1.
38const EDDY_RATE: f32 = 2.0;
39/// Most of a cell's velocity difference to its neighbours viscosity evens
40/// out in one substep (keeps the explicit diffusion stable).
41const VISCOUS_STEP_MAX: f32 = 0.2;
42
43/// Bed friction per unit kinematic viscosity (1/(m^2)): 3 / (h min(h, delta))
44/// over water `h` deep, where delta = sqrt(2 nu / omega) is the oscillating
45/// boundary layer of the waves the grid carries (about eight cells long):
46/// laminar sheet flow when the layer fills the depth, a thin Stokes layer
47/// at the bed when it doesn't.
48pub fn bed_friction(h: f32, cell: f32, gravity: f32, viscosity: f32) -> f32 {
49    let k = std::f32::consts::PI / (4.0 * cell.max(1e-3));
50    let omega = (gravity * k * (k * h).min(10.0).tanh()).max(1e-4).sqrt();
51    let delta = (2.0 * viscosity / omega).sqrt();
52    3.0 / (h * h.min(delta).max(1e-3))
53}
54/// Rate (1/s) the level eases back to rest, so sources and outflow balance.
55const LEVEL_RETURN: f32 = 0.02;
56/// A push's dent (m) and added water (m/s) per m/s of push.
57const PUSH_DENT: f32 = 0.03;
58const PUSH_RATE: f32 = 0.05;
59/// Fastest water velocity the solver reports (m/s).
60const MAX_SPEED: f32 = 8.0;
61/// Field cell kinds (`SimField` cells' w).
62pub const CELL_WATER: f32 = 0.0;
63/// Field cell kind: shore (shallow water or a dry beach).
64pub const CELL_SHORE: f32 = 1.0;
65/// Field cell kind: obstacle (a solid blocks the cell).
66pub const CELL_OBSTACLE: f32 = 2.0;
67/// Static wall whose steady current response is already in the recording.
68const CELL_RECORDED_OBSTACLE:f32=3.0;
69/// Shallower ground than this (m) starts dry.
70const MIN_WET_DEPTH: f32 = 0.03;
71/// Ground up to this high above the rest surface (m) is a beach waves can
72/// run up onto and drain back from; higher is a bank (a wall).
73pub const RUNUP_HEIGHT: f32 = 0.6;
74/// Water thinner than this (m) over a beach cell is drawn as dry (sunk
75/// under the ground) and holds no dispersive ripples.
76pub const DRY_FILM: f32 = 0.01;
77/// How far under the ground (m) a dry beach cell's surface is put.
78const DRY_SINK: f32 = 0.15;

80/// The square of cells a simulation covers (water-local XZ).
81#[derive(Clone, Copy, Debug, PartialEq)]
82pub struct SimWindow {
83    /// The first cell, in whole cells from the local origin.
84    pub origin: IVec2,
85    /// Cells per side.
86    pub n: u32,
87    /// Cell width (m).
88    pub cell: f32,
89    /// The window covers the whole body: its edges are walls. Otherwise it
90    /// follows a focus and its border absorbs what leaves.
91    pub closed: bool,
92}

94impl SimWindow {
95    /// Window width (m).
96    pub fn size(&self) -> f32 {
97        self.n as f32 * self.cell
98    }
99
100    /// Window centre (water-local XZ).
101    pub fn center(&self) -> Vec2 {
102        (self.origin.as_vec2() + Vec2::splat(self.n as f32 * 0.5)) * self.cell
103    }
104
105    /// Local XZ of the window's first corner.
106    pub fn lo(&self) -> Vec2 {
107        self.origin.as_vec2() * self.cell
108    }
109
110    /// Absorbing border width in cells (0 when closed).
111    pub fn sponge(&self) -> u32 {
112        if self.closed { 0 } else { (self.n / 16).max(2) }
113    }

116/// Cells per side the simulation runs for a requested resolution (a
117/// multiple of 8, 32..512).
118pub fn sim_resolution(r: u32) -> u32 {
119    (r.clamp(32, 512) + 4) / 8 * 8
120}

122/// The window for `water` this frame: the whole body when it fits in
123/// `ripple_area` (cells sized to it), else `ripple_area` around `focus`
124/// (local XZ), kept until the focus is an eighth of the window away.
125pub fn choose_window(water: &Water, focus: Vec2, prev: Option<SimWindow>) -> SimWindow {
126    let n = sim_resolution(water.ripple_resolution);
127    let area = water.ripple_area.max(1.0);
128    if let Some((lo, hi)) = water.extent() {
129        // A margin past the fade the renderer applies at the window border.
130        let size = (hi - lo).max_element().max(1.0) * 1.2;
131        if size <= area {
132            let cell = size / n as f32;
133            let origin = ((lo + hi) * 0.5 / cell - Vec2::splat(n as f32 * 0.5)).round().as_ivec2();
134            return SimWindow { origin, n, cell, closed: true };
135        }
136    }
137    let cell = area / n as f32;
138    let focus = if focus.is_finite() { focus } else { Vec2::ZERO };
139    let want = (focus / cell - Vec2::splat(n as f32 * 0.5)).round().as_ivec2();
140    if let Some(p) = prev
141        && !p.closed
142        && p.n == n
143        && (p.cell - cell).abs() <= cell * 1e-5
144        && (want - p.origin).abs().max_element() <= (n / 8) as i32
145    {
146        return p;
147    }
148    SimWindow { origin: want, n, cell, closed: false }
149}

151/// Substeps for a frame of `dt` on `cell`-metre cells with waves running at
152/// `depth` and water moving up to `max_speed`.
153pub fn pipe_substeps(dt: f32, cell: f32, depth: f32, max_speed: f32) -> u32 {
154    let c = (GRAVITY * depth.max(0.01)).sqrt() + max_speed.max(0.0);
155    ((dt.max(0.0) * c / (COURANT * cell.max(1e-4))).ceil() as u32).clamp(1, MAX_SIM_SUBSTEPS)
156}
157
158/// Longest frame the substep cap keeps stable.
159pub fn max_sim_dt(cell: f32, depth: f32, max_speed: f32) -> f32 {
160    let c = (GRAVITY * depth.max(0.01)).sqrt() + max_speed.max(0.0);
161    MAX_SIM_SUBSTEPS as f32 * COURANT * cell / c
162}

164/// Something occupying the water this frame (water-local XZ): a body's
165/// footprint and how deep it sits, or a push (falling water) as a short
166/// dent with water added.
167#[derive(Clone, Copy, Debug, PartialEq)]
168pub struct WaterDisplacer {
169    /// Centre (water-local XZ).
170    pub center: Vec2,
171    /// Half extents along `axis` and across it.
172    pub half: Vec2,
173    /// Unit direction along the body's length.
174    pub axis: Vec2,
175    /// Thickness of the body within the water column (m).
176    pub thickness: f32,
177    /// Rounded (ellipsoid) rather than boxy.
178    pub round: bool,
179    /// Velocity relative to the water's base flow (m/s).
180    pub velocity: Vec2,
181    /// Water added under it (m/s of column).
182    pub rate: f32,
183    /// How much it drags the water toward its velocity (0..1).
184    pub momentum: f32,
185    /// A one-frame push rather than a body: its dent is not scaled by the
186    /// displacement strength, and the dispersive layer takes it as an
187    /// impulse instead of following its occupancy.
188    pub impulse: bool,
189}

191impl WaterDisplacer {
192    /// A box (its eight corners in water-local space, `axis` its local X
193    /// there) in water whose surface is at local `surface`; only the part
194    /// within `reach` below the surface counts. None when it is clear of it.
195    pub fn from_corners(corners: &[Vec3; 8], axis: Vec3, surface: f32, reach: f32, velocity: Vec2, round: bool) -> Option<Self> {
196        let c = corners.iter().copied().sum::<Vec3>() / 8.0;
197        let ax = Vec2::new(axis.x, axis.z).try_normalize().unwrap_or(Vec2::X);
198        let side = ax.perp();
199        let mut half = Vec2::ZERO;
200        let (mut lo, mut hi) = (f32::MAX, f32::MIN);
201        for p in corners {
202            let d = Vec2::new(p.x - c.x, p.z - c.z);
203            half = half.max(Vec2::new(d.dot(ax).abs(), d.dot(side).abs()));
204            lo = lo.min(p.y);
205            hi = hi.max(p.y);
206        }
207        let thickness = (surface.min(hi) - lo.max(surface - reach.max(0.0))).max(0.0);
208        if thickness <= 1e-4 || half.min_element() <= 1e-4 {
209            return None;
210        }
211        Some(Self { center: Vec2::new(c.x, c.z), half, axis: ax, thickness, round, velocity, rate: 0.0, momentum: 1.0, impulse: false })
212    }
213
214    /// A push (falling water landing, a wake eddy) as a brief dent that adds
215    /// a little water.
216    pub fn from_push(p: &WaterPush) -> Self {
217        let r = p.radius.max(0.05);
218        Self {
219            center: p.at,
220            half: Vec2::splat(r),
221            axis: Vec2::X,
222            thickness: p.strength * PUSH_DENT,
223            round: true,
224            velocity: Vec2::ZERO,
225            rate: p.strength.max(0.0) * PUSH_RATE,
226            momentum: 0.0,
227            impulse: true,
228        }
229    }
230
231    /// Footprints under ~a cell grow to one, thinner, keeping their volume.
232    pub fn fit_to_cell(mut self, cell: f32) -> Self {
233        let min = Vec2::splat(cell * 0.75);
234        let grown = self.half.max(min);
235        let keep = (self.half.x * self.half.y) / (grown.x * grown.y).max(1e-8);
236        self.thickness *= keep;
237        self.rate *= keep;
238        self.half = grown;
239        self
240    }
241
242    /// Volume it occupies (m^3).
243    pub fn volume(&self) -> f32 {
244        let area = if self.round { std::f32::consts::PI * self.half.x * self.half.y * (2.0 / 3.0) } else { 4.0 * self.half.x * self.half.y };
245        area * self.thickness
246    }
247
248    /// The occupancy profile (0..1) at local `p`, `cell` the grid spacing.
249    /// Matches `water_sim.comp`'s splat.
250    pub fn profile(&self, p: Vec2, cell: f32) -> f32 {
251        let rel = p - self.center;
252        let l = Vec2::new(rel.dot(self.axis), rel.dot(self.axis.perp())) / self.half.max(Vec2::splat(1e-4));
253        if self.round {
254            // A paraboloid of the ellipsoid's volume: its rim rises gently
255            // where the ellipsoid's is vertical (a jolt per cell crossed).
256            (4.0 / 3.0) * (1.0 - l.dot(l)).max(0.0)
257        } else {
258            let m = l.x.abs().max(l.y.abs());
259            let soft = cell / self.half.min_element().max(1e-4);
260            ((1.0 - m) / soft + 0.5).clamp(0.0, 1.0)
261        }
262    }
263
264    /// The occupancy the solver splats at `p`: a body's profile softened
265    /// over its neighbouring cells, so its edge slides smoothly across the
266    /// grid as it moves instead of stepping a cell at a time (each step
267    /// would start a ripple). Pushes stay sharp. Matches `water_sim.comp`.
268    pub fn splat_profile(&self, p: Vec2, cell: f32) -> f32 {
269        if self.impulse {
270            return self.profile(p, cell);
271        }
272        let side = [Vec2::X, -Vec2::X, Vec2::Y, -Vec2::Y].iter().map(|o| self.profile(p + *o * cell, cell)).sum::<f32>();
273        0.5 * self.profile(p, cell) + 0.125 * side
274    }
275
276    /// GPU layout (`Displacer` in water_sim.comp).
277    pub fn to_gpu(&self) -> [f32; 12] {
278        let shape = if self.round { 1.0 } else { 0.0 } + if self.impulse { 2.0 } else { 0.0 };
279        [
280            self.center.x, self.center.y, self.half.x, self.half.y,
281            self.axis.x, self.axis.y, self.thickness, shape,
282            self.velocity.x, self.velocity.y, self.rate, self.momentum,
283        ]
284    }
285}

287/// Per-cell input of a window: rest depth (m), base flow x/z (m/s, local)
288/// and kind (`CELL_*`), row-major from the window's first cell.
289#[derive(Clone, Debug, PartialEq)]
290pub struct SimField {
291    /// Bumped on every change.
292    pub version: u64,
293    /// The window the cells cover.
294    pub window: SimWindow,
295    /// Per-cell rest depth, base flow and kind (row-major from the first cell).
296    pub cells: Arc<Vec<[f32; 4]>>,
297    /// Some cell has a base flow (the solver then advects).
298    pub flowing: bool,
299    /// Deepest rest depth, and the depth cells outside the field take.
300    pub max_depth: f32,
301    /// Depth cells outside the field take.
302    pub default_depth: f32,
303}
304
305impl SimField {
306    /// The field of `water` over `window`; `blocked` (row-major, same window)
307    /// marks cells obstacles fill.
308    pub fn build(water: &Water, window: SimWindow, blocked: Option<&[bool]>) -> Self {
309        Self::build_with_bed(water, window, blocked, None)
310    }
311
312    /// The same over the ground: `bed` (row-major, same window) is the depth
313    /// of the floor under each cell's rest surface (m; NaN = none found, the
314    /// water's own depth). Ground just above the surface is a dry beach the
315    /// water can run up (`RUNUP_HEIGHT`); higher ground is a bank.
316    pub fn build_with_bed(water: &Water, window: SimWindow, blocked: Option<&[bool]>, bed: Option<&[f32]>) -> Self {
317        let n = window.n as usize;
318        let cols = water.columns_in(window.lo(), n, window.cell);
319        let mut flowing = false;
320        let mut max_depth = 0.0f32;
321        let cells: Vec<[f32; 4]> = cols
322            .iter()
323            .enumerate()
324            .map(|(i, c)| match c {
325                Some(c) if blocked.is_some_and(|b| b.get(i).copied().unwrap_or(false)) => [c.depth.max(0.05), 0.0, 0.0, if water.cache_data.is_some(){CELL_RECORDED_OBSTACLE}else{CELL_OBSTACLE}],
326                Some(c) => {
327                    let depth = bed.and_then(|b| b.get(i).copied()).filter(|d| d.is_finite()).unwrap_or(c.depth);
328                    if depth < -RUNUP_HEIGHT {
329                        return [0.05, 0.0, 0.0, CELL_SHORE];
330                    }
331                    if depth < MIN_WET_DEPTH {
332                        // A beach: dry ground (negative rest depth) the
333                        // water can run up onto.
334                        return [depth.min(0.0), 0.0, 0.0, CELL_WATER];
335                    }
336                    flowing |= c.flow.x.abs() + c.flow.z.abs() > 1e-4;
337                    max_depth = max_depth.max(depth);
338                    [depth.max(0.05), c.flow.x, c.flow.z, CELL_WATER]
339                }
340                None => [0.05, 0.0, 0.0, CELL_SHORE],
341            })
342            .collect();
343        let default_depth = if water.body == WaterBody::River { 0.05 } else { water.depth.max(0.05) };
344        Self { version: 0, window, cells: Arc::new(cells), flowing, max_depth: max_depth.max(0.05), default_depth }
345    }
346}

375impl WaterSimFeed {
376    /// The field for `window`, rebuilt when the window, the water's shape or
377    /// the obstacles change; true when the water itself changed under the
378    /// same window (the simulation then restarts from rest).
379    pub fn field(&mut self, water: &Water, window: SimWindow) -> (Arc<SimField>, bool) {
380        let key = field_key(water);
381        let blocked = self.blocked.as_ref().filter(|(w, _)| *w == window).map(|(_, b)| b.clone());
382        let same_window = self.field.as_ref().is_some_and(|f| f.window == window);
383        let blocked_same = match (&blocked, &self.field_blocked) {
384            (None, None) => true,
385            (Some(a), Some(b)) => Arc::ptr_eq(a, b),
386            _ => false,
387        };
388        let bed = self.bed.as_ref().filter(|(w, _)| *w == window).map(|(_, b)| b.clone());
389        let bed_same = match (&bed, &self.field_bed) {
390            (None, None) => true,
391            (Some(a), Some(b)) => Arc::ptr_eq(a, b),
392            _ => false,
393        };
394        if let Some(f) = &self.field
395            && same_window
396            && key == self.field_key
397            && blocked_same
398            && bed_same
399        {
400            return (f.clone(), false);
401        }
402        let reset = same_window && key != self.field_key;
403        let version = self.field.as_ref().map_or(1, |f| f.version + 1);
404        let mut field = SimField::build_with_bed(water, window, blocked.as_deref().map(|b| b.as_slice()), bed.as_deref().map(|b| b.as_slice()));
405        field.version = version;
406        let field = Arc::new(field);
407        self.field = Some(field.clone());
408        self.field_key = key;
409        self.field_blocked = blocked;
410        self.field_bed = bed;
411        (field, reset)
412    }
413
414    /// This frame's displacers for `cell`-metre cells (bodies first, then
415    /// the queued pushes, at most `max`); the pushes are used up. The
416    /// solver scales the bodies by `PipeParams::displacement`.
417    pub fn take_displacers(&mut self, cell: f32, max: usize) -> Vec<WaterDisplacer> {
418        let mut out: Vec<WaterDisplacer> = self.displacers.iter().map(|d| d.fit_to_cell(cell)).collect();
419        for p in self.pushes.drain(..) {
420            if out.len() >= max {
421                break;
422            }
423            out.push(WaterDisplacer::from_push(&p).fit_to_cell(cell));
424        }
425        out.truncate(max);
426        self.pushes.clear();
427        out
428    }
429}

431fn field_key(water: &Water) -> u64 {
432    let mut h = std::collections::hash_map::DefaultHasher::new();
433    water.geometry_key().hash(&mut h);
434    water.cache_data.is_some().hash(&mut h);
435    h.finish()
436}

438/// Solver constants for a frame.
439#[derive(Clone, Copy, Debug, PartialEq)]
440pub struct PipeParams {
441    /// Gravity (m/s^2).
442    pub gravity: f32,
443    /// Share of pipe flow lost per second.
444    pub damping: f32,
445    /// Rate (1/s) displacers drag the water toward their velocity.
446    pub momentum: f32,
447    /// Rate (1/s) the level eases back to rest.
448    pub level_return: f32,
449    /// Waves travel as if the water were at most this deep (m).
450    pub wave_depth: f32,
451    /// Scales the bodies' occupancy (not the pushes').
452    pub displacement: f32,
453    /// Kinematic viscosity (m^2/s): flow over the bed loses 3 nu / h^2 per
454    /// second (laminar sheet flow).
455    pub viscosity: f32,
456    /// Eddy viscosity as a rate (1/s): nu_eddy = eddy * cell^2, so it acts
457    /// on what the grid resolves (the shortest waves most).
458    pub eddy: f32,
459}

461impl PipeParams {
462    /// The solver constants for a Water body's settings.
463    pub fn for_water(water: &Water) -> Self {
464        Self {
465            gravity: GRAVITY,
466            damping: water.ripple_damping.clamp(0.0, 0.99),
467            momentum: water.sim_momentum.max(0.0) * MOMENTUM_RATE,
468            level_return: LEVEL_RETURN,
469            wave_depth: water.sim_wave_depth.max(0.05),
470            displacement: water.sim_displacement.max(0.0),
471            viscosity: water.kinematic_viscosity(),
472            eddy: water.sim_chop_smoothing.max(0.0) * EDDY_RATE,
473        }
474    }

476    /// Per-substep pipe flow kept after damping.
477    pub fn keep(&self, dt: f32) -> f32 {
478        (1.0 - self.damping.clamp(0.0, 0.99)).powf(dt)
479    }

496    field_origin: IVec2,

511    moved_from: Option<IVec2>,
512    /// The rest depth each column's `h` is measured from (toroidal, like
513    /// `h`): what the last RESET or REBASE set it to.
514    rest: Vec<f32>,
515    /// The field was replaced since the last step: the next one rebases.
516    rebase: bool,
517}

519const DIRS: [IVec2; 4] = [IVec2::new(-1, 0), IVec2::new(1, 0), IVec2::new(0, -1), IVec2::new(0, 1)];

521impl PipeSim {
522    /// A fresh simulation over this field.
523    pub fn new(field: &SimField) -> Self {
524        let w = field.window;
525        let cells = (w.n * w.n) as usize;
526        let mut sim = Self {
527            window: w,
528            h: vec![0.0; cells],
529            flux: vec![[0.0; 4]; cells],
530            vel: vec![Vec2::ZERO; cells],
531            field: Vec::new(),
532            field_origin: w.origin,
533            default_depth: 0.05,
534            flowing: false,
535            max_depth: 0.05,
536            couple: vec![[0.0; 4]; cells],
537            src: vec![0.0; cells],
538            ew_in: vec![[0.0; 2]; cells],
539            ewave: None,
540            moved_from: None,
541            rest: vec![0.0; cells],
542            rebase: false,
543        };
544        sim.set_field(field);
545        sim.rebase = false;
546        sim.reset(w.origin + IVec2::splat(w.n as i32 * 4));
547        sim

550    /// Replace the field (rest depth, flow, kind per cell). The water keeps
551    /// its offset from rest: where the rest depth changed (the ground or a
552    /// wall under the window was rebuilt) the next step moves each column
553    /// by the change, so the surface does not jump.
554    pub fn set_field(&mut self, field: &SimField) {
555        self.rebase = true;
556        self.field = field.cells.as_ref().clone();
557        self.field_origin = field.window.origin;
558        self.default_depth = field.default_depth;
559        self.flowing = field.flowing;
560        self.max_depth = field.max_depth;
561    }

563    /// Run the dispersive layer with `params` (None stops it; starting it

610    /// How dry cell `w` is (0 wet .. 1 dry): only beach cells (ground above
611    /// the rest surface) with less than a film of water on them.
612    fn dryness(&self, w: IVec2) -> f32 {
613        if !self.inside(w) || self.field_at(w)[0] > 0.0 {
614            return 0.0;
615        }
616        let h = self.h[self.store(w)];
617        1.0 - (h / DRY_FILM).clamp(0.0, 1.0)
618    }

620    fn solid(&self, w: IVec2) -> bool {
621        if self.inside(w) { self.field_at(w)[3] > 0.5 } else { self.window.closed }
622    }

624    fn head(&self, w: IVec2) -> f32 {
625        self.h[self.store(w)] + self.couple[self.win(w)][0] - self.field_at(w)[0]
626    }

628    fn sponge_at(&self, w: IVec2) -> f32 {
629        let s = self.window.sponge();
630        if s == 0 {
631            return 0.0;
632        }
633        let e = w.x.min(w.y).min(self.n() - 1 - w.x).min(self.n() - 1 - w.y) as f32;
634        let s = s as f32;
635        if e < s { 0.25 * (1.0 - e / s).powi(2) } else { 0.0 }
636    }

638    fn cells(&self) -> impl Iterator<Item = IVec2> + use<> {
639        let n = self.n();
640        (0..n * n).map(move |i| IVec2::new(i % n, i / n))
641    }

643    /// RESET: cells whose world position was outside the window at `prev`
644    /// go back to rest, around this frame's bodies (the column under a body
645    /// already holds its occupancy, so a body the window moves onto is
646    /// level with the water round it instead of a mound that collapses).
647    fn reset(&mut self, prev: IVec2) {
648        for w in self.cells() {
649            let p = self.window.origin + w - prev;
650            if !self.inside(p) {
651                let s = self.store(w);
652                self.h[s] = (self.field_at(w)[0] - self.couple[self.win(w)][0]).max(0.0);
653                self.rest[s] = self.field_at(w)[0];
654                self.flux[s] = [0.0; 4];
655                self.vel[s] = Vec2::ZERO;
656                if let Some((_, l)) = &mut self.ewave {
657                    l.reset_cell(s);
658                }
659            }
660        }
661    }

663    /// REBASE: after the field changed, every column moves by its change of
664    /// rest depth (a solid cell takes its new depth), keeping its offset.
665    /// A beach cell (ground above the level) whose ground drops under the
666    /// level fills to the level: its offset was the ground's height, not
667    /// water (keeping it made a lump that ran off as waves).
668    fn rebase(&mut self) {
669        for w in self.cells() {
670            let s = self.store(w);
671            let now = self.field_at(w)[0];
672            let was = self.rest[s];
673            self.h[s] = if self.solid(w) {
674                now
675            } else if was <= 0.0 && now > 0.0 {
676                now + self.h[s]
677            } else {
678                (self.h[s] + now - was).max(0.0)
679            };
680            self.rest[s] = now;
681        }
682    }

684    /// SPLAT: this frame's occupancy and drag from the displacers (bodies
685    /// scaled by `displacement`).
686    fn splat(&mut self, displacers: &[WaterDisplacer], displacement: f32) {
687        let cell = self.window.cell;
688        for w in self.cells() {
689            let p = (self.window.origin + w).as_vec2() * cell + Vec2::splat(cell * 0.5);
690            let (mut b, mut wsum, mut v, mut src) = (0.0, 0.0, Vec2::ZERO, 0.0);
691            let (mut body, mut dent) = (0.0, 0.0);
692            for d in displacers {
693                let prof = d.splat_profile(p, cell);
694                if prof <= 0.0 {
695                    continue;
696                }
697                if d.impulse {
698                    dent += d.thickness * prof;
699                } else {
700                    b += d.thickness * prof * displacement;
701                    body += d.thickness * prof;
702                }
703                src += d.rate * prof;
704                // Only the part of the column the body fills is dragged along.
705                let share = (d.thickness * prof / self.field_at(w)[0].max(0.05)).clamp(0.0, 1.0);
706                let wt = prof * d.momentum * share;
707                v += d.velocity * wt;
708                wsum += wt;
709            }
710            let v = if wsum > 1e-5 { v / wsum } else { Vec2::ZERO };
711            // Bodies fill at most the column there (see water_sim.comp).
712            let depth = self.field_at(w)[0].max(0.0);
713            let b = b.min(depth) + dent;
714            let body = body.min(depth);
715            let i = self.win(w);
716            self.couple[i] = [b, wsum.min(1.0), v.x, v.y];
717            self.src[i] = src;
718            self.ew_in[i] = [body, dent];
719        }
720    }

722    /// FLUX: update every cell's outflow pipes from the head differences.
723    fn flux_pass(&mut self, dt: f32, keep: f32, p: &PipeParams) {
724        let cell = self.window.cell;
725        let mut out = self.flux.clone();
726        for w in self.cells() {
727            let s = self.store(w);
728            if self.solid(w) {
729                out[s] = [0.0; 4];
730                continue;
731            }
732            let hs = self.h[s];
733            let head = self.head(w);
734            let mut f = self.flux[s].map(|x| x * keep);
735            for (d, off) in DIRS.iter().enumerate() {
736                let nb = w + *off;
737                if self.solid(nb) {
738                    f[d] = 0.0;
739                    continue;
740                }
741                let (hn, nh) = if self.inside(nb) { (self.head(nb), self.h[self.store(nb)]) } else { (0.0, self.default_depth) };
742                let face = (0.5 * (hs + nh)).clamp(0.0, p.wave_depth);
743                f[d] = (f[d] + dt * p.gravity * face * (head - hn)).max(0.0);
744                if p.viscosity > 0.0 {
745                    let h = (0.5 * (hs + nh)).max(0.02);
746                    f[d] *= (-p.viscosity * bed_friction(h, cell, p.gravity, p.viscosity) * dt).exp();
747                }
748            }
749            let c = self.couple[self.win(w)];
750            let mut dq = Vec2::ZERO;
751            if c[1] > 0.0 && p.momentum > 0.0 {
752                dq += (Vec2::new(c[2], c[3]) - self.vel[s]) * (c[1] * (p.momentum * dt).min(1.0) * cell * hs.clamp(0.01, p.wave_depth));
753            }
754            if p.viscosity > 0.0 || p.eddy > 0.0 {
755                // Viscous shallow water: momentum spreads to the neighbours
756                // (nu laplacian u). Walls hold the water still (no slip);
757                // past the window it carries on as it is.
758                let u = self.vel[s];
759                let around: Vec2 = DIRS
760                    .iter()
761                    .map(|o| {
762                        let nb = w + *o;
763                        if !self.inside(nb) {
764                            u
765                        } else if self.solid(nb) {
766                            Vec2::ZERO
767                        } else {
768                            self.vel[self.store(nb)]
769                        }
770                    })
771                    .sum();
772                let rate = (p.viscosity * dt / (cell * cell) + p.eddy * dt).min(VISCOUS_STEP_MAX);
773                dq += (around - 4.0 * u) * (rate * cell * hs.clamp(0.01, p.wave_depth));
774            }
775            if dq != Vec2::ZERO {
776                if dq.x > 0.0 {
777                    if !self.solid(w + DIRS[1]) {
778                        f[1] += dq.x;
779                    }
780                } else if !self.solid(w + DIRS[0]) {
781                    f[0] -= dq.x;
782                }
783                if dq.y > 0.0 {
784                    if !self.solid(w + DIRS[3]) {
785                        f[3] += dq.y;
786                    }
787                } else if !self.solid(w + DIRS[2]) {
788                    f[2] -= dq.y;
789                }
790            }
791            let sp = 1.0 - self.sponge_at(w);
792            let total = (f[0] + f[1] + f[2] + f[3]) * sp * dt;
793            let k = if total > 0.0 { (hs * cell * cell / total).min(1.0) } else { 1.0 };
794            out[s] = f.map(|x| x * sp * k);
795        }
796        self.flux = out;
797    }

799    /// HEIGHT: move the water through the pipes, add sources, update the
800    /// velocity, push back from obstacles in a base flow, and relax toward
801    /// rest at the border.
802    fn height_pass(&mut self, dt: f32, p: &PipeParams) {
803        let cell = self.window.cell;
804        for w in self.cells() {
805            let s = self.store(w);
806            let fl = self.field_at(w);
807            if self.solid(w) {
808                self.h[s] = fl[0];
809                self.vel[s] = Vec2::ZERO;
810                continue;
811            }
812            let f = self.flux[s];
813            let inflow = |d: usize, from: usize| {
814                let nb = w + DIRS[d];
815                if self.inside(nb) && !self.solid(nb) { self.flux[self.store(nb)][from] } else { 0.0 }
816            };
817            let (il, ir, ib, it) = (inflow(0, 1), inflow(1, 0), inflow(2, 3), inflow(3, 2));
818            let h_old = self.h[s];
819            let c = self.couple[self.win(w)];
820            let mut hn = h_old + dt * ((il + ir + ib + it) - (f[0] + f[1] + f[2] + f[3])) / (cell * cell);
821            hn += self.src[self.win(w)] * dt;
822            let dw = Vec2::new(il - f[0] + f[1] - ir, ib - f[2] + f[3] - it) * 0.5;
823            let havg = (0.5 * (h_old + hn)).clamp(0.05, p.wave_depth);
824            let mut u = dw / (cell * havg);
825            if u.length() > MAX_SPEED {
826                u *= MAX_SPEED / u.length();
827            }
828            let base = Vec2::new(fl[1], fl[2]);
829            if base != Vec2::ZERO {
830                // A base flow into an obstacle piles water up in front of it
831                // and draws it down behind (capped near the stagnation head).
832                let eta = h_old + c[0] - fl[0];
833                for off in DIRS {
834                    let nb = w + off;
835                    if !self.inside(nb) || self.field_at(nb)[3] < 1.5 || self.field_at(nb)[3]>2.5 {
836                        continue;
837                    }
838                    let un = base.dot(off.as_vec2());
839                    let cap = un * un / p.gravity;
840                    if (un > 0.0 && eta < cap) || (un < 0.0 && eta > -cap) {
841                        hn += dt * un * h_old.clamp(0.0, p.wave_depth) / cell;
842                    }
843                }
844            }
845            let target = (fl[0] - c[0]).max(0.0);
846            // Level return pulls the whole surface back to rest. The sponge
847            // at an open border only lets water out: a crest there drains
848            // away, a trough is filled by its neighbours, never from outside
849            // (the open window must never gain water: edge_bc tests).
850            hn += (target - hn) * (p.level_return * dt).min(1.0);
851            let sponge = self.sponge_at(w);
852            if sponge > 0.0 && hn > target {
853                hn += (target - hn) * sponge.min(1.0);
854            }
855            self.h[s] = hn.max(0.0);
856            self.vel[s] = u;
857        }
858    }

860    /// ADVECT: carry the disturbance along the base flow (semi-Lagrangian).
861    fn advect(&mut self, dt: f32) {
862        let cell = self.window.cell;
863        let mut h = self.h.clone();
864        let mut flux = self.flux.clone();
865        for w in self.cells() {
866            let s = self.store(w);
867            let fl = self.field_at(w);
868            if self.solid(w) || fl[1] * fl[1] + fl[2] * fl[2] < 1e-8 {
869                continue;
870            }
871            let pos = w.as_vec2() - Vec2::new(fl[1], fl[2]) * dt / cell;
872            let i0 = pos.floor().as_ivec2();
873            let t = pos - i0.as_vec2();
874            let (mut eta, mut fx) = (0.0, [0.0f32; 4]);
875            for (dx, dz) in [(0, 0), (1, 0), (0, 1), (1, 1)] {
876                let c = i0 + IVec2::new(dx, dz);
877                let wt = if dx == 0 { 1.0 - t.x } else { t.x } * if dz == 0 { 1.0 - t.y } else { t.y };
878                let (e, f) = if self.inside(c) && !self.solid(c) {
879                    let sc = self.store(c);
880                    (self.h[sc] - self.field_at(c)[0], self.flux[sc])
881                } else if self.solid(c) && self.inside(c) {
882                    (self.h[s] - fl[0], self.flux[s])
883                } else {
884                    (0.0, [0.0; 4])
885                };
886                eta += wt * e;
887                for k in 0..4 {
888                    fx[k] += wt * f[k];
889                }
890            }
891            h[s] = (fl[0] + eta).max(0.0);
892            flux[s] = fx;
893        }
894        self.h = h;
895        self.flux = flux;
896    }

898    /// SOURCE: the dispersive layer takes this frame's occupancy change
899    /// and dents (only remembered when not `stepping`).
900    fn ewave_source(&mut self, dt: f32) {
901        let stepping = dt > 0.0;
902        let Some((p, mut layer)) = self.ewave.take() else { return };
903        for w in self.cells() {
904            let i = self.win(w);
905            layer.inject(self.store(w), self.ew_in[i][0], self.ew_in[i][1], &p, p.follow(dt), stepping && !self.solid(w));
906        }
907        self.ewave = Some((p, layer));
908    }

910    /// The dispersive layer's step, then solid cells to rest and the
911    /// sponge (MASK).
912    fn ewave_step(&mut self, dt: f32) {
913        let Some((p, mut layer)) = self.ewave.take() else { return };
914        layer.propagate(dt, self.window.cell, &p);
915        let sponge = self.window.sponge();
916        for w in self.cells() {
917            // By the smoothed footprint the source follows, so what a body
918            // pushes in ahead and lets go behind are blocked alike.
919            let block = (layer.smoothed(self.store(w)) / crate::water_ewave::BLOCK_DRAFT).max(self.dryness(w));
920            layer.mask_cell(self.store(w), self.solid(w), sponge_depth(w, self.n(), sponge), block, dt);
921        }
922        self.ewave = Some((p, layer));
923    }

925    /// Advance by `dt` with this frame's displacers.
926    pub fn step(&mut self, dt: f32, p: &PipeParams, displacers: &[WaterDisplacer]) {
927        self.splat(displacers, p.displacement);
928        if let Some(prev) = self.moved_from.take() {
929            self.reset(prev);
930        }
931        if std::mem::take(&mut self.rebase) {
932            self.rebase();
933        }
934        self.ewave_source(dt);
935        if dt <= 0.0 {
936            return;
937        }
938        let max_speed = self.vel.iter().map(|v| v.length()).fold(0.0, f32::max);
939        let max_eta = self.cells().map(|w| (self.h[self.store(w)] - self.field_at(w)[0]).abs()).fold(0.0, f32::max);
940        let depth = self.max_depth.min(p.wave_depth) + max_eta;
941        let dt = dt.min(max_sim_dt(self.window.cell, depth, max_speed));
942        if self.flowing {
943            self.advect(dt);
944        }
945        let steps = pipe_substeps(dt, self.window.cell, depth, max_speed);
946        let sub = dt / steps as f32;
947        let keep = p.keep(sub);
948        for _ in 0..steps {
949            self.flux_pass(sub, keep, p);
950            self.height_pass(sub, p);
951        }
952        self.ewave_step(dt);
953    }

956    fn eta(&self, w: IVec2) -> f32 {
957        if self.solid(w) {
958            return 0.0;
959        }
960        self.h[self.store(w)] + self.couple[self.win(w)][0] - self.field_at(w)[0]
961    }

963    /// OUTPUT: per window cell (height offset, d/dx, d/dz, speed) as the
964    /// ripple texture holds it, and (height offset, velocity x, z, drag
965    /// weight) as the readback does. The dispersive layer adds its height,
966    /// slope and orbital velocity here; other surface layers would too.
967    pub fn output(&self) -> (Vec<[f32; 4]>, Vec<[f32; 4]>) {
968        let cell = self.window.cell;
969        let mut tex = Vec::with_capacity(self.couple.len());
970        let mut back = Vec::with_capacity(self.couple.len());
971        for w in self.cells() {
972            let mut e = self.eta(w);
973            let at = |nb: IVec2| if self.inside(nb) && !self.solid(nb) { self.eta(nb) } else { e };
974            let mut g = Vec2::new(at(w + DIRS[1]) - at(w + DIRS[0]), at(w + DIRS[3]) - at(w + DIRS[2])) / (2.0 * cell);
975            let mut u = if self.solid(w) { Vec2::ZERO } else { self.vel[self.store(w)] };
976            if let Some((ep, l)) = &self.ewave
977                && !self.solid(w)
978            {
979                let m = ep.mix;
980                let c = l.state[self.store(w)];
981                let at = |nb: IVec2| if self.inside(nb) && !self.solid(nb) { l.state[self.store(nb)] } else { c };
982                let (x0, x1, z0, z1) = (at(w + DIRS[0]), at(w + DIRS[1]), at(w + DIRS[2]), at(w + DIRS[3]));
983                e += c[0] * m;
984                g += Vec2::new(x1[0] - x0[0], z1[0] - z0[0]) * m / (2.0 * cell);
985                u += Vec2::new(x1[1] - x0[1], z1[1] - z0[1]).clamp_length_max(MAX_ORBITAL * 2.0 * cell) * m / (2.0 * cell);
986            }
987            // Inside a body's footprint there is no free surface (see
988            // water_sim.comp): drawn flattened, the readback keeps it.
989            let inside_body = (self.ew_in[self.win(w)][0] / crate::water_ewave::BLOCK_DRAFT).clamp(0.0, 1.0);
990            // A dry beach: the surface sinks under the ground there, for
991            // drawing and for what floats on it, but never below the rest
992            // level (see water_sim.comp).
993            let dry = self.dryness(w);
994            let e = if dry > 0.0 { (e - DRY_SINK * dry).max(e.min(0.0)) } else { e };
995            let shown = if e > 0.0 { e * (1.0 - inside_body) } else { e };
996            tex.push([shown, g.x * (1.0 - inside_body), g.y * (1.0 - inside_body), u.length()]);
997            back.push([e, u.x, u.y, self.couple[self.win(w)][1]]);
998        }
999        (tex, back)
1000    }

1233    /// Grid-scale roughness of the dispersive layer (each cell against its
1234    /// neighbours' mean, squared and summed) after a body crossed it, and
1235    /// its total energy (heights squared, summed).
1236    fn checker(smoothing: f32, fine_damping: f32) -> (f32, f32) {
1237        let (mut water, field) = tank(128, 16.0, 2.0);
1238        water.sim_ripple_smoothing = smoothing;
1239        water.sim_fine_damping = fine_damping;
1240        water.sim_wake_ripples = 0.35;
1241        let p = PipeParams::for_water(&water);
1242        let mut sim = PipeSim::new(&field);
1243        sim.set_ewave(EwaveParams::for_water(&water, sim.window, 2.0));
1244        let mut x = -4.0;
1245        for _ in 0..270 {
1246            // A frame-locked step: the body moves once a frame, the water
1247            // takes three substeps.
1248            x += 1.0 / 90.0;
1249            let d = WaterDisplacer { center: Vec2::new(x, 0.0), half: Vec2::splat(0.4), axis: Vec2::X, thickness: 0.3, round: true, velocity: Vec2::new(1.0, 0.0), rate: 0.0, momentum: 1.0, impulse: false };
1250            for _ in 0..3 {
1251                sim.step(1.0 / 270.0, &p, &[d]);
1252            }
1253        }
1254        let layer = sim.ewave().unwrap();
1255        let n = sim.window.n as i32;
1256        let h = |x: i32, z: i32| layer.state[sim.store(IVec2::new(x.rem_euclid(n), z.rem_euclid(n)))][0];
1257        let (mut sum, mut all) = (0.0f32, 0.0f32);
1258        for z in 0..n {
1259            for x in 0..n {
1260                let r = h(x, z) - 0.25 * (h(x + 1, z) + h(x - 1, z) + h(x, z + 1) + h(x, z - 1));
1261                sum += r * r;
1262                all += h(x, z) * h(x, z);
1263            }
1264        }
1265        (sum, all)
1266    }
