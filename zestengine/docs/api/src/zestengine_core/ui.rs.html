1//! Retained in-game UI components (HUDs, menus). Data only: the engine lays
2//! them out and draws them over the game view each frame.
3//!
4//! A [`UiCanvas`] object is a root; UI components on it and on its descendants
5//! (scene hierarchy) are laid out inside the screen. Each element's
6//! [`UiRect`] places it relative to its parent's rect with anchors, a pivot,
7//! a position and a size (Unity RectTransform style, but y points DOWN: anchor
8//! (0,0) is the top-left corner). An element without a `UiRect` fills its
9//! parent. Children draw on top of parents, in hierarchy order.
10

108/// Placement inside the parent rect. With `anchor_min == anchor_max` the
109/// element has a fixed `size` and sits at `position` from the anchor point;
110/// with anchors apart it stretches and `size` is added to the anchored span.
111#[derive(Serialize, Deserialize, Clone, Copy)]
112#[serde(default)]
113pub struct UiRect {
114    /// Anchor point of the element's minimum corner (0..1 of the parent).
115    pub anchor_min: [f32; 2],
116    /// Anchor point of the element's maximum corner (0..1 of the parent).
117    pub anchor_max: [f32; 2],
118    /// Point of the element (0..1) that sits at `position`.
119    pub pivot: [f32; 2],
120    /// Offset of the pivot point from the anchor (reference pixels).
121    pub position: [f32; 2],
122    /// Size in reference pixels (added to the anchored span).
123    pub size: [f32; 2],
124}
125
126impl Default for UiRect {
127    fn default() -> Self {
128        Self {
129            anchor_min: [0.5, 0.5],
130            anchor_max: [0.5, 0.5],
131            pivot: [0.5, 0.5],
132            position: [0.0, 0.0],
133            size: [200.0, 60.0],
134        }
135    }
136}
137
138impl UiRect {
139    /// Resolve against a parent rect `(min, size)`; `scale` converts
140    /// reference pixels to screen pixels. Returns `(min, size)`.
141    pub fn resolve(&self, parent_min: [f32; 2], parent_size: [f32; 2], scale: f32) -> ([f32; 2], [f32; 2]) {
142        let mut min = [0.0; 2];
143        let mut size = [0.0; 2];
144        for a in 0..2 {
145            let a0 = parent_min[a] + self.anchor_min[a] * parent_size[a];
146            let a1 = parent_min[a] + self.anchor_max[a] * parent_size[a];
147            let s = ((a1 - a0) + self.size[a] * scale).max(0.0);
148            let pivot_point = a0 + (a1 - a0) * self.pivot[a] + self.position[a] * scale;
149            min[a] = pivot_point - s * self.pivot[a];
150            size[a] = s;
151        }
152        (min, size)
153    }
154
155    /// Presets for the editor's anchor picker: (label, anchor_min, anchor_max).
156    pub const PRESETS: [(&'static str, [f32; 2], [f32; 2]); 10] = [
157        ("Center", [0.5, 0.5], [0.5, 0.5]),
158        ("Top left", [0.0, 0.0], [0.0, 0.0]),
159        ("Top", [0.5, 0.0], [0.5, 0.0]),
160        ("Top right", [1.0, 0.0], [1.0, 0.0]),
161        ("Left", [0.0, 0.5], [0.0, 0.5]),
162        ("Right", [1.0, 0.5], [1.0, 0.5]),
163        ("Bottom left", [0.0, 1.0], [0.0, 1.0]),
164        ("Bottom", [0.5, 1.0], [0.5, 1.0]),
165        ("Bottom right", [1.0, 1.0], [1.0, 1.0]),
166        ("Stretch", [0.0, 0.0], [1.0, 1.0]),
167    ];
168}

502impl UiButton {
503    fn update_native(&mut self, ctx: &mut ObjectContext<'_, '_>) {
504        if (!self.on_click.is_empty() || !self.on_click_actions.is_empty()) && ctx.ui_clicked(ctx.self_id()) {

515/// Built-in button behaviour. Targets are object names (first match).
516#[derive(Serialize, Deserialize, Clone, PartialEq, Debug)]
517pub enum UiAction {
518    /// Load the scene at the given project-relative path.
519    LoadScene(String),
520    /// Activate the named object.
521    Show(String),
522    /// Deactivate the named object.
523    Hide(String),
524    /// Flip the named object's active state.
525    Toggle(String),
526    /// Quit the game.
527    Quit,
528}
529
530impl UiAction {
531    /// One of each kind, for the inspector's picker.
532    pub fn kinds() -> [UiAction; 5] {
533        [
534            UiAction::LoadScene(String::new()),
535            UiAction::Show(String::new()),
536            UiAction::Hide(String::new()),
537            UiAction::Toggle(String::new()),
538            UiAction::Quit,
539        ]
540    }
541
542    /// Inspector label of this action.
543    pub fn label(&self) -> &'static str {
544        match self {
545            UiAction::LoadScene(_) => "Load scene",
546            UiAction::Show(_) => "Show object",
547            UiAction::Hide(_) => "Hide object",
548            UiAction::Toggle(_) => "Toggle object",
549            UiAction::Quit => "Quit game",
550        }
551    }
552
553    /// The scene path / object name the action uses, if it takes one.
554    pub fn target_mut(&mut self) -> Option<&mut String> {
555        match self {
556            UiAction::LoadScene(t) | UiAction::Show(t) | UiAction::Hide(t) | UiAction::Toggle(t) => Some(t),
557            UiAction::Quit => None,
558        }
559    }
560
561    /// Same kind, keeping the target where both take one.
562    pub fn with_kind_of(&self, kind: &UiAction) -> UiAction {
563        let mut out = kind.clone();
564        if let (Some(dst), Some(src)) = (out.target_mut(), self.clone().target_mut().cloned()) {
565            *dst = src;
566        }
567        out
568    }
569
570    /// Queue the action using native scene metadata.
571    pub fn run(&self, ctx: &mut ObjectContext<'_, '_>) {
572        match self {
573            UiAction::LoadScene(path) if !path.is_empty() => ctx.load_scene(path.clone()),
574            UiAction::LoadScene(_) => {}
575            UiAction::Show(name) | UiAction::Hide(name) | UiAction::Toggle(name) => {
576                let Some(id) = ctx.find_object(name) else {
577                    return;
578                };
579                let on = match self {
580                    UiAction::Show(_) => true,
581                    UiAction::Hide(_) => false,
582                    _ => !ctx.is_active(id),
583                };
584                ctx.set_active(id, on);
585            }
586            UiAction::Quit => ctx.quit(),
587        }
588    }
589}

630impl UiToggle {
631    fn update_native(&mut self, ctx: &mut ObjectContext<'_, '_>) {
632        let action = self.invert_y_action.as_str();
633        if action.is_empty() || self.applied == Some(self.on) {
634            return;
635        }
636        if self.applied.is_none() {
637            if let Some(a) = ctx.control_scheme().and_then(|s| s.action(action)) {
638                self.on = a.invert_y;
639            }
640            self.applied = Some(self.on);
641            return;
642        }
643        let schemes: Vec<String> =
644            ctx.input_bindings().into_iter().flat_map(|bindings| &bindings.schemes).filter(|s| s.action(action).is_some()).map(|s| s.name.clone()).collect();
645        let tuning = crate::ActionTuning { invert_y: Some(self.on), ..Default::default() };
646        for scheme in schemes {
647            ctx.tune_action(&scheme, action, tuning);
648        }
649        self.applied = Some(self.on);
650    }
651}

655#[serde(default)]

670    pub fill_color: [f32; 4],

672    pub handle_color: [f32; 4],

681    fn default() -> Self {

683            value: 0.5,
684            min: 0.0,
685            max: 1.0,
686            whole_numbers: false,
687            interactable: true,
688            track_color: [0.2, 0.2, 0.25, 1.0],
689            fill_color: [0.95, 0.75, 0.2, 1.0],
690            handle_color: [1.0, 1.0, 1.0, 1.0],
691            audio_bus: String::new(),
692            applied: None,
693        }
694    }
695}
696
697impl UiSlider {
698    /// `value` as 0..1 along the track.
699    pub fn fraction(&self) -> f32 {
700        let span = self.max - self.min;
701        if span.abs() < 1e-9 {
702            0.0
703        } else {
704            ((self.value - self.min) / span).clamp(0.0, 1.0)
705        }
706    }
707
708    /// Set `value` from a 0..1 fraction along the track (snaps when
709    /// `whole_numbers`).
710    pub fn set_fraction(&mut self, f: f32) {
711        let mut v = self.min + (self.max - self.min) * f.clamp(0.0, 1.0);
712        if self.whole_numbers {
713            v = v.round();
714        }
715        self.value = v;
716    }
717}

720    const NAME: &'static str = "UI Slider";
721}

723impl UiSlider {
724    fn update_native(&mut self, ctx: &mut ObjectContext<'_, '_>) {
725        if !self.audio_bus.is_empty() && self.applied != Some(self.value) {
726            ctx.set_bus_volume(self.audio_bus.clone(), self.fraction());
727            self.applied = Some(self.value);
728        }
729    }

799/// Escape / gamepad back toggles `panel` (an object name), pausing time and
800/// freeing the cursor while it's open. Put it on an object that stays active
801/// (the canvas), not on the panel itself. A Resume button is `Hide <panel>`.
802#[derive(Serialize, Deserialize)]
803#[serde(default)]
804pub struct UiPauseMenu {
805    /// Object name of the panel shown while paused.
806    pub panel: String,
807    /// Set the time scale to 0 while the menu is open.
808    pub pause_time: bool,
809    /// Re-lock the cursor when the menu closes (first-person games).
810    pub lock_cursor_in_game: bool,
811    /// The menu's state the engine last saw (engine-filled, not saved).
812    #[serde(skip)]
813    pub open: Option<bool>,
814}
815
816impl Default for UiPauseMenu {
817    fn default() -> Self {
818        Self {
819            panel: "Pause Panel".into(),
820            pause_time: true,
821            lock_cursor_in_game: false,
822            open: None,
823        }
824    }
825}
826
827impl TypedComponent for UiPauseMenu {
828    const NAME: &'static str = "UI Pause Menu";
829}
830
831impl UiPauseMenu {
832    fn update_native(&mut self, ctx: &mut ObjectContext<'_, '_>) {
833        let Some(id) = ctx.find_object(&self.panel) else {
834            return;
835        };
836        let open = ctx.is_active(id);
837        if ctx.ui_cancel_pressed() {
838            ctx.set_active(id, !open);
839        }
840        if self.open != Some(open) {
841            // The first frame only acts when the panel starts open.
842            if self.open.is_some() || open {
843                if self.pause_time {
844                    ctx.set_time_scale(if open { 0.0 } else { 1.0 });
845                }
846                ctx.set_cursor_locked(!open && self.lock_cursor_in_game);
847            }
848            self.open = Some(open);
849        }
850    }
851}

887            placeholder_color: [1.0, 1.0, 1.0, 0.4],
888            password: false,
889            interactable: true,
890            editing: false,
891        }

894
895impl UiTextField {
896    /// Apply typed text / backspace, respecting `max_length`.
897    pub fn type_text(&mut self, typed: &str, backspaces: u32) {
898        for _ in 0..backspaces {
899            self.text.pop();
900        }
901        for c in typed.chars().filter(|c| !c.is_control()) {
902            if self.max_length > 0 && self.text.chars().count() >= self.max_length as usize {
903                break;
904            }
905            self.text.push(c);
906        }
907    }
908
909    /// What is drawn: bullets instead of characters when `password` is on.
910    pub fn display_text(&self) -> String {
911        if self.password {
912            "\u{2022}".repeat(self.text.chars().count())
913        } else {
914            self.text.clone()
915        }
916    }
917}
