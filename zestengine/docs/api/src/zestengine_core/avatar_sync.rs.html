1//! Wire format for synced avatar parameters (`P_AVPARAM`).
2//!
3//! Every peer wearing an avatar has the same declared parameters (they ship in
4//! the avatar package), so the packet carries no names: the *schema* is the
5//! synced parameters sorted by name, identified by a 32-bit hash of their
6//! names and kinds. A packet is `[schema hash: u32 LE][payload]`, and the
7//! payload is one byte per Int / Float (Int 0..=255; Float -1..=1 as a signed
8//! byte over 127, the same quantisation remote copies show) followed by the
9//! Bools packed eight to a byte. A full 256-bit budget is therefore at most
10//! 32 payload bytes. A receiver whose schema hash differs (another version of
11//! the avatar) drops the packet instead of misreading it.
12//!
13//! The codec is pure: it reads values from, and writes them to, plain slices
14//! or an [`AvatarParamStore`] through slots resolved once when the schema is
15//! built. [`SyncClock`] holds the send cadence: changes go out at 10 Hz,
16//! 30 Hz while a puppet is open, and the full state repeats every second so
17//! a lost packet or a late joiner catches up.
18
19use crate::avatar_params::{AvatarParam, AvatarParamKind, AvatarParamStore};
20
21/// Seconds between packets while values are changing.
22pub const SEND_INTERVAL: f64 = 0.1;
23/// Seconds between packets while a puppet or radial control is being moved.
24pub const FAST_INTERVAL: f64 = 1.0 / 30.0;
25/// Seconds between repeats of an unchanged state.
26pub const KEYFRAME_INTERVAL: f64 = 1.0;
27/// Bytes of the schema hash at the start of every packet.
28pub const HEADER_BYTES: usize = 4;
29
30/// Why a synced avatar-parameter packet was dropped.
31#[derive(Clone, Debug, PartialEq, Eq)]
32pub enum SyncError {
33    /// The sender's parameters differ from ours.
34    SchemaMismatch {
35        /// The sender's schema hash.
36        got: u32,
37        /// Our schema hash.
38        want: u32,
39    },
40    /// The packet is shorter or longer than the schema needs.
41    BadLength {
42        /// The packet's byte length.
43        got: usize,
44        /// The byte length the schema needs.
45        want: usize,
46    },
47}
48
49impl std::fmt::Display for SyncError {
50    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
51        match self {
52            Self::SchemaMismatch { got, want } => write!(f, "avatar parameter schema {got:08x} does not match {want:08x}"),
53            Self::BadLength { got, want } => write!(f, "avatar parameter packet is {got} bytes, expected {want}"),
54        }
55    }
56}

58/// One parameter in the wire schema.
59#[derive(Clone, Debug, PartialEq)]
60pub struct SchemaEntry {
61    /// Parameter name.
62    pub name: String,
63    /// Parameter kind.
64    pub kind: AvatarParamKind,
65    /// Store slot (`usize::MAX` when the schema was built from declarations).
66    pub slot: usize,
67}
68
69/// The synced parameters of one avatar in wire order.
70#[derive(Clone, Debug, Default, PartialEq)]
71pub struct AvParamSchema {
72    entries: Vec<SchemaEntry>,
73    hash: u32,
74    /// Entries that are numbers (Int / Float), in schema order, as indices
75    /// into `entries`; then the Bools.
76    numbers: Vec<usize>,
77    bools: Vec<usize>,
78}
79
80/// 32-bit FNV-1a over names and kinds.
81fn schema_hash(entries: &[SchemaEntry]) -> u32 {
82    let mut h: u32 = 0x811c_9dc5;
83    let mut eat = |b: u8| {
84        h ^= b as u32;
85        h = h.wrapping_mul(0x0100_0193);
86    };
87    for e in entries {
88        e.name.bytes().for_each(&mut eat);
89        eat(0);
90        eat(match e.kind {
91            AvatarParamKind::Bool => 1,
92            AvatarParamKind::Int => 2,
93            AvatarParamKind::Float => 3,
94        });
95        eat(b'\n');
96    }
97    h
98}
99
100impl AvParamSchema {
101    fn from_entries(mut entries: Vec<SchemaEntry>) -> Self {
102        entries.sort_by(|a, b| a.name.cmp(&b.name));
103        entries.dedup_by(|a, b| a.name == b.name);
104        let hash = schema_hash(&entries);
105        let numbers = (0..entries.len()).filter(|&i| entries[i].kind != AvatarParamKind::Bool).collect();
106        let bools = (0..entries.len()).filter(|&i| entries[i].kind == AvatarParamKind::Bool).collect();
107        Self { entries, hash, numbers, bools }
108    }
109
110    /// The synced parameters among `params` (local-only, nameless and
111    /// duplicate ones are not part of the schema).
112    pub fn from_params(params: &[AvatarParam]) -> Self {
113        Self::from_entries(
114            params
115                .iter()
116                .filter(|p| p.is_synced() && !p.name.is_empty())
117                .map(|p| SchemaEntry { name: p.name.clone(), kind: p.kind, slot: usize::MAX })
118                .collect(),
119        )
120    }
121
122    /// The same schema with store slots filled in, for `encode` / `apply`.
123    pub fn from_store(store: &AvatarParamStore) -> Self {
124        Self::from_entries(
125            store
126                .synced_slots()
127                .into_iter()
128                .map(|slot| SchemaEntry { name: store.name(slot).to_owned(), kind: store.kind(slot), slot })
129                .collect(),
130        )
131    }
132
133    /// The schema's hash (identifies the parameter set).
134    pub fn hash(&self) -> u32 {
135        self.hash
136    }
137    /// Number of entries.
138    pub fn len(&self) -> usize {
139        self.entries.len()
140    }
141    /// True when the schema has no entries.
142    pub fn is_empty(&self) -> bool {
143        self.entries.is_empty()
144    }
145    /// The entries in wire order.
146    pub fn entries(&self) -> &[SchemaEntry] {
147        &self.entries
148    }
149    /// Payload size in bytes (the header is extra).
150    pub fn payload_bytes(&self) -> usize {
151        self.numbers.len() + self.bools.len().div_ceil(8)
152    }
153    /// Whole packet size in bytes.
154    pub fn packet_bytes(&self) -> usize {
155        HEADER_BYTES + self.payload_bytes()
156    }
157
158    /// Append a packet for `values` (parallel to [`Self::entries`]).
159    pub fn encode_values(&self, values: &[f32], out: &mut Vec<u8>) {
160        out.extend_from_slice(&self.hash.to_le_bytes());
161        for &i in &self.numbers {
162            let v = values.get(i).copied().filter(|v| v.is_finite()).unwrap_or(0.0);
163            out.push(match self.entries[i].kind {
164                AvatarParamKind::Int => v.round().clamp(0.0, 255.0) as u8,
165                _ => ((v.clamp(-1.0, 1.0) * 127.0).round() as i8) as u8,
166            });
167        }
168        let mut byte = 0u8;
169        for (n, &i) in self.bools.iter().enumerate() {
170            if values.get(i).copied().unwrap_or(0.0) != 0.0 {
171                byte |= 1 << (n % 8);
172            }
173            if n % 8 == 7 {
174                out.push(byte);
175                byte = 0;
176            }
177        }
178        if !self.bools.len().is_multiple_of(8) {
179            out.push(byte);
180        }
181    }
182
183    /// Read a packet into `out` (resized to one value per entry).
184    pub fn decode_values(&self, data: &[u8], out: &mut Vec<f32>) -> Result<(), SyncError> {
185        if data.len() < HEADER_BYTES {
186            return Err(SyncError::BadLength { got: data.len(), want: self.packet_bytes() });
187        }
188        let got = u32::from_le_bytes([data[0], data[1], data[2], data[3]]);
189        if got != self.hash {
190            return Err(SyncError::SchemaMismatch { got, want: self.hash });
191        }
192        if data.len() != self.packet_bytes() {
193            return Err(SyncError::BadLength { got: data.len(), want: self.packet_bytes() });
194        }
195        out.clear();
196        out.resize(self.entries.len(), 0.0);
197        let body = &data[HEADER_BYTES..];
198        for (n, &i) in self.numbers.iter().enumerate() {
199            out[i] = match self.entries[i].kind {
200                AvatarParamKind::Int => body[n] as f32,
201                _ => (body[n] as i8) as f32 / 127.0,
202            };
203        }
204        let bits = &body[self.numbers.len()..];
205        for (n, &i) in self.bools.iter().enumerate() {
206            out[i] = ((bits[n / 8] >> (n % 8)) & 1) as f32;
207        }
208        Ok(())
209    }
210
211    /// Append a packet with the store's current values.
212    pub fn encode(&self, store: &AvatarParamStore, out: &mut Vec<u8>) {
213        let values: Vec<f32> = self.entries.iter().map(|e| if e.slot < store.len() { store.get(e.slot) } else { 0.0 }).collect();
214        self.encode_values(&values, out);
215    }
216
217    /// Write decoded values into the store (through the normal coercion).
218    pub fn apply(&self, values: &[f32], store: &mut AvatarParamStore) {
219        for (e, &v) in self.entries.iter().zip(values) {
220            if e.slot < store.len() {
221                store.set(e.slot, v);
222            }
223        }
224    }
225
226    /// The values as a remote copy would see them: floats quantised, ints and
227    /// bools rounded. What `encode_values` + `decode_values` give back.
228    pub fn quantise(&self, values: &[f32]) -> Vec<f32> {
229        let mut bytes = Vec::with_capacity(self.packet_bytes());
230        self.encode_values(values, &mut bytes);
231        let mut out = Vec::new();
232        let _ = self.decode_values(&bytes, &mut out);
233        out
234    }
235}

237/// When to send: 10 Hz while values change (30 Hz with a puppet open), and
238/// the full state once a second regardless.
239#[derive(Clone, Copy, Debug, Default)]
240pub struct SyncClock {
241    last_send: f64,
242    sent_once: bool,
243}
244
245impl SyncClock {
246    /// Whether to send at `now` (seconds on any steady clock). `changed`: a
247    /// synced value differs from what was last sent. `fast`: a puppet is open.
248    pub fn poll(&mut self, now: f64, changed: bool, fast: bool) -> bool {
249        let since = now - self.last_send;
250        let due = !self.sent_once || since >= KEYFRAME_INTERVAL || (changed && since >= if fast { FAST_INTERVAL } else { SEND_INTERVAL });
251        if due {
252            self.last_send = now;
253            self.sent_once = true;
254        }
255        due
256    }
257}
258
259/// One step of a remote float catching up with the last received value over
260/// the send interval, so radial changes look smooth between packets.
261pub fn smooth_step(current: f32, target: f32, dt: f32, interval: f32) -> f32 {
262    let k = if interval <= 1e-4 { 1.0 } else { (dt / interval).clamp(0.0, 1.0) };
263    current + (target - current) * k
264}

285    #[test]
286    fn schema_is_sorted_synced_only_and_order_independent() {
287        let a = AvParamSchema::from_params(&sample());
288        assert_eq!(a.entries().iter().map(|e| e.name.as_str()).collect::<Vec<_>>(), vec!["Cape", "Hat", "Outfit", "Smile"]);
289        let mut rev = sample();
290        rev.reverse();
291        assert_eq!(AvParamSchema::from_params(&rev).hash(), a.hash());
292        // 2 numbers + 2 bools in one byte.
293        assert_eq!(a.payload_bytes(), 3);
294        assert_eq!(a.packet_bytes(), 7);

297    #[test]
298    fn round_trip_quantises_like_a_remote_copy() {
299        let s = AvParamSchema::from_params(&sample());
300        // Cape, Hat, Outfit, Smile
301        let values = [1.0, 0.0, 200.0, 0.5];
302        let mut bytes = Vec::new();
303        s.encode_values(&values, &mut bytes);
304        assert_eq!(bytes.len(), s.packet_bytes());
305        let mut back = Vec::new();
306        s.decode_values(&bytes, &mut back).unwrap();
307        assert_eq!(back[0], 1.0);
308        assert_eq!(back[1], 0.0);
309        assert_eq!(back[2], 200.0);
310        assert!((back[3] - 0.5).abs() < 0.005 && back[3] != 0.5, "0.5 arrives as 64/127: {}", back[3]);
311        assert_eq!(s.quantise(&values), back);
312        // Out-of-range values clamp to what the wire can hold.
313        let mut b2 = Vec::new();
314        s.encode_values(&[1.0, 1.0, 999.0, -7.0], &mut b2);
315        s.decode_values(&b2, &mut back).unwrap();
316        assert_eq!((back[2], back[3]), (255.0, -1.0));
317        // Negative floats keep their sign.
318        let mut b3 = Vec::new();
319        s.encode_values(&[0.0, 0.0, 0.0, -0.25], &mut b3);
320        s.decode_values(&b3, &mut back).unwrap();
321        assert!((back[3] + 0.25).abs() < 0.005);
322    }

338    #[test]
339    fn schema_hash_mismatch_is_rejected() {
340        let a = AvParamSchema::from_params(&sample());
341        let mut other = sample();
342        other.push(p("Extra", AvatarParamKind::Bool));
343        let b = AvParamSchema::from_params(&other);
344        assert_ne!(a.hash(), b.hash());
345        let mut bytes = Vec::new();
346        a.encode_values(&[0.0; 4], &mut bytes);
347        let mut out = Vec::new();
348        assert_eq!(b.decode_values(&bytes, &mut out), Err(SyncError::SchemaMismatch { got: a.hash(), want: b.hash() }));
349        // A kind change alone also changes the hash.
350        let mut kinds = sample();
351        kinds[0].kind = AvatarParamKind::Int;
352        assert_ne!(AvParamSchema::from_params(&kinds).hash(), a.hash());
353        // Truncated and padded packets are refused too.
354        let mut ok = Vec::new();
355        a.encode_values(&[0.0; 4], &mut ok);
356        assert!(matches!(a.decode_values(&ok[..ok.len() - 1], &mut out), Err(SyncError::BadLength { .. })));
357        ok.push(0);
358        assert!(matches!(a.decode_values(&ok, &mut out), Err(SyncError::BadLength { .. })));
359        assert!(matches!(a.decode_values(&[1, 2], &mut out), Err(SyncError::BadLength { .. })));
360    }

382    #[test]
383    fn clock_sends_changes_at_ten_hz_puppets_faster_and_repeats_every_second() {
384        let mut c = SyncClock::default();
385        assert!(c.poll(0.0, false, false), "the first poll sends the state");
386        assert!(!c.poll(0.05, true, false), "coalesced inside 100 ms");
387        assert!(c.poll(0.11, true, false));
388        assert!(!c.poll(0.15, false, false), "nothing changed");
389        assert!(!c.poll(0.13, true, true), "fast mode still needs 33 ms");
390        assert!(c.poll(0.11 + FAST_INTERVAL + 1e-6, true, true));
391        let t = 0.11 + FAST_INTERVAL + 1e-6;
392        assert!(!c.poll(t + 0.9, false, false));
393        assert!(c.poll(t + 1.01, false, false), "unchanged state repeats once a second");
394    }
