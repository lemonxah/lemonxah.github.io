6impl ClothPanel{
7    /// Create a simple, possibly concave panel. Subdivision shares edge
8    /// midpoints, so the resulting mesh has no cracks or duplicate seam edges.
9    pub fn from_outline(outline:&[[f32;2]],subdivisions:u32)->Result<Self,String>{
10        if !(3..=512).contains(&outline.len())||subdivisions>6||outline.iter().flatten().any(|v|!v.is_finite()){
11            return Err("An outline needs 3–512 finite points and 0–6 subdivisions".into());
12        }
13        let mut p:Vec<_>=outline.iter().map(|&p|Vec2::from(p).as_dvec2()).collect();
14        let extent=p.iter().copied().fold(glam::DVec2::splat(f64::NEG_INFINITY),glam::DVec2::max)-p.iter().copied().fold(glam::DVec2::splat(f64::INFINITY),glam::DVec2::min);
15        let epsilon=extent.length_squared()*1e-12;
16        if epsilon<=1e-24{return Err("The outline has no area".into());}
17        // Closing the outline with a repeated first point is harmless.
18        if p[0].distance_squared(*p.last().unwrap())<=epsilon{p.pop();}
19        if p.len()<3||p.iter().enumerate().any(|(i,p0)|p0.distance_squared(p[(i+1)%p.len()])<=epsilon){return Err("Outline points must be distinct".into());}
20        for i in 0..p.len(){for j in i+1..p.len(){
21            if j==i+1||(i==0&&j+1==p.len()){continue;}
22            let(a,b,c,d)=(p[i],p[(i+1)%p.len()],p[j],p[(j+1)%p.len()]);
23            if a.min(b).cmple(c.max(d)+glam::DVec2::splat(epsilon.sqrt())).all()&&c.min(d).cmple(a.max(b)+glam::DVec2::splat(epsilon.sqrt())).all()
24                &&orient(a,b,c)*orient(a,b,d)<=epsilon*epsilon&&orient(c,d,a)*orient(c,d,b)<=epsilon*epsilon{
25                return Err("The outline crosses or touches itself".into());
26            }
27        }}
28        // Collinear clicks do not need their own simulation vertex; generated
29        // boundary subdivision supplies the requested edge density instead.
30        loop{let redundant=(0..p.len()).find(|&i|orient(p[(i+p.len()-1)%p.len()],p[i],p[(i+1)%p.len()]).abs()<=epsilon);
31            if let Some(i)=redundant{if p.len()<=3{return Err("The outline has no area".into());}p.remove(i);}else{break;}
32        }
33        let area=(0..p.len()).map(|i|p[i].perp_dot(p[(i+1)%p.len()])).sum::<f64>();
34        if area.abs()<=epsilon{return Err("The outline has no area".into());}if area<0.0{p.reverse();}
35        if (p.len()-2)*4usize.pow(subdivisions)>500_000{return Err("The subdivided panel exceeds 500,000 triangles".into());}
36        let mut polygon=(0..p.len()).collect::<Vec<_>>();let mut triangles=Vec::new();
37        while polygon.len()>3{
38            let ear=(0..polygon.len()).find(|&i|{
39                let ids=[polygon[(i+polygon.len()-1)%polygon.len()],polygon[i],polygon[(i+1)%polygon.len()]];let[a,b,c]=ids.map(|i|p[i]);
40                orient(a,b,c)>epsilon&&!polygon.iter().any(|j|!ids.contains(j)&&orient(a,b,p[*j])>=-epsilon&&orient(b,c,p[*j])>=-epsilon&&orient(c,a,p[*j])>=-epsilon)
41            }).ok_or("The outline cannot be triangulated; remove overlapping or very thin edges")?;
42            triangles.extend([polygon[(ear+polygon.len()-1)%polygon.len()]as u32,polygon[ear]as u32,polygon[(ear+1)%polygon.len()]as u32]);polygon.remove(ear);
43        }
44        triangles.extend(polygon.into_iter().map(|i|i as u32));let mut vertices=p.into_iter().map(|p|p.as_vec2().to_array()).collect::<Vec<_>>();
45        for _ in 0..subdivisions{
46            let mut midpoints=BTreeMap::new();let mut refined=Vec::with_capacity(triangles.len()*4);
47            for t in triangles.chunks_exact(3){let mut edge=[0;3];for i in 0..3{let(a,b)=(t[i],t[(i+1)%3]);let key=(a.min(b),a.max(b));
48                edge[i]=*midpoints.entry(key).or_insert_with(||{let i=vertices.len()as u32;vertices.push(((Vec2::from(vertices[a as usize])+Vec2::from(vertices[b as usize]))*0.5).to_array());i});
49            }
50            let[a,b,c]=[t[0],t[1],t[2]];let[ab,bc,ca]=edge;refined.extend([a,ab,ca,ab,b,bc,ca,bc,c,ab,bc,ca]);}
51            if vertices.len()>250_000{return Err("The subdivided panel exceeds 250,000 vertices".into());}triangles=refined;
52        }
53        Ok(Self{name:"Outline panel".into(),vertices,triangles,..Default::default()})
54    }
55    /// Move one custom vertex without changing its identity, painted masks or
56    /// seam references. Reject collapsed/inverted incident triangles atomically.
57    pub fn move_vertex(&mut self,index:usize,position:[f32;2])->Result<(),String>{
58        if index>=self.vertices.len()||position.iter().any(|v|!v.is_finite()){return Err("Choose a valid vertex and finite position".into());}
59        let target=Vec2::from(position).as_dvec2();
60        for t in self.triangles.chunks_exact(3).filter(|t|t.contains(&(index as u32))){
61            let mut p=[glam::DVec2::ZERO;3];for k in 0..3{p[k]=Vec2::from(*self.vertices.get(t[k]as usize).ok_or("Invalid panel triangle")?).as_dvec2();}
62            let old=orient(p[0],p[1],p[2]);let extent=(p[1]-p[0]).length_squared().max((p[2]-p[0]).length_squared());
63            for k in 0..3{if t[k]as usize==index{p[k]=target;}}
64            let new=orient(p[0],p[1],p[2]);if old*new<=0.0||new.abs()<=extent*1e-8{return Err("That move would collapse or invert a triangle".into());}
65        }
66        self.vertices[index]=position;Ok(())
67    }
68    pub fn boundary_loops(&self)->Result<Vec<Vec<u32>>,String>{
69        let (_,triangles)=self.mesh()?;let mut counts=BTreeMap::new();
70        for t in triangles.chunks_exact(3){for i in 0..3{let(a,b)=(t[i],t[(i+1)%3]);*counts.entry((a.min(b),a.max(b))).or_insert(0usize)+=1;}}
71        if counts.values().any(|&n|n>2){return Err("Panel edges must be manifold".into());}
72        let mut adjacent:BTreeMap<u32,Vec<u32>>=BTreeMap::new();for (&(a,b),&count)in &counts{if count==1{adjacent.entry(a).or_default().push(b);adjacent.entry(b).or_default().push(a);}}
73        if adjacent.values().any(|v|v.len()!=2){return Err("Panel boundaries must form simple loops".into());}
74        let mut remaining=adjacent.keys().copied().collect::<std::collections::BTreeSet<_>>();let mut loops=Vec::new();
75        while let Some(&first)=remaining.first(){let mut chain=vec![first];remaining.remove(&first);let mut last=first;let mut next=adjacent[&first][0];
76            while next!=first{if !remaining.remove(&next){return Err("Panel boundary overlaps itself".into());}chain.push(next);let choices=&adjacent[&next];let after=if choices[0]==last{choices[1]}else{choices[0]};last=next;next=after;}
77            loops.push(chain);
78        }Ok(loops)
79    }
80    /// Ordered seam vertices between two boundary picks. The default is the
81    /// shorter boundary arc in metres; alternate selects the other arc.
82    pub fn boundary_path(&self,start:u32,end:u32,alternate:bool)->Result<Vec<u32>,String>{
83        if start==end{return Err("Choose different seam endpoints".into());}
84        let chain=self.boundary_loops()?.into_iter().find(|c|c.contains(&start)&&c.contains(&end)).ok_or("Seam endpoints must be on the same boundary")?;
85        let index=chain.iter().position(|&i|i==start).unwrap();let mut paths=[Vec::new(),Vec::new()];
86        for side in 0..2{for k in 0..chain.len(){let at=if side==0{(index+k)%chain.len()}else{(index+chain.len()-k)%chain.len()};let v=chain[at];paths[side].push(v);if v==end{break;}}}
87        let(p,_)=self.mesh()?;let length=|path:&[u32]|path.windows(2).map(|e|p[e[0]as usize].distance(p[e[1]as usize])).sum::<f32>();
88        let shortest=usize::from(length(&paths[1])<length(&paths[0]));Ok(paths[shortest^usize::from(alternate)].clone())
89    }

91impl ClothPattern{
92    /// Turn a generated grid into editable vertices without invalidating its
93    /// side-based seams or changing any existing vertex/stitch identity.
94    pub fn make_panel_editable(&mut self,index:usize)->Result<(),String>{
95        let panel=self.panels.get(index).ok_or("Missing panel")?;if !panel.vertices.is_empty(){return Ok(());}
96        let(vertices,triangles)=panel.mesh()?;let mut seams=self.seams.clone();
97        for seam in &mut seams{for edge in [&mut seam.a,&mut seam.b]{if edge.panel==index{edge.vertices=panel.edge(edge,vertices.len(),&triangles)?;}}}
98        self.seams=seams;self.panels[index].vertices=vertices.into_iter().map(|p|p.to_array()).collect();self.panels[index].triangles=triangles;Ok(())
99    }

107        let outline=[[0.0,0.0],[2.0,0.0],[2.0,1.0],[1.0,1.0],[1.0,2.0],[0.0,2.0]];
108        for reverse in [false,true]{let mut points=outline.to_vec();if reverse{points.reverse();}
109            let panel=ClothPanel::from_outline(&points,3).unwrap();let(p,t)=panel.mesh().unwrap();
110            let area=t.chunks_exact(3).map(|t|{let a=orient(p[t[0]as usize].as_dvec2(),p[t[1]as usize].as_dvec2(),p[t[2]as usize].as_dvec2())*0.5;assert!(a>0.0);a}).sum::<f64>();assert!((area-3.0).abs()<1e-6);
111            let loops=panel.boundary_loops().unwrap();assert_eq!(loops.len(),1);assert_eq!(loops[0].len(),outline.len()*8);
