1//! Destructible objects: health and deterministic debris scatter when an
2//! object breaks.
3
4use glam::{Quat, Vec3};
5use serde::{Deserialize, Serialize};
6
7/// A destructible object's health.
8#[derive(Clone, Copy, Debug, PartialEq, Serialize, Deserialize)]
9#[serde(default)]
10pub struct Destructible {
11    /// Current health.
12    pub health: f32,
13    /// Health when intact.
14    pub max_health: f32,
15}
16
17impl Default for Destructible {
18    fn default() -> Self {
19        Self { health: 100.0, max_health: 100.0 }
20    }
21}
22
23impl Destructible {
24    /// Apply damage; returns true when the object breaks (health reaches zero).
25    pub fn damage(&mut self, amount: f32) -> bool {
26        self.health = (self.health - amount.max(0.0)).max(0.0);
27        self.health <= 0.0
28    }
29}
30
31/// One debris piece after an object breaks.
32#[derive(Clone, Copy, Debug, PartialEq)]
33pub struct Debris {
34    /// Launch velocity (m/s).
35    pub velocity: Vec3,
36    /// Spin orientation.
37    pub rotation: Quat,
38}
39
40fn hash01(seed: u32, i: usize) -> f32 {
41    let mut n = seed.wrapping_add((i as u32).wrapping_mul(0x9E37_79B9));
42    n = (n ^ (n >> 16)).wrapping_mul(0x7FEB_352D);
43    n ^= n >> 15;
44    (n & 0xF_FFFF) as f32 / 0xF_FFFF as f32
45}
46
47fn unit_dir(seed: u32, i: usize) -> Vec3 {
48    let u = hash01(seed, i) * 2.0 - 1.0;
49    let phi = hash01(seed, i.wrapping_add(0x4000_0000)) * std::f32::consts::TAU;
50    let s = (1.0 - u * u).max(0.0).sqrt();
51    Vec3::new(s * phi.cos(), u, s * phi.sin())
52}
53
54/// Deterministically scatter `count` debris pieces: an isotropic direction with
55/// a speed in `[0.5, 1.0] × strength`, and a random spin, seeded for replay.
56pub fn scatter_debris(strength: f32, count: usize, seed: u32) -> Vec<Debris> {
57    (0..count)
58        .map(|i| {
59            let speed = strength * (0.5 + 0.5 * hash01(seed, i.wrapping_add(0x2000_0000)));
60            let spin = hash01(seed, i.wrapping_add(0x6000_0000)) * std::f32::consts::TAU;
61            Debris { velocity: unit_dir(seed, i) * speed, rotation: Quat::from_rotation_x(spin) }
62        })
63        .collect()
64}

66#[cfg(test)]
67mod tests {
68    use super::*;
69
70    #[test]
71    fn destructible_breaks_when_health_runs_out() {
72        let mut d = Destructible::default();
73        assert!(!d.damage(50.0));
74        assert_eq!(d.health, 50.0);
75        assert!(d.damage(60.0));
76        assert_eq!(d.health, 0.0);
77    }
78
79    #[test]
80    fn debris_is_deterministic_and_bounded() {
81        let a = scatter_debris(10.0, 8, 42);
82        let b = scatter_debris(10.0, 8, 42);
83        assert_eq!(a.len(), 8);
84        assert_eq!(a, b, "same seed must scatter identically");
85        for d in &a {
86            assert!(d.velocity.length() <= 10.0 + 1e-4);
87            assert!(d.velocity.length() >= 5.0 - 1e-4);
88        }
89    }
