9/// How curved segments are cut into line pieces.
10#[derive(Clone, Copy, PartialEq, Debug)]
11pub enum TessMode {
12    /// Subdivide until the polyline stays within `tolerance` of the curve.
13    Adaptive {
14        /// Largest allowed distance between the curve and its polyline.
15        tolerance: f32,
16    },
17    /// `detail` equal parameter steps per curved segment.
18    Fixed {
19        /// Pieces per curved segment.
20        detail: u32,
21    },
22}

24/// The tessellated spline. Segment `i` runs `pts[i] -> pts[i + 1]` (and, for a
25/// closed polyline, the last point back to the first).
26#[derive(Clone, Debug, Default, PartialEq)]
27pub struct Poly {
28    /// The polyline's points.
29    pub pts: Vec<Vec2>,
30    /// Interpolated point height at every vertex.
31    pub height: Vec<f32>,
32    /// The control point a vertex is, `None` for points inside a curve.
33    pub ctrl: Vec<Option<usize>>,
34    /// Per segment, the control point whose segment it belongs to (its sprite
35    /// variant, height owner).
36    pub owner: Vec<usize>,
37    /// Closed (a ring) or open (a strip).
38    pub closed: bool,
39}
40
41impl Poly {
42    /// Number of points.
43    pub fn len(&self) -> usize {
44        self.pts.len()
45    }
46    /// True when the polyline has no points.
47    pub fn is_empty(&self) -> bool {
48        self.pts.is_empty()
49    }
50    /// Number of segments (a closed ring has one per point).
51    pub fn segment_count(&self) -> usize {
52        if self.pts.len() < 2 { 0 } else if self.closed { self.pts.len() } else { self.pts.len() - 1 }
53    }
54    /// End vertex of segment `s`.
55    pub fn next(&self, s: usize) -> usize {
56        (s + 1) % self.pts.len()
57    }
58    /// Total length of the polyline.
59    pub fn length(&self) -> f32 {
60        (0..self.segment_count()).map(|s| self.pts[s].distance(self.pts[self.next(s)])).sum()
61    }
62    /// The same polyline walked backwards (owners and control flags follow).
63    pub fn reversed(&self) -> Poly {
64        let n = self.pts.len();
65        let segs = self.segment_count();
66        let owner = (0..segs)
67            .map(|j| {
68                // Reversed segment j runs old vertex n-1-j -> n-2-j, i.e. old segment n-2-j (mod n).
69                let old = if self.closed { (2 * n - 2 - j) % n } else { n - 2 - j };
70                self.owner[old]
71            })
72            .collect();
73        Poly {
74            pts: self.pts.iter().rev().copied().collect(),
75            height: self.height.iter().rev().copied().collect(),
76            ctrl: self.ctrl.iter().rev().copied().collect(),
77            owner,
78            closed: self.closed,
79        }
80    }
81}

87/// The four bezier points of the segment from control point `a` to `b`.
88pub fn segment_bezier(a: &ShapePoint, b: &ShapePoint) -> [Vec2; 4] {
89    let (p0, p3) = (v(a.position), v(b.position));
90    [p0, p0 + v(a.out_handle()), p3 + v(b.in_handle()), p3]
91}

93/// A cubic bezier point at parameter `t` (0..1).
94pub fn bezier_at(c: &[Vec2; 4], t: f32) -> Vec2 {
95    let u = 1.0 - t;
96    c[0] * (u * u * u) + c[1] * (3.0 * u * u * t) + c[2] * (3.0 * u * t * t) + c[3] * (t * t * t)
97}

99fn is_straight(c: &[Vec2; 4]) -> bool {
100    c[0].distance_squared(c[1]) < 1e-12 && c[2].distance_squared(c[3]) < 1e-12
101}

103fn dist_to_line(p: Vec2, a: Vec2, b: Vec2) -> f32 {
104    let ab = b - a;
105    let l2 = ab.length_squared();
106    if l2 < 1e-18 {
107        return p.distance(a);
108    }
109    ((p - a).perp_dot(ab)).abs() / l2.sqrt()
110}

112const MAX_DEPTH: u32 = 18;
113
114/// Flatten one bezier into `(point, t)` pieces after the start point. The
115/// control polygon's distance from the chord bounds the curve's by 3/4, so the
116/// test `0.75 * max(d1, d2) <= tol` keeps every piece within `tol` of the curve.
117pub fn flatten_bezier(c: &[Vec2; 4], tol: f32, out: &mut Vec<(Vec2, f32)>) {
118    fn rec(c: [Vec2; 4], t0: f32, t1: f32, tol: f32, depth: u32, out: &mut Vec<(Vec2, f32)>) {
119        let d = dist_to_line(c[1], c[0], c[3]).max(dist_to_line(c[2], c[0], c[3]));
120        if depth >= MAX_DEPTH || 0.75 * d <= tol {
121            out.push((c[3], t1));
122            return;
123        }
124        let (a01, a12, a23) = ((c[0] + c[1]) * 0.5, (c[1] + c[2]) * 0.5, (c[2] + c[3]) * 0.5);
125        let (b0, b1) = ((a01 + a12) * 0.5, (a12 + a23) * 0.5);
126        let m = (b0 + b1) * 0.5;
127        let tm = (t0 + t1) * 0.5;
128        rec([c[0], a01, b0, m], t0, tm, tol, depth + 1, out);
129        rec([m, b1, a23, c[3]], tm, t1, tol, depth + 1, out);
130    }
131    if is_straight(c) {
132        out.push((c[3], 1.0));
133        return;
134    }
135    rec(*c, 0.0, 1.0, tol.max(1e-6), 0, out);
136}

138/// Equal parameter steps.
139pub fn flatten_fixed(c: &[Vec2; 4], detail: u32, out: &mut Vec<(Vec2, f32)>) {
140    if is_straight(c) {
141        out.push((c[3], 1.0));
142        return;
143    }
144    let n = detail.max(1);
145    for i in 1..=n {
146        let t = i as f32 / n as f32;
147        out.push((if i == n { c[3] } else { bezier_at(c, t) }, t));
148    }
149}

151/// Tessellate a spline. Straight segments stay one piece. `max_points` bounds
152/// the vertex count: the adaptive tolerance is doubled (and the fixed detail
153/// reduced) until the result fits, down to one piece per segment, so the cap
154/// can only be exceeded by a spline with more control points than the cap.
155/// Vertices closer than 1e-5 are merged.
156pub fn tessellate(points: &[ShapePoint], closed: bool, mode: TessMode, max_points: u32) -> Poly {
157    let n = points.len();
158    if n == 0 {
159        return Poly { closed, ..Default::default() };
160    }
161    if n == 1 {
162        return Poly { pts: vec![v(points[0].position)], height: vec![points[0].height], ctrl: vec![Some(0)], owner: vec![], closed };
163    }
164    let segs = if closed { n } else { n - 1 };
165    let curves: Vec<[Vec2; 4]> = (0..segs).map(|k| segment_bezier(&points[k], &points[(k + 1) % n])).collect();
166    let cap = (max_points as usize).max(segs + 1);
167    type Piece = (Vec2, f32);
168    let build = |pieces: &dyn Fn(&[Vec2; 4], &mut Vec<Piece>)| -> Vec<Vec<Piece>> {
169        curves
170            .iter()
171            .map(|c| {
172                let mut o = Vec::new();
173                pieces(c, &mut o);
174                o
175            })
176            .collect()
177    };
178    let count = |s: &[Vec<(Vec2, f32)>]| s.iter().map(Vec::len).sum::<usize>() + 1 - closed as usize;
179    let pieces = match mode {
180        TessMode::Adaptive { tolerance } => {
181            let mut tol = tolerance.max(1e-6);
182            loop {
183                let s = build(&|c, o| flatten_bezier(c, tol, o));
184                if count(&s) <= cap || tol > 1e7 {
185                    break s;
186                }
187                tol *= 2.0;
188            }
189        }
190        TessMode::Fixed { detail } => {
191            let mut d = detail.max(1);
192            loop {
193                let s = build(&|c, o| flatten_fixed(c, d, o));
194                if count(&s) <= cap || d == 1 {
195                    break s;
196                }
197                // Scale down in proportion, at least by one.
198                let over = count(&s) as f32 / cap as f32;
199                d = ((d as f32 / over).floor() as u32).clamp(1, d - 1);
200            }
201        }
202    };
203    let mut poly = Poly { closed, ..Default::default() };
204    let mut arrive: Vec<usize> = Vec::new(); // owner of the segment arriving at each vertex
205    let push = |poly: &mut Poly, p: Vec2, h: f32, ctrl: Option<usize>, owner: usize, arrive: &mut Vec<usize>| {
206        if let Some(&last) = poly.pts.last()
207            && last.distance_squared(p) < 1e-10
208        {
209            if poly.ctrl.last() == Some(&None) {
210                *poly.ctrl.last_mut().unwrap() = ctrl;
211            }
212            return;
213        }
214        poly.pts.push(p);
215        poly.height.push(h);
216        poly.ctrl.push(ctrl);
217        arrive.push(owner);
218    };
219    push(&mut poly, curves[0][0], points[0].height, Some(0), 0, &mut arrive);
220    let mut closing_owner = 0;
221    for (k, ps) in pieces.iter().enumerate() {
222        let (ha, hb) = (points[k].height, points[(k + 1) % n].height);
223        for (i, &(p, t)) in ps.iter().enumerate() {
224            let last = i + 1 == ps.len();
225            if closed && k + 1 == segs && last {
226                closing_owner = k; // the wrap-around back to point 0: no new vertex
227                continue;
228            }
229            let ctrl = last.then_some((k + 1) % n);
230            push(&mut poly, p, ha + (hb - ha) * t, ctrl, k, &mut arrive);
231        }
232    }
233    // Closing vertex coincident with the first one (a zero-length last curve).
234    if closed && poly.pts.len() > 1 && poly.pts[0].distance_squared(*poly.pts.last().unwrap()) < 1e-10 {
235        poly.pts.pop();
236        poly.height.pop();
237        poly.ctrl.pop();
238        arrive.pop();
239    }
240    let m = poly.pts.len();
241    let segs_out = if m < 2 { 0 } else if closed { m } else { m - 1 };
242    poly.owner = (0..segs_out).map(|j| if j + 1 < m { arrive[j + 1] } else { closing_owner }).collect();
243    poly
244}

250/// Shoelace area, positive for counter-clockwise rings.
251pub fn signed_area(p: &[Vec2]) -> f32 {
252    let n = p.len();
253    if n < 3 {
254        return 0.0;
255    }
256    let mut a = 0.0f64;
257    for i in 0..n {
258        let (u, w) = (p[i], p[(i + 1) % n]);
259        a += u.x as f64 * w.y as f64 - w.x as f64 * u.y as f64;
260    }
261    (a * 0.5) as f32
262}

264fn cross(o: Vec2, a: Vec2, b: Vec2) -> f32 {
265    (a - o).perp_dot(b - o)
266}

268fn in_triangle(p: Vec2, a: Vec2, b: Vec2, c: Vec2) -> bool {
269    let (d1, d2, d3) = (cross(a, b, p), cross(b, c, p), cross(c, a, p));
270    d1 >= 0.0 && d2 >= 0.0 && d3 >= 0.0
271}

273/// Ear-clipping triangulation of a simple polygon (no holes) in either
274/// orientation. Returns CCW triangles as indices into `pts`. Collinear vertices
275/// are dropped; a self-intersecting ring still terminates (the least-bad ear is
276/// clipped) but its fill is wrong where it crosses itself.
277pub fn triangulate(pts: &[Vec2]) -> Vec<[u32; 3]> {
278    let n = pts.len();
279    if n < 3 {
280        return Vec::new();
281    }
282    let mut idx: Vec<u32> = (0..n as u32).collect();
283    if signed_area(pts) < 0.0 {
284        idx.reverse();
285    }
286    // Drop vertices that add nothing: duplicates and collinear points.
287    let scale = pts.iter().fold(0.0f32, |m, p| m.max(p.x.abs()).max(p.y.abs())).max(1.0);
288    let eps = 1e-7 * scale * scale;
289    loop {
290        let m = idx.len();
291        if m < 3 {
292            return Vec::new();
293        }
294        let before = m;
295        let mut i = 0;
296        while i < idx.len() {
297            let m = idx.len();
298            let (a, b, c) = (pts[idx[(i + m - 1) % m] as usize], pts[idx[i] as usize], pts[idx[(i + 1) % m] as usize]);
299            if cross(a, b, c).abs() <= eps && (b - a).dot(c - b) >= 0.0 {
300                idx.remove(i);
301            } else {
302                i += 1;
303            }
304            if idx.len() < 3 {
305                return Vec::new();
306            }
307        }
308        if idx.len() == before {
309            break;
310        }
311    }
312    let mut tris = Vec::with_capacity(idx.len().saturating_sub(2));
313    let mut i = 0;
314    let mut stalled = 0;
315    while idx.len() > 3 {
316        let m = idx.len();
317        i %= m;
318        let (ia, ib, ic) = (idx[(i + m - 1) % m], idx[i], idx[(i + 1) % m]);
319        let (a, b, c) = (pts[ia as usize], pts[ib as usize], pts[ic as usize]);
320        let convex = cross(a, b, c) > eps;
321        let ear = convex && !idx.iter().any(|&k| k != ia && k != ib && k != ic && {
322            let p = pts[k as usize];
323            // Vertices coincident with a corner (a pinched ring) don't block.
324            p != a && p != b && p != c && in_triangle(p, a, b, c)
325        });
326        if ear {
327            tris.push([ia, ib, ic]);
328            idx.remove(i);
329            stalled = 0;
330        } else {
331            i += 1;
332            stalled += 1;
333            if stalled > m {
334                // No ear anywhere: the ring crosses itself. Clip the sharpest convex corner.
335                let best = (0..m)
336                    .filter(|&k| cross(pts[idx[(k + m - 1) % m] as usize], pts[idx[k] as usize], pts[idx[(k + 1) % m] as usize]) > 0.0)
337                    .max_by(|&x, &y| {
338                        let f = |k: usize| cross(pts[idx[(k + m - 1) % m] as usize], pts[idx[k] as usize], pts[idx[(k + 1) % m] as usize]);
339                        f(x).total_cmp(&f(y))
340                    })
341                    .unwrap_or(0);
342                let (ia, ib, ic) = (idx[(best + m - 1) % m], idx[best], idx[(best + 1) % m]);
343                if cross(pts[ia as usize], pts[ib as usize], pts[ic as usize]) > 0.0 {
344                    tris.push([ia, ib, ic]);
345                }
346                idx.remove(best);
347                stalled = 0;
348            }
349        }
350    }
351    if idx.len() == 3 && cross(pts[idx[0] as usize], pts[idx[1] as usize], pts[idx[2] as usize]) > 0.0 {
352        tris.push([idx[0], idx[1], idx[2]]);
353    }
354    tris
355}

359/// Even-odd point in ring.
360pub fn point_in_ring(ring: &[Vec2], p: Vec2) -> bool {
361    let n = ring.len();
362    let mut inside = false;
363    let mut j = n.wrapping_sub(1);
364    for i in 0..n {
365        let (a, b) = (ring[i], ring[j]);
366        if (a.y > p.y) != (b.y > p.y) && p.x < (b.x - a.x) * (p.y - a.y) / (b.y - a.y) + a.x {
367            inside = !inside;
368        }
369        j = i;
370    }
371    inside
372}

374/// Whether segments `ab` and `cd` cross at a point interior to both (touching at shared
375/// endpoints or collinear overlap are not crossings here).
376fn proper_cross(a: Vec2, b: Vec2, c: Vec2, d: Vec2) -> bool {
377    let (d1, d2) = (cross(a, b, c), cross(a, b, d));
378    let (d3, d4) = (cross(c, d, a), cross(c, d, b));
379    ((d1 > 0.0 && d2 < 0.0) || (d1 < 0.0 && d2 > 0.0)) && ((d3 > 0.0 && d4 < 0.0) || (d3 < 0.0 && d4 > 0.0))
380}

382/// Whether the diagonal from hole vertex `m` to ring vertex `v` stays inside the filled region:
383/// it crosses no edge of the working ring or of the hole itself, touches no other vertex, and its
384/// midpoint is inside the ring but not inside the hole.
385fn bridge_ok(ring: &[Vec2], hole: &[Vec2], m: Vec2, v: Vec2) -> bool {
386    let scale = (m - v).length().max(1e-9);
387    let near_seg = |q: Vec2| {
388        if q == m || q == v {
389            return false;
390        }
391        let ab = v - m;
392        let t = ((q - m).dot(ab) / ab.length_squared()).clamp(0.0, 1.0);
393        q.distance(m + ab * t) < 1e-6 * scale.max(1.0)
394    };
395    for set in [ring, hole] {
396        let n = set.len();
397        for i in 0..n {
398            let (c, d) = (set[i], set[(i + 1) % n]);
399            if proper_cross(m, v, c, d) || near_seg(c) {
400                return false;
401            }
402        }
403    }
404    let mid = (m + v) * 0.5;
405    point_in_ring(ring, mid) && !point_in_ring(hole, mid)
406}

408/// Triangulate `outer` minus `holes`: each hole is joined to the outer ring by a zero-width
409/// bridge (two coincident edges), holes taken left to right, and the single weakly simple ring
410/// is ear clipped. A hole with no valid bridge (outside the outer ring, crossing it) is
411/// skipped. Returns the ring's vertices (bridge ends duplicated) and CCW triangles into them,
412/// so the triangle area equals the outer area minus the holes' (for holes that do not overlap
413/// each other).
414pub fn triangulate_with_holes(outer: &[Vec2], holes: &[Vec<Vec2>]) -> (Vec<Vec2>, Vec<[u32; 3]>) {
415    let mut ring: Vec<Vec2> = outer.to_vec();
416    if signed_area(&ring) < 0.0 {
417        ring.reverse();
418    }
419    // Holes clockwise, leftmost first.
420    let mut hs: Vec<Vec<Vec2>> = holes.iter().filter(|h| h.len() >= 3 && signed_area(h).abs() > 1e-12).cloned().collect();
421    for h in &mut hs {
422        if signed_area(h) > 0.0 {
423            h.reverse();
424        }
425    }
426    let leftmost = |h: &Vec<Vec2>| h.iter().enumerate().fold((0usize, f32::INFINITY), |a, (i, p)| if p.x < a.1 { (i, p.x) } else { a });
427    hs.sort_by(|a, b| leftmost(a).1.total_cmp(&leftmost(b).1));
428    for hole in &hs {
429        let (li, _) = leftmost(hole);
430        let m = hole[li];
431        // Nearest ring vertices first; the first that sees the hole's vertex wins.
432        let mut order: Vec<usize> = (0..ring.len()).collect();
433        order.sort_by(|&a, &b| ring[a].distance_squared(m).total_cmp(&ring[b].distance_squared(m)));
434        let Some(bi) = order.into_iter().find(|&i| bridge_ok(&ring, hole, m, ring[i])) else { continue };
435        let mut merged = Vec::with_capacity(ring.len() + hole.len() + 2);
436        merged.extend_from_slice(&ring[..=bi]);
437        for k in 0..=hole.len() {
438            merged.push(hole[(li + k) % hole.len()]);
439        }
440        merged.extend_from_slice(&ring[bi..]);
441        ring = merged;
442    }
443    let tris = triangulate(&ring);
444    (ring, tris)
445}

447/// Move every vertex of a ring along its outward miter normal by `d`
448/// (positive grows the polygon, either orientation). The miter is limited to
449/// twice the distance so a sharp spike cannot shoot off.
450pub fn offset_ring(pts: &[Vec2], d: f32) -> Vec<Vec2> {
451    let n = pts.len();
452    if n < 3 || d == 0.0 {
453        return pts.to_vec();
454    }
455    let ccw = signed_area(pts) >= 0.0;
456    // Outward normal of an edge: right of travel for CCW, left for CW.
457    let out = |a: Vec2, b: Vec2| -> Vec2 {
458        let e = (b - a).normalize_or_zero();
459        if ccw { Vec2::new(e.y, -e.x) } else { Vec2::new(-e.y, e.x) }
460    };
461    (0..n)
462        .map(|i| {
463            let (p, a, b) = (pts[i], pts[(i + n - 1) % n], pts[(i + 1) % n]);
464            let (n0, n1) = (out(a, p), out(p, b));
465            let denom = 1.0 + n0.dot(n1);
466            let mut m = if denom > 1e-3 { (n0 + n1) / denom } else { n0 };
467            if m.length() > 2.0 {
468                m = m.normalize() * 2.0;
469            }
470            p + m * d
471        })
472        .collect()
473}

475fn seg_dist(p: Vec2, a: Vec2, b: Vec2) -> f32 {
476    let ab = b - a;
477    let l2 = ab.length_squared();
478    if l2 < 1e-18 {
479        return p.distance(a);
480    }
481    let t = ((p - a).dot(ab) / l2).clamp(0.0, 1.0);
482    p.distance(a + ab * t)
483}

485/// Distance from `p` to the polyline `pts` (closed = also the last-to-first edge).
486pub fn distance_to_polyline(p: Vec2, pts: &[Vec2], closed: bool) -> f32 {
487    let n = pts.len();
488    match n {
489        0 => f32::INFINITY,
490        1 => p.distance(pts[0]),
491        _ => (0..if closed { n } else { n - 1 }).map(|i| seg_dist(p, pts[i], pts[(i + 1) % n])).fold(f32::INFINITY, f32::min),
492    }
493}

495/// Douglas-Peucker. An open polyline keeps both ends; a closed ring is split at
496/// its two farthest-apart vertices so it never collapses. Every dropped vertex
497/// stays within `tol` of the result.
498pub fn simplify(pts: &[Vec2], tol: f32, closed: bool) -> Vec<Vec2> {
499    let n = pts.len();
500    if tol <= 0.0 || n < 3 {
501        return pts.to_vec();
502    }
503    fn dp(pts: &[Vec2], lo: usize, hi: usize, tol: f32, keep: &mut [bool]) {
504        if hi <= lo + 1 {
505            return;
506        }
507        let (mut far, mut fd) = (lo, 0.0f32);
508        for i in lo + 1..hi {
509            let d = seg_dist(pts[i], pts[lo], pts[hi]);
510            if d > fd {
511                (far, fd) = (i, d);
512            }
513        }
514        if fd > tol {
515            keep[far] = true;
516            dp(pts, lo, far, tol, keep);
517            dp(pts, far, hi, tol, keep);
518        }
519    }
520    let mut keep = vec![false; n];
521    if !closed {
522        keep[0] = true;
523        keep[n - 1] = true;
524        dp(pts, 0, n - 1, tol, &mut keep);
525    } else {
526        // Anchor: vertex 0 and the vertex farthest from it.
527        let far = (1..n).max_by(|&a, &b| pts[a].distance_squared(pts[0]).total_cmp(&pts[b].distance_squared(pts[0]))).unwrap();
528        keep[0] = true;
529        keep[far] = true;
530        dp(pts, 0, far, tol, &mut keep);
531        let mut tail: Vec<Vec2> = pts[far..].to_vec();
532        tail.push(pts[0]);
533        let mut tk = vec![false; tail.len()];
534        tk[0] = true;
535        *tk.last_mut().unwrap() = true;
536        dp(&tail, 0, tail.len() - 1, tol, &mut tk);
537        for (k, &flag) in tk.iter().enumerate().take(tail.len() - 1) {
538            if flag {
539                keep[far + k] = true;
540            }
541        }
542    }
543    pts.iter().zip(&keep).filter(|(_, k)| **k).map(|(p, _)| *p).collect()
544}

653    #[test]
654    fn triangulation_area_matches_shoelace() {
655        let v = |a: &[[f32; 2]]| a.iter().map(|&p| Vec2::from(p)).collect::<Vec<_>>();
656        let square = v(&[[0., 0.], [4., 0.], [4., 4.], [0., 4.]]);
657        let l_shape = v(&[[0., 0.], [4., 0.], [4., 2.], [2., 2.], [2., 4.], [0., 4.]]);
658        let comb = v(&[[0., 0.], [9., 0.], [9., 3.], [7., 3.], [7., 1.], [5., 1.], [5., 3.], [3., 3.], [3., 1.], [1., 1.], [1., 3.], [0., 3.]]);
659        let with_collinear = v(&[[0., 0.], [2., 0.], [4., 0.], [4., 4.], [0., 4.], [0., 2.]]);
660        for (name, poly) in [("square", square), ("L", l_shape), ("comb", comb), ("collinear", with_collinear)] {
661            let want = signed_area(&poly);
662            let t = triangulate(&poly);
663            assert!((poly_area(&poly, &t) - want).abs() < 1e-4, "{name} ccw: {} vs {want}", poly_area(&poly, &t));
664            let mut cw = poly.clone();
665            cw.reverse();
666            let t = triangulate(&cw);
667            assert!((poly_area(&cw, &t) - want.abs()).abs() < 1e-4, "{name} cw");
668            assert!(t.iter().all(|tri| cross(cw[tri[0] as usize], cw[tri[1] as usize], cw[tri[2] as usize]) > 0.0), "{name}: all triangles CCW");
669        }
670        // A tessellated blob (open curve closed by a straight segment) - the realistic input.
671        let mut pts = vec![ShapePoint::at(-3.0, 0.0), ShapePoint::at(0.0, 3.0), ShapePoint::at(3.0, 0.0), ShapePoint::at(0.0, -3.0)];
672        for p in pts.iter_mut() {
673            let r = Vec2::from(p.position);
674            p.set_mode(ShapeTangentMode::Continuous, [r.y, -r.x], 1.6); // clockwise round blob
675        }
676        let blob = tessellate(&pts, true, TessMode::Adaptive { tolerance: 0.01 }, 1000);
677        let t = triangulate(&blob.pts);
678        assert!(blob.pts.len() > 12);
679        assert!((poly_area(&blob.pts, &t) - signed_area(&blob.pts).abs()).abs() < 1e-3, "CCW triangles sum to the ring's absolute area");
680    }
