1//! Spline edit operations of the Scene-view tool, as a pure model: every
2//! operation applies to a [`SpriteShapeRenderer`] and returns its exact
3//! inverse, so an undo entry is just the inverse stored on a stack. (The editor
4//! also records the frame's whole-component diff, so these are the testable
5//! core of what the tool does, not a second undo system.)

12/// One spline edit with an exact inverse.
13#[derive(Clone, PartialEq, Debug)]
14pub enum SplineOp {
15    /// Insert `point` so it ends up at `index` (`index == len` appends).
16    Insert {
17        /// Where the point ends up.
18        index: usize,
19        /// The point.
20        point: ShapePoint,
21    },
22    /// Remove the point at `index`.
23    Remove {
24        /// Index of the point to remove.
25        index: usize,
26    },
27    /// Replace the point at `index` (move, tangents, mode, height, corner, variant).
28    Set {
29        /// Index of the point to replace.
30        index: usize,
31        /// Its new state.
32        point: ShapePoint,
33    },
34    /// Open or close the spline.
35    SetClosed(bool),
36    /// Several operations as one undo step (a segment split changes three points).
37    Batch(Vec<SplineOp>),
38}
39
40/// Apply `op`; returns the operation that undoes it, or `None` (and changes
41/// nothing) when an index is out of range.
42pub fn apply_op(shape: &mut SpriteShapeRenderer, op: &SplineOp) -> Option<SplineOp> {
43    match op {
44        SplineOp::Insert { index, point } => {
45            if *index > shape.points.len() {
46                return None;
47            }
48            shape.points.insert(*index, *point);
49            Some(SplineOp::Remove { index: *index })
50        }
51        SplineOp::Remove { index } => {
52            if *index >= shape.points.len() {
53                return None;
54            }
55            let point = shape.points.remove(*index);
56            Some(SplineOp::Insert { index: *index, point })
57        }
58        SplineOp::Set { index, point } => {
59            let slot = shape.points.get_mut(*index)?;
60            let old = std::mem::replace(slot, *point);
61            Some(SplineOp::Set { index: *index, point: old })
62        }
63        SplineOp::SetClosed(c) => {
64            let old = std::mem::replace(&mut shape.closed, *c);
65            Some(SplineOp::SetClosed(old))
66        }
67        SplineOp::Batch(ops) => {
68            let mut inverse = Vec::with_capacity(ops.len());
69            for o in ops {
70                match apply_op(shape, o) {
71                    Some(i) => inverse.push(i),
72                    None => {
73                        // Roll back what already applied.
74                        for i in inverse.iter().rev() {
75                            apply_op(shape, i);
76                        }
77                        return None;
78                    }
79                }
80            }
81            inverse.reverse();
82            Some(SplineOp::Batch(inverse))
83        }
84    }
85}

87/// An undo / redo history of operations on one shape.
88#[derive(Default, Debug)]
89pub struct SplineHistory {
90    undo: Vec<SplineOp>,
91    redo: Vec<SplineOp>,
92}
93
94impl SplineHistory {
95    /// Apply and remember `op` (clears the redo stack).
96    pub fn apply(&mut self, shape: &mut SpriteShapeRenderer, op: SplineOp) -> bool {
97        match apply_op(shape, &op) {
98            Some(inv) => {
99                self.undo.push(inv);
100                self.redo.clear();
101                true
102            }
103            None => false,
104        }
105    }
106    /// Undo the last applied operation; false when the stack is empty.
107    pub fn undo(&mut self, shape: &mut SpriteShapeRenderer) -> bool {
108        let Some(op) = self.undo.pop() else { return false };
109        match apply_op(shape, &op) {
110            Some(inv) => {
111                self.redo.push(inv);
112                true
113            }
114            None => false,
115        }
116    }
117    /// Redo the last undone operation; false when the stack is empty.
118    pub fn redo(&mut self, shape: &mut SpriteShapeRenderer) -> bool {
119        let Some(op) = self.redo.pop() else { return false };
120        match apply_op(shape, &op) {
121            Some(inv) => {
122                self.undo.push(inv);
123                true
124            }
125            None => false,
126        }
127    }
128    /// True when an undo is available.
129    pub fn can_undo(&self) -> bool {
130        !self.undo.is_empty()
131    }
132    /// True when a redo is available.
133    pub fn can_redo(&self) -> bool {
134        !self.redo.is_empty()
135    }
136}

138/// Number of control segments.
139pub fn segment_count(shape: &SpriteShapeRenderer) -> usize {
140    let n = shape.points.len();
141    if n < 2 { 0 } else if shape.closed { n } else { n - 1 }
142}

144/// Move a point (tangents travel with it, being relative).
145pub fn move_point(shape: &SpriteShapeRenderer, index: usize, to: [f32; 2]) -> Option<SplineOp> {
146    let mut p = *shape.points.get(index)?;
147    p.position = to;
148    Some(SplineOp::Set { index, point: p })
149}

151/// Set the outgoing (`out`) or incoming tangent from an absolute handle position.
152pub fn move_tangent(shape: &SpriteShapeRenderer, index: usize, out: bool, handle: [f32; 2]) -> Option<SplineOp> {
153    let mut p = *shape.points.get(index)?;
154    let t = [handle[0] - p.position[0], handle[1] - p.position[1]];
155    // Dragging a handle of a Linear point turns it into a smooth one.
156    if p.mode == ShapeTangentMode::Linear {
157        p.mode = ShapeTangentMode::Continuous;
158        p.tangent_in = [0.0; 2];
159        p.tangent_out = [0.0; 2];
160    }
161    if out { p.set_tangent_out(t) } else { p.set_tangent_in(t) }
162    Some(SplineOp::Set { index, point: p })
163}

165/// Change a point's tangent mode (a fresh curve gets tame default handles).
166pub fn set_mode(shape: &SpriteShapeRenderer, index: usize, mode: ShapeTangentMode) -> Option<SplineOp> {
167    let n = shape.points.len();
168    let mut p = *shape.points.get(index)?;
169    let prev = shape.points.get(if index == 0 { if shape.closed { n - 1 } else { 0 } } else { index - 1 })?;
170    let next = shape.points.get(if index + 1 < n { index + 1 } else if shape.closed { 0 } else { index })?;
171    let along = [next.position[0] - prev.position[0], next.position[1] - prev.position[1]];
172    let len = Vec2::from(along).length();
173    // A third of the way to the nearer neighbour keeps a fresh curve tame.
174    let reach = Vec2::from(p.position).distance(Vec2::from(prev.position)).min(Vec2::from(p.position).distance(Vec2::from(next.position)));
175    p.set_mode(mode, along, if len > 1e-6 { reach / 3.0 } else { 0.0 });
176    Some(SplineOp::Set { index, point: p })
177}

179/// Cycle Linear -> Continuous -> Broken -> Linear.
180pub fn cycle_mode(shape: &SpriteShapeRenderer, index: usize) -> Option<SplineOp> {
181    let next = match shape.points.get(index)?.mode {
182        ShapeTangentMode::Linear => ShapeTangentMode::Continuous,
183        ShapeTangentMode::Continuous => ShapeTangentMode::Broken,
184        ShapeTangentMode::Broken => ShapeTangentMode::Linear,
185    };
186    set_mode(shape, index, next)
187}

189/// Flip a point's corner flag.
190pub fn toggle_corner(shape: &SpriteShapeRenderer, index: usize) -> Option<SplineOp> {
191    let mut p = *shape.points.get(index)?;
192    p.corner = !p.corner;
193    Some(SplineOp::Set { index, point: p })
194}

196/// Set a point's edge-thickness scale (clamped to 0+).
197pub fn set_height(shape: &SpriteShapeRenderer, index: usize, height: f32) -> Option<SplineOp> {
198    let mut p = *shape.points.get(index)?;
199    p.height = height.max(0.0);
200    Some(SplineOp::Set { index, point: p })
201}

203/// Split control segment `seg` at parameter `t` so the curve keeps its shape
204/// (de Casteljau): the new point sits at `index = seg + 1`, and the neighbours'
205/// tangents shrink to the halves. A straight segment gets a Linear point.
206pub fn split_segment(shape: &SpriteShapeRenderer, seg: usize, t: f32) -> Option<SplineOp> {
207    let n = shape.points.len();
208    if seg >= segment_count(shape) {
209        return None;
210    }
211    let t = t.clamp(0.001, 0.999);
212    let (ia, ib) = (seg, (seg + 1) % n);
213    let (a, b) = (shape.points[ia], shape.points[ib]);
214    let c = segment_bezier(&a, &b);
215    let straight = c[1] == c[0] && c[2] == c[3];
216    let lerp = |x: Vec2, y: Vec2| x + (y - x) * t;
217    let (q0, q1, q2) = (lerp(c[0], c[1]), lerp(c[1], c[2]), lerp(c[2], c[3]));
218    let (r0, r1) = (lerp(q0, q1), lerp(q1, q2));
219    let m = lerp(r0, r1);
220    let mut point = ShapePoint {
221        position: m.to_array(),
222        height: a.height + (b.height - a.height) * t,
223        sprite_index: a.sprite_index,
224        corner: true,
225        ..Default::default()
226    };
227    let mut ops = vec![];
228    if !straight {
229        point.mode = ShapeTangentMode::Broken;
230        point.tangent_in = (r0 - m).to_array();
231        point.tangent_out = (r1 - m).to_array();
232        // Shrunk neighbours (a Linear end keeps no handle, which stays correct: its handle was zero).
233        let mut na = a;
234        if a.mode != ShapeTangentMode::Linear {
235            na.tangent_out = (q0 - c[0]).to_array();
236            if a.mode == ShapeTangentMode::Continuous {
237                // The other side of a continuous point keeps its length; mirror would shorten it.
238                na.mode = ShapeTangentMode::Broken;
239            }
240        }
241        let mut nb = b;
242        if b.mode != ShapeTangentMode::Linear {
243            nb.tangent_in = (q2 - c[3]).to_array();
244            if b.mode == ShapeTangentMode::Continuous {
245                nb.mode = ShapeTangentMode::Broken;
246            }
247        }
248        ops.push(SplineOp::Set { index: ia, point: na });
249        ops.push(SplineOp::Set { index: ib, point: nb });
250    }
251    // Insert after the sets (indices of the neighbours don't move before it).
252    // A wrap-around segment (closed, last -> first) appends at the end.
253    ops.push(SplineOp::Insert { index: if ib == 0 { n } else { ib }, point });
254    Some(SplineOp::Batch(ops))
255}

257/// Where `p` is closest to the spline: `(segment, t, distance)`.
258pub fn nearest_on_spline(shape: &SpriteShapeRenderer, p: Vec2) -> Option<(usize, f32, f32)> {
259    let n = shape.points.len();
260    let mut best: Option<(usize, f32, f32)> = None;
261    for seg in 0..segment_count(shape) {
262        let c = segment_bezier(&shape.points[seg], &shape.points[(seg + 1) % n]);
263        const STEPS: usize = 32;
264        let mut prev = c[0];
265        for i in 1..=STEPS {
266            let (t0, t1) = ((i - 1) as f32 / STEPS as f32, i as f32 / STEPS as f32);
267            let cur = bezier_at(&c, t1);
268            let ab = cur - prev;
269            let l2 = ab.length_squared();
270            let u = if l2 > 1e-12 { ((p - prev).dot(ab) / l2).clamp(0.0, 1.0) } else { 0.0 };
271            let d = p.distance(prev + ab * u);
272            if best.is_none_or(|b| d < b.2) {
273                best = Some((seg, t0 + (t1 - t0) * u, d));
274            }
275            prev = cur;
276        }
277    }
278    best
279}

281/// Snap a coordinate pair to a grid (`step <= 0` = off).
282pub fn snap(p: [f32; 2], step: f32) -> [f32; 2] {
283    if step <= 0.0 { p } else { [(p[0] / step).round() * step, (p[1] / step).round() * step] }
284}

286/// Where a click at `p` adds a point to an open spline: after the last point.
287/// Closed shapes insert on the nearest segment instead (see [`split_segment`]).
288pub fn append_point(shape: &SpriteShapeRenderer, p: [f32; 2]) -> SplineOp {
289    let mut pt = ShapePoint::at(p[0], p[1]);
290    if let Some(last) = shape.points.last() {
291        pt.height = last.height;
292    }
293    SplineOp::Insert { index: shape.points.len(), point: pt }
294}

307    #[test]
308    fn every_op_undoes_exactly() {
309        let base = shape();
310        let ops = vec![
311            append_point(&base, [9.0, 9.0]),
312            SplineOp::Insert { index: 1, point: ShapePoint::at(2.0, 1.0) },
313            SplineOp::Remove { index: 1 },
314            move_point(&base, 2, [5.0, 5.0]).unwrap(),
315            move_tangent(&base, 1, true, [4.0, 2.0]).unwrap(),
316            move_tangent(&base, 0, false, [-1.0, 0.0]).unwrap(),
317            cycle_mode(&base, 0).unwrap(),
318            cycle_mode(&base, 1).unwrap(),
319            toggle_corner(&base, 2).unwrap(),
320            set_height(&base, 1, 2.5).unwrap(),
321            SplineOp::SetClosed(true),
322            split_segment(&base, 0, 0.3).unwrap(),
323            split_segment(&base, 1, 0.5).unwrap(),
324        ];
325        for op in ops {
326            let mut s = base.clone();
327            let inv = apply_op(&mut s, &op).unwrap_or_else(|| panic!("{op:?} applies"));
328            assert_ne!(s, base, "{op:?} changed something");
329            let redo = apply_op(&mut s, &inv).unwrap();
330            assert_eq!(s, base, "undo of {op:?}");
331            apply_op(&mut s, &redo);
332            let mut again = base.clone();
333            apply_op(&mut again, &op);
334            assert_eq!(s, again, "redo of {op:?}");
335        }
336    }

350    #[test]
351    fn history_undoes_and_redoes_a_sequence() {
352        let base = shape();
353        let mut s = base.clone();
354        let mut h = SplineHistory::default();
355        assert!(!h.undo(&mut s) && !h.can_undo());
356        let ops = vec![append_point(&s, [8.0, 0.0]), SplineOp::SetClosed(true), set_height(&s, 0, 3.0).unwrap()];
357        let mut states = vec![s.clone()];
358        for op in ops {
359            assert!(h.apply(&mut s, op));
360            states.push(s.clone());
361        }
362        for want in states.iter().rev().skip(1) {
363            assert!(h.undo(&mut s));
364            assert_eq!(&s, want);
365        }
366        assert_eq!(s, base);
367        assert!(h.redo(&mut s) && h.redo(&mut s));
368        assert_eq!(s, states[2]);
369        // A new edit drops the redo branch.
370        assert!(h.apply(&mut s, SplineOp::SetClosed(false)));
371        assert!(!h.can_redo());
372    }

375    fn splitting_a_segment_keeps_the_curve() {
376        let mut base = SpriteShapeRenderer { points: vec![ShapePoint::at(0.0, 0.0), ShapePoint::at(6.0, 0.0), ShapePoint::at(6.0, 5.0)], ..Default::default() };
377        base.points[0].set_mode(ShapeTangentMode::Broken, [0.0, 1.0], 3.0);

379        for closed in [false, true] {
380            for seg in 0..(if closed { 3 } else { 2 }) {
381                for t in [0.2, 0.5, 0.8] {
382                    let mut s = base.clone();
383                    s.closed = closed;
384                    let before = tessellate(&s.points, closed, TessMode::Fixed { detail: 64 }, 10_000);
385                    let op = split_segment(&s, seg, t).unwrap();
386                    apply_op(&mut s, &op).unwrap();

388                    let after = tessellate(&s.points, closed, TessMode::Fixed { detail: 64 }, 10_000);

390                    let fine = tessellate(&base.points.to_vec(), closed, TessMode::Fixed { detail: 512 }, 100_000);
391                    let worst = after.pts.iter().map(|p| crate::sprite_shape::spline::distance_to_polyline(*p, &fine.pts, closed)).fold(0.0f32, f32::max);
392                    assert!(worst < 2e-3, "closed {closed} seg {seg} t {t}: {worst}");
393                    assert!((after.length() - before.length()).abs() / before.length() < 5e-3);
394                }
395            }
