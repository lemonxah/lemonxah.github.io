1//! Sprite Shape (2D-8): a spline-driven sprite mesh. A closed spline is filled
2//! (a tiled texture or a stretched sprite) and every spline segment gets an
3//! edge sprite chosen by the direction the segment faces; sharp corners get
4//! corner sprites and an open spline can end in caps.
5//!
6//! * [`SpriteShapeRenderer`] (component): the spline (control points with
7//!   in / out tangents, per-point height, corner flag and sprite variant),
8//!   open or closed, the tessellation settings, sorting and the collider
9//!   settings. It names a `.spriteshape` profile.
10//! * [`SpriteShapeProfile`] (asset, RON, extension [`SPRITE_SHAPE_EXTENSION`]):
11//!   angle ranges mapping a segment's normal angle to a list of edge sprites,
12//!   corner and cap sprites, the fill texture.
13//! * [`spline`]: pure geometry (bezier tessellation with an error bound and a
14//!   point cap, ear-clipping triangulation, polygon offset, Douglas-Peucker).
15//! * [`generate`]: spline + profile -> [`generate::ShapeMesh`] (position / UV /
16//!   index groups, one per texture), and the collider outline.
17//! * [`edit`]: the spline edit operations the Scene-view tool applies, each with
18//!   an exact inverse (pure model, unit tested).
19//! * [`presets`]: starter profiles and splines (grass strip, ground with fill,
20//!   rope, road).
21//!
22//! The engine gathers shapes in `engine/src/scene/shape2d.rs`: it rebuilds a
23//! shape only when [`SpriteShapeRenderer::content_hash`] or the profile changed
24//! and writes the groups into rewritable meshes of the sprite pool, so a scene
25//! with no shapes pays nothing. Behaviour follows the Unity Sprite Shape
26//! package as documented in `docs/research/unity7/phase5b-2d-tilemaps-physics.md`
27//! section 5; the code is our own.
28//!
29//! Conventions: object-local XY plane, y up. A segment's *normal* is the left of
30//! its travel direction; a closed spline is walked clockwise (a counter-clockwise
31//! one is reversed) so the normal points out of the shape. The *angle* of a
32//! segment is the normal's angle from +Y, counter-clockwise, in degrees
33//! `[0, 360)`: 0 faces up, 90 left, 180 down, 270 right.
34

52/// How a control point's tangents behave.
53#[derive(Clone, Copy, PartialEq, Eq, Debug, Default, Serialize, Deserialize)]
54pub enum ShapeTangentMode {
55    /// No tangents: both neighbouring segments leave / arrive straight.
56    #[default]
57    Linear,
58    /// `tangent_in` is always the mirror of `tangent_out` (smooth curve).
59    Continuous,
60    /// The two tangents are independent (a cusp is possible).
61    Broken,
62}
63
64impl ShapeTangentMode {
65    /// Every mode, in declaration order.
66    pub const ALL: [ShapeTangentMode; 3] = [Self::Linear, Self::Continuous, Self::Broken];
67    /// Inspector label of this mode.
68    pub fn label(self) -> &'static str {
69        match self {
70            Self::Linear => "Linear",
71            Self::Continuous => "Continuous",
72            Self::Broken => "Broken",
73        }
74    }
75}

84/// One control point of the spline. Tangents are offsets from `position` (the
85/// bezier handles), in object-local units.
86#[derive(Clone, Copy, PartialEq, Debug, Serialize, Deserialize)]
87#[serde(default)]
88pub struct ShapePoint {
89    /// Control point position (object-local).
90    pub position: [f32; 2],
91    /// Incoming tangent offset from `position`.
92    pub tangent_in: [f32; 2],
93    /// Outgoing tangent offset from `position`.
94    pub tangent_out: [f32; 2],
95    /// How the tangents behave.
96    pub mode: ShapeTangentMode,
97    /// Scales the edge sprites' thickness at this point (interpolated between
98    /// points); 1 = the sprite's own height.
99    #[serde(default = "one")]
100    pub height: f32,
101    /// Edge sprite variant used by the segment that starts at this point
102    /// (index into the angle range's sprite list, wrapped).
103    pub sprite_index: u32,
104    /// Allowed to become a corner when the turn exceeds the shape's threshold.
105    #[serde(default = "yes")]
106    pub corner: bool,
107}
108
109impl Default for ShapePoint {
110    fn default() -> Self {
111        Self { position: [0.0; 2], tangent_in: [0.0; 2], tangent_out: [0.0; 2], mode: ShapeTangentMode::Linear, height: 1.0, sprite_index: 0, corner: true }
112    }
113}
114
115impl ShapePoint {
116    /// A point at (x, y) with default handles.
117    pub fn at(x: f32, y: f32) -> Self {
118        Self { position: [x, y], ..Default::default() }
119    }
120
121    /// The handle offset the bezier really uses (a Linear point has none).
122    pub fn out_handle(&self) -> [f32; 2] {
123        if self.mode == ShapeTangentMode::Linear { [0.0; 2] } else { self.tangent_out }
124    }
125
126    /// The incoming handle offset the bezier really uses (none when Linear).
127    pub fn in_handle(&self) -> [f32; 2] {
128        if self.mode == ShapeTangentMode::Linear { [0.0; 2] } else { self.tangent_in }
129    }
130
131    /// Set the outgoing tangent; a Continuous point mirrors it into the incoming one.
132    pub fn set_tangent_out(&mut self, t: [f32; 2]) {
133        self.tangent_out = t;
134        if self.mode == ShapeTangentMode::Continuous {
135            self.tangent_in = [-t[0], -t[1]];
136        }
137    }
138
139    /// Set the incoming tangent; a Continuous point mirrors it into the
140    /// outgoing one.
141    pub fn set_tangent_in(&mut self, t: [f32; 2]) {
142        self.tangent_in = t;
143        if self.mode == ShapeTangentMode::Continuous {
144            self.tangent_out = [-t[0], -t[1]];
145        }
146    }
147
148    /// Change the mode. Leaving Linear gives the point default handles along
149    /// `along` (the direction between its neighbours) so it actually curves;
150    /// entering Continuous mirrors the outgoing tangent; entering Linear keeps
151    /// the stored handles (they come back if the mode is switched again).
152    pub fn set_mode(&mut self, mode: ShapeTangentMode, along: [f32; 2], handle_len: f32) {
153        if self.mode == ShapeTangentMode::Linear && mode != ShapeTangentMode::Linear && self.tangent_in == [0.0; 2] && self.tangent_out == [0.0; 2] {
154            let len = (along[0] * along[0] + along[1] * along[1]).sqrt();
155            if len > 1e-6 {
156                let d = [along[0] / len * handle_len, along[1] / len * handle_len];
157                self.tangent_out = d;
158                self.tangent_in = [-d[0], -d[1]];
159            }
160        }
161        self.mode = mode;
162        if mode == ShapeTangentMode::Continuous {
163            self.tangent_in = [-self.tangent_out[0], -self.tangent_out[1]];
164        }
165    }
166}

190/// How edge sprites are laid along a run of segments of one angle range.
191#[derive(Clone, Copy, PartialEq, Eq, Debug, Default, Serialize, Deserialize)]
192pub enum EdgeFit {
193    /// A whole number of tiles (nearest), stretched or squeezed to fill the run exactly.
194    #[default]
195    Stretch,
196    /// A whole number of tiles that never shrink (round down), widened to fill the run exactly.
197    Fit,
198    /// Tiles at their own width; the last one is cut off (its UVs too).
199    Tile,
200}
201
202impl EdgeFit {
203    /// Every fit, in declaration order.
204    pub const ALL: [EdgeFit; 3] = [Self::Stretch, Self::Fit, Self::Tile];
205    /// Inspector label of this fit.
206    pub fn label(self) -> &'static str {
207        match self {
208            Self::Stretch => "Stretch",
209            Self::Fit => "Fit (never shrink)",
210            Self::Tile => "Tile (cut last)",
211        }
212    }
213}

231/// An inner closed spline that cuts a hole out of the fill of a closed shape (and gets its own
232/// edge sprites, facing into the hole, and its own collider path). Same tessellation settings as
233/// the outer spline. A hole should lie inside the outer outline and not overlap another hole.
234#[derive(Clone, PartialEq, Debug, Default, Serialize, Deserialize)]
235#[serde(default)]
236pub struct ShapeHole {
237    /// The hole's control points (an inner closed spline).
238    pub points: Vec<ShapePoint>,
239}
240
241impl ShapeHole {
242    /// A regular `n`-gon (linear points) of radius `r` around `c`.
243    pub fn ngon(c: [f32; 2], r: f32, n: usize) -> Self {
244        let n = n.max(3);
245        Self { points: (0..n).map(|i| {
246            let a = std::f32::consts::TAU * i as f32 / n as f32;
247            ShapePoint::at(c[0] + r * a.cos(), c[1] + r * a.sin())
248        }).collect() }
249    }
250}

252/// The Sprite Shape component: spline + settings of one shape. Needs a
253/// `.spriteshape` profile for sprites; without one it draws nothing.
254#[derive(Clone, PartialEq, Debug, Serialize, Deserialize)]
255#[serde(default)]
256pub struct SpriteShapeRenderer {
257    /// Project-relative `.spriteshape` path.
258    pub profile: String,
259    /// The spline's control points.
260    pub points: Vec<ShapePoint>,
261    /// Inner closed splines subtracted from the fill of a closed shape.
262    pub holes: Vec<ShapeHole>,
263    /// Closed shapes are filled and walked clockwise; open ones are a strip.
264    pub closed: bool,
265    /// Draw the fill of a closed shape.
266    pub fill: bool,

268    pub tessellation: Tessellation,

270    pub spline_detail: u32,
271    /// Largest distance between the curve and its polyline, local units (`Adaptive`).
272    pub tolerance: f32,
273    /// Upper bound of tessellated points (the tolerance is relaxed to honour it).
274    pub max_points: u32,
275    /// A control point turning more than this (degrees) may become a corner.
276    pub corner_angle_threshold: f32,
277    /// How edge sprites lay along a run of one angle range.
278    pub fit: EdgeFit,
279    /// Walk the spline the other way: edge sprites go on the other side.
280    pub flip_edge: bool,
281    /// Shifts the edge sprites along the normal as a fraction of their height
282    /// (0 = centred on the spline, 0.5 = fully outside).
283    pub border_pivot: f32,
284    /// Tint, linear RGBA.
285    pub color: [f32; 4],
286    /// Sorting layer.
287    pub sorting_layer: SortingLayerId,
288    /// Sort key inside the layer.
289    pub order_in_layer: i16,
290    /// How sprite masks affect the shape.
291    pub mask_interaction: MaskInteraction2D,
292    /// Write the outline into the Polygon (closed) / Edge (open) Collider 2D
293    /// on this object whenever the shape changes.
294    pub auto_update_collider: bool,
295    /// Pieces per curved segment of the collider outline.
296    pub collider_detail: u32,
297    /// Outline offset (closed shapes), local units; positive grows the polygon.
298    pub collider_offset: f32,
299    /// Douglas-Peucker tolerance of the collider outline, 0 = keep every point.
300    pub collider_optimize: f32,
301}

335/// FNV-1a over raw bits, the shape's "dirty version".
336struct Fnv(u64);
337
338impl Fnv {
339    fn new() -> Self {
340        Self(0xcbf2_9ce4_8422_2325)
341    }
342    fn bytes(&mut self, b: &[u8]) {
343        for &x in b {
344            self.0 = (self.0 ^ x as u64).wrapping_mul(0x0000_0100_0000_01b3);
345        }
346    }
347    fn f(&mut self, v: f32) {
348        self.bytes(&v.to_bits().to_le_bytes());
349    }
350    fn u(&mut self, v: u32) {
351        self.bytes(&v.to_le_bytes());
352    }
353}
354
355impl SpriteShapeRenderer {
356    /// Hash of everything that changes the generated geometry (not the tint,
357    /// sorting or mask settings, which are per-frame instance data). The engine
358    /// rebuilds the mesh only when this or the profile changed.
359    pub fn content_hash(&self) -> u64 {
360        let mut h = Fnv::new();
361        h.bytes(self.profile.as_bytes());
362        h.u(self.points.len() as u32);
363        for p in &self.points {
364            for v in p.position.iter().chain(&p.tangent_in).chain(&p.tangent_out) {
365                h.f(*v);
366            }
367            h.u(p.mode as u32);
368            h.f(p.height);
369            h.u(p.sprite_index);
370            h.u(p.corner as u32);
371        }
372        h.u(self.holes.len() as u32);
373        for hole in &self.holes {
374            h.u(hole.points.len() as u32);
375            for p in &hole.points {
376                for v in p.position.iter().chain(&p.tangent_in).chain(&p.tangent_out) {
377                    h.f(*v);
378                }
379                h.u(p.mode as u32);
380                h.f(p.height);
381                h.u(p.sprite_index);
382                h.u(p.corner as u32);
383            }
384        }
385        h.u(self.closed as u32);
386        h.u(self.fill as u32);
387        h.u(self.tessellation as u32);
388        h.u(self.spline_detail);
389        h.f(self.tolerance);
390        h.u(self.max_points);
391        h.f(self.corner_angle_threshold);
392        h.u(self.fit as u32);
393        h.u(self.flip_edge as u32);
394        h.f(self.border_pivot);
395        h.0
396    }
397
398    /// Hash of the collider-relevant settings and the spline.
399    pub fn collider_hash(&self) -> u64 {
400        let mut h = Fnv(self.content_hash());
401        h.u(self.collider_detail);
402        h.f(self.collider_offset);
403        h.f(self.collider_optimize);
404        h.0
405    }
406
407    /// The tessellation settings as the geometry code takes them.
408    pub fn tess_mode(&self) -> TessMode {
409        match self.tessellation {
410            Tessellation::Adaptive => TessMode::Adaptive { tolerance: self.tolerance.max(1e-5) },
411            Tessellation::Fixed => TessMode::Fixed { detail: self.spline_detail.max(1) },
412        }
413    }
414}

416/// An angle range of a profile: segments whose normal angle falls in
417/// `start..end` (degrees, counter-clockwise from up, wrapping allowed: a range
418/// of `-40..40` is the top) use the range's sprites. Overlapping ranges: the
419/// higher `order` wins, then the earlier in the list.
420#[derive(Clone, PartialEq, Debug, Serialize, Deserialize)]
421#[serde(default)]
422pub struct AngleRange {
423    /// First angle of the range (degrees, counter-clockwise from up).
424    pub start: f32,
425    /// Last angle of the range (degrees).
426    pub end: f32,
427    /// Higher order wins when ranges overlap.
428    pub order: i32,
429    /// Edge sprite variants (`path#id` references); the segment's point picks one.
430    pub sprites: Vec<String>,
431}
432
433impl Default for AngleRange {
434    fn default() -> Self {
435        Self { start: -40.0, end: 40.0, order: 0, sprites: Vec::new() }
436    }
437}
438
439impl AngleRange {
440    /// Whether `angle` (degrees, any value) lies in the range. A range whose
441    /// span reaches 360 degrees covers everything; `start == end` covers nothing
442    /// but that exact angle.
443    pub fn contains(&self, angle: f32) -> bool {
444        let span = self.end - self.start;
445        if span >= 360.0 {
446            return true;
447        }
448        if span < 0.0 {
449            // Written with end before start (170..-170): the arc runs counter-clockwise
450            // from start through the wrap (here 170 -> 190), the same as the range 170..190.
451            let span = span.rem_euclid(360.0);
452            return (angle - self.start).rem_euclid(360.0) <= span;
453        }
454        (angle - self.start).rem_euclid(360.0) <= span + 1e-4
455    }
456}
457
458/// The angle range that draws a segment facing `angle`, or `None` (no edge
459/// sprite there).
460pub fn select_range(ranges: &[AngleRange], angle: f32) -> Option<usize> {
461    let mut best: Option<usize> = None;
462    for (i, r) in ranges.iter().enumerate() {
463        if r.contains(angle) && best.is_none_or(|b| r.order > ranges[b].order) {
464            best = Some(i);
465        }
466    }
467    best
468}

470fn ppu_default() -> f32 {
471    100.0

474/// The `.spriteshape` asset.
475#[derive(Clone, PartialEq, Debug, Serialize, Deserialize)]
476#[serde(default)]
477pub struct SpriteShapeProfile {
478    /// Which edge sprites each segment angle uses.
479    pub angle_ranges: Vec<AngleRange>,
480    /// Sprite for corners that turn away from the shape (a convex corner of a
481    /// clockwise outline). Empty = a mitered join.
482    pub outer_corner: String,
483    /// Sprite for corners that turn into the shape (concave).
484    pub inner_corner: String,
485    /// Open shapes: sprite placed before the first / after the last point.
486    pub start_cap: String,
487    /// Open shapes: sprite placed after the last point.
488    pub end_cap: String,
489    /// Fill texture (project-relative path), tiled by world units: it repeats
490    /// every `size_px / fill_pixels_per_unit` units (needs a Repeat wrap mode).
491    pub fill_texture: String,
492    /// Fill texture pixels per world unit (tiling scale).
493    pub fill_pixels_per_unit: f32,
494    /// Shifts the tiling pattern, world units.
495    pub fill_uv_offset: [f32; 2],
496    /// Fill used when there is no `fill_texture`: one sprite stretched over the
497    /// shape's bounds (`path#id`).
498    pub fill_sprite: String,
499    /// Moves the fill outline along its normal, world units (negative tucks
500    /// the fill under the edge sprites).
501    pub fill_edge_offset: f32,
502}

521impl SpriteShapeProfile {
522    /// Hash of the whole profile (the engine compares it, not file times, for
523    /// the live profile editor).
524    pub fn content_hash(&self) -> u64 {
525        let mut h = Fnv::new();
526        for r in &self.angle_ranges {
527            h.f(r.start);
528            h.f(r.end);
529            h.u(r.order as u32);
530            for s in &r.sprites {
531                h.bytes(s.as_bytes());
532                h.bytes(&[0]);
533            }
534            h.bytes(&[1]);
535        }
536        for s in [&self.outer_corner, &self.inner_corner, &self.start_cap, &self.end_cap, &self.fill_texture, &self.fill_sprite] {
537            h.bytes(s.as_bytes());
538            h.bytes(&[2]);
539        }
540        h.f(self.fill_pixels_per_unit);
541        h.f(self.fill_uv_offset[0]);
542        h.f(self.fill_uv_offset[1]);
543        h.f(self.fill_edge_offset);
544        h.0
545    }
546
547    /// Every sprite reference the profile can draw (edge variants, corners, caps, fill).
548    pub fn all_sprites(&self) -> Vec<&str> {
549        let mut v: Vec<&str> = self.angle_ranges.iter().flat_map(|r| r.sprites.iter().map(String::as_str)).collect();
550        v.extend([&self.outer_corner, &self.inner_corner, &self.start_cap, &self.end_cap, &self.fill_sprite].iter().map(|s| s.as_str()));
551        v.retain(|s| !s.is_empty());
552        v
553    }
554
555    /// The circle of angles split into arcs `(start, end, range)` by the range
556    /// that wins there (`None` = no range covers it): a partition of
557    /// `[0, 360)` at whole-degree resolution, in order, for the dial preview.
558    pub fn resolved_arcs(&self) -> Vec<(f32, f32, Option<usize>)> {
559        let pick = |d: u32| select_range(&self.angle_ranges, d as f32 + 0.5);
560        let mut out: Vec<(f32, f32, Option<usize>)> = Vec::new();
561        let mut start = 0u32;
562        let mut cur = pick(0);
563        for d in 1..=360u32 {
564            let next = if d < 360 { pick(d) } else { None };
565            if d == 360 || next != cur {
566                out.push((start as f32, d as f32, cur));
567                start = d;
568                cur = next;
569            }
570        }
571        out
572    }
573
574    /// Angles (degrees, `[0, 360)`) no range covers, as `(start, end)` arcs.
575    pub fn uncovered_arcs(&self) -> Vec<(f32, f32)> {
576        self.resolved_arcs().into_iter().filter(|a| a.2.is_none()).map(|a| (a.0, a.1)).collect()
577    }
578}

580/// Read and parse a `.spriteshape` profile.
581pub fn load_spriteshape(path: impl AsRef<std::path::Path>) -> Result<SpriteShapeProfile, String> {
582    let path = path.as_ref();
583    let text = std::fs::read_to_string(path).map_err(|e| format!("reading {}: {e}", path.display()))?;
584    ron::from_str(&text).map_err(|e| format!("parsing {}: {e}", path.display()))

587/// Serialize and write a `.spriteshape` profile.
588pub fn save_spriteshape(path: impl AsRef<std::path::Path>, profile: &SpriteShapeProfile) -> Result<(), String> {
589    let path = path.as_ref();
590    let text = ron::ser::to_string_pretty(profile, ron::ser::PrettyConfig::default()).map_err(|e| e.to_string())?;
591    std::fs::write(path, text).map_err(|e| format!("writing {}: {e}", path.display()))
592}
