1//! Authored dry regions and their renderer-independent spatial queries.
2use glam::{Mat4, Vec3};
3use serde::{Deserialize, Serialize};
4use crate::{ObjectId, ObjectRef, TypedComponent};
5
6/// Removes water inside a transformed box. Parent this to a boat or building
7/// to keep its interior dry without removing the water beneath its floor.
8#[derive(Clone, Debug, Serialize, Deserialize)]
9#[serde(default)]
10pub struct WaterExclusionVolume {
11    pub half_extents: [f32; 3],
12    /// Empty affects every water body; otherwise only the listed objects.
13    pub waters: Vec<ObjectRef>,
14}
15
16impl Default for WaterExclusionVolume {
17    fn default() -> Self { Self { half_extents: [1.0; 3], waters: Vec::new() } }
18}
19impl TypedComponent for WaterExclusionVolume { const NAME: &'static str = "Water Exclusion Volume"; }
20
21impl WaterExclusionVolume {
22    pub fn region(&self, world: Mat4) -> Option<WaterExclusionRegion> {
23        let half = Vec3::from(self.half_extents);
24        if !half.is_finite() || half.cmple(Vec3::ZERO).any() || !world.is_finite() || world.determinant().abs() < 1e-8 { return None; }
25        Some(WaterExclusionRegion { to_unit: Mat4::from_scale(half.recip()) * world.inverse() })
26    }
27    pub fn affects(&self, water: ObjectId) -> bool { self.waters.is_empty() || self.waters.iter().any(|r| r.id() == Some(water)) }
28}
29
30/// Transform from the query's space to a unit box [-1,1]. Engine-resolved
31/// copies on Water use water-local space; this has no rendering dependency.
32#[derive(Clone, Copy, Debug, PartialEq)]
33pub struct WaterExclusionRegion { pub to_unit: Mat4 }
34
35impl WaterExclusionRegion {
36    pub fn contains(&self, point: Vec3) -> bool {
37        let p = self.to_unit.transform_point3(point);
38        p.is_finite() && p.abs().cmple(Vec3::ONE).all()
39    }
40
41    /// Ray parameters inside the box. Direction need not be normalized: a
42    /// transformed world ray retains world-metre parameters, even with scale.
43    pub fn interval(&self, origin: Vec3, direction: Vec3, range: [f32; 2]) -> Option<[f32; 2]> {
44        let p = self.to_unit.transform_point3(origin);
45        let d = self.to_unit.transform_vector3(direction);
46        if !p.is_finite() || !d.is_finite() { return None; }
47        let [mut lo, mut hi] = range;
48        for axis in 0..3 {
49            if d[axis].abs() < 1e-12 {
50                if p[axis].abs() > 1.0 { return None; }
51            } else {
52                let a = (-1.0 - p[axis]) / d[axis];
53                let b = (1.0 - p[axis]) / d[axis];
54                lo = lo.max(a.min(b)); hi = hi.min(a.max(b));
55            }
56        }
57        (hi > lo).then_some([lo, hi])
58    }
59}

61#[cfg(test)]
62mod tests {
63    use super::*;
64    #[test]
65    fn transformed_boxes_clip_world_metres_and_keep_water_below() {
66        let volume = WaterExclusionVolume { half_extents: [2.0, 1.0, 0.5], ..Default::default() };
67        let world = Mat4::from_scale_rotation_translation(Vec3::new(2.0, 1.0, 3.0), glam::Quat::from_rotation_y(0.7), Vec3::new(4.0, 2.0, -3.0));
68        let region = volume.region(world).unwrap();
69        assert!(region.contains(world.transform_point3(Vec3::ZERO)));
70        assert!(!region.contains(world.transform_point3(Vec3::new(0.0, -1.01, 0.0))));
71        let origin = world.transform_point3(Vec3::new(-3.0, 0.0, 0.0));
72        let dir = world.transform_vector3(Vec3::X).normalize();
73        let hit = region.interval(origin, dir, [0.0, 100.0]).unwrap();
74        assert!((hit[0]-2.0).abs() < 1e-5 && (hit[1]-10.0).abs() < 1e-5);
75        assert!(volume.region(Mat4::ZERO).is_none());
76    }
77
78    #[test]
79    fn dry_water_queries_leave_the_wet_floor_and_do_not_serialize_runtime_regions() {
80        let volume = WaterExclusionVolume { half_extents: [1.0, 0.5, 1.0], ..Default::default() };
81        let mut water = crate::Water { waves: Vec::new(), current: [2.0, 0.0, 0.0], ..Default::default() };
82        water.exclusions.push(volume.region(Mat4::IDENTITY).unwrap());
83        assert!(water.height_at(Mat4::IDENTITY, Vec3::ZERO, 0.0).is_none());
84        assert_eq!(water.flow_at(Mat4::IDENTITY, Vec3::ZERO), Vec3::ZERO);
85        assert!(water.height_at(Mat4::IDENTITY, Vec3::new(0.0,-1.0,0.0),0.0).is_some());
86        assert_eq!(water.flow_at(Mat4::IDENTITY, Vec3::new(0.0,-1.0,0.0)), Vec3::new(2.0,0.0,0.0));
87        let saved = ron::to_string(&water).unwrap();
88        let loaded: crate::Water = ron::from_str(&saved).unwrap();
89        assert!(loaded.exclusions.is_empty());
90        let authored: WaterExclusionVolume = ron::from_str(&ron::to_string(&volume).unwrap()).unwrap();
91        assert_eq!(authored.half_extents, volume.half_extents);
92    }
