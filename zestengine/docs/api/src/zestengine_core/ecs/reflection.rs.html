9/// Disjoint values lent by a checked query that already owns their column.
10trait ComponentLoans {
11    fn get(&self, entity: Entity) -> Option<&dyn Component>;
12}
13impl ComponentLoans for HashMap<Entity, &dyn Component> {
14    fn get(&self, entity: Entity) -> Option<&dyn Component> { HashMap::get(self, &entity).copied() }
15}
16
17struct QueryLoans<'a, 'q, T> {
18    before: &'a [(Entity, (&'q Owner, &'q mut T))],
19    after: &'a [(Entity, (&'q Owner, &'q mut T))],
20    current: usize,
21    positions: &'a HashMap<Entity, usize>,
22}
23impl<T: Component> ComponentLoans for QueryLoans<'_, '_, T> {
24    fn get(&self, entity: Entity) -> Option<&dyn Component> {
25        let index = *self.positions.get(&entity)?;
26        let (_, (_, value)) = if index < self.current { self.before.get(index)? }
27            else { self.after.get(index.checked_sub(self.current + 1)?)? };
28        Some(&**value)
29    }
30}

55/// Reflection over the actual native world. During an action the executing
56/// instance is excluded, and other instances of its type are lent by that
57/// action's query so same-archetype property reads remain borrow-safe.
58pub struct NativeProperties<'a, 'w> {
59    pub world: &'a SystemWorld<'w>,
60    pub index: &'a dyn ComponentIndex,
61    excluded: Option<(Entity, TypeId)>,
62    lent: Option<(TypeId, &'a dyn ComponentLoans)>,
63}
64
65impl<'a, 'w> NativeProperties<'a, 'w> {
66    pub fn new(world: &'a SystemWorld<'w>, index: &'a dyn ComponentIndex) -> Self {
67        Self { world, index, excluded: None, lent: None }
68    }
69    pub fn object(&self, id: ObjectId) -> Option<Entity> {
70        let entity = self.world.entity(id)?;
71        self.index.component_count(entity).map(|_| entity)
72    }
73    pub fn with_component<R>(&self, slot: ComponentSlot, read: impl FnOnce(&dyn Component) -> R) -> Option<R> {
74        if self.excluded == Some((slot.entity, slot.type_id())) { return None; }
75        if let Some((_, values)) = self.lent.filter(|(ty, _)| *ty == slot.type_id()) {
76            return values.get(slot.entity).map(read);
77        }
78        let value = slot.read_in(self.world)?;
79        Some(read(&*value))
80    }
81    pub fn component(&self, owner: Entity, name: &str, sel: InstanceSel<'_>) -> Option<(usize, ComponentSlot)> {
82        let mut seen = 0;
83        let mut fallback = None;
84        for position in 0..self.index.component_count(owner)? {
85            let slot = self.index.component_slot(owner, position)?;
86            if slot.name != name || self.excluded == Some((slot.entity, slot.type_id())) { continue; }
87            if sel.label.is_empty() && seen == sel.index { return Some((position, slot)); }
88            if !sel.label.is_empty() {
89                if self.with_component(slot, |value| value.instance_label().as_deref() == Some(sel.label)) == Some(true) { return Some((position, slot)); }
90                if seen == sel.index { fallback = Some((position, slot)); }
91            }
92            seen += 1;
93        }
94        fallback
95    }
96}

98/// Update typed instances while lending all other instances to reflection.
99/// The positions table contains identities only; values remain borrowed in
100/// their native columns, with no per-instance copies or unchecked references.
101pub(crate) fn run_with_properties<T: Component>(world: &SystemWorld<'_>, ctx: &mut SystemContext<'_>, mut update: impl FnMut(&mut T, Entity, &mut SystemContext<'_>, &NativeProperties<'_, '_>)) {
102    let mut query = world.query::<(&Owner, &mut T)>();
103    let mut actors: Vec<_> = query.iter().filter(|(_, (owner, _))| ctx.is_visible(world, owner.0)).collect();
104    actors.sort_unstable_by_key(|(instance, (owner, _))| ctx.instance_order::<T>(world, owner.0, *instance));
105    let positions: HashMap<_, _> = actors.iter().enumerate().map(|(index, (entity, _))| (*entity, index)).collect();
106    for current in 0..actors.len() {
107        let (before, rest) = actors.split_at_mut(current);
108        let ((entity, (owner, value)), after) = rest.split_first_mut().unwrap();
109        if !ctx.is_active(world, owner.0) { continue; }
110        let loans = QueryLoans { before, after, current, positions: &positions };
111        let reader = NativeProperties { world, index: ctx.services.components.unwrap_or(&MetadataOnly),
112            excluded: Some((*entity, TypeId::of::<T>())), lent: Some((TypeId::of::<T>(), &loans)) };
113        update(value, owner.0, ctx, &reader);
114    }
115}

117impl PropertyReader for NativeProperties<'_, '_> {
118    fn read(&self, object: ObjectId, component: &str, property: &str) -> Result<ArgValue, String> {
119        let owner = self.object(object).ok_or("object not found")?;
120        if component.is_empty() || component == crate::OBJECT_COMPONENT {
121            let name = self.world.get::<Name>(owner).ok_or("object has no name")?;
122            let position = self.world.world_transform(owner).ok_or("object has no position")?.w_axis.truncate();
123            return crate::object_property(object, &name.0, self.world.current_active(owner), position, property);
124        }
125        let (_, slot) = self.component(owner, component, InstanceSel::FIRST)
126            .ok_or_else(|| format!("object has no {component} component (a component cannot read itself during its own action)"))?;
127        self.with_component(slot, |value| value.read_property(property)).flatten()
128            .ok_or_else(|| format!("{component} has no property '{property}'"))
129    }
130}

132pub fn invoke_action_in(world: &SystemWorld<'_>, index: &dyn ComponentIndex, mut call: NativeActionCall<'_>, out: &mut Vec<ComponentCommand>) -> Result<(), String> {
133    let reader = NativeProperties::new(world, index);
134    let owner = reader.object(call.object).ok_or("target object is gone")?;
135    let slot = match index.component_slot(owner, call.slot as usize) {
136        Some(slot) if slot.name == call.component => slot,
137        _ => {
138            let (position, slot) = reader.component(owner, call.component, InstanceSel::FIRST)
139                .ok_or_else(|| format!("target has no {} component", call.component))?;
140            call.slot = position as u32;
141            slot
142        }
143    };
144    slot.invoke_in(world, index, call, out)
145}

147pub(super) fn invoke_typed<T: Component>(world: &SystemWorld<'_>, index: &dyn ComponentIndex, slot: ComponentSlot, call: NativeActionCall<'_>, out: &mut Vec<ComponentCommand>) -> Result<(), String> {
148    // hecs borrows columns. Lend the disjoint other instances from the same
149    // checked query instead of attempting to reborrow that column, copying
150    // payloads, or creating unchecked transient component references.
151    let mut query = world.query::<&mut T>();
152    let mut target = None;
153    let mut values: HashMap<Entity, &dyn Component> = HashMap::new();
154    for (entity, value) in query.iter() {
155        if entity == slot.entity { target = Some(value); }
156        else { values.insert(entity, value); }
157    }
158    let target = target.ok_or("target component is gone")?;
159    let reader = NativeProperties { world, index, excluded: Some((slot.entity, slot.type_id())), lent: Some((TypeId::of::<T>(), &values)) };
160    target.invoke_action(&mut ActionCtx { self_id: call.object, instigator: call.instigator, commands: out, reader: Some(&reader) }, call.action, call.args)
161}
