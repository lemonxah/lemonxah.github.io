9/// A disabled `T`, stored densely in its own typed ECS column. Queries for `T`
10/// see enabled instances; query this type explicitly to inspect disabled data.
11/// Authoring slots preserve their order and expose `T` through component_inner.
12#[repr(transparent)]
13pub struct DisabledComponent<T: Component> {
14    value: T,
15}
16
17impl<T: Component> DisabledComponent<T> {
18    pub(crate) fn new(value: T) -> Self { Self { value } }
19    pub fn value(&self) -> &T { &self.value }
20    pub fn value_mut(&mut self) -> &mut T { &mut self.value }
21    pub fn into_value(self) -> T { self.value }
22}
23impl<T: Component> Deref for DisabledComponent<T> {
24    type Target = T;
25    fn deref(&self) -> &T { &self.value }
26}
27impl<T: Component> DerefMut for DisabledComponent<T> {
28    fn deref_mut(&mut self) -> &mut T { &mut self.value }
29}
30
31impl<T: Component> Component for DisabledComponent<T> {
32    fn into_any(self: Box<Self>) -> Box<dyn Any> { self }
33    fn into_storage(self: Box<Self>) -> Box<dyn Component> { self }
34    fn into_enabled(self: Box<Self>) -> Box<dyn Component> { Box::new(self.value) }
35    fn into_disabled(self: Box<Self>) -> Box<dyn Component> { self }
36    fn inner_data(&self) -> &dyn Component { self.value.inner_data() }
37    fn inner_data_mut(&mut self) -> &mut dyn Component { self.value.inner_data_mut() }
38    fn is_enabled(&self) -> bool { false }
39    fn storage_type(&self) -> ComponentType { ComponentType::of::<Self>() }
40    fn type_name(&self) -> &'static str { self.value.type_name() }
41    fn save(&self) -> String { self.value.save() }
42    fn actions(&self) -> &'static [ActionDesc] { self.value.actions() }
43    fn invoke_action(&mut self, _ctx: &mut ActionCtx, _name: &str, _args: &[ArgValue]) -> Result<(), String> {
44        Err(format!("{} is disabled", self.value.type_name()))
45    }
46    fn properties(&self) -> &'static [PropertyDesc] { self.value.properties() }
47    fn read_property(&self, name: &str) -> Option<ArgValue> { self.value.read_property(name) }
48    fn instance_label(&self) -> Option<String> { self.value.instance_label() }
49    fn as_any(&self) -> &dyn Any { self }
50    fn as_any_mut(&mut self) -> &mut dyn Any { self }
51}

60    #[derive(Default, Serialize, Deserialize)]
61    struct Probe {
62        id: u32,
63        data: Vec<u8>,
64        #[serde(skip)]
65        live: u32,
66    }
67    impl TypedComponent for Probe { const NAME: &'static str = "Disabled Storage Probe"; }
68    fn probe(id: u32) -> Box<dyn Component> { Box::new(Probe { id, data: vec![1, 2, 3], live: 40 }) }

70    #[test]
71    fn disabled_payload_is_inline_and_native_queries_and_editor_share_unsaved_state() {
72        let mut scene = SceneWorld::new();
73        let owner = scene.spawn("Object");
74        let mut components = ComponentSet::new();
75        components.push(Box::new(Disabled(probe(1))));
76        components.attach(&scene, owner);
77        let allocation = components.get_including_disabled::<Probe>().unwrap().data.as_ptr();
78        assert_eq!(std::mem::size_of::<DisabledComponent<Probe>>(), std::mem::size_of::<Probe>());
79        assert!(components.get_typed::<Probe>().is_none());
80        scene.with_world(|world| {
81            assert_eq!(world.query::<&Disabled>().iter().count(), 0, "construction boxes cannot remain in ECS");
82            assert_eq!(world.query::<&Probe>().iter().count(), 0);
83            for (_, (owner_data, value)) in world.query::<(&Owner, &mut DisabledComponent<Probe>)>().iter() {
84                assert_eq!(owner_data.0, owner);
85                value.live += 2;
86                assert_eq!(value.data.as_ptr(), allocation);
87            }
88        });
89        assert_eq!(components.get_including_disabled::<Probe>().unwrap().live, 42);
90        components.get_including_disabled_mut::<Probe>().unwrap().id = 7;
91        for _ in 0..8 {
92            assert!(components.set_enabled(0, true));
93            let value = components.get_typed::<Probe>().unwrap();
94            assert_eq!((value.id, value.live), (7, 42));
95            assert_eq!(value.data.as_ptr(), allocation, "toggle moves the payload without cloning buffers");
96            drop(value);
97            assert!(components.set_enabled(0, false));
98            assert!(!components.set_enabled(0, false));
99        }
100        let mut next = SceneWorld::new();
101        let next_owner = next.spawn("Next");
102        components.detach();
103        components.attach(&next, next_owner);
104        let value = components.get_including_disabled::<Probe>().unwrap();
105        assert_eq!((value.id, value.live, value.data.as_ptr()), (7, 42, allocation));
106        drop(value);
107        next.with_world(|world| assert_eq!(world.get::<DisabledComponent<Probe>>(next_owner).unwrap().live, 42));
108    }

110    #[test]
111    fn duplicate_enable_edits_preserve_order_and_the_primary_enabled_and_disabled_values() {
112        let mut scene = SceneWorld::new();
113        let owner = scene.spawn("Object");
114        let mut components = ComponentSet::from_vec(vec![Box::new(Disabled(probe(1))), probe(2), Box::new(Disabled(probe(3))), probe(4)]);
115        components.attach(&scene, owner);
116        let values = |components: &ComponentSet| components.iter().map(|value| {
117            (component_enabled(value.as_ref()), component_inner(value.as_ref()).as_any().downcast_ref::<Probe>().unwrap().id)
118        }).collect::<Vec<_>>();
119        assert_eq!(values(&components), [(false, 1), (true, 2), (false, 3), (true, 4)]);
120        scene.with_world(|world| {
121            assert_eq!(world.query::<&Probe>().iter().count(), 2);
122            assert_eq!(world.query::<&DisabledComponent<Probe>>().iter().count(), 2);
123            assert_eq!(world.get::<Probe>(owner).unwrap().id, 2);
124            assert_eq!(world.get::<DisabledComponent<Probe>>(owner).unwrap().id, 1);
125        });
126        components.set_enabled(0, true);
127        components.set_enabled(1, false);
128        assert_eq!(values(&components), [(true, 1), (false, 2), (false, 3), (true, 4)]);
129        scene.with_world(|world| {
130            assert_eq!(world.get::<Probe>(owner).unwrap().id, 1);
131            assert_eq!(world.get::<DisabledComponent<Probe>>(owner).unwrap().id, 2);
132        });
133        {
134            let mut value = components.get_mut(1).unwrap();
135            component_inner_data_mut(value.as_mut()).as_any_mut().downcast_mut::<Probe>().unwrap().live = 9;
136        }
137        components.remove(1);
138        assert_eq!(values(&components), [(true, 1), (false, 3), (true, 4)]);
139        scene.with_world(|world| assert_eq!(world.get::<DisabledComponent<Probe>>(owner).unwrap().id, 3));
140    }

142    #[test]
143    fn native_disabled_insertion_removal_and_nested_compatibility_inputs_keep_values_typed() {
144        let mut scene = SceneWorld::new();
145        let owner = scene.spawn("Object");
146        scene.insert_disabled(owner, Probe { id: 1, live: 2, ..Default::default() }).unwrap();
147        let duplicate = scene.insert_disabled(owner, Probe { id: 3, live: 4, ..Default::default() }).unwrap();
148        assert_ne!(duplicate, owner);
149        assert_eq!(scene.remove_disabled::<Probe>(duplicate).unwrap().live, 4);
150        assert!(scene.get::<Owner>(duplicate).is_none());
151        assert_eq!(scene.remove_disabled::<Probe>(owner).unwrap().live, 2);
152        assert!(scene.object_id(owner).is_some());
153        let mut components = ComponentSet::new();
154        components.push(Box::new(Disabled(Box::new(Disabled(probe(8))))));
155        components.attach(&scene, owner);
156        assert_eq!(components.get_including_disabled::<Probe>().unwrap().id, 8);
157        scene.with_world(|world| {
158            assert_eq!(world.query::<&Disabled>().iter().count(), 0);
159            assert_eq!(world.query::<&DisabledComponent<Probe>>().iter().count(), 1);
160        });
161        assert!(components.set_enabled(0, true));
162        assert_eq!(components.get_typed::<Probe>().unwrap().id, 8);
163    }
