1//! Portable wave-asset contract and deterministic Gerstner wave generation.
2use serde::{Deserialize, Serialize};
3use crate::{GerstnerWave, Water};
4
5#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
6#[serde(default)]
7pub struct WaterWaveSpectrum {
8    pub seed: u32,
9    pub count: u32,
10    pub direction: f32,
11    pub spread: f32,
12    pub min_wavelength: f32,
13    pub max_wavelength: f32,
14    /// Significant wave height (approximately four times RMS surface height).
15    pub height: f32,
16    pub steepness: f32,
17    /// Jitter within each wavelength stratum, 0 = evenly spaced, 1 = random.
18    pub randomness: f32,
19    /// Exponent of the logarithmic wavelength distribution; >1 favours short waves.
20    pub wavelength_falloff: f32,
21    /// Relative amplitude scales as (wavelength / longest)^falloff before
22    /// normalisation to significant height. Zero gives equal wave energy.
23    pub amplitude_falloff: f32,
24    /// Multipliers of steepness at the shortest/longest wavelength.
25    pub small_wave_steepness: f32,
26    pub large_wave_steepness: f32,
27}
28impl Default for WaterWaveSpectrum {
29    fn default() -> Self {
30        Self { seed: 17, count: 6, direction: 20.0, spread: 65.0, min_wavelength: 0.8,
31            max_wavelength: 16.0, height: 0.65, steepness: 0.55,
32            randomness: 1.0, wavelength_falloff: 1.0, amplitude_falloff: 0.5,
33            small_wave_steepness: 1.0, large_wave_steepness: 1.0 }
34    }
35}
36impl WaterWaveSpectrum {
37    pub fn generate(&self) -> Vec<GerstnerWave> {
38        let finite = |v: f32, d: f32| if v.is_finite() { v } else { d };
39        let n = self.count.clamp(1, Water::MAX_WAVES as u32) as usize;
40        let lo = finite(self.min_wavelength, 0.8).clamp(0.05, 1000.0);
41        let hi = finite(self.max_wavelength, 16.0).clamp(lo, 1000.0);
42        let height = finite(self.height, 0.65).clamp(0.0, 100.0);
43        let mut state = self.seed;
44        let mut random = || { state = state.wrapping_mul(1664525).wrapping_add(1013904223); (state >> 8) as f32 / 16777216.0 };
45        let mut waves: Vec<_> = (0..n).map(|i| {
46            let jitter = 0.5 + (random() - 0.5) * finite(self.randomness, 1.0).clamp(0.0, 1.0);
47            let t = ((i as f32 + jitter) / n as f32).powf(finite(self.wavelength_falloff, 1.0).clamp(0.05, 10.0));
48            let wavelength = lo * (hi / lo).powf(t);
49            GerstnerWave {
50                direction: finite(self.direction, 0.0) + (random() - 0.5) * finite(self.spread, 65.0).clamp(0.0, 360.0),
51                wavelength,
52                amplitude: (wavelength / hi).powf(finite(self.amplitude_falloff, 0.5).clamp(0.0, 10.0)),
53                steepness: (finite(self.steepness, 0.55) * (
54                    finite(self.small_wave_steepness, 1.0).clamp(0.0, 2.0) * (1.0 - t)
55                    + finite(self.large_wave_steepness, 1.0).clamp(0.0, 2.0) * t)).clamp(0.0, 1.0), speed: 1.0,
56            }
57        }).collect();
58        // Avoid energy underflow for steep falloff across wide spectra.
59        let sum = waves.iter().map(|w| (w.amplitude as f64).powi(2)).sum::<f64>();
60        let scale = height as f64 / (4.0 * (sum * 0.5).sqrt()).max(f64::MIN_POSITIVE);
61        for wave in &mut waves { wave.amplitude = (wave.amplitude as f64 * scale) as f32; }
62        waves
63    }
64}

66/// Engine extensions generate the same portable waves used by rendering and
67/// headless physics. Implementations must be deterministic for the same settings.
68/// Register on LoadedScene before resolving assets; this API has no renderer or
69/// editor dependencies. Output is validated and bounded by WaterWaveProfile.
70pub trait WaterWaveGenerator: Send + Sync {
71    fn generate(&self, settings: &std::collections::BTreeMap<String, f32>) -> Result<Vec<GerstnerWave>, String>;
72}
73
74#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
75#[serde(default)]
76pub struct WaterWaveGeneratorConfig {
77    pub id: String,
78    pub settings: std::collections::BTreeMap<String, f32>,
79}
80
81#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
82#[serde(default)]
83pub struct WaterWaveProfile {
84    /// Registered engine generator. None uses the built-in spectrum/list below.
85    pub generator: Option<WaterWaveGeneratorConfig>,
86    /// None uses the authored list verbatim (subject to runtime safety limits).
87    pub spectrum: Option<WaterWaveSpectrum>,
88    pub waves: Vec<GerstnerWave>,
89    /// Optional FFT patch shared by every body using this profile.
90    pub ocean: Option<crate::OceanSpectrum>,
91}
92impl Default for WaterWaveProfile {
93    fn default() -> Self { Self { generator: None, spectrum: Some(WaterWaveSpectrum::default()), waves: Vec::new(), ocean: None } }
94}
95impl WaterWaveProfile {
96    pub fn resolved(&self) -> Vec<GerstnerWave> {
97        if let Some(spectrum) = &self.spectrum { return spectrum.generate(); }
98        Self::validated(&self.waves)
99    }
100    pub fn generated(&self, generator: &dyn WaterWaveGenerator) -> Result<Vec<GerstnerWave>, String> {
101        let config = self.generator.as_ref().ok_or("profile has no custom generator")?;
102        let waves = generator.generate(&config.settings)?;
103        if waves.iter().any(|w| [w.direction, w.wavelength, w.amplitude, w.steepness, w.speed].iter().any(|v| !v.is_finite())) {
104            return Err(format!("wave generator '{}' produced non-finite values", config.id));
105        }
106        Ok(Self::validated(&waves))
107    }
108    fn validated(waves: &[GerstnerWave]) -> Vec<GerstnerWave> {
109        waves.iter().filter(|w| [w.direction, w.wavelength, w.amplitude, w.steepness, w.speed].iter().all(|v| v.is_finite())).take(Water::MAX_WAVES)
110            .map(|w| GerstnerWave { direction: w.direction.rem_euclid(360.0), wavelength: w.wavelength.clamp(0.05, 1000.0),
111                amplitude: w.amplitude.clamp(0.0, 50.0), steepness: w.steepness.clamp(0.0, 1.0), speed: w.speed.clamp(0.0, 10.0) }).collect()
112    }
113}

117    use super::*;
118    #[test]
119    fn generated_sea_is_repeatable_and_has_the_authored_energy() {
120        let spectrum = WaterWaveSpectrum::default();
121        let a = spectrum.generate();
122        assert_eq!(a, spectrum.generate());
123        assert_ne!(a, WaterWaveSpectrum { seed: 18, ..spectrum.clone() }.generate());
124        let hs = 4.0 * (a.iter().map(|w| w.amplitude.powi(2)).sum::<f32>() * 0.5).sqrt();
125        assert!((hs - spectrum.height).abs() < 1e-5);
126        assert!(a.iter().all(|w| w.wavelength >= spectrum.min_wavelength && w.wavelength <= spectrum.max_wavelength));
127    }
128
129    #[test]
130    fn legacy_profiles_keep_their_distribution_and_new_controls_keep_energy() {
131        let legacy: WaterWaveProfile = ron::from_str("(spectrum:Some((seed:17,count:6,direction:20.0,spread:65.0,min_wavelength:0.8,max_wavelength:16.0,height:0.65,steepness:0.55)),waves:[])").unwrap();
132        assert_eq!(legacy, WaterWaveProfile::default());
133        let base = WaterWaveSpectrum { count: 32, spread: 0.0, randomness: 0.0, ..Default::default() };
134        let regular = base.generate();
135        assert_eq!(regular.len(), 32);
136        let altered = WaterWaveSpectrum { wavelength_falloff: 3.0, amplitude_falloff: 2.0,
137            small_wave_steepness: 0.0, large_wave_steepness: 1.5, ..base.clone() }.generate();
138        assert!(altered.iter().zip(&regular).all(|(a,b)| a.wavelength < b.wavelength));
139        assert!(altered.first().unwrap().steepness < altered.last().unwrap().steepness);
140        let hs = 4.0 * (altered.iter().map(|w| w.amplitude.powi(2)).sum::<f32>() * 0.5).sqrt();
141        assert!((hs - base.height).abs() < 1e-5);
142        assert_eq!(regular, WaterWaveSpectrum { seed: 42, ..base }.generate());
143    }

157    fn generator_output_is_bounded_and_nonfinite_output_is_rejected() {
158        struct Generator(bool);
159        impl WaterWaveGenerator for Generator {
160            fn generate(&self, _: &std::collections::BTreeMap<String, f32>) -> Result<Vec<GerstnerWave>, String> {
161                Ok(vec![GerstnerWave { amplitude: if self.0 { f32::NAN } else { 100.0 }, ..Default::default() }; 100])

163        }
164        let profile = WaterWaveProfile { generator: Some(Default::default()), ..Default::default() };
165        let waves = profile.generated(&Generator(false)).unwrap();
166        assert_eq!(waves.len(), Water::MAX_WAVES);
167        assert!(waves.iter().all(|w| w.amplitude == 50.0));
168        assert!(profile.generated(&Generator(true)).is_err());
