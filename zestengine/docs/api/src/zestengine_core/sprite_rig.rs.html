1//! 2D skeletal animation components (2D-7): authoring data only. The skinning,
2//! IK and sprite-swap logic lives in `zestengine-assets` (`rig2d.rs`, `ik2d.rs`,
3//! `sprite_library.rs`) and runs in the engine's sprite gather
4//! (`engine/src/scene/rig2d.rs`); a scene with none of these components pays
5//! nothing.
6//!
7//! * [`SpriteBone`] marks an object as a bone of a `.rig2d` skeleton. Bones are
8//!   ordinary scene objects, so the existing animation tracks (position,
9//!   rotation, scale by object path) animate them with no extra track kind.
10//! * [`SpriteSkin`] sits next to a `SpriteRenderer`: it names the rig asset and
11//!   the bone objects (index = skeleton bone index). The renderer's sprite id
12//!   picks the skinned mesh inside the rig.
13//! * [`SpriteLibraryRef`] / [`SpriteResolver`]: a sprite swap. A clip's
14//!   `SpriteSwap` track writes `hash`; the engine looks the hash up in the
15//!   library and changes what the sibling `SpriteRenderer` shows.
16//! * [`IkSolver2D`]: a CCD / FABRIK / two-bone limb solver over the bone chain
17//!   ending at `tip`, run after animation and before skinning.
18

23/// Mask of the sprite-swap hash: 24 bits, exactly representable as an `f32`
24/// clip key.
25pub const SPRITE_SWAP_HASH_MASK: u32 = 0x00FF_FFFF;
26
27/// Hash of a library `category` + `label` (FNV-1a, 24 bits, never 0). The
28/// assets crate's `sprite_library::entry_hash` is this function.
29pub fn sprite_swap_hash(category: &str, label: &str) -> u32 {
30    let mut h: u32 = 0x811c_9dc5;
31    for b in category.bytes().chain(std::iter::once(0)).chain(label.bytes()) {
32        h = (h ^ b as u32).wrapping_mul(0x0100_0193);
33    }
34    let h = (h ^ (h >> 24)) & SPRITE_SWAP_HASH_MASK;
35    if h == 0 { 1 } else { h }
36}

38/// A bone of a sprite rig.
39#[derive(Clone, PartialEq, Debug, Serialize, Deserialize)]
40#[serde(default)]
41pub struct SpriteBone {
42    /// The skeleton bone's stable id (`Bone2D::id`); 0 = match by name.
43    pub bone_id: u64,
44    /// IK: clamp this bone's rotation relative to its parent.
45    pub limit_rotation: bool,
46    /// Degrees.
47    pub min_angle: f32,
48    /// Maximum rotation relative to the parent (degrees).
49    pub max_angle: f32,
50}
51
52impl Default for SpriteBone {
53    fn default() -> Self {
54        Self { bone_id: 0, limit_rotation: false, min_angle: -90.0, max_angle: 90.0 }
55    }
56}

62/// Skins the `SpriteRenderer` on the same object with a rig's weighted mesh.
63#[derive(Clone, PartialEq, Debug, Serialize, Deserialize)]
64#[serde(default)]
65pub struct SpriteSkin {
66    /// Project-relative `.rig2d` path.
67    pub rig: String,
68    /// Bone objects, by skeleton bone index. Entries that are unset or missing
69    /// are rebound automatically when `auto_rebind` is on.
70    pub bones: Vec<ObjectRef>,
71    /// The object whose descendants carry the [`SpriteBone`]s; unset = this object.
72    pub root: ObjectRef,
73    /// Rebind `bones` by bone id (then name) among the root's descendants
74    /// whenever one is unresolved.
75    pub auto_rebind: bool,
76    /// Skin even when the sprite is not drawn (offscreen / culled).
77    pub always_update: bool,
78}
79
80impl Default for SpriteSkin {
81    fn default() -> Self {
82        Self { rig: String::new(), bones: Vec::new(), root: ObjectRef::NONE, auto_rebind: true, always_update: false }
83    }
84}

86impl TypedComponent for SpriteSkin {
87    const NAME: &'static str = "Sprite Skin";
88}

102/// Shows the library entry `hash` on the `SpriteRenderer` of this object. The
103/// library is the nearest [`SpriteLibraryRef`] on this object or an ancestor.
104/// `hash` is what `SpriteSwap` animation tracks write.
105#[derive(Clone, PartialEq, Debug, Serialize, Deserialize)]
106#[serde(default)]
107#[derive(Default)]
108pub struct SpriteResolver {
109    /// Library category (e.g. "Head").
110    pub category: String,
111    /// Library entry label (e.g. "Happy").
112    pub label: String,
113    /// 24-bit swap hash; 0 = none yet. [`Self::set_entry`] keeps it in step
114    /// with `category` / `label`; an animation overrides it directly.
115    pub hash: u32,
116}
117
118
119impl SpriteResolver {
120    /// Set the category and label, recomputing the 24-bit swap hash.
121    pub fn set_entry(&mut self, category: &str, label: &str) {
122        self.category = category.to_owned();
123        self.label = label.to_owned();
124        self.hash = sprite_swap_hash(category, label);
125    }
126}
127
128impl TypedComponent for SpriteResolver {
129    const NAME: &'static str = "Sprite Resolver";
130}

132/// Which IK algorithm solves the chain.
133#[derive(Clone, Copy, PartialEq, Eq, Debug, Default, Serialize, Deserialize)]
134pub enum IkSolverKind {
135    /// Cyclic coordinate descent.
136    #[default]
137    Ccd,
138    /// Forward and backward reaching.
139    Fabrik,
140    /// Closed-form two-bone limb.
141    Limb,
142}
143
144impl IkSolverKind {
145    /// Every solver, in declaration order.
146    pub const ALL: [IkSolverKind; 3] = [Self::Ccd, Self::Fabrik, Self::Limb];
147    /// Inspector label of this solver.
148    pub fn label(self) -> &'static str {
149        match self {
150            Self::Ccd => "CCD",
151            Self::Fabrik => "FABRIK",
152            Self::Limb => "Limb (two bones)",
153        }
154    }
155}

157/// 2D inverse kinematics for the bone chain that ends at `tip` and runs up
158/// `chain_length` bones through the parents.
159#[derive(Clone, PartialEq, Debug, Serialize, Deserialize)]
160#[serde(default)]
161pub struct IkSolver2D {
162    /// Which IK algorithm solves the chain.
163    pub solver: IkSolverKind,
164    /// The last bone of the chain (its end is the effector).
165    pub tip: ObjectRef,
166    /// Bones in the chain, counted up from `tip` (Limb uses 2).
167    pub chain_length: u32,
168    /// The target the tip reaches toward.
169    pub target: ObjectRef,
170    /// 0 = animation only, 1 = fully solved.
171    pub weight: f32,
172    /// Solver iterations.
173    pub iterations: u32,
174    /// Solve until the effector is within this distance of the target.
175    pub tolerance: f32,
176    /// Limb: bend the other way.
177    pub flip: bool,
178}
179
180impl Default for IkSolver2D {
181    fn default() -> Self {
182        Self { solver: IkSolverKind::Ccd, tip: ObjectRef::NONE, chain_length: 2, target: ObjectRef::NONE, weight: 1.0, iterations: 10, tolerance: 0.001, flip: false }
183    }
184}

186impl TypedComponent for IkSolver2D {
187    const NAME: &'static str = "IK Solver 2D";
188}

194    #[test]
195    fn swap_hash_is_24_bit_stable_and_nonzero() {
196        let h = sprite_swap_hash("Head", "Happy");
197        assert!(h != 0 && h <= SPRITE_SWAP_HASH_MASK);
198        assert_eq!(h as f32 as u32, h);
199        assert_eq!(h, sprite_swap_hash("Head", "Happy"));
200        assert_ne!(h, sprite_swap_hash("Head", "Sad"));
201        let mut r = SpriteResolver::default();
202        r.set_entry("Head", "Happy");
203        assert_eq!(r.hash, h);
204    }

206    #[test]
207    fn old_files_load_with_defaults() {
208        let s: SpriteSkin = ron::from_str("()").unwrap();
209        assert!(s.auto_rebind && s.bones.is_empty() && !s.always_update);
210        let b: SpriteBone = ron::from_str("(bone_id: 5)").unwrap();
211        assert_eq!((b.bone_id, b.limit_rotation, b.min_angle, b.max_angle), (5, false, -90.0, 90.0));
212        let i: IkSolver2D = ron::from_str("(solver: Limb)").unwrap();
213        assert_eq!((i.solver, i.chain_length, i.weight, i.iterations), (IkSolverKind::Limb, 2, 1.0, 10));
214        let r: SpriteResolver = ron::from_str("(hash: 9)").unwrap();
215        assert_eq!(r.hash, 9);
216        let l: SpriteLibraryRef = ron::from_str("()").unwrap();
217        assert!(l.library.is_empty());
218    }
