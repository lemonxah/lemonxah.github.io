1//! Homogeneous water optics, in inverse metres. Shared by surface rendering,
2//! submerged views and tools; no backend resources or light selection here.
3use glam::Vec3;
4use serde::{Deserialize, Serialize};
5
6#[derive(Clone, Copy, Debug, PartialEq, Serialize, Deserialize)]
7#[serde(default)]
8pub struct WaterOptics {
9    /// RGB absorption coefficients (1/m). Red usually attenuates first.
10    pub absorption: [f32; 3],
11    /// RGB scattering coefficients (1/m).
12    pub scattering: [f32; 3],
13    /// Refractive index relative to air.
14    pub ior: f32,
15    /// Henyey–Greenstein anisotropy: positive scatters light forward.
16    pub anisotropy: f32,
17}
18
19impl Default for WaterOptics {
20    fn default() -> Self {
21        Self { absorption: [0.32, 0.065, 0.025], scattering: [0.015, 0.035, 0.045], ior: 1.333, anisotropy: 0.65 }
22    }
23}
24
25impl WaterOptics {
26    pub fn sanitized(self) -> Self {
27        let finite = |v: f32, fallback: f32, lo: f32, hi: f32| if v.is_finite() { v.clamp(lo, hi) } else { fallback };
28        let d = Self::default();
29        Self {
30            absorption: std::array::from_fn(|i| finite(self.absorption[i], d.absorption[i], 0.0, 100.0)),
31            scattering: std::array::from_fn(|i| finite(self.scattering[i], d.scattering[i], 0.0, 100.0)),
32            ior: finite(self.ior, d.ior, 1.0, 2.5),
33            anisotropy: finite(self.anisotropy, d.anisotropy, -0.95, 0.95),
34        }
35    }
36
37    pub fn extinction(self) -> Vec3 {
38        let v = self.sanitized();
39        Vec3::from(v.absorption) + Vec3::from(v.scattering)
40    }
41
42    /// Beer–Lambert transmittance along the ray, not vertical water depth.
43    pub fn transmittance(self, distance: f32) -> Vec3 {
44        let distance = if distance.is_nan() { 0.0 } else { distance.clamp(0.0, 1.0e6) };
45        (-self.extinction() * distance).exp()
46    }
47
48    /// Integral of scattering along a segment under unit incident radiance.
49    /// Multiplying by a phase function and incident light gives radiance.
50    pub fn integrated_scattering(self, distance: f32) -> Vec3 {
51        Vec3::from(self.sanitized().scattering) / self.extinction().max(Vec3::splat(1e-6))
52            * (Vec3::ONE - self.transmittance(distance))
53    }
54
55    pub fn normal_reflectance(self) -> f32 {
56        let n = self.sanitized().ior;
57        ((n - 1.0) / (n + 1.0)).powi(2)
58    }
59
60    /// Unpolarized dielectric Fresnel, including the underwater critical
61    /// angle. `cosine` is the absolute view/normal cosine; `exiting` means
62    /// travelling from water into air. The result partitions reflected and
63    /// transmitted energy at the surface.
64    pub fn surface_reflectance(self, cosine: f32, exiting: bool) -> f32 {
65        let ior = self.sanitized().ior;
66        if ior == 1.0 { return 0.0; }
67        let c = if cosine.is_finite() { cosine.abs().clamp(0.0, 1.0) } else { 1.0 };
68        let eta = if exiting { ior } else { 1.0 / ior };
69        let sin2 = eta * eta * (1.0 - c * c);
70        if sin2 >= 1.0 { return 1.0; }
71        let ct = (1.0 - sin2).sqrt();
72        let rs = (eta * c - ct) / (eta * c + ct).max(1e-6);
73        let rp = (eta * ct - c) / (eta * ct + c).max(1e-6);
74        (0.5 * (rs * rs + rp * rp)).clamp(0.0, 1.0)
75    }
76
77    /// Normalized Henyey–Greenstein phase, integral over the sphere = 1.
78    pub fn phase(self, cosine: f32) -> f32 {
79        let g = self.sanitized().anisotropy;
80        let c = if cosine.is_finite() { cosine.clamp(-1.0, 1.0) } else { 0.0 };
81        (1.0 - g * g) / (4.0 * std::f32::consts::PI * (1.0 + g * g - 2.0 * g * c).max(1e-4).powf(1.5))
82    }
83
84    /// Convert older colour/depth controls once at the rendering boundary.
85    /// All scenes then use the same optical model, including non-water fluids.
86    pub fn from_legacy(look: &crate::WaterLook) -> Self {
87        let depth = if look.absorption_depth.is_finite() { look.absorption_depth.max(0.01) } else { 4.0 };
88        let rate = 3.0 / depth;
89        let colour = Vec3::from(look.deep_color).clamp(Vec3::ZERO, Vec3::splat(0.98));
90        let extinction = Vec3::new(1.55, 1.0, 0.8) * rate;
91        Self {
92            absorption: (extinction * (Vec3::ONE - colour)).to_array(),
93            scattering: (extinction * colour).to_array(),
94            ior: 1.333,
95            anisotropy: 0.65,
96        }.sanitized()
97    }
98}

100#[cfg(test)]
101mod tests {
102    use super::*;
103    #[test]
104    fn surface_fresnel_has_a_clear_snell_window_and_no_reflection_at_matched_ior() {
105        let water = WaterOptics::default();
106        for exiting in [false, true] {
107            assert!((water.surface_reflectance(1.0, exiting) - water.normal_reflectance()).abs() < 1e-6);
108        }
109        assert_eq!(water.surface_reflectance(0.5, true), 1.0);
110        assert!(water.surface_reflectance(0.9, true) < 0.04);
111        let clear = WaterOptics { ior: 1.0, ..water };
112        for i in 0..=100 {
113            let c=i as f32/100.0;
114            assert_eq!(clear.surface_reflectance(c, false), 0.0);
115            for exiting in [false,true] { assert!((0.0..=1.0).contains(&water.surface_reflectance(c, exiting))); }
116        }
117    }
118    #[test]
119    fn old_water_looks_keep_legacy_controls_and_new_looks_roundtrip() {
120        let old: crate::WaterLook = ron::from_str("(deep_color: (0.1, 0.2, 0.3), absorption_depth: 7.0)").unwrap();
121        assert!(old.optics.is_none());
122        assert!(WaterOptics::from_legacy(&old).transmittance(10.0).is_finite());
123        let new = crate::WaterLook::default();
124        let saved = ron::to_string(&new).unwrap();
125        assert_eq!(ron::from_str::<crate::WaterLook>(&saved).unwrap(), new);
126        assert!(new.optics.is_some());
127    }
128    #[test]
129    fn optical_segments_compose_and_conserve_energy() {
130        let o = WaterOptics::default();
131        assert_eq!(o.transmittance(0.0), Vec3::ONE);
132        assert_eq!(o.integrated_scattering(0.0), Vec3::ZERO);
133        assert!((o.transmittance(7.0) - o.transmittance(3.0) * o.transmittance(4.0)).length() < 1e-6);
134        let t = o.transmittance(10.0);
135        assert!(t.x < t.y && t.y < t.z);
136        assert!((t + o.integrated_scattering(10.0)).cmple(Vec3::ONE).all());
137        assert!((o.normal_reflectance() - 0.02037).abs() < 1e-4);
138        let clear = WaterOptics { absorption: [0.0; 3], scattering: [0.0; 3], ..o };
139        assert_eq!(clear.transmittance(100.0), Vec3::ONE);
140        assert_eq!(clear.integrated_scattering(100.0), Vec3::ZERO);
141    }
142
143    #[test]
144    fn phase_is_normalized_and_finite_for_authored_extremes() {
145        for g in [-0.95, 0.0, 0.65, 0.95] {
146            let o = WaterOptics { anisotropy: g, ..Default::default() };
147            let n = 100_000;
148            let integral: f32 = (0..n).map(|i| o.phase(-1.0 + (i as f32 + 0.5) * 2.0 / n as f32)).sum::<f32>()
149                * 4.0 * std::f32::consts::PI / n as f32;
150            assert!((integral - 1.0).abs() < 0.002, "{g}: {integral}");
151        }
152        let bad = WaterOptics { absorption: [f32::NAN, -1.0, f32::INFINITY], scattering: [-1.0; 3], ior: f32::NAN, anisotropy: 9.0 };
153        assert!(bad.transmittance(10.0).is_finite());
154        assert!(bad.phase(f32::NAN).is_finite());
155    }
