1//! Sandboxed user-generated logic: the `Script` component.
2//!
3//! A `Script` hosts one WebAssembly module (compiled by `wasmtime`) and calls
4//! its exported `start` / `update` hooks. The module sees only the narrow
5//! imported-function API below; no WASI is linked, so guest code has no
6//! file-system or network access. Execution is bounded per hook call with
7//! fuel, and linear-memory growth is capped.
8//!
9//! # Host API (import module `env`)
10//!
11//! | import                        | signature                          |
12//! |-------------------------------|------------------------------------|
13//! | `env.get_dt`                  | `() -> f32`                        |
14//! | `env.get_time`                | `() -> f32`                        |
15//! | `env.set_position`            | `(f32, f32, f32) -> ()`            |
16//! | `env.get_position`            | `() -> (f32, f32, f32)`            |
17//! | `env.set_rotation`            | `(f32, f32, f32, f32) -> ()`       |
18//! | `env.get_rotation`            | `() -> (f32, f32, f32, f32)`       |
19//! | `env.set_scale`               | `(f32, f32, f32) -> ()`            |
20//! | `env.get_scale`               | `() -> (f32, f32, f32)`            |
21//! | `env.get_prop`                | `(i32) -> f32`                     |
22//! | `env.set_prop`                | `(i32, f32) -> ()`                 |
23//!
24//! Position/rotation/scale are the owning object's **local** transform (the
25//! fields stored in ECS). The eight `props` are the component's
26//! own numeric fields, so scripts have some per-object state. Reads happen
27//! against a copy written into the store before each hook and written back
28//! afterwards, so a trap leaves the object unchanged.
29//!
30//! # Module format
31//!
32//! The `module` field is a path to a `.wasm` binary. Content packages have
33//! packaged paths rewritten to absolute extraction paths at load time
34//! (see `docs/user-content.md`); base-game scenes resolve relative to the
35//! process working directory. The module must export at least one of
36//! `start` or `update` (both `() -> ()`).
37

44/// Per-hook fuel budget. Wasmtime consumes one unit per instruction, so this
45/// is a hard bound on guest work per `start` / `update` call.
46const FUEL_PER_CALL: u64 = 100_000;
47
48/// Cap on guest linear-memory growth (1 MiB). A module whose declared or grown
49/// memory exceeds this fails to instantiate / grow.
50const MEMORY_LIMIT_BYTES: usize = 1 << 20;
51
52/// Epoch-deadline backstop. The engine never increments the epoch, so this is
53/// wired but not a wall-clock limit yet; fuel is the active per-frame bound.
54const EPOCH_DEADLINE_TICKS: u64 = 1 << 40;
55
56/// Number of `props` slots exposed through `env.get_prop` / `env.set_prop`.
57pub const SCRIPT_PROP_COUNT: usize = 8;
58
59/// The current host API version. Future package manifests can refuse modules
60/// written against a newer ABI (see `docs/user-content.md`).
61pub const SCRIPT_API_VERSION: u32 = 1;
62
63#[derive(Clone, Copy)]
64enum Hook {
65    Start,
66    Update,
67}
68
69/// Host state owned by a script's `wasmtime::Store`. Imported functions read
70/// and write this; the component copies it in before a hook and out after it.
71struct ScriptState {
72    dt: f32,
73    time: f32,
74    translation: Vec3,
75    rotation: Quat,
76    scale: Vec3,
77    props: [f32; SCRIPT_PROP_COUNT],
78    limits: wasmtime::StoreLimits,
79}
80
81impl ScriptState {
82    fn new() -> Self {
83        Self {
84            dt: 0.0,
85            time: 0.0,
86            translation: Vec3::ZERO,
87            rotation: Quat::IDENTITY,
88            scale: Vec3::ONE,
89            props: [0.0; SCRIPT_PROP_COUNT],
90            limits: wasmtime::StoreLimitsBuilder::new()
91                .memory_size(MEMORY_LIMIT_BYTES)
92                .memories(1)
93                .instances(1)
94                .tables(1)
95                .build(),
96        }
97    }
98}

100/// A sandboxed WebAssembly behaviour component (see module docs).
101#[derive(Serialize, Deserialize)]
102pub struct Script {
103    /// Path to the `.wasm` module (absolute after content extraction).
104    pub module: String,
105    /// Script-visible numeric state (`env.get_prop` / `env.set_prop`).
106    pub props: [f32; SCRIPT_PROP_COUNT],
107    #[serde(skip)]
108    engine: Option<wasmtime::Engine>,
109    #[serde(skip)]
110    store: Option<wasmtime::Store<ScriptState>>,
111    #[serde(skip)]
112    update: Option<wasmtime::TypedFunc<(), ()>>,
113    #[serde(skip)]
114    start: Option<wasmtime::TypedFunc<(), ()>>,
115    /// Last compile / instantiate / trap error; the script is disabled once set.
116    #[serde(skip)]
117    error: Option<String>,
118    /// Startup belongs to this native instance, including late additions.
119    #[serde(skip)]
120    started: bool,
121}

131            start: None,

133            started: false,
134        }

138impl Script {
139    /// Compile `bytes` and instantiate it against the `env` API. Exposed as a
140    /// private helper so the loader (path) and the tests (raw bytes) share one
141    /// code path.
142    fn build_runtime(&mut self, bytes: &[u8]) -> Result<(), String> {
143        let mut config = wasmtime::Config::new();
144        config.consume_fuel(true);
145        config.epoch_interruption(true);
146
147        let engine = wasmtime::Engine::new(&config).map_err(|e| e.to_string())?;
148        let module = wasmtime::Module::new(&engine, bytes).map_err(|e| e.to_string())?;
149        let mut linker = wasmtime::Linker::<ScriptState>::new(&engine);
150
151        linker
152            .func_wrap("env", "get_dt", |caller: wasmtime::Caller<'_, ScriptState>| {
153                caller.data().dt
154            })
155            .map_err(|e| e.to_string())?;
156        linker
157            .func_wrap("env", "get_time", |caller: wasmtime::Caller<'_, ScriptState>| {
158                caller.data().time
159            })
160            .map_err(|e| e.to_string())?;
161        linker
162            .func_wrap(
163                "env",
164                "set_position",
165                |mut caller: wasmtime::Caller<'_, ScriptState>, x: f32, y: f32, z: f32| {
166                    caller.data_mut().translation = Vec3::new(x, y, z);
167                },
168            )
169            .map_err(|e| e.to_string())?;
170        linker
171            .func_wrap("env", "get_position", |caller: wasmtime::Caller<'_, ScriptState>| {
172                let p = caller.data().translation;
173                (p.x, p.y, p.z)
174            })
175            .map_err(|e| e.to_string())?;
176        linker
177            .func_wrap(
178                "env",
179                "set_rotation",
180                |mut caller: wasmtime::Caller<'_, ScriptState>, x: f32, y: f32, z: f32, w: f32| {
181                    caller.data_mut().rotation = Quat::from_xyzw(x, y, z, w);
182                },
183            )
184            .map_err(|e| e.to_string())?;
185        linker
186            .func_wrap("env", "get_rotation", |caller: wasmtime::Caller<'_, ScriptState>| {
187                let q = caller.data().rotation;
188                (q.x, q.y, q.z, q.w)
189            })
190            .map_err(|e| e.to_string())?;
191        linker
192            .func_wrap(
193                "env",
194                "set_scale",
195                |mut caller: wasmtime::Caller<'_, ScriptState>, x: f32, y: f32, z: f32| {
196                    caller.data_mut().scale = Vec3::new(x, y, z);
197                },
198            )
199            .map_err(|e| e.to_string())?;
200        linker
201            .func_wrap("env", "get_scale", |caller: wasmtime::Caller<'_, ScriptState>| {
202                let s = caller.data().scale;
203                (s.x, s.y, s.z)
204            })
205            .map_err(|e| e.to_string())?;
206        linker
207            .func_wrap(
208                "env",
209                "get_prop",
210                |caller: wasmtime::Caller<'_, ScriptState>, index: i32| {
211                    caller.data().props.get(index as usize).copied().unwrap_or(0.0)
212                },
213            )
214            .map_err(|e| e.to_string())?;
215        linker
216            .func_wrap(
217                "env",
218                "set_prop",
219                |mut caller: wasmtime::Caller<'_, ScriptState>, index: i32, value: f32| {
220                    if let Some(slot) = caller.data_mut().props.get_mut(index as usize) {
221                        *slot = value;
222                    }
223                },
224            )
225            .map_err(|e| e.to_string())?;
226
227        let mut store = wasmtime::Store::new(&engine, ScriptState::new());
228        store.limiter(|state| &mut state.limits);
229        store.epoch_deadline_trap();
230        store.set_epoch_deadline(EPOCH_DEADLINE_TICKS);
231
232        let instance = linker.instantiate(&mut store, &module).map_err(|e| e.to_string())?;
233        let update = instance.get_typed_func::<(), ()>(&mut store, "update").ok();
234        let start = instance.get_typed_func::<(), ()>(&mut store, "start").ok();
235        if update.is_none() && start.is_none() {
236            return Err("Script module exports neither `start` nor `update`".into());
237        }
238
239        self.engine = Some(engine);
240        self.store = Some(store);
241        self.update = update;
242        self.start = start;
243        Ok(())
244    }
245
246    /// Load and instantiate the module named by `self.module`, once.
247    fn ensure_loaded(&mut self) {
248        if self.engine.is_some() || self.error.is_some() {
249            return;
250        }
251        if self.module.trim().is_empty() {
252            self.error = Some("Script module path is empty".into());
253            return;
254        }
255        let bytes = match std::fs::read(&self.module) {
256            Ok(bytes) => bytes,
257            Err(e) => {
258                self.error = Some(format!("Script {}: {e}", self.module));
259                return;
260            }
261        };
262        if let Err(e) = self.build_runtime(&bytes) {
263            self.error = Some(e);
264        }
265    }
266
267    /// Run one exported hook (`start` / `update`) against the current frame.
268    fn run_hook(&mut self, ctx: &EcsMoveHost<'_, '_>, hook: Hook) {
269        if self.error.is_some() {
270            return;
271        }
272        let func = match hook {
273            Hook::Start => self.start.clone(),
274            Hook::Update => self.update.clone(),
275        };
276        let Some(func) = func else { return };
277        let Some(store) = self.store.as_mut() else { return };
278
279        let Some(mut transform) = ctx.world.get_mut::<Transform>(ctx.owner) else { return };
280        {
281            let state = store.data_mut();
282            state.dt = ctx.dt;
283            state.time = ctx.time;
284            state.translation = transform.translation;
285            state.rotation = transform.rotation;
286            state.scale = transform.scale;
287            state.props = self.props;
288        }
289
290        if let Err(e) = store.set_fuel(FUEL_PER_CALL) {
291            self.error = Some(format!("Script fuel: {e}"));
292            return;
293        }
294        if let Err(e) = func.call(&mut *store, ()) {
295            self.error = Some(e.to_string());
296            return;
297        }
298
299        let state = store.data();
300        transform.translation = state.translation;
301        transform.rotation = state.rotation;
302        transform.scale = state.scale;
303        self.props = state.props;
304    }
305}

311impl Script {
312    fn start_native(&mut self, ctx: &mut EcsMoveHost<'_, '_>) {
313        if self.started { return; }
314        self.started = true;
315        self.ensure_loaded();
316        self.run_hook(ctx, Hook::Start);
317    }
318
319    fn update_native(&mut self, ctx: &mut EcsMoveHost<'_, '_>) {
320        self.start_native(ctx);
321        self.ensure_loaded();
322        self.run_hook(ctx, Hook::Update);
323    }
324}

326pub(crate) fn register_systems(registry: &mut crate::ComponentRegistry) {
327    registry.register_ecs::<Script>();
328    registry.register_system_if("zestengine::script_start", Stage::Start,
329        |world, ctx| crate::ecs::run_ordered(world, ctx, Script::start_native),
330        |world| world.query::<&Script>().iter().any(|(_, script)| !script.started));
331    registry.register_system_for::<Script>("zestengine::script_update", Stage::Gameplay,
332        |world, ctx| crate::ecs::run_ordered(world, ctx, Script::update_native));
333}

385    #[test]
386    fn native_script_start_tracks_new_instances_and_survives_disable_reenable() {
387        use crate::ecs::Enabled;
388        const COUNTS: &str = r#"(module
389            (import "env" "get_prop" (func $get (param i32) (result f32)))
390            (import "env" "set_prop" (func $set (param i32 f32)))
391            (func (export "start") (call $set (i32.const 0) (f32.add (call $get (i32.const 0)) (f32.const 1))))
392            (func (export "update") (call $set (i32.const 1) (f32.add (call $get (i32.const 1)) (f32.const 1)))))"#;
393        let mut scene = SceneWorld::new();
394        let parent = scene.spawn("Hidden parent");
395        let entity = scene.spawn("Scripted");
396        scene.set_parent(entity, Some(parent), false).unwrap();
397        scene.get_mut::<Enabled>(parent).unwrap().0 = false;
398        scene.insert(entity, compiled(COUNTS)).unwrap();
399        run(&mut scene, Stage::Start);
400        assert!(!scene.get::<Script>(entity).unwrap().started);
401        scene.get_mut::<Enabled>(parent).unwrap().0 = true;
402        for _ in 0..2 { run(&mut scene, Stage::Start); run(&mut scene, Stage::Gameplay); }
403        assert_eq!(&scene.get::<Script>(entity).unwrap().props[..2], &[1.0, 2.0]);
404        let script = scene.remove::<Script>(entity).unwrap();
405        scene.insert_disabled(entity, script).unwrap();
406        run(&mut scene, Stage::Start); run(&mut scene, Stage::Gameplay);
407        let script = scene.remove_disabled::<Script>(entity).unwrap();
408        assert_eq!(&script.props[..2], &[1.0, 2.0]);
409        scene.insert(entity, script).unwrap();
410        run(&mut scene, Stage::Start); run(&mut scene, Stage::Gameplay);
411        assert_eq!(&scene.get::<Script>(entity).unwrap().props[..2], &[1.0, 3.0]);
412        scene.remove::<Script>(entity).unwrap();
413        scene.insert(entity, compiled(COUNTS)).unwrap();
414        run(&mut scene, Stage::Start); run(&mut scene, Stage::Gameplay);
415        assert_eq!(&scene.get::<Script>(entity).unwrap().props[..2], &[1.0, 1.0], "replacement instances get their own start");
416        let saved = crate::Component::save(&*scene.get::<Script>(entity).unwrap());
417        let restored: Script = ron::from_str(&saved).unwrap();
418        assert!(!restored.started && restored.store.is_none() && restored.error.is_none());
419    }
