17pub struct ClothColliderMotion {
18    pub start: ClothShapeWorld,
19    pub end: ClothShapeWorld,
20}
21impl ClothColliderMotion {
22    pub fn stationary(shape:ClothShapeWorld)->Self{Self{start:shape.clone(),end:shape}}
23    /// Prepare field motion once per collider/substep rather than per particle.
24    pub fn prepare(&self)->ClothColliderQuery<'_>{ClothColliderQuery{motion:self,field:self.level_set()}}
25    fn level_set(&self)->Option<crate::cloth_sdf::ClothSdfMotion>{
26        let ClothShapeWorld::LevelSet(end)=&self.end else{return None};
27        let start=if let ClothShapeWorld::LevelSet(start)=&self.start{start.clone()}else{end.clone()};
28        // A handedness flip cannot be swept through a nonsingular frame.
29        // Treat it as replacement geometry and still enforce its final body.
30        crate::cloth_sdf::ClothSdfMotion::new(start,end.clone()).or_else(|_|crate::cloth_sdf::ClothSdfMotion::new(end.clone(),end.clone())).ok()
31    }
32    pub fn at(&self,t:f32)->ClothShapeWorld {
33        use ClothShapeWorld::*;
34        match (self.start.clone(),self.end.clone()) {
35            (Sphere{c:a,r:ra},Sphere{c:b,r:rb})=>Sphere{c:a.lerp(b,t),r:ra+(rb-ra)*t},
36            (Capsule{a:a0,b:b0,ra:ra0,rb:rb0},Capsule{a:a1,b:b1,ra:ra1,rb:rb1})=>Capsule{a:a0.lerp(a1,t),b:b0.lerp(b1,t),ra:ra0+(ra1-ra0)*t,rb:rb0+(rb1-rb0)*t},
37            (Plane{p:a,n:na},Plane{p:b,n:nb})=>{
38                let na=na.normalize_or(Vec3::Y);let nb=nb.normalize_or(Vec3::Y);
39                let n=Quat::IDENTITY.slerp(Quat::from_rotation_arc(na,nb),t)*na;
40                Plane{p:a.lerp(b,t),n}
41            },
42            (LevelSet(a),LevelSet(b))=>{
43                let world=crate::cloth_sdf::ClothSdfMotion::new(a,b.clone()).map(|m|m.at(t)).unwrap_or(b.transform());
44                crate::cloth_sdf::ClothSdfShape::new(b.field.clone(),world).map(|s|LevelSet(std::sync::Arc::new(s))).unwrap_or(Disabled)
45            },
46            // A replacement is a new collider, not a sweep through unrelated geometry.
47            _=>self.end.clone(),
48        }
49    }
50    pub fn interval(&self,a:f32,b:f32)->Self{Self{start:self.at(a),end:self.at(b)}}
51    fn speed_bound(&self,p0:Vec3,p1:Vec3)->f32 {
52        use ClothShapeWorld::*;
53        let v=p1-p0;
54        match (self.start.clone(),self.end.clone()) {
55            (Sphere{c:a,r:ra},Sphere{c:b,r:rb})=>(v-(b-a)).length()+(rb-ra).abs(),
56            (Capsule{a:a0,b:b0,ra:ra0,rb:rb0},Capsule{a:a1,b:b1,ra:ra1,rb:rb1})=>
57                (v-(a1-a0)).length().max((v-(b1-b0)).length())+(ra1-ra0).abs().max((rb1-rb0).abs()),
58            (Plane{p:a,n:na},Plane{p:b,n:nb})=>{
59                let angle=na.normalize_or(Vec3::Y).dot(nb.normalize_or(Vec3::Y)).clamp(-1.0,1.0).acos();
60                (v-(b-a)).length()+angle*(p0-a).length().max((p1-b).length())
61            },
62            _=>v.length(),
63        }
64    }
65}

68impl ClothColliderQuery<'_>{
69    pub fn end(&self)->&ClothShapeWorld{&self.motion.end}
70    pub fn resolve(&self,p0:Vec3,p1:Vec3,radius:f32)->Option<(Vec3,Vec3,Vec3)>{
71        if let Some(field)=&self.field{field.resolve(p0,p1,radius)}else{resolve_cloth_collider(p0,p1,radius,self.motion.clone())}
72    }

75    }

93/// Returns the first proximity along a continuous path. Distance is Lipschitz
94/// with `speed`; all skipped intervals are therefore proven separated.
95fn advance(mut distance:impl FnMut(f32)->f32,speed:f32,margin:f32)->Option<(f32,bool)> {
96    let tolerance=(margin*1e-4).max(1e-7);
97    let mut t=0.0;
98    for _ in 0..192 {
99        let gap=distance(t)-margin;
100        if !gap.is_finite(){return None;}
101        if gap<=tolerance{return Some((t,false));}
102        if speed<=1e-12{return None;}
103        let safe=gap/speed;
104        if safe>1.0-t{return None;}
105        let next=t+safe*0.9;
106        if next<=t{return Some((t,true));}
107        t=next;
108    }
109    Some((t,true))
110}

112/// Signed distance to the union of linearly tapered spheres. Unlike projecting
113/// onto the axis, this finds the true closest taper parameter.
114fn shape_sample(p:Vec3,shape:ClothShapeWorld)->(f32,Vec3,f32) {
115    match shape {
116        ClothShapeWorld::Disabled=>(f32::INFINITY,Vec3::Y,0.0),
117        ClothShapeWorld::LevelSet(sdf)=>{let(d,n)=sdf.sample(p);(d,n,0.0)},
118        ClothShapeWorld::Sphere{c,r}=>{let d=p-c;(d.length()-r.max(0.0),d.normalize_or(Vec3::Y),0.0)},
119        ClothShapeWorld::Capsule{a,b,ra,rb}=>{
120            let e=b-a;let length=e.length();let dr=rb-ra;
121            let t=if length<1e-8 {if rb>ra{1.0}else{0.0}}
122                else if dr.abs()>=length{if dr>0.0{1.0}else{0.0}}
123                else{let axis=e/length;let x=(p-a).dot(axis);let h=((p-a)-axis*x).length();((x+dr*h/(length*length-dr*dr).sqrt())/length).clamp(0.0,1.0)};
124            let d=p-a.lerp(b,t);(d.length()-(ra+dr*t).max(0.0),d.normalize_or(e.normalize_or(Vec3::Y).any_orthonormal_vector()),t)
125        },
126        ClothShapeWorld::Plane{p:o,n}=>{let n=n.normalize_or(Vec3::Y);((p-o).dot(n),n,0.0)},
127    }
128}
129pub fn cloth_shape_distance(p:Vec3,shape:ClothShapeWorld)->(f32,Vec3){let(d,n,_)=shape_sample(p,shape);(d,n)}

131pub fn sweep_cloth_collider(p0:Vec3,p1:Vec3,radius:f32,motion:ClothColliderMotion)->Option<ClothSweep>{
132    if !p0.is_finite()||!p1.is_finite(){return None;}
133    if let Some(sdf)=motion.level_set(){return sdf.first_contact(p0,p1,radius);}
134    let(t,conservative)=advance(|t|shape_sample(p0.lerp(p1,t),motion.at(t)).0,motion.speed_bound(p0,p1),radius.max(0.0))?;
135    Some(ClothSweep{time:t,normal:shape_sample(p0.lerp(p1,t),motion.at(t)).1,conservative})
136}

138/// Position response against a moving primitive, and its contact displacement
139/// over the whole interval (used for friction in the collider's moving frame).
140pub fn resolve_cloth_collider(p0:Vec3,p1:Vec3,radius:f32,motion:ClothColliderMotion)->Option<(Vec3,Vec3,Vec3)> {
141    if let Some(sdf)=motion.level_set(){return sdf.resolve(p0,p1,radius);}
142    let hit=sweep_cloth_collider(p0,p1,radius,motion.clone())?;
143    respond_to_collider(p0,p1,radius,motion,hit)
144}

146fn respond_to_collider(p0:Vec3,p1:Vec3,radius:f32,motion:ClothColliderMotion,hit:ClothSweep)->Option<(Vec3,Vec3,Vec3)> {
147    let at=p0.lerp(p1,hit.time);let (_,normal,u)=shape_sample(at,motion.at(hit.time));
148    let surface=|shape| match shape {
149        ClothShapeWorld::Disabled | ClothShapeWorld::LevelSet(_) => at,
150        ClothShapeWorld::Sphere{c,r}=>c+normal*(r+radius),
151        ClothShapeWorld::Capsule{a,b,ra,rb}=>a.lerp(b,u)+normal*(ra+(rb-ra)*u+radius),
152        ClothShapeWorld::Plane{p,n}=>at-n.normalize_or(Vec3::Y)*((at-p).dot(n.normalize_or(Vec3::Y))-radius),
153    };
154    let support=surface(motion.end.clone());let displacement=surface(motion.end.clone())-surface(motion.start.clone());
155    let mut out=p1;
156    let depth=(out-support).dot(normal);
157    if depth<0.0{out-=normal*depth;}
158    let (distance,end_normal)=cloth_shape_distance(out,motion.end);
159    if distance<radius{out+=end_normal*(radius-distance);}
160    if out.distance_squared(p1)<1e-18 && (p1-p0-displacement).dot(normal)>=0.0{return None;}
161    Some((out,normal,displacement))
162}

164/// Swept contact of the whole segment, including its interior. The radius may
165/// vary linearly along a bone chain. Convex distance along the segment is
166/// minimized by its derivative; no fixed spatial samples can miss a thin body.
167pub fn resolve_cloth_segment(start:[Vec3;2],end:[Vec3;2],radii:[f32;2],motion:ClothColliderMotion)->Option<(f32,Vec3,Vec3,Vec3)> {
168    if start.iter().chain(&end).any(|p|!p.is_finite()){return None;}
169    if let Some(sdf)=motion.level_set(){return sdf.resolve_segment(start,end,radii);}
170    let radii=radii.map(|r|r.max(0.0));
171    let sample=|time:f32|{
172        let a=start[0].lerp(end[0],time);let b=start[1].lerp(end[1],time);let edge=b-a;let dr=radii[1]-radii[0];let shape=motion.at(time);
173        let derivative=|u:f32|edge.dot(shape_sample(a+edge*u,shape.clone()).1)-dr;
174        let u=if derivative(0.0)>=0.0{0.0}else if derivative(1.0)<=0.0{1.0}else{
175            let (mut lo,mut hi)=(0.0,1.0);for _ in 0..32{let mid=(lo+hi)*0.5;if derivative(mid)<0.0{lo=mid}else{hi=mid}}
176            (lo+hi)*0.5
177        };
178        let (distance,normal,_)=shape_sample(a+edge*u,shape.clone());
179        (distance-radii[0]-dr*u,normal,u)
180    };
181    let speed=motion.speed_bound(start[0],end[0]).max(motion.speed_bound(start[1],end[1]));
182    let (time,conservative)=advance(|t|sample(t).0,speed,0.0)?;
183    let (_,normal,u)=sample(time);let p0=start[0].lerp(start[1],u);let p1=end[0].lerp(end[1],u);let radius=radii[0]+(radii[1]-radii[0])*u;
184    let (out,normal,carried)=respond_to_collider(p0,p1,radius,motion,ClothSweep{time,normal,conservative})?;
185    Some((u,out-p1,normal,carried))
186}

188/// Closest triangle point, including edges and degenerate triangles.
189pub(crate) fn closest_triangle(p:Vec3,t:[Vec3;3])->(Vec3,[f32;3]) {
190    let [a,b,c]=t;let ab=b-a;let ac=c-a;
191    let denom=ab.length_squared()*ac.length_squared()-ab.dot(ac).powi(2);
192    if denom>1e-20 {
193        let ap=p-a;let u=(ap.dot(ab)*ac.length_squared()-ap.dot(ac)*ab.dot(ac))/denom;
194        let v=(ap.dot(ac)*ab.length_squared()-ap.dot(ab)*ab.dot(ac))/denom;
195        if u>=0.0&&v>=0.0&&u+v<=1.0{return(a+ab*u+ac*v,[1.0-u-v,u,v]);}
196    }
197    let mut best=(a,[1.0,0.0,0.0]);let mut distance=p.distance_squared(a);
198    for (i,j) in [(0,1),(1,2),(2,0)]{let e=t[j]-t[i];let u=if e.length_squared()>1e-20{((p-t[i]).dot(e)/e.length_squared()).clamp(0.0,1.0)}else{0.0};let q=t[i]+e*u;let d=p.distance_squared(q);
199        if d<distance{distance=d;let mut w=[0.0;3];w[i]=1.0-u;w[j]=u;best=(q,w);}
200    }best
201}
202fn closest_edges(a:[Vec3;2],b:[Vec3;2])->(Vec3,Vec3,f32,f32){
203    let u=a[1]-a[0];let v=b[1]-b[0];let r=a[0]-b[0];let aa=u.dot(u);let bb=v.dot(v);let uv=u.dot(v);let ur=u.dot(r);let vr=v.dot(r);
204    let mut s;let mut t;
205    if aa<1e-20{s=0.0;t=if bb<1e-20{0.0}else{(vr/bb).clamp(0.0,1.0)};}
206    else if bb<1e-20{s=(-ur/aa).clamp(0.0,1.0);t=0.0;}
207    else{let d=aa*bb-uv*uv;s=if d>1e-20{((uv*vr-ur*bb)/d).clamp(0.0,1.0)}else{0.0};t=(uv*s+vr)/bb;
208        if t<0.0{t=0.0;s=(-ur/aa).clamp(0.0,1.0);}else if t>1.0{t=1.0;s=((uv-ur)/aa).clamp(0.0,1.0);}}
209    (a[0]+u*s,b[0]+v*t,s,t)
210}

212pub fn sweep_cloth_vertex_face(start:[Vec3;4],end:[Vec3;4],thickness:f32)->Option<ClothFeatureContact>{
213    let velocity: [Vec3;4]=std::array::from_fn(|i|end[i]-start[i]);
214    let speed=(1..4).map(|i|(velocity[i]-velocity[0]).length()).fold(0.0,f32::max);
215    let sample=|time|{let p: [Vec3;4]=std::array::from_fn(|i|start[i].lerp(end[i],time));let(q,w)=closest_triangle(p[0],[p[1],p[2],p[3]]);(p,p[0]-q,w)};
216    let (time,conservative)=advance(|t|sample(t).1.length(),speed,thickness.max(1e-7))?;
217    let (p,d,w)=sample(time);let normal=d.try_normalize().unwrap_or_else(||{
218        let n=(p[2]-p[1]).cross(p[3]-p[1]).normalize_or(Vec3::Y);
219        if (start[0]-start[1]).dot(n)<0.0{-n}else{n}
220    });
221    Some(ClothFeatureContact{sweep:ClothSweep{time,normal,conservative},coefficients:[1.0,-w[0],-w[1],-w[2]]})
222}
223pub fn sweep_cloth_edges(start:[Vec3;4],end:[Vec3;4],thickness:f32)->Option<ClothFeatureContact>{
224    let velocity: [Vec3;4]=std::array::from_fn(|i|end[i]-start[i]);
225    let speed=(0..2).flat_map(|i|(2..4).map(move |j|(velocity[i]-velocity[j]).length())).fold(0.0,f32::max);
226    let sample=|time|{let p: [Vec3;4]=std::array::from_fn(|i|start[i].lerp(end[i],time));let(a,b,s,t)=closest_edges([p[0],p[1]],[p[2],p[3]]);(p,a-b,s,t)};
227    let(time,conservative)=advance(|t|sample(t).1.length(),speed,thickness.max(1e-7))?;
228    let(p,d,s,t)=sample(time);let normal=d.try_normalize().unwrap_or_else(||{
229        let n=(p[1]-p[0]).cross(p[3]-p[2]).normalize_or((p[1]-p[0]).normalize_or(Vec3::X).any_orthonormal_vector());
230        if (start[0].lerp(start[1],s)-start[2].lerp(start[3],t)).dot(n)<0.0{-n}else{n}
231    });
232    Some(ClothFeatureContact{sweep:ClothSweep{time,normal,conservative},coefficients:[1.0-s,s,-(1.0-t),-t]})
233}

235/// Mass-weighted contact projection. Fixed particles never move. The contact
236/// normal is retained from impact so an end-of-step crossing cannot flip sides.
237pub fn project_cloth_contact(positions:&mut[Vec3],inverse_mass:&[f32],ids:[usize;4],contact:ClothFeatureContact,thickness:f32)->f32 {
238    let c=contact.coefficients;let n=contact.sweep.normal;
239    let gap=(0..4).map(|k|positions[ids[k]]*c[k]).sum::<Vec3>().dot(n);
240    let denom=(0..4).map(|k|inverse_mass[ids[k]].max(0.0)*c[k]*c[k]).sum::<f32>();
241    if denom<1e-20||gap>=thickness{return 0.0;}
242    let correction=(thickness-gap)/denom;
243    for k in 0..4 {positions[ids[k]]+=n*(correction*inverse_mass[ids[k]].max(0.0)*c[k]);}
244    thickness-gap
245}

250#[derive(Clone, Debug, Default)]
251pub struct ClothContactTopology {
252    pub triangles:Vec<[usize;3]>,
253    pub edges:Vec<[usize;2]>,
254    adjacent:std::collections::HashSet<(usize,usize)>,
255    excluded:std::collections::HashSet<(usize,usize)>,
256}
257impl ClothContactTopology {
258    pub fn new(triangles:Vec<[usize;3]>)->Self {
259        let mut edges=std::collections::BTreeSet::new();
260        for t in &triangles {for k in 0..3 {let(a,b)=(t[k],t[(k+1)%3]);if a!=b{edges.insert([a.min(b),a.max(b)]);}}}
261        let adjacent=edges.iter().map(|e|(e[0],e[1])).collect();
262        Self{triangles,edges:edges.into_iter().collect(),adjacent,excluded:Default::default()}
263    }
264    pub fn exclude_pair(&mut self,a:usize,b:usize){let pair=(a.min(b),a.max(b));self.adjacent.insert(pair);self.excluded.insert(pair);}
265    fn neighbours(&self,a:usize,b:usize)->bool{a==b||self.adjacent.contains(&(a.min(b),a.max(b)))}
266    /// Swept broad phase covers the complete interval. The same routine handles
267    /// a combined mesh of several garments: no adjacency crosses their islands.
268    pub fn solve(&self,positions:&mut[Vec3],previous:&[Vec3],inverse_mass:&[f32],thickness:f32,continuous:bool,untangle:bool)->ClothContactReport {
269        self.solve_filtered(positions,previous,inverse_mass,thickness,continuous,untangle,None,None,None)
270    }
271    /// Self-contact on a visible mesh whose vertices share simulation drivers.
272    /// Topological neighbours still use the original surface connectivity.
273    pub fn solve_mapped(&self,positions:&mut[Vec3],previous:&[Vec3],inverse_mass:&[f32],thickness:f32,drivers:&ClothContactMapping,continuous:bool,untangle:bool)->ClothContactReport{
274        if !drivers.valid_weights(inverse_mass){return ClothContactReport::default();}
275        if drivers.trivial(){return self.solve(positions,previous,inverse_mass,thickness,continuous,untangle);}
276        self.solve_filtered(positions,previous,inverse_mass,thickness,continuous,untangle,None,None,Some(drivers))
277    }
278    /// Solve only between distinct garments. Radii are half the desired cloth
279    /// thickness; pair gaps add the two garments' radii rather than inflating
280    /// every garment to the thickest member of the group.
281    pub fn solve_between(&self,positions:&mut[Vec3],previous:&[Vec3],inverse_mass:&[f32],owners:&[usize],radii:&[f32],continuous:bool,untangle:bool)->ClothContactReport {
282        if owners.len()!=positions.len()||radii.len()!=positions.len(){return ClothContactReport::default();}
283        let thickness=radii.iter().copied().fold(0.0,f32::max)*2.0;
284        self.solve_filtered(positions,previous,inverse_mass,thickness,continuous,untangle,Some(owners),Some(radii),None)
285    }
286    /// Contact the visible surface while maintaining its simulation drivers.
287    /// Each vertex carries its driver's inverse mass, not a separate mass.
288    pub fn solve_between_mapped(&self,positions:&mut[Vec3],previous:&[Vec3],inverse_mass:&[f32],owners:&[usize],radii:&[f32],drivers:&ClothContactMapping,continuous:bool,untangle:bool)->ClothContactReport{
289        if !drivers.valid(inverse_mass,owners)||radii.len()!=positions.len(){return ClothContactReport::default();}
290        if drivers.trivial(){return self.solve_between(positions,previous,inverse_mass,owners,radii,continuous,untangle);}
291        let thickness=radii.iter().copied().fold(0.0,f32::max)*2.0;
292        self.solve_filtered(positions,previous,inverse_mass,thickness,continuous,untangle,Some(owners),Some(radii),Some(drivers))
293    }
294    #[allow(clippy::too_many_arguments)]
295    fn solve_filtered(&self,positions:&mut[Vec3],previous:&[Vec3],inverse_mass:&[f32],thickness:f32,continuous:bool,untangle:bool,owners:Option<&[usize]>,radii:Option<&[f32]>,mapping:Option<&ClothContactMapping>)->ClothContactReport {
296        let mut report=ClothContactReport::default();
297        if thickness<=0.0||!thickness.is_finite()||positions.len()!=previous.len()||positions.len()!=inverse_mass.len()
298            || self.triangles.iter().flatten().chain(self.edges.iter().flatten()).any(|&i|i>=positions.len())
299            || positions.iter().chain(previous).any(|p|!p.is_finite())||inverse_mass.iter().any(|w|!w.is_finite())
300            || radii.is_some_and(|r|r.iter().any(|r|!r.is_finite()||*r<0.0)){return report;}
301        let start=if continuous{previous.to_vec()}else{positions.to_vec()};
302        let end=positions.to_vec();
303        let point_bounds:Vec<_>=start.iter().zip(&end).map(|(a,b)|Bounds::points(&[*a,*b]).expanded(thickness)).collect();
304        let face_bounds:Vec<_>=self.triangles.iter().map(|t|t.iter().map(|&i|point_bounds[i]).reduce(Bounds::union).unwrap()).collect();
305        let face_tree=Tree::new(&face_bounds);
306        let mut candidates=Vec::new();
307        for i in 0..positions.len(){candidates.clear();face_tree.query(point_bounds[i],&mut candidates);
308            for &ti in &candidates {let t=self.triangles[ti];if owners.is_some_and(|o|o[i]==o[t[0]])||t.iter().any(|&j|self.neighbours(i,j)){continue;}
309                let ids=[i,t[0],t[1],t[2]];if ids.iter().all(|&j|inverse_mass[j]<=0.0){continue;}
310                let gap=radii.map_or(thickness,|r|r[i]+t.iter().map(|&j|r[j]).fold(0.0,f32::max));
311                if let Some(contact)=sweep_cloth_vertex_face(ids.map(|i|start[i]),ids.map(|i|end[i]),gap){
312                    if match mapping{Some(m)=>m.project(positions,inverse_mass,ids,contact,gap),None=>project_cloth_contact(positions,inverse_mass,ids,contact,gap)}>0.0{report.vertex_face+=1;report.conservative+=usize::from(contact.sweep.conservative);}
313                }
314            }
315        }
316        let edge_bounds:Vec<_>=self.edges.iter().map(|e|point_bounds[e[0]].union(point_bounds[e[1]])).collect();
317        let edge_tree=Tree::new(&edge_bounds);
318        for (ei,&a) in self.edges.iter().enumerate(){candidates.clear();edge_tree.query(edge_bounds[ei],&mut candidates);
319            for &ej in &candidates{if ej<=ei{continue;}let b=self.edges[ej];if owners.is_some_and(|o|o[a[0]]==o[b[0]])||a.iter().any(|&i|b.iter().any(|&j|self.neighbours(i,j))){continue;}
320                let ids=[a[0],a[1],b[0],b[1]];if ids.iter().all(|&j|inverse_mass[j]<=0.0){continue;}
321                let gap=radii.map_or(thickness,|r|r[a[0]].max(r[a[1]])+r[b[0]].max(r[b[1]]));
322                if let Some(contact)=sweep_cloth_edges(ids.map(|i|start[i]),ids.map(|i|end[i]),gap){
323                    if match mapping{Some(m)=>m.project(positions,inverse_mass,ids,contact,gap),None=>project_cloth_contact(positions,inverse_mass,ids,contact,gap)}>0.0{report.edge_edge+=1;report.conservative+=usize::from(contact.sweep.conservative);}
324                }
325            }
326        }
327        let before=mapping.map(|_|positions.to_vec());
328        if untangle {report.untangled=self.untangle_filtered(positions,inverse_mass,thickness,owners,radii);}
329        untangling::project_fixed(self,positions,inverse_mass,thickness,owners,radii);
330        if let Some(m)=mapping{m.cohere(before.as_ref().unwrap(),positions);}
331        let coherent_previous=mapping.map(|m|{let mut origin=previous.to_vec();m.cohere(positions,&mut origin);origin});
332        if continuous&&untangling::preserve_valid_pose(self,positions,coherent_previous.as_deref().unwrap_or(previous),inverse_mass,owners){report.conservative+=1;}
333        report
334    }
335    /// Resolve intersections already present at the start (for example an
336    /// imported garment intersecting itself). Partition along intersected edges
337    /// and choose a coherent movable side; fixed particles retain zero weight.
338    pub fn untangle(&self,p:&mut[Vec3],weights:&[f32],thickness:f32)->usize {
339        self.untangle_filtered(p,weights,thickness,None,None)
340    }
341    fn untangle_filtered(&self,p:&mut[Vec3],weights:&[f32],thickness:f32,owners:Option<&[usize]>,radii:Option<&[f32]>)->usize {
342        untangling::recover(self,p,weights,thickness,owners,radii)
343    }
344}

346#[derive(Clone, Copy, Debug)]
347struct Bounds{lo:Vec3,hi:Vec3}
348impl Bounds {
349    fn points(p:&[Vec3])->Self{Self{lo:p.iter().copied().fold(Vec3::splat(f32::INFINITY),Vec3::min),hi:p.iter().copied().fold(Vec3::splat(f32::NEG_INFINITY),Vec3::max)}}
350    fn union(self,b:Self)->Self{Self{lo:self.lo.min(b.lo),hi:self.hi.max(b.hi)}}
351    fn expanded(self,r:f32)->Self{Self{lo:self.lo-Vec3::splat(r),hi:self.hi+Vec3::splat(r)}}
352    fn overlaps(self,b:Self)->bool{self.lo.cmple(b.hi).all()&&b.lo.cmple(self.hi).all()}
353}
354struct Node{bounds:Bounds,children:Option<[usize;2]>,items:Vec<usize>}
355struct Tree{nodes:Vec<Node>,items:Vec<Bounds>}
356impl Tree {
357    fn new(bounds:&[Bounds])->Self {let mut tree=Self{nodes:Vec::new(),items:bounds.to_vec()};if !bounds.is_empty(){tree.build(bounds,(0..bounds.len()).collect());}tree}
358    fn build(&mut self,bounds:&[Bounds],mut items:Vec<usize>)->usize{
359        let b=items.iter().map(|&i|bounds[i]).reduce(Bounds::union).unwrap();let index=self.nodes.len();
360        self.nodes.push(Node{bounds:b,children:None,items:Vec::new()});
361        if items.len()<=8{self.nodes[index].items=items;}else{
362            let extent=b.hi-b.lo;let axis=if extent.x>=extent.y&&extent.x>=extent.z{0}else if extent.y>=extent.z{1}else{2};
363            items.sort_unstable_by(|&i,&j|(bounds[i].lo[axis]+bounds[i].hi[axis]).total_cmp(&(bounds[j].lo[axis]+bounds[j].hi[axis])).then(i.cmp(&j)));
364            let right=items.split_off(items.len()/2);let a=self.build(bounds,items);let b=self.build(bounds,right);self.nodes[index].children=Some([a,b]);
365        }index
366    }
367    fn query(&self,bounds:Bounds,out:&mut Vec<usize>){if !self.nodes.is_empty(){self.query_node(0,bounds,out);}}
368    fn query_node(&self,index:usize,bounds:Bounds,out:&mut Vec<usize>){
369        let node=&self.nodes[index];if !node.bounds.overlaps(bounds){return;}
370        if let Some([a,b])=node.children{self.query_node(a,bounds,out);self.query_node(b,bounds,out);}
371        else{out.extend(node.items.iter().copied().filter(|&i|self.items[i].overlaps(bounds)));}
372    }
373}
