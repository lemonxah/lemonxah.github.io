1//! Avatar parameters and expression-menu data.
2//!
3//! [`AvatarParameters`] declares the typed values an avatar exposes (Bool / Int
4//! / Float, with a default, a saved flag and synced-or-local) and owns the
5//! runtime [`AvatarParamStore`]. The store is the avatar's one parameter
6//! namespace: engine-filled built-ins ([`BuiltinParam`]) first, then the
7//! declared ones. It pushes only the values that changed since the last flush
8//! into the Animator's parameter map, so a quiet avatar costs nothing per
9//! frame, and reads back what state-behaviour drivers wrote.
10//! [`AvatarMenuItem`] is the menu data (the hierarchy is the menu); the radial
11//! menu UI and the network sync are built on top of it elsewhere.
12//!
13//! Sync budget: a Bool costs 1 bit, an Int or Float 8; only `Synced`
14//! parameters count (decision D6: 256 bits per avatar by default, set per
15//! project through `ContentLimits::max_param_bits` and per avatar through
16//! `AvatarParameters::budget_bits`).
17

25/// Synced-bit budget when nothing says otherwise.
26pub const DEFAULT_PARAM_BUDGET_BITS: u32 = 256;
27/// Controls one menu page holds.
28pub const MAX_MENU_CONTROLS: usize = 8;
29/// Bits a Bool costs on the wire.
30pub const BOOL_BITS: u32 = 1;
31/// Bits an Int or Float costs on the wire.
32pub const NUMBER_BITS: u32 = 8;

34/// What a declared parameter can hold.
35#[derive(Serialize, Deserialize, Clone, Copy, Debug, PartialEq, Eq, Default)]
36pub enum AvatarParamKind {
37    /// On / off (stored as 0 / 1).
38    Bool,
39    /// Whole number.
40    Int,
41    /// Continuous value.
42    #[default]
43    Float,
44}
45
46impl AvatarParamKind {
47    /// Every kind, in declaration order.
48    pub const ALL: [AvatarParamKind; 3] = [Self::Bool, Self::Int, Self::Float];
49    /// Inspector label of this kind.
50    pub fn label(self) -> &'static str {
51        match self {
52            Self::Bool => "Bool",
53            Self::Int => "Int",
54            Self::Float => "Float",
55        }
56    }
57    /// Bits this kind occupies in the synced payload.
58    pub fn bits(self) -> u32 {
59        match self {
60            Self::Bool => BOOL_BITS,
61            Self::Int | Self::Float => NUMBER_BITS,
62        }
63    }
64    /// Round / threshold `v` to what this kind can hold (no range clamp).
65    pub fn coerce_local(self, v: f32) -> f32 {
66        match self {
67            Self::Bool => (v != 0.0) as u8 as f32,
68            Self::Int => v.round(),
69            Self::Float => v,
70        }
71    }
72}

96/// How a parameter declared below the avatar root is named in the finished
97/// avatar (see `avatar_build`'s scoping pass). Parameters declared on the root
98/// are always global.
99#[derive(Serialize, Deserialize, Clone, Debug, PartialEq, Eq, Default)]
100pub enum AvatarParamScope {

102    /// parameter the avatar already has).

105    /// Private to the declaring object's subtree: renamed

107    Internal,
108    /// Renamed to this name everywhere in the subtree.
109    RenameTo(String),
110}

112/// One declared avatar parameter.
113#[derive(Serialize, Deserialize, Clone, Debug, PartialEq)]
114#[serde(default)]
115pub struct AvatarParam {
116    /// Parameter name (also what controllers and menus reference).
117    pub name: String,
118    /// What the parameter holds.
119    pub kind: AvatarParamKind,

148impl AvatarParam {
149    /// Declare a parameter of this kind with default values.
150    pub fn new(name: &str, kind: AvatarParamKind) -> Self {
151        Self { name: name.to_owned(), kind, ..Default::default() }
152    }
153    /// True when the value replicates to other players.
154    pub fn is_synced(&self) -> bool {
155        self.sync == AvatarParamSync::Synced
156    }
157    /// Bits this parameter costs against the budget (0 when local only).
158    pub fn bit_cost(&self) -> u32 {
159        if self.is_synced() { self.kind.bits() } else { 0 }
160    }
161    /// The value a write of `v` leaves: Bool 0/1, Int rounded, and synced
162    /// values held to the wire range (Int 0..=255, Float -1..=1).
163    pub fn coerce(&self, v: f32) -> f32 {
164        let v = self.kind.coerce_local(v);
165        match (self.kind, self.is_synced()) {
166            (AvatarParamKind::Int, true) => v.clamp(0.0, 255.0),
167            (AvatarParamKind::Float, true) => v.clamp(-1.0, 1.0),
168            _ => v,
169        }
170    }
171}

173/// What a remote copy sees of a synced float: clamped to -1..1 and stored as a
174/// signed byte, so 0.5 arrives as 0.504.
175pub fn quantise_float(v: f32) -> f32 {
176    (v.clamp(-1.0, 1.0) * 127.0).round() / 127.0
177}

345/// The wire cost of `params` in bits: synced parameters only, each name once,
346/// built-ins never.
347pub fn synced_bits(params: &[AvatarParam]) -> u32 {
348    let mut seen: HashSet<&str> = HashSet::with_capacity(params.len());
349    params
350        .iter()
351        .filter(|p| !p.name.is_empty() && BuiltinParam::from_name(&p.name).is_none() && seen.insert(p.name.as_str()))
352        .map(AvatarParam::bit_cost)
353        .sum()
354}

356/// How full a budget is, for the meter (amber past half, red past the limit).
357#[derive(Clone, Copy, Debug, PartialEq, Eq)]
358pub enum BudgetLevel {
359    /// Within budget.
360    Ok,
361    /// Past half the budget.
362    Warn,
363    /// Past the budget.
364    Over,
365}
366
367/// How much of a synced-bit budget a set of parameters uses.
368#[derive(Clone, Copy, Debug, PartialEq, Eq)]
369pub struct ParamBudget {
370    /// Bits used.
371    pub used: u32,
372    /// Bit limit.
373    pub max: u32,
374}
375
376impl ParamBudget {
377    /// Measure `params` against a bit limit.
378    pub fn of(params: &[AvatarParam], max: u32) -> Self {
379        Self { used: synced_bits(params), max }
380    }
381    /// Bits left before the limit.
382    pub fn remaining(&self) -> u32 {
383        self.max.saturating_sub(self.used)
384    }
385    /// True when the used bits exceed the limit.
386    pub fn is_over(&self) -> bool {
387        self.used > self.max
388    }
389    /// 0..1+ (above 1 = over budget).
390    pub fn fraction(&self) -> f32 {
391        if self.max == 0 { if self.used == 0 { 0.0 } else { f32::INFINITY } } else { self.used as f32 / self.max as f32 }
392    }
393    /// Ok / Warn / Over, for the budget meter.
394    pub fn level(&self) -> BudgetLevel {
395        if self.is_over() {
396            BudgetLevel::Over
397        } else if self.used * 2 > self.max {
398            BudgetLevel::Warn
399        } else {
400            BudgetLevel::Ok
401        }
402    }
403    /// "cost/MAX Synced Bits".
404    pub fn label(&self) -> String {
405        format!("{}/{} synced bits", self.used, self.max)
406    }
407    /// Whether adding `extra` still fits.
408    pub fn fits(&self, extra: &AvatarParam) -> bool {
409        self.used + extra.bit_cost() <= self.max
410    }
411}

413/// Values worth keeping between sessions: the saved parameters, by name.
414#[derive(Serialize, Deserialize, Clone, Debug, Default, PartialEq)]
415#[serde(default)]
416pub struct SavedParams {
417    /// Saved values by parameter name.
418    pub values: BTreeMap<String, f32>,
419}
420
421impl SavedParams {
422    /// Serialize to RON.
423    pub fn to_ron(&self) -> String {
424        ron::to_string(self).unwrap_or_default()
425    }
426    /// Parse from RON.
427    pub fn from_ron(text: &str) -> Result<Self, String> {
428        ron::from_str(text).map_err(|e| e.to_string())
429    }
430    /// `<dir>/avatar_params/<package hash>.ron` (the hash is reduced to
431    /// file-name-safe characters).
432    pub fn path_for(dir: &Path, package_hash: &str) -> PathBuf {
433        let safe: String = package_hash.chars().filter(|c| c.is_ascii_alphanumeric() || *c == '-' || *c == '_').collect();
434        dir.join("avatar_params").join(format!("{safe}.ron"))
435    }
436    /// Write to `path` (creating parent directories).
437    pub fn save(&self, path: &Path) -> std::io::Result<()> {
438        if let Some(parent) = path.parent() {
439            std::fs::create_dir_all(parent)?;
440        }
441        std::fs::write(path, self.to_ron())
442    }
443    /// Missing or unreadable files give an empty set (first run).
444    pub fn load(path: &Path) -> Self {
445        std::fs::read_to_string(path).ok().and_then(|t| Self::from_ron(&t).ok()).unwrap_or_default()
446    }
447}

449/// An avatar's live parameter values. Slots are resolved once (`index_of`);
450/// reads and writes by slot are array accesses.
451#[derive(Clone, Debug)]
452pub struct AvatarParamStore {
453    names: Vec<String>,
454    /// Declared parameters (`None` for built-ins).
455    decl: Vec<Option<AvatarParam>>,
456    kinds: Vec<AvatarParamKind>,
457    values: Vec<f32>,
458    /// One bit per slot: changed since the last flush to the Animator.
459    dirty: Vec<u64>,
460    index: HashMap<String, usize>,
461}
462
463impl Default for AvatarParamStore {
464    fn default() -> Self {
465        Self::new(&[])
466    }
467}
468
469impl AvatarParamStore {
470    /// Built-ins, then `params` (duplicates, empty names and built-in names
471    /// are skipped), every value at its default.
472    pub fn new(params: &[AvatarParam]) -> Self {
473        let n = BuiltinParam::ALL.len();
474        let mut s = Self {
475            names: Vec::with_capacity(n + params.len()),
476            decl: Vec::with_capacity(n + params.len()),
477            kinds: Vec::with_capacity(n + params.len()),
478            values: Vec::with_capacity(n + params.len()),
479            dirty: Vec::new(),
480            index: HashMap::with_capacity(n + params.len()),
481        };
482        for b in BuiltinParam::ALL {
483            s.index.insert(b.name().to_owned(), s.names.len());
484            s.names.push(b.name().to_owned());
485            s.decl.push(None);
486            s.kinds.push(b.kind());
487            s.values.push(b.default_value());
488        }
489        for p in params {
490            if p.name.is_empty() || s.index.contains_key(&p.name) {
491                continue;
492            }
493            s.index.insert(p.name.clone(), s.names.len());
494            s.names.push(p.name.clone());
495            s.kinds.push(p.kind);
496            s.values.push(p.coerce(p.default));
497            s.decl.push(Some(p.clone()));
498        }
499        // Everything is new to the Animator: the first flush writes it all.
500        s.dirty = vec![u64::MAX; s.names.len().div_ceil(64)];
501        s
502    }
503
504    /// An allocation-free stand-in with no slots, for swapping a store out of
505    /// a component for a moment. Not usable until the real one is put back.
506    pub fn placeholder() -> Self {
507        Self { names: Vec::new(), decl: Vec::new(), kinds: Vec::new(), values: Vec::new(), dirty: Vec::new(), index: HashMap::new() }
508    }
509
510    /// Number of slots (built-ins included).
511    pub fn len(&self) -> usize {
512        self.names.len()
513    }
514    /// True when the store has no slots.
515    pub fn is_empty(&self) -> bool {
516        self.names.is_empty()
517    }
518    /// The slot of the named parameter.
519    pub fn index_of(&self, name: &str) -> Option<usize> {
520        self.index.get(name).copied()
521    }
522    /// The parameter name at `slot`.
523    pub fn name(&self, slot: usize) -> &str {
524        &self.names[slot]
525    }
526    /// The kind at `slot`.
527    pub fn kind(&self, slot: usize) -> AvatarParamKind {
528        self.kinds[slot]
529    }
530    /// The value at `slot`.
531    pub fn get(&self, slot: usize) -> f32 {
532        self.values[slot]
533    }
534    /// The built-in's current value.
535    pub fn get_builtin(&self, b: BuiltinParam) -> f32 {
536        self.values[b as usize]
537    }
538    /// True when `slot` is a built-in (not a declared parameter).
539    pub fn is_builtin(&self, slot: usize) -> bool {
540        slot < BuiltinParam::ALL.len()
541    }
542    /// The declaration behind `slot` (`None` for built-ins).
543    pub fn declared(&self, slot: usize) -> Option<&AvatarParam> {
544        self.decl.get(slot).and_then(Option::as_ref)
545    }
546    /// Slots of the synced declared parameters, in name order: the wire layout.
547    pub fn synced_slots(&self) -> Vec<usize> {
548        let mut v: Vec<usize> = (0..self.decl.len()).filter(|&s| self.decl[s].as_ref().is_some_and(AvatarParam::is_synced)).collect();
549        v.sort_by(|&a, &b| self.names[a].cmp(&self.names[b]));
550        v
551    }
552
553    fn mark(&mut self, slot: usize) {
554        self.dirty[slot / 64] |= 1 << (slot % 64);
555    }
556
557    /// Write a declared parameter, coerced to its kind and range.
558    pub fn set(&mut self, slot: usize, v: f32) {
559        if !v.is_finite() {
560            return;
561        }
562        let v = match &self.decl[slot] {
563            Some(p) => p.coerce(v),
564            None => self.kinds[slot].coerce_local(v),
565        };
566        if self.values[slot] != v {
567            self.values[slot] = v;
568            self.mark(slot);
569        }
570    }
571    /// Write a built-in, coerced to its kind.
572    pub fn set_builtin(&mut self, b: BuiltinParam, v: f32) {
573        self.set(b as usize, v);
574    }
575
576    /// Back to the declared defaults (built-ins are the engine's to set).
577    pub fn reset_to_defaults(&mut self) {
578        for slot in BuiltinParam::ALL.len()..self.names.len() {
579            let d = self.decl[slot].as_ref().map_or(0.0, |p| p.default);
580            self.set(slot, d);
581        }
582    }
583
584    /// Write the slots that changed since the last flush into the Animator's
585    /// parameter map, then clear the marks.
586    pub fn flush_to(&mut self, map: &mut HashMap<String, f32>) {
587        for w in 0..self.dirty.len() {
588            let mut bits = self.dirty[w];
589            while bits != 0 {
590                let slot = w * 64 + bits.trailing_zeros() as usize;
591                bits &= bits - 1;
592                if slot >= self.names.len() {
593                    break;
594                }
595                match map.get_mut(&self.names[slot]) {
596                    Some(x) => *x = self.values[slot],
597                    None => {
598                        map.insert(self.names[slot].clone(), self.values[slot]);
599                    }
600                }
601            }
602            self.dirty[w] = 0;
603        }
604    }
605
606    /// Take values state-behaviour drivers (or code) wrote into the Animator's
607    /// map for declared parameters. A name lookup per declared parameter, so
608    /// call it only when something may have written (`params_touched`).
609    pub fn absorb_from(&mut self, map: &HashMap<String, f32>) {
610        for slot in BuiltinParam::ALL.len()..self.names.len() {
611            if let Some(&v) = map.get(&self.names[slot]) {
612                self.set(slot, v);
613            }
614        }
615    }
616
617    /// Slots of the declared parameters flagged `saved` (never built-ins).
618    pub fn saved_slots(&self) -> Vec<usize> {
619        (0..self.decl.len()).filter(|&s| self.decl[s].as_ref().is_some_and(|p| p.saved)).collect()
620    }
621
622    /// The saved parameters' current values.
623    pub fn saved_values(&self) -> SavedParams {
624        let mut out = SavedParams::default();
625        for (slot, d) in self.decl.iter().enumerate() {
626            if let Some(p) = d
627                && p.saved
628            {
629                out.values.insert(self.names[slot].clone(), self.values[slot]);
630            }
631        }
632        out
633    }
634
635    /// Restore saved values. Names that are not (or no longer) saved
636    /// parameters are ignored.
637    pub fn load_saved(&mut self, saved: &SavedParams) {
638        for (name, &v) in &saved.values {
639            if let Some(slot) = self.index_of(name)
640                && self.decl[slot].as_ref().is_some_and(|p| p.saved)
641            {
642                self.set(slot, v);
643            }
644        }
645    }
646}

648/// The parameters one avatar (or one part of it) declares. Put it on the
649/// avatar root; the store below is what Play runs on.
650#[derive(Serialize, Deserialize)]
651#[serde(default)]
652pub struct AvatarParameters {
653    /// The declared parameters.
654    pub params: Vec<AvatarParam>,
655    /// Synced-bit budget for this avatar; the project's content limit
656    /// (`ContentLimits::max_param_bits`) caps it.
657    pub budget_bits: u32,
658    /// Live values (built from `params`; not saved).
659    #[serde(skip)]
660    pub store: AvatarParamStore,
661    /// `store` was built from `params` (Play builds it on first use).
662    #[serde(skip)]
663    pub ready: bool,
664    /// A puppet is open on this avatar's radial menu: its parameters sync at
665    /// the fast rate (set by the menu, read by the network sync).
666    #[serde(skip)]
667    pub fast_sync: bool,
668}
669
670impl Default for AvatarParameters {
671    fn default() -> Self {
672        Self { params: Vec::new(), budget_bits: DEFAULT_PARAM_BUDGET_BITS, store: AvatarParamStore::default(), ready: false, fast_sync: false }
673    }
674}

676impl AvatarParameters {
677    /// Rebuild the store from `params` (Play start, or after an edit).
678    pub fn rebuild_store(&mut self) {
679        self.store = AvatarParamStore::new(&self.params);
680        self.ready = true;
681    }
682    /// This avatar's synced-bit use against `max`.
683    pub fn budget(&self, max: u32) -> ParamBudget {
684        ParamBudget::of(&self.params, max)
685    }
686    /// The budget this avatar is held to, given the project's cap.
687    pub fn effective_budget(&self, project_cap: u32) -> ParamBudget {
688        ParamBudget::of(&self.params, self.budget_bits.min(project_cap))
689    }
690    /// Names declared more than once, empty, or shadowing a built-in.
691    pub fn problems(&self) -> Vec<String> {
692        let mut out = Vec::new();
693        let mut seen: HashSet<&str> = HashSet::new();
694        for p in &self.params {
695            if p.name.is_empty() {
696                out.push("A parameter has no name.".to_owned());
697            } else if BuiltinParam::from_name(&p.name).is_some() {
698                out.push(format!("'{}' is a built-in parameter and is ignored.", p.name));
699            } else if !seen.insert(p.name.as_str()) {
700                out.push(format!("'{}' is declared more than once.", p.name));
701            }
702        }
703        out
704    }
705}

711/// What an expression-menu item does when used.
712#[derive(Serialize, Deserialize, Clone, Copy, Debug, PartialEq, Eq, Default)]
713pub enum MenuItemKind {
714    /// Sets `parameter` to `value` while held (at least 0.2 s), 0 on release.
715    Button,
716    /// Click sets `parameter` to `value`, click again sets 0.
717    #[default]
718    Toggle,
719    /// Opens a page of the item's children. `parameter` (optional) is `value`
720    /// while the page is open.
721    SubMenu,
722    /// One axis 0..1 from the stick angle.
723    RadialPuppet,
724    /// Two axes, -1..1 each.
725    TwoAxisPuppet,
726    /// Four directions, 0..1 each.
727    FourAxisPuppet,
728}
729
730impl MenuItemKind {
731    /// Every kind, in declaration order.
732    pub const ALL: [MenuItemKind; 6] =
733        [Self::Button, Self::Toggle, Self::SubMenu, Self::RadialPuppet, Self::TwoAxisPuppet, Self::FourAxisPuppet];
734    /// Inspector label of this kind.
735    pub fn label(self) -> &'static str {
736        match self {
737            Self::Button => "Button",
738            Self::Toggle => "Toggle",
739            Self::SubMenu => "Sub Menu",
740            Self::RadialPuppet => "Radial Puppet",
741            Self::TwoAxisPuppet => "Two-Axis Puppet",
742            Self::FourAxisPuppet => "Four-Axis Puppet",
743        }
744    }
745    /// Sub-parameters a puppet of this kind needs (0 for the others).
746    pub fn sub_parameter_count(self) -> usize {
747        match self {
748            Self::RadialPuppet => 1,
749            Self::TwoAxisPuppet => 2,
750            Self::FourAxisPuppet => 4,
751            _ => 0,
752        }
753    }
754    /// True for the puppet kinds (they drive sub-parameters).
755    pub fn is_puppet(self) -> bool {
756        self.sub_parameter_count() > 0
757    }
758    /// Range of each sub-parameter.
759    pub fn sub_parameter_range(self) -> (f32, f32) {
760        if self == Self::TwoAxisPuppet { (-1.0, 1.0) } else { (0.0, 1.0) }
761    }
762}

764/// One entry of the avatar's expression menu. The object hierarchy is the menu:
765/// a Sub Menu item's children are its page, and the Sub Menu on the avatar
766/// root is the main page.
767#[derive(Serialize, Deserialize, Clone, Debug, PartialEq)]
768#[serde(default)]
769pub struct AvatarMenuItem {
770    /// Text shown in the menu.
771    pub label: String,
772    /// Project-relative icon (capped at 256 px when built).
773    pub icon: String,
774    /// What the item does when used.
775    pub kind: MenuItemKind,
776    /// Parameter this item drives (empty = let the build pass pick one).
777    pub parameter: String,
778    /// Value written for Button / Toggle, and the "page open" value of
779    /// Sub Menu and puppets. 0 reads as "off", so use a non-zero value.
780    pub value: f32,
781    /// Puppet axes: 1, 2 or 4 parameter names by kind.
782    pub sub_parameters: Vec<String>,
783    /// Puppet direction labels (up, right, down, left).
784    pub labels: Vec<String>,
785    /// Sub Menu from a hand-authored `.avmenu` asset instead of children.
786    pub sub_menu_asset: String,
787    /// Remember the value between sessions.
788    pub saved: bool,
789    /// Send the value to other players.
790    pub synced: bool,
791    /// Starts selected.
792    pub is_default: bool,
793}
794
795impl Default for AvatarMenuItem {
796    fn default() -> Self {
797        Self {
798            label: "Item".into(),
799            icon: String::new(),
800            kind: MenuItemKind::Toggle,
801            parameter: String::new(),
802            value: 1.0,
803            sub_parameters: Vec::new(),
804            labels: Vec::new(),
805            sub_menu_asset: String::new(),
806            saved: true,
807            synced: true,
808            is_default: false,
809        }
810    }
811}

813impl AvatarMenuItem {
814    /// What a click does to a Toggle whose parameter holds `current`.
815    pub fn toggled(&self, current: f32) -> f32 {
816        if (current - self.value).abs() < 1e-4 { 0.0 } else { self.value }
817    }
818    /// Whether the Toggle / Button is on at `current`.
819    pub fn is_on(&self, current: f32) -> bool {
820        (current - self.value).abs() < 1e-4
821    }
822    /// Declared parameters this item needs but `known` lacks, plus shape
823    /// problems (wrong puppet axis count, value 0).
824    pub fn problems(&self, known: &dyn Fn(&str) -> bool) -> Vec<String> {
825        let mut out = Vec::new();
826        let drives = matches!(self.kind, MenuItemKind::Button | MenuItemKind::Toggle);
827        if drives && self.parameter.is_empty() {
828            out.push(format!("'{}' has no parameter.", self.label));
829        }
830        if !self.parameter.is_empty() && !known(&self.parameter) {
831            out.push(format!("'{}' uses '{}', which is not declared.", self.label, self.parameter));
832        }
833        if drives && self.value == 0.0 {
834            out.push(format!("'{}' sets its parameter to 0, which cannot be told apart from off.", self.label));
835        }
836        let need = self.kind.sub_parameter_count();
837        if self.sub_parameters.len() != need && self.kind.is_puppet() {
838            out.push(format!("'{}' needs {need} sub-parameter(s), has {}.", self.label, self.sub_parameters.len()));
839        }
840        for sp in self.sub_parameters.iter().filter(|s| !s.is_empty()) {
841            if !known(sp) {
842                out.push(format!("'{}' uses '{}', which is not declared.", self.label, sp));
843            }
844        }
845        out
846    }
847}
