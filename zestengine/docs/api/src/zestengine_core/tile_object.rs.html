1//! GameObject tiles (2D-5): a `.tile` whose `game_object` names a prefab puts one instance of it in
2//! every cell that uses the tile.
3//!
4//! The instances are ordinary scene objects, children of the tilemap's object, each carrying a
5//! [`TileObject`] that records the cell and the prefab it was made for (that is the per-cell storage:
6//! it is saved with the scene, so a loaded map finds its instances again and does not make them twice).
7//! The engine keeps them in step with the map (`engine/src/scene/tile_objects.rs`), in the editor and
8//! in play: painting a cell makes its instance, erasing the cell or changing its tile removes or
9//! replaces it, and undo of a stroke restores the map, after which the same step restores the
10//! instances. [`plan_tile_objects`] is that step's pure core.
11

17/// Marks a scene object as the instance of a tile's prefab in one cell of its parent tilemap.
18#[derive(Clone, PartialEq, Eq, Debug, Default, Serialize, Deserialize)]
19#[serde(default)]
20pub struct TileObject {
21    /// The cell (x, y) of the parent tilemap.
22    pub cell: [i32; 2],
23    /// The tile asset that asked for it (informational).
24    pub tile: String,
25    /// The prefab it is an instance of (project-relative `.prefab`).
26    pub prefab: String,
27}
28
29impl TypedComponent for TileObject {
30    const NAME: &'static str = "Tile Object";
31}

33/// An instance the map wants: this cell shows a tile whose prefab is `prefab`.
34#[derive(Clone, PartialEq, Eq, Debug)]
35pub struct WantObject {
36    /// The cell to fill.
37    pub cell: IVec2,
38    /// The prefab the cell's tile names.
39    pub prefab: String,
40}
41
42/// An instance that exists: scene object `object` (an index the caller understands) for `cell`.
43#[derive(Clone, PartialEq, Eq, Debug)]
44pub struct HaveObject {
45    /// Scene object index the caller understands.
46    pub object: usize,
47    /// The cell the instance sits in.
48    pub cell: IVec2,
49    /// The prefab it was made for.
50    pub prefab: String,
51}
52
53/// What to do to make the instances match the map.
54#[derive(Clone, PartialEq, Eq, Debug, Default)]
55pub struct TileObjectPlan {
56    /// Indices into `want` that have no instance yet.
57    pub spawn: Vec<usize>,
58    /// Indices into `have` to delete: instances of erased cells, of another prefab than the cell
59    /// now asks for, and duplicates of one cell.
60    pub remove: Vec<usize>,
61}
62
63/// Match `have` against `want` by (cell, prefab).
64pub fn plan_tile_objects(want: &[WantObject], have: &[HaveObject]) -> TileObjectPlan {
65    use std::collections::{HashMap, HashSet};
66    let wanted: HashMap<(IVec2, &str), usize> = want.iter().enumerate().map(|(i, w)| ((w.cell, w.prefab.as_str()), i)).collect();
67    let mut kept: HashSet<(IVec2, &str)> = HashSet::new();
68    let mut plan = TileObjectPlan::default();
69    for (i, h) in have.iter().enumerate() {
70        let key = (h.cell, h.prefab.as_str());
71        if wanted.contains_key(&key) && kept.insert(key) {
72            continue;
73        }
74        plan.remove.push(i);
75    }
76    for (i, w) in want.iter().enumerate() {
77        if !kept.contains(&(w.cell, w.prefab.as_str())) {
78            plan.spawn.push(i);
79        }
80    }
81    plan
82}

84#[cfg(test)]
85mod tests {
86    use super::*;

88    fn w(x: i32, y: i32, p: &str) -> WantObject {
89        WantObject { cell: IVec2::new(x, y), prefab: p.into() }
90    }
91    fn h(o: usize, x: i32, y: i32, p: &str) -> HaveObject {
92        HaveObject { object: o, cell: IVec2::new(x, y), prefab: p.into() }
93    }
94
95    #[test]
96    fn painting_erasing_and_replacing_cells_plan_the_right_edits() {
97        // Nothing yet: everything spawns.
98        let p = plan_tile_objects(&[w(0, 0, "a"), w(1, 0, "a")], &[]);
99        assert_eq!((p.spawn.clone(), p.remove.clone()), (vec![0, 1], vec![]));
100        // In step: nothing to do.
101        let p = plan_tile_objects(&[w(0, 0, "a"), w(1, 0, "a")], &[h(7, 0, 0, "a"), h(9, 1, 0, "a")]);
102        assert_eq!(p, TileObjectPlan::default());
103        // (1, 0) was erased and (2, 0) painted.
104        let p = plan_tile_objects(&[w(0, 0, "a"), w(2, 0, "a")], &[h(7, 0, 0, "a"), h(9, 1, 0, "a")]);
105        assert_eq!((p.spawn, p.remove), (vec![1], vec![1]));
106        // The cell's tile changed to one with another prefab: the old instance goes, a new one is made.
107        let p = plan_tile_objects(&[w(0, 0, "b")], &[h(7, 0, 0, "a")]);
108        assert_eq!((p.spawn, p.remove), (vec![0], vec![0]));
109        // Duplicates of one cell (a hand-merged scene) collapse to one; the rest are removed.
110        let p = plan_tile_objects(&[w(0, 0, "a")], &[h(1, 0, 0, "a"), h(2, 0, 0, "a"), h(3, 0, 0, "a")]);
111        assert_eq!((p.spawn.len(), p.remove), (0, vec![1, 2]));
112        // A map with no game-object tiles removes every instance.
113        let p = plan_tile_objects(&[], &[h(1, 0, 0, "a"), h(2, 4, 4, "a")]);
114        assert_eq!((p.spawn.len(), p.remove), (0, vec![0, 1]));
115    }
