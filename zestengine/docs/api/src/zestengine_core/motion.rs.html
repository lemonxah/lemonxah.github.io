1//! Replayable movement, for networked prediction (Network Object >
2//! Predicted): the state a controller's movement rewinds to, the world it
3//! moves through ([`MotionContext`]) and the controller side ([`Predictable`]).
4//! [`crate::ecs::ObjectContext`] is a `MotionContext`, and so is the engine's rewind / replay
5//! simulation, so a live frame and its replay run the same step.
6

14/// A controller's movement state: what prediction rewinds to and replays
15/// inputs from.
16#[derive(Clone, Copy, Debug, Default, PartialEq)]
17pub struct MotionState {
18    /// Current velocity (m/s).
19    pub vel: Vec3,
20    /// View yaw / pitch (degrees). They follow the owner's input, so
21    /// reconciliation keeps the owner's.
22    pub yaw: f32,
23    pub pitch: f32,
24    /// Facing of a body that turns apart from the view (third person).
25    pub body_yaw: f32,
26    /// On the ground.
27    pub grounded: bool,
28    /// Floor height when there is no Character.
29    pub floor_y: f32,
30    /// Crouching.
31    pub crouched: bool,
32    /// The Character capsule is shrunk for the crouch.
33    pub crouch_capsule: bool,
34    /// In water.
35    pub swimming: bool,
36    /// Seconds of breath left while it has been under (None = full).
37    pub breath: Option<f32>,
38    /// Current climb, if climbing.
39    pub climb: Option<Climb>,
40}

42/// What a movement step reads and moves: the object, its Character
43/// Controller and the world around it.
44pub trait MotionContext {
45    /// The object's id.
46    fn self_id(&self) -> ObjectId;
47    /// Local translation / rotation (the object's TRS fields).
48    fn translation(&self) -> Vec3;
49    /// Set the local translation.
50    fn set_translation(&mut self, t: Vec3);
51    /// Local rotation.
52    fn rotation(&self) -> Quat;
53    /// Local scale, including the calibrated VR tracking-space scale.
54    fn scale(&self) -> Vec3 { Vec3::ONE }
55    /// Set the local rotation.
56    fn set_rotation(&mut self, r: Quat);
57    /// World matrix of the parent (None for a root object).
58    fn parent_world(&self) -> Option<Mat4>;
59    /// A Character moves the object through the physics world.
60    fn has_character(&self) -> bool;
61    /// Move the Character by a world delta, colliding.
62    fn move_character(&mut self, delta: Vec3);
63    /// The Character touched the ground on its last move.
64    fn grounded(&self) -> bool;
65    /// The Character's (radius, height, center).
66    fn capsule(&self) -> Option<(f32, f32, [f32; 3])>;
67    /// Resize the capsule settings; physics reshapes it on the next move.
68    fn set_capsule(&mut self, height: f32, center: [f32; 3]);
69    /// The object's swim tuning (defaults without a Swimmer).
70    fn swimmer(&self) -> Swimmer;
71    /// Height of the water surface above or below `p`.
72    fn water_surface(&self, p: Vec3) -> Option<f32>;
73    /// Ray-cast against the physics world.
74    fn raycast(&self, origin: Vec3, dir: Vec3, max_dist: f32, filter: &RayFilter) -> Option<RayHit>;
75    /// Set an Animator parameter on the object.
76    fn set_animator_bool(&mut self, name: &str, value: bool);
77
78    /// World matrix of the object, including calibrated tracking-space scale.
79    fn world(&self) -> Mat4 {
80        let local = Mat4::from_scale_rotation_translation(self.scale(), self.rotation(), self.translation());
81        self.parent_world().map_or(local, |p| p * local)
82    }
83
84    /// World position.
85    fn world_position(&self) -> Vec3 {
86        let t = self.translation();
87        self.parent_world().map_or(t, |p| p.transform_point3(t))
88    }
89
90    /// Set the world position (converting through the parent).
91    fn set_world_position(&mut self, p: Vec3) {
92        let local = self.parent_world().map_or(p, |m| m.inverse().transform_point3(p));
93        self.set_translation(local);
94    }
95}

97/// A controller whose movement networked prediction can rewind and replay.
98pub trait Predictable: 'static {
99    /// The movement state prediction rewinds to.
100    fn motion(&self) -> MotionState;
101    /// Put the movement back into `m` (resizing the capsule through `obj`
102    /// to match).
103    fn set_motion(&mut self, m: &MotionState, obj: &mut dyn MotionContext);
104    /// What the frame just simulated took from outside the input actions
105    /// (view angles, headset pose, a teleport), carried by its input command.
106    fn frame_extra(&self) -> Vec<f32>;
107    /// Simulate one recorded input frame. `extra` is the frame's
108    /// [`Self::frame_extra`] (empty when it had none).
109    fn replay(&mut self, obj: &mut dyn MotionContext, input: &InputState, extra: &[f32], dt: f32);
110    /// The local player drives it (a Pawn only while possessed).
111    fn locally_driven(&self) -> bool {
112        true
113    }
114}

116/// The component as a [`Predictable`] controller, if it is one.
117pub fn as_predictable(c: &mut dyn Component) -> Option<&mut dyn Predictable> {
118    c.predictable_mut()
119}
120
121/// True when the component implements [`Predictable`] (has a predictable
122/// controller).
123pub fn is_predictable(c: &dyn Component) -> bool {
124    c.predictable().is_some()
125}

127/// A networked copy another peer drives (its owner, or the host simulating
128/// the owner's predicted inputs): local input mustn't move it.
129pub fn driven_remotely(ctx: &crate::ecs::ObjectContext<'_, '_>) -> bool { ctx.driven_remotely() }

131/// The view (yaw, pitch) a frame's extra input carries, if it does.
132pub(crate) fn view_of(extra: &[f32]) -> Option<(f32, f32)> {
133    match extra {
134        [yaw, pitch, ..] => Some((*yaw, *pitch)),
135        _ => None,
136    }
