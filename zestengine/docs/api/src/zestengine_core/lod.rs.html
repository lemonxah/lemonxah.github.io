1//! Screen-size level of detail: pure selection logic shared by the renderer's
2//! mesh swap and the non-geometry distance LOD (animation rate, IK).
3//!
4//! "Coverage" is the share of the screen height an object's bounding sphere
5//! covers. Unlike a distance it already accounts for the object's size, the
6//! camera FOV and (through the projection) the window height, so one set of
7//! thresholds works for a ring and a building, on a monitor and in a headset.
8
9use crate::LodLevel;
10
11/// Fraction of the screen height covered by a bounding sphere of `radius` at
12/// `distance`, for a camera with `tan_half_fov` = tan(vertical FOV / 2).
13/// An object the camera is inside of covers everything (`f32::MAX`).
14pub fn screen_coverage(radius: f32, distance: f32, tan_half_fov: f32) -> f32 {
15    if distance <= radius || tan_half_fov <= 0.0 {
16        return f32::MAX;
17    }
18    // Projected diameter over the frustum height at that distance.
19    radius / (distance * tan_half_fov)
20}
21
22/// Relative margin around a threshold inside which the current level holds.
23pub const COVERAGE_HYSTERESIS: f32 = 0.1;
24
25/// Level for `coverage` out of `thresholds` (the minimum coverage each level
26/// serves, descending; level 0 is the most detailed). `previous` holds while
27/// the coverage stays within [`COVERAGE_HYSTERESIS`] of its band so an object
28/// does not flicker at a boundary. Empty thresholds give `None`.
29pub fn select_by_coverage(thresholds: &[f32], coverage: f32, previous: Option<usize>) -> Option<usize> {
30    if thresholds.is_empty() {
31        return None;
32    }
33    let pick = |c: f32| thresholds.iter().position(|&t| c >= t).unwrap_or(thresholds.len() - 1);
34    if let Some(prev) = previous.filter(|&p| p < thresholds.len()) {
35        let (grown, shrunk) = (coverage * (1.0 + COVERAGE_HYSTERESIS), coverage * (1.0 - COVERAGE_HYSTERESIS));
36        if pick(grown) == prev || pick(shrunk) == prev {
37            return Some(prev);
38        }
39    }
40    Some(pick(coverage))
41}
42
43/// The automatic chain every model gets: the share of triangles kept and the
44/// minimum screen coverage that level serves.
45pub const AUTO_LOD_STEPS: [(f32, f32); 4] = [(1.0, 0.30), (0.5, 0.12), (0.25, 0.05), (0.1, 0.0)];
46
47/// Below this coverage (the last automatic level serves everything under 0.05)
48/// a static mesh may draw as an octahedral impostor billboard instead.
49pub const IMPOSTOR_COVERAGE: f32 = 0.015;
50
51/// From this coverage down a mesh gets its impostor captured, so the atlas is
52/// ready before the switch.
53pub const IMPOSTOR_PREPARE_COVERAGE: f32 = 0.04;
54
55/// Whether a mesh at `coverage` draws as its impostor; `was` (it did last
56/// frame) holds it within [`COVERAGE_HYSTERESIS`] of the boundary.
57pub fn use_impostor(coverage: f32, was: bool) -> bool {
58    let limit = if was { IMPOSTOR_COVERAGE * (1.0 + COVERAGE_HYSTERESIS) } else { IMPOSTOR_COVERAGE };
59    coverage < limit
60}
61
62/// Camera distance at which a bounding sphere of `radius` reaches [`IMPOSTOR_COVERAGE`].
63pub fn impostor_distance(radius: f32, tan_half_fov: f32) -> f32 {
64    radius / (IMPOSTOR_COVERAGE * tan_half_fov)
65}
66
67/// Models with fewer triangles than this get no automatic chain: the draw and
68/// memory overhead of extra levels outweighs the saving.
69pub const AUTO_LOD_MIN_TRIANGLES: usize = 1500;
70
71impl LodLevel {
72    /// A level selected by screen size, generated from the object's own model.
73    pub fn by_screen(keep: f32, screen_size: f32) -> Self {
74        Self { max_distance: 0.0, model: String::new(), keep, screen_size }
75    }
76}
77
78/// Minimum coverages of an automatic chain (descending), for selection.
79pub fn auto_thresholds() -> [f32; AUTO_LOD_STEPS.len()] {
80    AUTO_LOD_STEPS.map(|s| s.1)
81}
82
83/// How often a far object's animation is re-evaluated, in frames (1 = every
84/// frame). Cheap tiers only: the pose between updates is held, so the steps
85/// stay small enough not to read as stutter at that size on screen.
86pub fn animation_interval(coverage: f32) -> u32 {
87    match coverage {
88        c if c >= 0.15 => 1,
89        c if c >= 0.06 => 2,
90        c if c >= 0.025 => 4,
91        _ => 8,
92    }
93}
94
95/// Below this coverage humanoid IK of a non-local avatar is skipped.
96pub const IK_MIN_COVERAGE: f32 = 0.02;

98#[cfg(test)]
99mod tests {
100    use super::*;
101
102    #[test]
103    fn coverage_falls_with_distance_and_grows_with_size() {
104        let t = (60f32.to_radians() / 2.0).tan();
105        let near = screen_coverage(1.0, 5.0, t);
106        let far = screen_coverage(1.0, 50.0, t);
107        assert!((near / far - 10.0).abs() < 1e-3);
108        assert!(screen_coverage(2.0, 50.0, t) > far);
109        // A narrower FOV (zoomed in) makes the same object cover more.
110        assert!(screen_coverage(1.0, 50.0, (20f32.to_radians() / 2.0).tan()) > far);
111        assert_eq!(screen_coverage(1.0, 0.5, t), f32::MAX);
112    }
113
114    #[test]
115    fn impostor_switch_has_hysteresis_and_matches_its_distance() {
116        assert!(!use_impostor(0.02, false) && use_impostor(0.01, false));
117        // Just over the boundary holds an impostor already in use, clearly over drops it.
118        assert!(use_impostor(IMPOSTOR_COVERAGE * 1.05, true) && !use_impostor(IMPOSTOR_COVERAGE * 1.05, false));
119        assert!(!use_impostor(IMPOSTOR_COVERAGE * 1.2, true));
120        let t = (60f32.to_radians() / 2.0).tan();
121        let d = impostor_distance(2.0, t);
122        assert!((screen_coverage(2.0, d, t) - IMPOSTOR_COVERAGE).abs() < 1e-6);
123        // The impostor takes over below the last automatic level.
124        assert!(IMPOSTOR_PREPARE_COVERAGE < AUTO_LOD_STEPS[AUTO_LOD_STEPS.len() - 2].1 && IMPOSTOR_COVERAGE < IMPOSTOR_PREPARE_COVERAGE);
125    }
126
127    #[test]
128    fn selection_follows_thresholds_with_hysteresis() {
129        let th = auto_thresholds();
130        assert_eq!(select_by_coverage(&th, 0.5, None), Some(0));
131        assert_eq!(select_by_coverage(&th, 0.2, None), Some(1));
132        assert_eq!(select_by_coverage(&th, 0.06, None), Some(2));
133        assert_eq!(select_by_coverage(&th, 0.001, None), Some(3));
134        assert_eq!(select_by_coverage(&[], 0.5, None), None);
135        // Just under the 0.30 boundary stays on level 0, clearly under switches.
136        assert_eq!(select_by_coverage(&th, 0.29, Some(0)), Some(0));
137        assert_eq!(select_by_coverage(&th, 0.2, Some(0)), Some(1));
138        // Just over the boundary from below holds level 1.
139        assert_eq!(select_by_coverage(&th, 0.31, Some(1)), Some(1));
140        assert_eq!(select_by_coverage(&th, 0.4, Some(1)), Some(0));
141        // A stale previous index out of range is ignored.
142        assert_eq!(select_by_coverage(&th, 0.5, Some(9)), Some(0));
143    }
144
145    #[test]
146    fn animation_rate_only_drops_with_size() {
147        let mut last = 0;
148        for c in [1.0, 0.1, 0.03, 0.001] {
149            let n = animation_interval(c);
150            assert!(n >= last);
151            last = n;
152        }
153        assert_eq!(animation_interval(0.5), 1);
154        assert_eq!(animation_interval(0.001), 8);
155    }
156
157    #[test]
158    fn auto_chain_is_descending_and_ends_open() {
159        let th = auto_thresholds();
160        assert!(th.windows(2).all(|w| w[0] > w[1]));
161        assert_eq!(*th.last().unwrap(), 0.0);
162        assert!(AUTO_LOD_STEPS.windows(2).all(|w| w[0].0 > w[1].0));
163    }
