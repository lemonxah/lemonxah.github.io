1//! Painted per-vertex cloth pins, limits and material/drive/backstop masks, and the
2//! brush math the editor's pin painter runs on it.
3//!
4//! Values are quantized to a byte per render vertex and saved as a short
5//! run-length + base64 string. Each set is keyed by the vertex count and a
6//! hash of the triangle list, and keeps quantized bind positions so a

16/// Bytes that serialize as a compact string: `r` + base64 of (run, value)
17/// pairs when that is shorter, else `b` + base64 of the bytes. The encoded
18/// string is cached until the bytes change.
19#[derive(Default)]
20pub struct PackedBytes {
21    data: Vec<u8>,
22    encoded: OnceLock<String>,
23}
24
25impl PackedBytes {
26    /// Build from raw bytes.
27    pub fn new(data: Vec<u8>) -> Self {
28        Self { data, encoded: OnceLock::new() }
29    }
30
31    /// The raw bytes.
32    pub fn as_slice(&self) -> &[u8] {
33        &self.data
34    }
35
36    /// Byte length.
37    pub fn len(&self) -> usize {
38        self.data.len()
39    }
40
41    /// True when empty.
42    pub fn is_empty(&self) -> bool {
43        self.data.is_empty()
44    }
45
46    /// Mutable access; drops the cached encoding.
47    pub fn make_mut(&mut self) -> &mut Vec<u8> {
48        self.encoded.take();
49        &mut self.data
50    }
51
52    /// Replace the bytes.
53    pub fn set(&mut self, data: Vec<u8>) {
54        *self = Self::new(data);
55    }
56
57    /// The serialized form (cached until the bytes change).
58    pub fn encoded(&self) -> &str {
59        self.encoded.get_or_init(|| pack(&self.data))
60    }
61}

66    }

100const B64: &[u8; 64] = b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
101
102fn base64(bytes: &[u8]) -> String {
103    let mut out = String::with_capacity(bytes.len().div_ceil(3) * 4);
104    for c in bytes.chunks(3) {
105        let n = ((c[0] as u32) << 16) | ((*c.get(1).unwrap_or(&0) as u32) << 8) | *c.get(2).unwrap_or(&0) as u32;
106        for k in 0..4 {
107            if k <= c.len() {
108                out.push(B64[((n >> (18 - 6 * k)) & 63) as usize] as char);
109            } else {
110                out.push('=');
111            }
112        }
113    }
114    out
115}
116
117fn unbase64(s: &str) -> Option<Vec<u8>> {
118    let mut out = Vec::with_capacity(s.len() / 4 * 3);
119    let (mut acc, mut bits) = (0u32, 0u32);
120    for ch in s.bytes() {
121        if ch == b'=' {
122            break;
123        }
124        let v = B64.iter().position(|&b| b == ch)? as u32;
125        acc = (acc << 6) | v;
126        bits += 6;
127        if bits >= 8 {
128            bits -= 8;
129            out.push(((acc >> bits) & 0xff) as u8);
130        }
131    }
132    Some(out)
133}
134
135fn rle(bytes: &[u8]) -> Vec<u8> {
136    let mut out = Vec::new();
137    let mut k = 0;
138    while k < bytes.len() {
139        let v = bytes[k];
140        let mut run = 1;
141        while k + run < bytes.len() && bytes[k + run] == v && run < 255 {
142            run += 1;
143        }
144        out.extend([run as u8, v]);
145        k += run;
146    }
147    out
148}
149
150fn pack(bytes: &[u8]) -> String {
151    if bytes.is_empty() {
152        return String::new();
153    }
154    let runs = rle(bytes);
155    if runs.len() < bytes.len() { format!("r{}", base64(&runs)) } else { format!("b{}", base64(bytes)) }
156}
157
158fn unpack(s: &str) -> Option<Vec<u8>> {
159    if s.is_empty() {
160        return Some(Vec::new());
161    }
162    let body = unbase64(&s[1..])?;
163    match s.as_bytes()[0] {
164        b'b' => Some(body),
165        b'r' => {
166            let mut out = Vec::new();
167            for pair in body.as_chunks::<2>().0 {
168                out.extend(std::iter::repeat_n(pair[1], pair[0] as usize));
169            }
170            Some(out)
171        }
172        _ => None,
173    }
174}

176/// FNV-1a over the vertex count and triangle list: stable across runs and
177/// builds, so saved paint recognises its mesh.
178pub fn topology_hash(vertex_count: usize, indices: &[u32]) -> u64 {
179    let mut h: u64 = 0xcbf2_9ce4_8422_2325;
180    let mut eat = |b: u8| {
181        h ^= b as u64;
182        h = h.wrapping_mul(0x0100_0000_01b3);
183    };
184    for b in (vertex_count as u64).to_le_bytes() {
185        eat(b);
186    }
187    for i in indices {
188        for b in i.to_le_bytes() {
189            eat(b);
190        }
191    }
192    h
193}

222/// How a brush fades from its centre to its rim.
223#[derive(Clone, Copy, Debug, PartialEq, Eq, Default, Serialize, Deserialize)]
224pub enum BrushFalloff {
225    /// Cosine falloff (smoothstep).
226    #[default]
227    Smooth,
228    /// Straight-line falloff.
229    Linear,
230    /// Full strength to the rim.
231    Constant,
232}
233
234impl BrushFalloff {
235    /// Every falloff, in declaration order.
236    pub const ALL: [BrushFalloff; 3] = [BrushFalloff::Smooth, BrushFalloff::Linear, BrushFalloff::Constant];
237
238    /// Inspector label of this falloff.
239    pub fn label(self) -> &'static str {
240        match self {
241            BrushFalloff::Smooth => "Smooth",
242            BrushFalloff::Linear => "Linear",
243            BrushFalloff::Constant => "Constant",
244        }
245    }
246
247    /// Brush weight at distance `d` from the centre (1 at the centre, 0 at
248    /// and past `radius`).
249    pub fn weight(self, d: f32, radius: f32) -> f32 {
250        if radius <= 0.0 || d >= radius {
251            return 0.0;
252        }
253        let t = (d / radius).max(0.0);
254        match self {
255            BrushFalloff::Smooth => 1.0 - t * t * (3.0 - 2.0 * t),
256            BrushFalloff::Linear => 1.0 - t,
257            BrushFalloff::Constant => 1.0,
258        }
259    }
260}

313/// Render vertices welded by position (UV seam copies share a group), with
314/// each group's edge neighbours: brushes and smoothing act per group so seams
315/// never split.
316#[derive(Clone, Debug, Default)]
317pub struct WeldTopology {
318    /// Vertex -> group.
319    pub group: Vec<u32>,
320    /// Group -> its vertices.
321    pub members: Vec<Vec<u32>>,
322    /// Group -> neighbouring groups.
323    pub neighbours: Vec<Vec<u32>>,
324    /// Group position (object space, bind pose).
325    pub positions: Vec<Vec3>,
326}
327
328impl WeldTopology {
329    /// Weld by position and link each group's edge neighbours.
330    pub fn build(positions: &[Vec3], indices: &[u32]) -> Self {
331        let cell = 1e-5;
332        let key = |p: Vec3| ((p.x / cell).round() as i64, (p.y / cell).round() as i64, (p.z / cell).round() as i64);
333        let mut by_key: HashMap<(i64, i64, i64), u32> = HashMap::new();
334        let mut group = Vec::with_capacity(positions.len());
335        let mut members: Vec<Vec<u32>> = Vec::new();
336        let mut group_pos = Vec::new();
337        for (v, p) in positions.iter().enumerate() {
338            let next = members.len() as u32;
339            let g = *by_key.entry(key(*p)).or_insert(next);
340            if g == next {
341                members.push(Vec::new());
342                group_pos.push(*p);
343            }
344            members[g as usize].push(v as u32);
345            group.push(g);
346        }
347        let mut neighbours: Vec<Vec<u32>> = vec![Vec::new(); members.len()];
348        for tri in indices.as_chunks::<3>().0 {
349            if tri.iter().any(|&v| v as usize >= group.len()) {
350                continue;
351            }
352            for k in 0..3 {
353                let (a, b) = (group[tri[k] as usize], group[tri[(k + 1) % 3] as usize]);
354                if a != b {
355                    neighbours[a as usize].push(b);
356                    neighbours[b as usize].push(a);
357                }
358            }
359        }
360        for n in &mut neighbours {
361            n.sort_unstable();
362            n.dedup();
363        }
364        Self { group, members, neighbours, positions: group_pos }
365    }
366
367    /// Render vertices in the welded mesh.
368    pub fn vertex_count(&self) -> usize {
369        self.group.len()
370    }
371
372    /// Apply a dab to per-vertex `values` (0 to 1). True when any changed.
373    pub fn apply(&self, values: &mut [f32], dab: &Dab) -> bool {
374        if values.len() != self.group.len() {
375            return false;
376        }
377        let r2 = dab.radius * dab.radius;
378        let hit: Vec<(usize, f32)> = self
379            .positions
380            .iter()
381            .enumerate()
382            .filter_map(|(g, p)| {
383                let w = dab
384                    .centers
385                    .iter()
386                    .filter(|c| (**c - *p).length_squared() < r2)
387                    .map(|c| dab.falloff.weight((*c - *p).length(), dab.radius))
388                    .fold(0.0f32, f32::max);
389                (w > 0.0).then_some((g, w * dab.strength.clamp(0.0, 1.0)))
390            })
391            .collect();
392        let value_of = |values: &[f32], g: usize| self.members[g].first().map_or(0.0, |&v| values[v as usize]);
393        let targets: Vec<f32> = match dab.mode {
394            BrushMode::Smooth => hit
395                .iter()
396                .map(|&(g, _)| {
397                    let n = &self.neighbours[g];
398                    if n.is_empty() {
399                        value_of(values, g)
400                    } else {
401                        n.iter().map(|&o| value_of(values, o as usize)).sum::<f32>() / n.len() as f32
402                    }
403                })
404                .collect(),
405            BrushMode::Paint => vec![1.0; hit.len()],
406            BrushMode::Erase => vec![0.0; hit.len()],
407            BrushMode::Set => vec![dab.value.clamp(0.0, 1.0); hit.len()],
408        };
409        let mut changed = false;
410        for (&(g, a), target) in hit.iter().zip(targets) {
411            let old = value_of(values, g);
412            let new = step_toward(old, target, a);
413            if new != old {
414                changed = true;
415                for &v in &self.members[g] {
416                    values[v as usize] = new;
417                }
418            }
419        }
420        changed
421    }
422}

424/// `old` moved `amount` of the way to `target`, but by at least one byte
425/// step when it moves at all, so light strokes still reach the target once
426/// quantized.
427fn step_toward(old: f32, target: f32, amount: f32) -> f32 {
428    if amount <= 0.0 || (old - target).abs() < 0.5 / 255.0 {
429        return old;
430    }
431    let new = old + (target - old) * amount.min(1.0);
432    let step = 1.0 / 255.0;
433    if (quantize(new) as i32 - quantize(old) as i32).abs() >= 1 {
434        new
435    } else if target > old {
436        (old + step).min(target)
437    } else {
438        (old - step).max(target)
439    }
440}

450/// For each vertex, the vertex nearest its mirror across local X (within
451/// `tolerance`), via a coarse grid.
452pub fn mirror_map(positions: &[Vec3], tolerance: f32) -> Vec<Option<u32>> {
453    let grid = Grid::new(positions, tolerance.max(1e-5));
454    positions.iter().map(|p| grid.nearest(positions, mirror_x(*p), tolerance)).collect()
455}

457/// Uniform hash grid over points for nearest lookups.
458struct Grid {
459    cell: f32,
460    cells: HashMap<(i32, i32, i32), Vec<u32>>,
461}
462
463impl Grid {
464    fn new(points: &[Vec3], cell: f32) -> Self {
465        let mut cells: HashMap<(i32, i32, i32), Vec<u32>> = HashMap::new();
466        for (k, p) in points.iter().enumerate() {
467            cells.entry(Self::key(cell, *p)).or_default().push(k as u32);
468        }
469        Self { cell, cells }
470    }
471
472    fn key(cell: f32, p: Vec3) -> (i32, i32, i32) {
473        ((p.x / cell).floor() as i32, (p.y / cell).floor() as i32, (p.z / cell).floor() as i32)
474    }
475
476    /// Nearest point within `max` (any distance when `max` is infinite),
477    /// searching outward ring by ring.
478    fn nearest(&self, points: &[Vec3], q: Vec3, max: f32) -> Option<u32> {
479        if points.is_empty() {
480            return None;
481        }
482        let (cx, cy, cz) = Self::key(self.cell, q);
483        let rings = if max.is_finite() { (max / self.cell).ceil() as i32 + 1 } else { 64 };
484        let mut best: Option<(f32, u32)> = None;
485        for ring in 0..=rings {
486            for dx in -ring..=ring {
487                for dy in -ring..=ring {
488                    for dz in -ring..=ring {
489                        if dx.abs().max(dy.abs()).max(dz.abs()) != ring {
490                            continue;
491                        }
492                        let Some(list) = self.cells.get(&(cx + dx, cy + dy, cz + dz)) else { continue };
493                        for &k in list {
494                            let d = (points[k as usize] - q).length_squared();
495                            if best.is_none_or(|(bd, _)| d < bd) {
496                                best = Some((d, k));
497                            }
498                        }
499                    }
500                }
501            }
502            // Anything in a later ring is at least `ring * cell` away.
503            if let Some((d, _)) = best
504                && d.sqrt() <= ring as f32 * self.cell
505            {
506                break;
507            }
508        }
509        if !max.is_finite() && best.is_none() {
510            return points
511                .iter()
512                .enumerate()
513                .min_by(|a, b| (*a.1 - q).length_squared().total_cmp(&(*b.1 - q).length_squared()))
514                .map(|(k, _)| k as u32);
515        }
516        best.filter(|(d, _)| d.sqrt() <= max).map(|(_, k)| k)
517    }
518}

531/// Painted Mesh Cloth values for one submesh, keyed by its topology.
532#[derive(Clone, Debug, Default, PartialEq, Serialize, Deserialize)]
533#[serde(default)]
534pub struct ClothPaint {
535    /// Render vertices the paint covers.
536    pub vertex_count: u32,
537    /// [`topology_hash`] of the mesh it was painted on.
538    pub topology: u64,
539    /// Pin weight per render vertex (255 = pinned).
540    pub pins: PackedBytes,
541    /// Max distance per render vertex as a share of the cloth's max distance;
542    /// empty = not limited.
543    pub max_distance: PackedBytes,
544    /// Independent material masks. Missing channel means a multiplier of one;
545    /// stored zero values disable that property locally. Pins remain independent.
546    pub maps: BTreeMap<PaintChannel,PackedBytes>,
547    /// Bind-pose positions, 3 x u16 per vertex across `bounds`, for remapping.
548    pub positions: PackedBytes,
549    /// Object-space bind bounds (min xyz, max xyz) the positions span.
550    pub bounds: [f32; 6],
551}
552
553impl ClothPaint {
554    /// True when nothing is painted.
555    pub fn is_empty(&self) -> bool {
556        self.vertex_count == 0
557    }
558
559    /// True when the paint was made for this mesh (same vertex count and
560    /// topology).
561    pub fn matches(&self, vertex_count: usize, topology: u64) -> bool {
562        self.vertex_count as usize == vertex_count
563            && self.topology == topology
564            && self.pins.len() == vertex_count
565            && (self.max_distance.is_empty() || self.max_distance.len() == vertex_count)
566            && self.maps.values().all(|map|map.is_empty()||map.len()==vertex_count)
567    }
568
569    /// Fresh paint for a mesh from per-vertex pins (0 to 1).
570    pub fn seed(bind: &[Vec3], indices: &[u32], pins: &[f32]) -> Self {
571        let mut out = Self {
572            vertex_count: bind.len() as u32,
573            topology: topology_hash(bind.len(), indices),
574            pins: PackedBytes::new((0..bind.len()).map(|v| quantize(pins.get(v).copied().unwrap_or(0.0))).collect()),
575            ..Default::default()
576        };
577        out.store_positions(bind);
578        out
579    }
580
581    fn store_positions(&mut self, bind: &[Vec3]) {
582        let (lo, hi) = bind.iter().fold((Vec3::splat(f32::MAX), Vec3::splat(f32::MIN)), |(lo, hi), p| (lo.min(*p), hi.max(*p)));
583        let (lo, hi) = if bind.is_empty() { (Vec3::ZERO, Vec3::ZERO) } else { (lo, hi) };
584        self.bounds = [lo.x, lo.y, lo.z, hi.x, hi.y, hi.z];
585        let span = (hi - lo).max(Vec3::splat(1e-9));
586        let mut bytes = Vec::with_capacity(bind.len() * 6);
587        for p in bind {
588            let t = ((*p - lo) / span).clamp(Vec3::ZERO, Vec3::ONE);
589            for c in [t.x, t.y, t.z] {
590                bytes.extend(((c * 65535.0).round() as u16).to_le_bytes());
591            }
592        }
593        self.positions.set(bytes);
594    }
595
596    /// Authoring edits can move vertices while preserving their identities.
597    /// Refresh remapping coordinates without transferring masks to neighbours.
598    pub fn rebind_positions(&mut self,bind:&[Vec3],indices:&[u32])->bool{
599        if !self.matches(bind.len(),topology_hash(bind.len(),indices))||bind.iter().any(|p|!p.is_finite()){return false;}
600        self.store_positions(bind);true
601    }
602    /// Preserve exact paint ownership through a known vertex deletion/reorder.
603    /// Use spatial `fit` only when that correspondence is unavailable.
604    pub fn reindex(&mut self,bind:&[Vec3],indices:&[u32],old_vertices:&[Option<usize>])->bool{
605        if bind.len()!=old_vertices.len()||bind.iter().any(|p|!p.is_finite())||old_vertices.iter().flatten().any(|&i|i>=self.vertex_count as usize)
606            ||!self.matches(self.vertex_count as usize,self.topology){return false;}
607        let pick=|values:&PackedBytes,neutral|PackedBytes::new(old_vertices.iter().map(|i|i.map_or(neutral,|i|values.as_slice()[i])).collect());
608        let mut next=Self::seed(bind,indices,&[]);next.pins=pick(&self.pins,0);
609        if !self.max_distance.is_empty(){next.max_distance=pick(&self.max_distance,255);}
610        for (&channel,values)in &self.maps{if !values.is_empty(){next.maps.insert(channel,pick(values,255));}}
611        *self=next;true
612    }

614    /// The saved bind positions, dequantized.
615    pub fn stored_positions(&self) -> Vec<Vec3> {
616        let lo = Vec3::new(self.bounds[0], self.bounds[1], self.bounds[2]);
617        let hi = Vec3::new(self.bounds[3], self.bounds[4], self.bounds[5]);
618        let span = hi - lo;
619        self.positions
620            .as_slice()
621            .as_chunks::<6>()
622            .0
623            .iter()
624            .map(|c| {
625                let u = |k: usize| u16::from_le_bytes([c[k], c[k + 1]]) as f32 / 65535.0;
626                lo + Vec3::new(u(0), u(2), u(4)) * span
627            })
628            .collect()
629    }

631    /// Make the paint fit this mesh: keep it, start it, or remap it by
632    /// nearest stored position when the topology changed.
633    pub fn fit(&mut self, bind: &[Vec3], indices: &[u32]) -> PaintFit {
634        let topology = topology_hash(bind.len(), indices);
635        let old = self.stored_positions();
636        let tolerance=((Vec3::from_slice(&self.bounds[3..])-Vec3::from_slice(&self.bounds[..3])).max_element()/65535.0*2.0).max(1e-5);
637        if self.matches(bind.len(),topology)&&old.len()==bind.len()&&old.iter().zip(bind).all(|(a,b)|a.distance(*b)<=tolerance){return PaintFit::Same;}
638        if self.is_empty() || old.is_empty() || old.len() != self.pins.len() {
639            *self = Self::seed(bind, indices, &[]);
640            return PaintFit::Fresh;
641        }
642        let grid = Grid::new(&old, ((Vec3::from_slice(&self.bounds[3..]) - Vec3::from_slice(&self.bounds[..3])).max_element() / 64.0).max(1e-4));
643        let nearest: Vec<usize> = bind.iter().map(|p| grid.nearest(&old, *p, f32::INFINITY).unwrap_or(0) as usize).collect();
644        let pick = |src: &PackedBytes| -> Vec<u8> { nearest.iter().map(|&k| src.as_slice().get(k).copied().unwrap_or(0)).collect() };
645        let pins = pick(&self.pins);
646        let max_distance = if self.max_distance.len() == old.len() { pick(&self.max_distance) } else { Vec::new() };
647        let maps=self.maps.iter().filter(|(channel,map)|channel.is_material()&&map.len()==old.len()).map(|(&channel,map)|(channel,PackedBytes::new(pick(map)))).collect();
648        *self = Self::seed(bind, indices, &[]);
649        self.pins.set(pins);
650        self.max_distance.set(max_distance);
651        self.maps=maps;
652        PaintFit::Remapped
653    }

655    /// Authored data, or None for the neutral unpainted value.
656    pub fn channel(&self,channel:PaintChannel)->Option<&PackedBytes>{
657        let bytes=match channel{PaintChannel::Pin=>Some(&self.pins),PaintChannel::MaxDistance=>Some(&self.max_distance),_=>self.maps.get(&channel)};
658        bytes.filter(|bytes|!bytes.is_empty())
659    }
660    /// Per-vertex normalized values. Unpainted masks read as one; pin defaults zero.
661    pub fn values(&self,channel:PaintChannel)->Vec<f32>{
662        match self.channel(channel).filter(|bytes|bytes.len()==self.vertex_count as usize){
663            Some(bytes)=>bytes.as_slice().iter().map(|&b|dequantize(b)).collect(),
664            None=>vec![if channel==PaintChannel::Pin{0.0}else{1.0};self.vertex_count as usize],
665        }
666    }
667    pub fn set_values(&mut self,channel:PaintChannel,values:&[f32]){
668        if values.len()!=self.vertex_count as usize{return;}
669        let bytes=values.iter().map(|&v|quantize(v)).collect();
670        match channel{PaintChannel::Pin=>self.pins.set(bytes),PaintChannel::MaxDistance=>self.max_distance.set(bytes),_=>{self.maps.insert(channel,PackedBytes::new(bytes));}}
671    }
672    /// Restore the neutral unpainted behavior, independently of other channels.
673    pub fn remove_channel(&mut self,channel:PaintChannel){
674        match channel{PaintChannel::Pin=>self.pins.set(vec![0;self.vertex_count as usize]),PaintChannel::MaxDistance=>self.max_distance.set(Vec::new()),_=>{self.maps.remove(&channel);}}
675    }
676
677    /// Pin weight per vertex (0..1).
678    pub fn pin_weights(&self) -> Vec<f32> {
679        self.values(PaintChannel::Pin)
680    }
681
682    /// Per-vertex distance limits in metres (`full` = a value of 1), or None
683    /// when the max distance isn't painted.
684    pub fn limits(&self, full: f32) -> Option<Vec<f32>> {
685        (!self.max_distance.is_empty() && full > 0.0)
686            .then(|| self.max_distance.as_slice().iter().map(|&b| dequantize(b) * full).collect())
687    }
688
689    /// Copy each vertex's value from its mirror on the +X side to the -X side
690    /// (`positive_to_negative`) or the other way round. Vertices without a
691    /// mirror keep their value.
692    pub fn symmetrize(&mut self, channel: PaintChannel, bind: &[Vec3], positive_to_negative: bool, tolerance: f32) {
693        let mut values = self.values(channel);
694        if values.len() != bind.len() {
695            return;
696        }
697        let map = mirror_map(bind, tolerance);
698        let src = values.clone();
699        for (v, m) in map.iter().enumerate() {
700            let take = if positive_to_negative { bind[v].x < 0.0 } else { bind[v].x > 0.0 };
701            if let Some(m) = m
702                && take
703            {
704                values[v] = src[*m as usize];
705            }
706        }
707        self.set_values(channel, &values);
708    }

711/// Vertices whose bind height (local Y) is in the top (`above`) or bottom
712/// `share` of the mesh.
713pub fn select_by_height(bind: &[Vec3], share: f32, above: bool) -> Vec<bool> {
714    let (lo, hi) = bind.iter().fold((f32::MAX, f32::MIN), |(lo, hi), p| (lo.min(p.y), hi.max(p.y)));
715    let share = share.clamp(0.0, 1.0);
716    bind.iter()
717        .map(|p| if above { p.y >= hi - (hi - lo) * share - 1e-6 } else { p.y <= lo + (hi - lo) * share + 1e-6 })
718        .collect()
719}

721/// Nearest hit of a ray with a triangle list (both faces): (distance, triangle
722/// index, barycentric weights of its three corners).
723pub fn ray_triangles(origin: Vec3, dir: Vec3, positions: &[Vec3], indices: &[u32]) -> Option<(f32, usize, Vec3)> {
724    let mut best: Option<(f32, usize, Vec3)> = None;
725    for (t, tri) in indices.as_chunks::<3>().0.iter().enumerate() {
726        let (Some(a), Some(b), Some(c)) = (positions.get(tri[0] as usize), positions.get(tri[1] as usize), positions.get(tri[2] as usize)) else {
727            continue;
728        };
729        let (e1, e2) = (*b - *a, *c - *a);
730        let p = dir.cross(e2);
731        let det = e1.dot(p);
732        if det.abs() < 1e-12 {
733            continue;
734        }
735        let inv = 1.0 / det;
736        let s = origin - *a;
737        let u = s.dot(p) * inv;
738        if !(0.0..=1.0).contains(&u) {
739            continue;
740        }
741        let q = s.cross(e1);
742        let v = dir.dot(q) * inv;
743        if v < 0.0 || u + v > 1.0 {
744            continue;
745        }
746        let d = e2.dot(q) * inv;
747        if d > 0.0 && best.is_none_or(|(bd, _, _)| d < bd) {
748            best = Some((d, t, Vec3::new(1.0 - u - v, u, v)));
749        }
750    }
751    best
752}
