1//! Independent XPBD fabric model. Constraints accumulate multipliers within a
2//! substep (Macklin et al., 2016); material rest metrics never follow animation.
3//! UV U selects warp, its in-surface perpendicular selects weft. Missing or
4//! degenerate UVs use the first triangle edge instead.
5use std::collections::{BinaryHeap, HashMap};
6use std::cmp::Reverse;
7use glam::{Mat4, Vec2, Vec3};
8use serde::{Deserialize, Serialize};
9use super::MeshParticle;

11#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
12#[serde(default)]
13pub struct ClothFabric {
14    /// Mass per undeformed square metre (kg/m²).
15    pub density: f32,
16    /// In-plane elastic stiffness in N/m (UV U, perpendicular, and shear).
17    pub warp: f32,
18    pub weft: f32,
19    pub shear: f32,
20    /// Angular fold stiffness (N m/rad). Zero disables bending.
21    pub bend: f32,
22    /// Multiplier on fold stiffness after buckling_angle degrees of folding.
23    pub buckling: f32,
24    pub buckling_angle: f32,
25    /// Follow animation with an implicit spring (Hz); zero disables the drive.
26    pub animation_hz: f32,
27    /// Damping ratio of the animation drive; one is critically damped.
28    pub animation_damping: f32,
29    pub drag: f32,
30    pub lift: f32,
31    /// Signed normal pressure (Pa); does not require a Wind Zone.
32    pub pressure: f32,
33    /// Animated normal backstop sphere (m); zero radius disables it.
34    pub backstop_radius: f32,
35    /// Gap allowed behind the animated surface (m).
36    pub backstop_distance: f32,
37    /// Expensive self-contact passes per substep, distributed across the solver
38    /// with one always at the end. Independent of material solver iterations.
39    pub self_collision_passes: u32,
40}
41impl Default for ClothFabric {
42    fn default()->Self { Self {density:0.2,warp:1500.0,weft:1000.0,shear:500.0,bend:0.02,
43        buckling:0.3,buckling_angle:45.0,animation_hz:0.0,animation_damping:1.0,
44        drag:1.0,lift:0.3,pressure:0.0,backstop_radius:0.0,backstop_distance:0.0,self_collision_passes:1} }
45}

60#[derive(Clone, Debug)]
61struct Triangle { ids:[usize;3], warp:Vec3, u:[f32;3], v:[f32;3], area:f32, lambda:[f32;3], material:[f32;3] }
62#[derive(Clone, Debug)]
63struct Hinge {ids:[usize;4], rest:f32, lambda:f32, material:f32}
64#[derive(Clone, Debug, Default)]
65pub(super) struct FabricState {
66    triangles:Vec<Triangle>, hinges:Vec<Hinge>,
67    pub weights:Vec<f32>,
68    /// Mesh-edge geodesic length; disconnected islands cannot borrow pins.
69    pub tethers:Vec<(usize,usize,f32)>,
70    pub normals:Vec<Vec3>,
71    contacts:crate::cloth_contact::ClothContactTopology,
72}
73impl FabricState {
74    pub fn contact_triangles(&self)->Vec<[usize;3]>{self.triangles.iter().map(|t|t.ids).collect()}
75    pub fn build(rest:&[Vec3],owner:&[u32],indices:&[u32],uv:&[Vec2],edges:&[(u32,u32)],pins:&[f32])->Self {
76        let mut out=Self::default();
77        let mut adjacent:HashMap<(usize,usize),Vec<usize>>=HashMap::new();
78        let mut seen=std::collections::HashSet::new();
79        for tri in indices.chunks_exact(3) {
80            let ids=[owner[tri[0]as usize]as usize,owner[tri[1]as usize]as usize,owner[tri[2]as usize]as usize];
81            if ids[0]==ids[1]||ids[1]==ids[2]||ids[0]==ids[2]{continue;}
82            let mut key=ids;key.sort_unstable();if !seen.insert(key){continue;}
83            let e=rest[ids[1]]-rest[ids[0]];let f=rest[ids[2]]-rest[ids[0]];
84            if e.cross(f).length_squared()<1e-16{continue;}
85            let mut warp=e.normalize();
86            if uv.len()==owner.len(){let a=uv[tri[1]as usize]-uv[tri[0]as usize];let b=uv[tri[2]as usize]-uv[tri[0]as usize];let det=a.perp_dot(b);
87                if det.is_finite()&&det.abs()>1e-10 {let u=(e*b.y-f*a.y)/det;if u.is_finite()&&u.length_squared()>1e-16{warp=u.normalize();}}
88            }
89            out.triangles.push(Triangle{ids,warp,u:[0.0;3],v:[0.0;3],area:0.0,lambda:[0.0;3],material:[1.0;3]});
90            for k in 0..3{let(a,b)=(ids[k],ids[(k+1)%3]);adjacent.entry((a.min(b),a.max(b))).or_default().push(ids[(k+2)%3]);}
91        }
92        let mut keys:Vec<_>=adjacent.keys().copied().collect();keys.sort_unstable();
93        for (a,b) in keys {let opposite=&adjacent[&(a,b)];if opposite.len()==2{out.hinges.push(Hinge {ids:[a,b,opposite[0],opposite[1]],rest:0.0,lambda:0.0,material:1.0});}}
94        out.contacts=crate::cloth_contact::ClothContactTopology::new(out.triangles.iter().map(|t|t.ids).collect());
95        out.rebuild_tethers(rest,edges,pins);out
96    }
97    pub fn exclude_seams(&mut self,pairs:&std::collections::HashSet<(u32,u32)>){self.contacts=crate::cloth_contact::ClothContactTopology::new(self.contact_triangles());for &(a,b) in pairs{self.contacts.exclude_pair(a as usize,b as usize);}}
98    pub fn set_material_masks(&mut self,masks:&[[f32;7]]){
99        for t in &mut self.triangles{for channel in 0..3{t.material[channel]=t.ids.iter().map(|&i|masks.get(i).map_or(1.0,|m|m[channel])).sum::<f32>()/3.0;}}
100        for h in &mut self.hinges{h.material=h.ids.iter().map(|&i|masks.get(i).map_or(1.0,|m|m[3])).sum::<f32>()/4.0;}
101    }
102    pub fn rebuild_tethers(&mut self,rest:&[Vec3],edges:&[(u32,u32)],pins:&[f32]) {
103        let mut links=vec![Vec::new();rest.len()];
104        for &(a,b) in edges{let(a,b)=(a as usize,b as usize);let d=rest[a].distance(rest[b]);links[a].push((b,d));links[b].push((a,d));}
105        let mut distance=vec![f32::INFINITY;rest.len()];let mut anchor=vec![usize::MAX;rest.len()];let mut queue=BinaryHeap::new();
106        for i in 0..rest.len(){if pins[i]>=0.999{distance[i]=0.0;anchor[i]=i;queue.push(Reverse((0u32,i,i)));}}
107        while let Some(Reverse((bits,root,i)))=queue.pop(){let d=f32::from_bits(bits);if d!=distance[i]||root!=anchor[i]{continue;}
108            for &(j,length) in &links[i]{let next=d+length;if next<distance[j]||(next==distance[j]&&root<anchor[j]){distance[j]=next;anchor[j]=root;queue.push(Reverse((next.to_bits(),root,j)));}}
109        }
110        self.tethers=(0..rest.len()).filter(|&i|pins[i]<0.999&&anchor[i]!=usize::MAX).map(|i|(i,anchor[i],distance[i])).collect();
111    }
112    pub fn prepare(&mut self,rest:&[Vec3],world:Mat4,p:&[MeshParticle],cfg:&ClothFabric) {
113        let rest:Vec<_>=rest.iter().map(|x|world.transform_point3(*x)).collect();
114        self.weights=vec![0.0;p.len()];self.normals=vec![Vec3::ZERO;p.len()];
115        for t in &mut self.triangles {let [a,b,c]=t.ids;let e=rest[b]-rest[a];let f=rest[c]-rest[a];let cross=e.cross(f);t.area=cross.length()*0.5;
116            let u=world.transform_vector3(t.warp).normalize_or_zero();let v=cross.normalize_or_zero().cross(u).normalize_or_zero();
117            let det=e.dot(u)*f.dot(v)-f.dot(u)*e.dot(v);
118            if det.abs()<1e-12{t.u=[0.0;3];t.v=[0.0;3];continue;}
119            let (ub,uc,vb,vc)=(f.dot(v)/det,-e.dot(v)/det,-f.dot(u)/det,e.dot(u)/det);
120            t.u=[-ub-uc,ub,uc];t.v=[-vb-vc,vb,vc];
121            for i in t.ids{self.weights[i]+=t.area/3.0;}
122        }
123        for (i,w) in self.weights.iter_mut().enumerate(){*w=if *w>1e-10 {(1.0-p[i].pin.clamp(0.0,1.0))/(*w*cfg.density.clamp(0.001,100.0))}else{0.0};}
124        for hinge in &mut self.hinges{hinge.rest=dihedral(hinge.ids.map(|i|rest[i])).map_or(0.0,|(a,_)|a);}
125    }
126    pub fn begin_step(&mut self,target:&[Vec3]) {
127        for t in &mut self.triangles{t.lambda=[0.0;3];}
128        for h in &mut self.hinges{h.lambda=0.0;}
129        self.normals.fill(Vec3::ZERO);
130        for t in &self.triangles {let[a,b,c]=t.ids;let n=(target[b]-target[a]).cross(target[c]-target[a]);for i in t.ids{self.normals[i]+=n;}}
131        for n in &mut self.normals{*n=n.normalize_or_zero();}
132    }
133    pub fn solve(&mut self,p:&mut [MeshParticle],cfg:&ClothFabric,h:f32) {
134        for t in &mut self.triangles {if t.area<=1e-10{continue;}
135            for (channel,stiffness) in [cfg.warp,cfg.weft,cfg.shear].into_iter().enumerate(){let stiffness=stiffness*t.material[channel];if stiffness<=0.0{continue;}
136                let u=(0..3).map(|k|p[t.ids[k]].pos*t.u[k]).sum::<Vec3>();let v=(0..3).map(|k|p[t.ids[k]].pos*t.v[k]).sum::<Vec3>();
137                let (c,g)=match channel {0=>(u.length()-1.0,t.u.map(|w|u.normalize_or_zero()*w)),1=>(v.length()-1.0,t.v.map(|w|v.normalize_or_zero()*w)),_=>(u.dot(v),std::array::from_fn(|k|v*t.u[k]+u*t.v[k]))};
138                project(p,&self.weights,t.ids,g,c,1.0/(stiffness.max(1e-6)*t.area*h*h),&mut t.lambda[channel]);
139            }
140        }
141        if cfg.bend>0.0 {for hinge in &mut self.hinges {
142            if hinge.material<=0.0{continue;}
143            let Some((angle,g))=dihedral(hinge.ids.map(|i|p[i].pos))else{continue};
144            let c=wrap_angle(angle-hinge.rest);
145            let strength=if c.abs()>cfg.buckling_angle.clamp(0.0,180.0).to_radians(){cfg.buckling.clamp(0.001,1.0)}else{1.0};
146            project(p,&self.weights,hinge.ids,g,c,1.0/(cfg.bend*hinge.material*strength*h*h),&mut hinge.lambda);
147        }}
148    }
149    pub fn forces(&self,p:&[MeshParticle],cfg:&ClothFabric,winds:&[crate::cloth::WindSample],wind:f32,time:f32)->Vec<Vec3> {
150        let mut force=vec![Vec3::ZERO;p.len()];
151        for t in &self.triangles {let[a,b,c]=t.ids;let center=(p[a].pos+p[b].pos+p[c].pos)/3.0;let cross=(p[b].pos-p[a].pos).cross(p[c].pos-p[a].pos);let area=cross.length()*0.5;
152            if area<1e-10{continue;}let n=cross.normalize();let velocity=(p[a].vel+p[b].vel+p[c].vel)/3.0;
153            let air=winds.iter().map(|w|w.at(center,time)).sum::<Vec3>()*wind.max(0.0);let relative=(air-velocity).clamp_length_max(100.0);let dir=relative.normalize_or_zero();let cosine=n.dot(dir);
154            // Drag follows relative flow; lift is perpendicular to it. Both
155            // sides of the sheet receive the same force under reversed winding.
156            let q=0.5*1.225*relative.length_squared()*area;
157            let f=q*(cfg.drag.clamp(0.0,10.0)*cosine.abs()*dir+cfg.lift.clamp(0.0,10.0)*cosine*(n-dir*cosine))+n*(cfg.pressure.clamp(-10000.0,10000.0)*area);
158            for i in t.ids{force[i]+=f/3.0;}
159        }
160        force
161    }
162    pub fn recover_intersections(&self,p:&mut[MeshParticle],thickness:f32){
163        let mut positions:Vec<_>=p.iter().map(|p|p.pos).collect();
164        for _ in 0..4{if self.contacts.untangle(&mut positions,&self.weights,thickness)==0{break;}}
165        for (p,position) in p.iter_mut().zip(positions){p.pos=position;}
166    }
167    #[cfg(test)]
168    pub fn self_collide(&self,p:&mut[MeshParticle],previous:&[Vec3],thickness:f32){
169        self.self_collide_swept(p,previous,thickness,true,true);
170    }
171    pub fn self_collide_swept(&self,p:&mut[MeshParticle],previous:&[Vec3],thickness:f32,continuous:bool,untangle:bool){
172        let mut positions:Vec<_>=p.iter().map(|p|p.pos).collect();
173        self.contacts.solve(&mut positions,previous,&self.weights,thickness,continuous,untangle);
174        for (p,position) in p.iter_mut().zip(positions){p.pos=position;}
175    }

178fn project<const N:usize>(p:&mut[MeshParticle],w:&[f32],ids:[usize;N],g:[Vec3;N],c:f32,alpha:f32,lambda:&mut f32){
179    let denom=alpha+(0..N).map(|k|w[ids[k]]*g[k].length_squared()).sum::<f32>();
180    if denom<=1e-12||!denom.is_finite()||!c.is_finite(){return;}
181    let delta=(-c-alpha* *lambda)/denom;*lambda+=delta;
182    for k in 0..N{p[ids[k]].pos+=g[k]*(w[ids[k]]*delta);}
183}
184fn wrap_angle(a:f32)->f32{(a+std::f32::consts::PI).rem_euclid(std::f32::consts::TAU)-std::f32::consts::PI}
185/// Signed fold angle and its analytic derivatives. The two faces use opposite
186/// edge directions, so a flat sheet has angle zero.
187fn dihedral(p:[Vec3;4])->Option<(f32,[Vec3;4])>{
188    let [a,b,c,d]=p;let e=b-a;let l=e.length();let n0=e.cross(c-a);let n1=(d-a).cross(e);
189    if l<1e-8||n0.length_squared()<1e-16||n1.length_squared()<1e-16{return None;}
190    let u=n0.normalize();let v=n1.normalize();let angle=(e/l).dot(u.cross(v)).atan2(u.dot(v));
191    let gc=-n0*(l/n0.length_squared());let gd=-n1*(l/n1.length_squared());
192    let gb=-gc*((c-a).dot(e)/(l*l))-gd*((d-a).dot(e)/(l*l));
193    Some((angle,[-gb-gc-gd,gb,gc,gd]))
194}
