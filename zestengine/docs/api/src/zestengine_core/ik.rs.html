1//! Inverse kinematics for VR full-body tracking: solve a skeleton's joint
2//! rotations so end effectors (head, hands, hips, feet) reach tracker targets.
3//!
4//! Two solvers, matching the VRChat/Unreal IK rig split:
5//! - [`solve_two_bone`]: analytic two-bone IK for limbs (upper arm + forearm →
6//!   hand, thigh + shin → foot), with a pole/hint for elbow/knee direction.
7//!   Exact, single-pass, no iteration — the right tool for a 3-joint limb.
8//! - [`solve_fabrik`]: FABRIK for longer chains (spine: hips → chest → neck →
9//!   head), iterative, handles arbitrary length.
10//!
11//! Both work in world space on joint positions; the caller converts the
12//! resulting positions back into the skeleton's local joint rotations.
13

19/// Full-body IK end-effector targets (world-space pose per limb). The source is
20/// open: VR trackers, gameplay (look-at / hand reach / foot planting), a
21/// cinematic, or a networked remote avatar — any humanoid avatar can be posed
22/// from these. `None` for a limb leaves it at its animated/rest pose.
23///
24/// (Named `TrackerTargets` for the VR path; [`IkTargets`] is the general alias.)
25#[derive(Clone, Copy, Debug, Default, PartialEq)]
26pub struct TrackerTargets {
27    /// World pose for the head; None keeps the animated pose.
28    pub head: Option<(Vec3, Quat)>,
29    /// World pose for the left hand; None keeps the animated pose.
30    pub left_hand: Option<(Vec3, Quat)>,
31    /// World pose for the right hand; None keeps the animated pose.
32    pub right_hand: Option<(Vec3, Quat)>,
33    /// World pose for the hips; None keeps the animated pose.
34    pub hips: Option<(Vec3, Quat)>,
35    /// World pose for the left foot; None keeps the animated pose.
36    pub left_foot: Option<(Vec3, Quat)>,
37    /// World pose for the right foot; None keeps the animated pose.
38    pub right_foot: Option<(Vec3, Quat)>,
39    /// World pose for the chest; None keeps the animated pose.
40    pub chest: Option<(Vec3, Quat)>,
41    /// World pose for the left elbow; None keeps the animated pose.
42    pub left_elbow: Option<(Vec3, Quat)>,
43    /// World pose for the right elbow; None keeps the animated pose.
44    pub right_elbow: Option<(Vec3, Quat)>,
45    /// World pose for the left knee; None keeps the animated pose.
46    pub left_knee: Option<(Vec3, Quat)>,
47    /// World pose for the right knee; None keeps the animated pose.
48    pub right_knee: Option<(Vec3, Quat)>,
49}

51/// Number of tracker roles (the length of [`TrackerTargets::slots`]).
52pub const TRACKER_SLOTS: usize = 11;
53
54/// Tracker role names in slot order.
55pub const TRACKER_ROLES: [&str; TRACKER_SLOTS] = [
56    "Head",
57    "Left hand",
58    "Right hand",
59    "Hips",
60    "Left foot",
61    "Right foot",
62    "Chest",
63    "Left elbow",
64    "Right elbow",
65    "Left knee",
66    "Right knee",
67];
68
69impl TrackerTargets {
70    /// Every role in slot order (head, hands, hips, feet, chest, elbows, knees).
71    pub fn slots(&self) -> [Option<(Vec3, Quat)>; TRACKER_SLOTS] {
72        [
73            self.head,
74            self.left_hand,
75            self.right_hand,
76            self.hips,
77            self.left_foot,
78            self.right_foot,
79            self.chest,
80            self.left_elbow,
81            self.right_elbow,
82            self.left_knee,
83            self.right_knee,
84        ]
85    }
86
87    /// Build from the slot array.
88    pub fn from_slots(s: [Option<(Vec3, Quat)>; TRACKER_SLOTS]) -> Self {
89        Self {
90            head: s[0],
91            left_hand: s[1],
92            right_hand: s[2],
93            hips: s[3],
94            left_foot: s[4],
95            right_foot: s[5],
96            chest: s[6],
97            left_elbow: s[7],
98            right_elbow: s[8],
99            left_knee: s[9],
100            right_knee: s[10],
101        }
102    }
103
104    /// True when any target is set.
105    pub fn any(&self) -> bool {
106        self.slots().iter().any(Option::is_some)
107    }
108
109    /// Every pose moved by the rigid transform `m` (scale is ignored for the
110    /// rotations, applied to the positions).
111    pub fn transformed(&self, m: glam::Mat4) -> Self {
112        let (_, r, _) = m.to_scale_rotation_translation();
113        let mut s = self.slots();
114        for p in s.iter_mut().flatten() {
115            *p = (m.transform_point3(p.0), (r * p.1).normalize());
116        }
117        Self::from_slots(s)
118    }
119}

163/// Full-body IK for a humanoid model: poses it from VR trackers (head, hands,
164/// hips, feet, chest, elbows, knees), a remote player's replicated trackers
165/// or scene objects, on top of its Animator (or rest) pose.
166#[derive(Clone, Serialize, Deserialize)]
167#[serde(default)]
168pub struct HumanoidIk {
169    /// Whether canned locomotion supplies the base pose under tracking.
170    pub locomotion: TrackingLocomotion,
171    /// Where the tracker targets come from.
172    pub source: IkSource,
173    /// Objects source: one object per tracker role, in `TRACKER_ROLES` order.

177    pub tracking_root: ObjectRef,
178    /// 0..1 mix over the incoming pose.
179    pub weight: f32,
180    /// Share of the head's turn the spine follows.
181    pub spine_twist: f32,
182    /// 0..1: shoulders turning toward reaching hands.
183    pub shoulder_rotation: f32,
184    /// Without foot trackers, keep feet planted and step procedurally.
185    pub plant_feet: bool,
186    /// A planted foot steps once this far from where it belongs (m).
187    pub step_distance: f32,
188    /// Or once the body turned this far from it (degrees).
189    pub step_angle: f32,
190    /// Seconds per step.
191    pub step_time: f32,
192    /// Peak step height (m).
193    pub step_height: f32,
194    /// Engine-filled state (not saved).
195    #[serde(skip)]
196    pub runtime: HumanoidIkRuntime,
197}

219/// Whether canned locomotion supplies the base pose under tracking.
220#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
221pub enum TrackingLocomotion {
222    /// Suppress locomotion when hips and both feet have tracker targets.
223    #[default]
224    Auto,
225    /// Locomotion always animates.
226    On,
227    /// Locomotion never animates.
228    Off,
229}
230
231impl TrackingLocomotion {
232    /// Whether canned locomotion should animate given the current targets.
233    pub fn animate(self, targets: Option<&TrackerTargets>) -> bool {
234        match self {
235            Self::On => true,
236            Self::Off => false,
237            Self::Auto => !targets.is_some_and(|t| t.hips.is_some() && t.left_foot.is_some() && t.right_foot.is_some()),
238        }
239    }
240}

262/// One tracker's calibration: the fixed rigid offset from the raw tracker pose to
263/// the avatar bone it drives, captured once while the player holds a known
264/// reference pose (T-pose). A SteamVR/SlimeVR tracker sits at an arbitrary spot on
265/// the limb at an arbitrary orientation; this offset cancels that so the bone
266/// follows the tracker naturally.
267///
268///   offset       = tracker_world⁻¹ · bone_world      (captured at T-pose)
269///   bone_target  = tracker_world   · offset           (every frame after)
270#[derive(Clone, Copy, Debug)]
271pub struct TrackerCalibration {
272    /// The rigid offset from tracker space to bone space.
273    pub offset: glam::Mat4,
274}
275
276impl TrackerCalibration {
277    /// Capture the offset from a tracker's world pose to its bone's world pose in
278    /// the calibration (T-pose) frame.
279    pub fn capture(tracker_world: glam::Mat4, bone_world: glam::Mat4) -> Self {
280        Self {
281            offset: tracker_world.inverse() * bone_world,
282        }
283    }
284
285    /// Apply the calibration to a live tracker pose, yielding the bone's target
286    /// world pose.
287    pub fn apply(&self, tracker_world: glam::Mat4) -> glam::Mat4 {
288        tracker_world * self.offset
289    }
290}

292/// Result of a two-bone solve: world-space mid (elbow/knee) and end positions,
293/// plus the rotations to apply to the root and mid joints.
294#[derive(Clone, Copy, Debug)]
295pub struct TwoBoneSolution {
296    /// Rotation to apply to the root joint.
297    pub root_rotation: Quat,
298    /// Rotation to apply to the mid joint.
299    pub mid_rotation: Quat,
300    /// Solved mid (elbow / knee) position.
301    pub mid_pos: Vec3,
302    /// Solved end (hand / foot) position.
303    pub end_pos: Vec3,
304}
305
306/// Analytic two-bone IK. `root`/`mid`/`end` are the current world joint
307/// positions (e.g. shoulder/elbow/wrist); `target` is where the end should
308/// reach; `pole` biases the bend plane (points roughly where the elbow/knee
309/// should aim). Lengths are taken from the current pose so bone lengths are
310/// preserved. Returns the rotations to apply to the root and mid joints.
311pub fn solve_two_bone(root: Vec3, mid: Vec3, end: Vec3, target: Vec3, pole: Vec3) -> TwoBoneSolution {
312    let l_upper = (mid - root).length().max(1e-5);
313    let l_lower = (end - mid).length().max(1e-5);
314    let to_target = target - root;
315    let mut dist = to_target.length().max(1e-5);
316    // Clamp the reach so the law-of-cosines stays valid (can't over-extend).
317    dist = dist.clamp((l_upper - l_lower).abs() + 1e-4, l_upper + l_lower - 1e-4);
318    let dir = to_target.normalize_or(Vec3::Y);
319
320    // Interior angle at the root (between the upper bone and the root→target line).
321    let cos_root = ((l_upper * l_upper + dist * dist - l_lower * l_lower)
322        / (2.0 * l_upper * dist))
323        .clamp(-1.0, 1.0);
324    let root_angle = cos_root.acos();
325
326    // Bend axis from the pole: the plane containing root→target and the pole hint.
327    let pole_dir = (pole - root).normalize_or(Vec3::Z);
328    let mut axis = dir.cross(pole_dir);
329    if axis.length_squared() < 1e-8 {
330        // Pole colinear with the limb; pick any perpendicular axis.
331        axis = dir.cross(Vec3::Y);
332        if axis.length_squared() < 1e-8 {
333            axis = dir.cross(Vec3::X);
334        }
335    }
336    axis = axis.normalize_or(Vec3::Z);
337
338    // Upper bone direction: rotate the root→target dir by root_angle about axis.
339    let upper_dir = Quat::from_axis_angle(axis, root_angle) * dir;
340    let mid_pos = root + upper_dir * l_upper;
341    let lower_dir = (target - mid_pos).normalize_or(upper_dir);
342    let end_pos = mid_pos + lower_dir * l_lower;
343
344    // Rotations that take the current bone directions to the solved ones.
345    let cur_upper = (mid - root).normalize_or(Vec3::Y);
346    let cur_lower = (end - mid).normalize_or(Vec3::Y);
347    let root_rotation = Quat::from_rotation_arc(cur_upper, upper_dir);
348    let mid_rotation = Quat::from_rotation_arc(cur_lower, lower_dir);
349
350    TwoBoneSolution {
351        root_rotation,
352        mid_rotation,
353        mid_pos,
354        end_pos,
355    }
356}

358/// FABRIK (Forward And Backward Reaching Inverse Kinematics) for a chain of
359/// joint world positions. The first joint is treated as the anchored root; the
360/// last reaches `target`. Bone lengths are preserved from the input. Runs up to
361/// `iterations` passes (8–10 is plenty for a spine). Mutates `joints` in place.
362pub fn solve_fabrik(joints: &mut [Vec3], target: Vec3, iterations: usize) {
363    let n = joints.len();
364    if n < 2 {
365        return;
366    }
367    let lengths: Vec<f32> = (0..n - 1)
368        .map(|i| (joints[i + 1] - joints[i]).length().max(1e-5))
369        .collect();
370    let total: f32 = lengths.iter().sum();
371    let root = joints[0];
372
373    // Target unreachable: stretch the chain straight toward it.
374    if (target - root).length() >= total {
375        let dir = (target - root).normalize_or(Vec3::Y);
376        for i in 1..n {
377            joints[i] = joints[i - 1] + dir * lengths[i - 1];
378        }
379        return;
380    }
381
382    let tol = 1e-3;
383    for _ in 0..iterations {
384        if (joints[n - 1] - target).length() < tol {
385            break;
386        }
387        // Backward pass: end → root, pulling the tip to the target.
388        joints[n - 1] = target;
389        for i in (0..n - 1).rev() {
390            let dir = (joints[i] - joints[i + 1]).normalize_or(Vec3::Y);
391            joints[i] = joints[i + 1] + dir * lengths[i];
392        }
393        // Forward pass: root → end, re-anchoring the root.
394        joints[0] = root;
395        for i in 0..n - 1 {
396            let dir = (joints[i + 1] - joints[i]).normalize_or(Vec3::Y);
397            joints[i + 1] = joints[i] + dir * lengths[i];
398        }
399    }
400}

406    #[test]
407    fn two_bone_reaches_target_in_range() {
408        let root = Vec3::ZERO;
409        let mid = Vec3::new(0.0, -1.0, 0.0);
410        let end = Vec3::new(0.0, -2.0, 0.0);
411        let target = Vec3::new(1.0, -1.0, 0.0);
412        let s = solve_two_bone(root, mid, end, target, Vec3::new(0.0, -1.0, 1.0));
413        // End effector should land very close to the (reachable) target.
414        assert!((s.end_pos - target).length() < 1e-2, "end {:?}", s.end_pos);
415    }

417    #[test]
418    fn calibration_roundtrips_then_follows_tracker() {
419        use glam::{Mat4, Quat};
420        // T-pose: tracker offset from the bone by some rigid transform.
421        let bone = Mat4::from_rotation_translation(Quat::IDENTITY, Vec3::new(0.5, 1.0, 0.0));
422        let tracker =
423            Mat4::from_rotation_translation(Quat::from_rotation_z(0.3), Vec3::new(0.55, 1.1, 0.02));
424        let cal = TrackerCalibration::capture(tracker, bone);
425        // At the calibration pose, applying recovers the bone exactly.
426        let back = cal.apply(tracker);
427        assert!((back.w_axis.truncate() - bone.w_axis.truncate()).length() < 1e-4);
428        // Move the tracker by a known world translation; the bone target moves with it.
429        let moved = Mat4::from_translation(Vec3::new(0.0, 0.2, 0.3)) * tracker;
430        let target = cal.apply(moved);
431        let expected = bone.w_axis.truncate() + Vec3::new(0.0, 0.2, 0.3);
432        assert!((target.w_axis.truncate() - expected).length() < 1e-4);
433    }

435    #[test]
436    fn fabrik_reaches_and_preserves_length() {
437        let mut joints = vec![
438            Vec3::ZERO,
