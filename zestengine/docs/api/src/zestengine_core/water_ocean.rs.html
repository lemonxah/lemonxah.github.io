1//! Deterministic periodic ocean spectrum. The same FFT fields drive visible
2//! displacement, normals, whitecaps and headless surface queries.

4use serde::{Deserialize,Serialize};

7#[derive(Clone,Debug,PartialEq,Serialize,Deserialize)]
8#[serde(default)]
9pub struct OceanSpectrum {
10    pub resolution:u32,
11    pub patch_size:f32,
12    pub wind_speed:f32,
13    pub wind_direction:f32,
14    pub significant_height:f32,
15    pub choppiness:f32,
16    pub depth:f32,
17    pub seed:u32,
18    /// Quantize dispersion to this repeat period; 0 uses free dispersion.
19    pub loop_seconds:f32,
20    pub foam_threshold:f32,
21    pub foam_gain:f32,
22}
23impl Default for OceanSpectrum {
24    fn default()->Self{Self{resolution:64,patch_size:128.0,wind_speed:10.0,wind_direction:20.0,significant_height:1.0,choppiness:0.8,depth:100.0,seed:17,loop_seconds:120.0,foam_threshold:0.65,foam_gain:2.0}}
25}
26impl OceanSpectrum {
27    pub fn sanitized(&self)->Self {
28        let f=|v:f32,d:f32,lo:f32,hi:f32|if v.is_finite(){v.clamp(lo,hi)}else{d};
29        Self{resolution:self.resolution.clamp(16,256).next_power_of_two(),patch_size:f(self.patch_size,128.0,4.0,100000.0),
30            wind_speed:f(self.wind_speed,10.0,0.0,100.0),wind_direction:f(self.wind_direction,20.0,-36000.0,36000.0),
31            significant_height:f(self.significant_height,1.0,0.0,50.0),choppiness:f(self.choppiness,0.8,0.0,2.0),depth:f(self.depth,100.0,0.05,10000.0),
32            loop_seconds:f(self.loop_seconds,120.0,0.0,3600.0),foam_threshold:f(self.foam_threshold,0.65,0.0,1.0),foam_gain:f(self.foam_gain,2.0,0.0,10.0),seed:self.seed}
33    }
34}

36#[derive(Clone,Debug)]
37pub struct OceanFrame {
38    pub n:usize,
39    pub size:f32,
40    pub time:f32,
41    /// Mips, fine to coarse. Two vec4s per texel: displacement xyz/foam,
42    /// then height slopes x/z and horizontal Jacobian. Periodic at edges.
43    pub fields:Vec<[f32;4]>,
44}
45impl OceanFrame {
46    pub fn sample(&self,p:Vec2)->(Vec3,Vec3,f32) {
47        let q=p/self.size*self.n as f32;let b=q.floor();let f=q-b;
48        let at=|dx:i32,dz:i32,lane:usize|{let x=(b.x as i32+dx).rem_euclid(self.n as i32)as usize;let z=(b.y as i32+dz).rem_euclid(self.n as i32)as usize;glam::Vec4::from(self.fields[(z*self.n+x)*2+lane])};
49        let sample=|lane|at(0,0,lane).lerp(at(1,0,lane),f.x).lerp(at(0,1,lane).lerp(at(1,1,lane),f.x),f.y);
50        let d=sample(0);let s=sample(1);
51        (d.truncate(),Vec3::new(-s.x,1.0,-s.y).normalize_or(Vec3::Y),d.w)
52    }
53}

55#[derive(Debug)]
56struct Spectrum {settings:OceanSpectrum,h0:Vec<[f32;2]>,omega:Vec<f32>}
57impl Spectrum {
58    fn new(settings:OceanSpectrum)->Self {
59        let n=settings.resolution as usize;let mut h0=vec![[0.0;2];n*n];let mut omega=vec![0.0;n*n];
60        let mut rng=settings.seed;let mut rand=||{rng=rng.wrapping_mul(1664525).wrapping_add(1013904223);((rng>>8)as f32+0.5)/16777216.0};
61        let angle=settings.wind_direction.to_radians();let wind=Vec2::new(angle.cos(),angle.sin());
62        let length=settings.wind_speed.powi(2)/9.81;
63        let freq=|i:usize|if i<n/2{i as f32}else{i as f32-n as f32};
64        for z in 0..n {for x in 0..n {
65            if x==n/2 || z==n/2 || x==0 && z==0 {continue;}
66            let k=Vec2::new(freq(x),freq(z))*(std::f32::consts::TAU/settings.patch_size);let k2=k.length_squared();let km=k2.sqrt();
67            let directional=k.dot(wind)/km;
68            let p=if length>0.0 {(-1.0/(k2*length*length)).exp()*(-k2*(settings.patch_size/n as f32*0.35).powi(2)).exp()*directional.powi(2)/k2.powi(2)*if directional<0.0{0.15}else{1.0}}else{0.0};
69            let radius=(-2.0*rand().ln()).sqrt();let angle=std::f32::consts::TAU*rand();let(s,c)=angle.sin_cos();
70            h0[z*n+x]=[radius*c*(p*0.5).sqrt(),radius*s*(p*0.5).sqrt()];
71            let w=(9.81*km*(km*settings.depth).tanh()).sqrt();
72            omega[z*n+x]=if settings.loop_seconds>0.0{let base=std::f32::consts::TAU/settings.loop_seconds;(w/base).round()*base}else{w};
73        }}
74        let energy=h0.iter().map(|h|h[0]as f64*h[0]as f64+h[1]as f64*h[1]as f64).sum::<f64>();
75        let gain=if energy>0.0{settings.significant_height as f64*(n*n)as f64/(4.0*(2.0*energy).sqrt())}else{0.0};
76        for h in &mut h0{h[0]=(h[0]as f64*gain)as f32;h[1]=(h[1]as f64*gain)as f32;}
77        Self{settings,h0,omega}
78    }
79    fn frame(&self,time:f32)->OceanFrame {
80        let n=self.settings.resolution as usize;let mut height=vec![[0.0;2];n*n];let mut dx=height.clone();let mut dz=height.clone();
81        let t=if self.settings.loop_seconds>0.0{time.rem_euclid(self.settings.loop_seconds)}else{time};
82        let freq=|i:usize|if i<n/2{i as f32}else{i as f32-n as f32};
83        for z in 0..n {for x in 0..n {
84            let i=z*n+x;let a=self.h0[i];let b=self.h0[((n-z)%n)*n+(n-x)%n];let(s,c)=(self.omega[i]*t).sin_cos();
85            let h=[(a[0]+b[0])*c-(a[1]+b[1])*s,(a[0]-b[0])*s+(a[1]-b[1])*c];height[i]=h;
86            let k=Vec2::new(freq(x),freq(z)).normalize_or_zero()*self.settings.choppiness;
87            dx[i]=[-h[1]*k.x,h[0]*k.x];dz[i]=[-h[1]*k.y,h[0]*k.y];
88        }}
89        for field in [&mut height,&mut dx,&mut dz]{crate::water_ewave::fft2(field,n,true);}
90        let mut fields=Vec::with_capacity(n*n*8/3);let inv=0.5*n as f32/self.settings.patch_size;
91        let derivative=|field:&[[f32;2]],x:usize,z:usize|Vec2::new(field[z*n+(x+1)%n][0]-field[z*n+(x+n-1)%n][0],field[((z+1)%n)*n+x][0]-field[((z+n-1)%n)*n+x][0])*inv;
92        for z in 0..n{for x in 0..n{let i=z*n+x;let h=derivative(&height,x,z);let a=derivative(&dx,x,z);let b=derivative(&dz,x,z);
93            let jac=(1.0+a.x)*(1.0+b.y)-a.y*b.x;
94            // Surface gradient after horizontal displacement (inverse Jacobian).
95            let denom=jac.max(0.1);let slope=Vec2::new((h.x*(1.0+b.y)-h.y*b.x)/denom,((1.0+a.x)*h.y-a.y*h.x)/denom);
96            let foam=((self.settings.foam_threshold-jac)*self.settings.foam_gain).clamp(0.0,1.0);
97            fields.push([dx[i][0],height[i][0],dz[i][0],foam]);fields.push([slope.x,slope.y,jac,slope.length_squared()]);
98        }}
99        crate::water_field::append_mips(&mut fields,n,2);
100        OceanFrame{n,size:self.settings.patch_size,time,fields}
101    }
102}

104#[derive(Debug,Default)]
105pub struct OceanCache {spectrum:Option<Spectrum>,frame:Option<Arc<OceanFrame>>}
106impl OceanCache {
107    pub fn frame(&mut self,settings:&OceanSpectrum,time:f32)->Arc<OceanFrame> {
108        let settings=settings.sanitized();let time=if time.is_finite(){time}else{0.0};
109        if self.spectrum.as_ref().is_none_or(|s|s.settings!=settings){self.spectrum=Some(Spectrum::new(settings));self.frame=None;}
110        if self.frame.as_ref().is_none_or(|f|f.time!=time){self.frame=Some(Arc::new(self.spectrum.as_ref().unwrap().frame(time)));}
111        self.frame.as_ref().unwrap().clone()
112    }
113}
