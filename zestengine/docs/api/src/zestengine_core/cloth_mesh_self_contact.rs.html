4pub(super) fn topology(indices:&[u32],welds:&[usize])->crate::cloth_contact::ClothContactTopology{
5    let mut topology=crate::cloth_contact::ClothContactTopology::new(indices.chunks_exact(3).map(|t|[t[0]as usize,t[1]as usize,t[2]as usize]).collect());
6    let mut groups=HashMap::<usize,Vec<usize>>::new();
7    for (v,&weld)in welds.iter().enumerate(){groups.entry(weld).or_default().push(v);}
8    // UV duplicates share adjacency even if their vertex indices differ.
9    for vertices in groups.values(){for &a in vertices{for &b in vertices{if a!=b{topology.exclude_pair(a,b);}}}}
10    for edge in topology.edges.clone(){for &a in &groups[&welds[edge[0]]]{for &b in &groups[&welds[edge[1]]]{if a!=b{topology.exclude_pair(a,b);}}}}
11    topology
12}

14impl MeshClothSim{
15    pub(super) fn rebuild_surface_contacts(&mut self){
16        self.surface_contacts=topology(&self.indices,&self.normal_owner);
17        for &(a,b)in &self.seam_neighbours{for &a in &self.members[a as usize]{for &b in &self.members[b as usize]{self.surface_contacts.exclude_pair(a as usize,b as usize);}}}
18    }
19    pub(super) fn collide_visible_self(&mut self,cloth:&MeshCloth,sweep:bool,previous:Option<&[Vec3]>,dt:f32){
20        let (mut geometry,_)=self.render_collision_geometry(cloth.fabric.is_some(),self.last_world);
21        let before=geometry.positions.clone();
22        if let Some(previous)=previous.filter(|p|p.len()==before.len()){geometry.previous=previous.to_vec();}
23        if !sweep||geometry.previous.len()!=before.len(){geometry.previous.clone_from(&before);}
24        let thickness=cloth.thickness*crate::cloth::max_scale(self.last_world);
25        // The shared body/self loop owns the iteration budget and refreshes
26        // the sweep after every correction. Do not multiply that budget here.
27        self.surface_contacts.solve_mapped(&mut geometry.positions,&geometry.previous,&geometry.inverse_mass,thickness,&self.surface_mapping,cloth.continuous_collision&&sweep,cloth.untangle);
28        for (p,members)in self.particles.iter_mut().zip(&self.members){
29            if p.pin>=1.0{continue;}
30            let delta=(members.iter().map(|&v|(geometry.positions[v as usize]-before[v as usize]).as_dvec3()).sum::<glam::DVec3>()/members.len()as f64).as_vec3();
31            p.pos+=delta;if dt>1e-6{p.vel=(p.vel+delta/dt).clamp_length_max(30.0);}
32        }
33    }
34}

44            for y in 0..n{for x in 0..n{let a=offset+y*(n+1)+x;indices.extend([a,a+1,a+n+1,a+1,a+n+2,a+n+1]);}}
45        }
46        let upper=((n+1)*(n+1))as usize;let mut animated=rest.clone();animated[upper+40].z=-0.18;
