1//! Shared water mesh policy. The engine builds one adaptive topology per zone;
2//! authored water bodies keep their geometry, material and simulation settings.
3use serde::{Deserialize,Serialize};
4use crate::TypedComponent;
5#[derive(Clone,Debug,Serialize,Deserialize)]
6#[serde(default)]
7pub struct WaterZone {
8    pub half_extents:[f32;2],
9    /// Smallest tile width in zone-local metres.
10    pub tile_size:f32,
11    /// Cells along each tile edge (power of two at runtime).
12    pub tile_quads:u32,
13    /// Split tiles closer than this many tile widths to the camera.
14    /// Runtime minimum 2 keeps successive geometric transition bands separate.
15    pub lod_scale:f32,
16    pub max_lod:u32,
17    /// Higher priority zones own bodies in overlapping zones.
18    pub priority:i32,
19    /// Ocean coverage outside the detailed zone, in local metres (0 disables).
20    pub far_distance:f32,
21}
22impl Default for WaterZone {
23    fn default()->Self {Self {half_extents:[512.0;2],tile_size:16.0,tile_quads:8,lod_scale:2.0,max_lod:6,priority:0,far_distance:20000.0}}
24}
25impl TypedComponent for WaterZone {const NAME:&'static str="Water Zone";}
