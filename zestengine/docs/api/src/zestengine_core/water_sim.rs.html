1//! Interactive water: a height-field wave simulation over a square window of
2//! a Water body that follows the action. Floating bodies and characters push
3//! the surface as they move, and the disturbance travels at the shallow-water
4//! wave speed for the local depth, so they leave ripples and wakes. Cells
5//! marked blocked (posts, walls, rocks through the surface) reflect waves.
6//! The result adds to the body's analytic waves for physics and rendering.
7
8use glam::Vec2;
9
10/// Something disturbing the water this step (water-local XZ).
11#[derive(Clone, Copy, Debug)]
12pub struct WaterPush {
13    /// Where the water is disturbed (water-local XZ).
14    pub at: Vec2,
15    /// Radius of the disturbance (m).
16    pub radius: f32,
17    /// Downward push speed (m/s): vertical motion plus a share of horizontal
18    /// speed for wakes.
19    pub strength: f32,
20}
21
22/// The height-field simulation over one Water's interactive window.
23#[derive(Clone, Debug)]
24pub struct RippleSim {
25    /// Cells per side.
26    pub n: usize,
27    /// Window width (m).
28    pub size: f32,
29    /// Window centre (local XZ), snapped to whole cells.
30    pub center: Vec2,
31    height: Vec<f32>,
32    velocity: Vec<f32>,
33    /// Cells something solid passes through (row-major, empty = none).
34    blocked: Vec<bool>,
35    /// Filled from the GPU simulation's readback (not stepped here).
36    pub gpu: bool,
37    /// Horizontal water velocity (local XZ, m/s) per cell; GPU-fed only.
38    flow: Vec<Vec2>,
39    /// Largest |height| and flow speed in the field (GPU-fed only).
40    pub max_height: f32,
41    /// Largest flow speed in the field (GPU-fed only).
42    pub max_speed: f32,
43}
44
45impl RippleSim {
46    /// An empty simulation window of `n` cells per side over `size` metres.
47    pub fn new(n: usize, size: f32) -> Self {
48        let n = n.clamp(8, 512);
49        Self {
50            n,
51            size: size.max(1.0),
52            center: Vec2::ZERO,
53            height: vec![0.0; n * n],
54            velocity: vec![0.0; n * n],
55            blocked: Vec::new(),
56            gpu: false,
57            flow: Vec::new(),
58            max_height: 0.0,
59            max_speed: 0.0,
60        }
61    }
62
63    /// An empty field owned by the GPU simulation, filled by [`Self::set_field`].
64    pub fn gpu_owned(n: usize, size: f32) -> Self {
65        Self { gpu: true, ..Self::new(n, size) }
66    }
67
68    /// Replace the field with a readback: `n` x `n` cells over `size` metres
69    /// around `center` (local XZ), each (height offset, flow x, flow z).
70    pub fn set_field(&mut self, n: usize, size: f32, center: Vec2, cells: &[[f32; 4]]) {
71        if cells.len() != n * n || n < 2 {
72            return;
73        }
74        self.n = n;
75        self.size = size.max(1e-3);
76        self.center = center;
77        self.height.clear();
78        self.flow.clear();
79        self.velocity.clear();
80        let (mut mh, mut ms) = (0.0f32, 0.0f32);
81        for c in cells {
82            self.height.push(c[0]);
83            let f = Vec2::new(c[1], c[2]);
84            self.flow.push(f);
85            mh = mh.max(c[0].abs());
86            ms = ms.max(f.length());
87        }
88        self.max_height = mh;
89        self.max_speed = ms;
90    }
91
92    /// Simulated horizontal water velocity at local `xz` (zero outside the
93    /// window or without GPU data).
94    pub fn flow(&self, xz: Vec2) -> Vec2 {
95        if self.flow.len() != self.n * self.n {
96            return Vec2::ZERO;
97        }
98        self.sample(xz, |i| self.flow[i]).unwrap_or(Vec2::ZERO)
99    }
100
101    fn sample<T: Copy + std::ops::Add<Output = T> + std::ops::Sub<Output = T> + std::ops::Mul<f32, Output = T>>(
102        &self,
103        xz: Vec2,
104        at: impl Fn(usize) -> T,
105    ) -> Option<T> {
106        let cell = self.cell();
107        let g = (xz - self.center + Vec2::splat(self.size * 0.5)) / cell - Vec2::splat(0.5);
108        let (x0, z0) = (g.x.floor(), g.y.floor());
109        if x0 < 0.0 || z0 < 0.0 || x0 + 1.0 >= self.n as f32 || z0 + 1.0 >= self.n as f32 {
110            return None;
111        }
112        let (fx, fz) = (g.x - x0, g.y - z0);
113        let (x0, z0) = (x0 as usize, z0 as usize);
114        let n = self.n;
115        let v = |x: usize, z: usize| at(z * n + x);
116        let top = v(x0, z0) + (v(x0 + 1, z0) - v(x0, z0)) * fx;
117        let bot = v(x0, z0 + 1) + (v(x0 + 1, z0 + 1) - v(x0, z0 + 1)) * fx;
118        Some(top + (bot - top) * fz)
119    }
120
121    /// The cell size (window width / cells per side, m).
122    pub fn cell(&self) -> f32 {
123        self.size / self.n as f32
124    }
125
126    /// Move the window toward `focus`, shifting the field by whole cells so
127    /// the water already disturbed stays put in the world.
128    pub fn follow(&mut self, focus: Vec2) {
129        let cell = self.cell();
130        let shift = ((focus - self.center) / cell).round();
131        if shift == Vec2::ZERO {
132            return;
133        }
134        let (sx, sz) = (shift.x as i64, shift.y as i64);
135        let n = self.n as i64;
136        let mut h = vec![0.0; self.height.len()];
137        let mut v = vec![0.0; self.velocity.len()];
138        for z in 0..n {
139            for x in 0..n {
140                let (ox, oz) = (x + sx, z + sz);
141                if (0..n).contains(&ox) && (0..n).contains(&oz) {
142                    h[(z * n + x) as usize] = self.height[(oz * n + ox) as usize];
143                    v[(z * n + x) as usize] = self.velocity[(oz * n + ox) as usize];
144                }
145            }
146        }
147        self.height = h;
148        self.velocity = v;
149        self.blocked.clear();
150        self.flow.clear();
151        self.center += shift * cell;
152    }
153
154    /// Local XZ of cell (`x`, `z`)'s centre.
155    pub fn cell_center(&self, x: usize, z: usize) -> Vec2 {
156        let cell = self.cell();
157        self.center - Vec2::splat(self.size * 0.5) + Vec2::new((x as f32 + 0.5) * cell, (z as f32 + 0.5) * cell)
158    }
159
160    /// Mark which cells are solid (row-major `n` x `n`; empty clears).
161    pub fn set_blocked(&mut self, blocked: Vec<bool>) {
162        self.blocked = if blocked.len() == self.n * self.n && blocked.iter().any(|b| *b) { blocked } else { Vec::new() };
163    }
164
165    /// Advance by `dt`. `depth` is the still-water depth (sets the wave
166    /// speed, sqrt(g h)); `damping` the share of motion lost per second.
167    pub fn step(&mut self, dt: f32, depth: f32, damping: f32, pushes: &[WaterPush]) {
168        if dt <= 0.0 {
169            return;
170        }
171        let n = self.n;
172        let cell = self.cell();
173        let half = self.size * 0.5;
174        for p in pushes {
175            let local = p.at - self.center + Vec2::splat(half);
176            let r = p.radius.max(cell);
177            let (lo, hi) = (((local - Vec2::splat(r)) / cell).floor(), ((local + Vec2::splat(r)) / cell).ceil());
178            for z in lo.y.max(0.0) as usize..(hi.y.max(0.0) as usize).min(n) {
179                for x in lo.x.max(0.0) as usize..(hi.x.max(0.0) as usize).min(n) {
180                    let c = Vec2::new((x as f32 + 0.5) * cell, (z as f32 + 0.5) * cell);
181                    let fall = (1.0 - c.distance(local) / r).max(0.0);
182                    self.velocity[z * n + x] -= p.strength * fall * fall * dt * 8.0;
183                }
184            }
185        }
186        let speed = (9.81 * depth.clamp(0.05, 4.0)).sqrt();
187        // Stay inside the explicit scheme's stability limit.
188        let steps = ((speed * dt / (cell * 0.5)).ceil() as usize).clamp(1, 16);
189        let h_dt = dt / steps as f32;
190        let k = speed * speed / (cell * cell);
191        let keep = (1.0 - damping.clamp(0.0, 1.0)).powf(h_dt);
192        let border = (n / 10).max(2);
193        for _ in 0..steps {
194            for z in 0..n {
195                for x in 0..n {
196                    let i = z * n + x;
197                    let here = self.height[i];
198                    let blocked = &self.blocked;
199                    // A solid neighbour mirrors this cell, so waves bounce off it.
200                    let at = |xx: usize, zz: usize| {
201                        let j = zz * n + xx;
202                        if blocked.get(j).copied().unwrap_or(false) { here } else { self.height[j] }
203                    };
204                    let l = if x > 0 { at(x - 1, z) } else { 0.0 };
205                    let r = if x + 1 < n { at(x + 1, z) } else { 0.0 };
206                    let d = if z > 0 { at(x, z - 1) } else { 0.0 };
207                    let u = if z + 1 < n { at(x, z + 1) } else { 0.0 };
208                    let lap = l + r + d + u - 4.0 * self.height[i];
209                    self.velocity[i] = (self.velocity[i] + k * lap * h_dt) * keep;
210                }
211            }
212            for z in 0..n {
213                for x in 0..n {
214                    let i = z * n + x;
215                    // A soft border absorbs waves leaving the window.
216                    let edge = x.min(z).min(n - 1 - x).min(n - 1 - z);
217                    let sponge = if edge < border { 0.85 + 0.15 * edge as f32 / border as f32 } else { 1.0 };
218                    if self.blocked.get(i).copied().unwrap_or(false) {
219                        self.velocity[i] = 0.0;
220                        self.height[i] = 0.0;
221                        continue;
222                    }
223                    self.velocity[i] *= sponge;
224                    self.height[i] = (self.height[i] + self.velocity[i] * h_dt) * sponge;
225                }
226            }
227        }
228    }
229
230    /// Simulated height offset at local `xz` (0 outside the window).
231    pub fn height(&self, xz: Vec2) -> f32 {
232        if self.height.len() != self.n * self.n {
233            return 0.0;
234        }
235        self.sample(xz, |i| self.height[i]).unwrap_or(0.0)
236    }
237
238    /// The raw field (row-major, `n` x `n`), for uploading to the GPU.
239    pub fn heights(&self) -> &[f32] {
240        &self.height
241    }
242}

248    #[test]
249    fn a_push_makes_a_ring_that_spreads_and_fades() {
250        let mut sim = RippleSim::new(96, 24.0);
251        let push = WaterPush { at: Vec2::ZERO, radius: 0.5, strength: 2.0 };
252        for _ in 0..6 {
253            sim.step(1.0 / 60.0, 1.0, 0.5, &[push]);
254        }
255        assert!(sim.height(Vec2::ZERO) < -0.01, "pushed down where it hit: {}", sim.height(Vec2::ZERO));
256        for _ in 0..40 {
257            sim.step(1.0 / 60.0, 1.0, 0.5, &[]);
258        }
259        let ring: f32 = (0..32).map(|k| {
260            let a = k as f32 / 32.0 * std::f32::consts::TAU;
261            sim.height(Vec2::new(a.cos(), a.sin()) * 2.0).abs()
262        }).sum::<f32>() / 32.0;
263        assert!(ring > 1e-4, "the ripple travelled outward: {ring}");
264        for _ in 0..1200 {
265            sim.step(1.0 / 60.0, 1.0, 0.5, &[]);
266        }
267        let left: f32 = sim.heights().iter().map(|h| h.abs()).fold(0.0, f32::max);
268        assert!(left < 0.002, "damping and the border settle it: {left}");
269    }

271    #[test]
272    fn a_wall_reflects_waves() {
273        let run = |wall: bool| {
274            let mut sim = RippleSim::new(64, 16.0);
275            if wall {
276                let mut b = vec![false; 64 * 64];
277                for z in 0..64 {
278                    b[z * 64 + 40] = true;
279                }
280                sim.set_blocked(b);
281            }
282            sim.step(1.0 / 60.0, 1.0, 0.0, &[WaterPush { at: Vec2::new(0.0, 0.0), radius: 0.5, strength: 3.0 }]);
283            for _ in 0..90 {
284                sim.step(1.0 / 60.0, 1.0, 0.0, &[]);
285            }
286            (sim.height(Vec2::new(3.5, 0.0)).abs(), sim.height(Vec2::new(4.6, 0.0)).abs())
287        };
288        let (open_near, _) = run(false);
289        let (walled_near, behind) = run(true);
290        assert!(behind < 1e-6, "nothing passes the wall: {behind}");
291        assert!((walled_near - open_near).abs() > 1e-4, "the reflection changes the water in front of it: {walled_near} vs {open_near}");
292    }

294    #[test]
295    fn following_keeps_disturbances_in_place() {
296        let mut sim = RippleSim::new(64, 16.0);
297        sim.step(1.0 / 60.0, 1.0, 0.0, &[WaterPush { at: Vec2::new(2.0, 1.0), radius: 0.5, strength: 3.0 }]);
298        let before = sim.height(Vec2::new(2.0, 1.0));
299        sim.follow(Vec2::new(3.0, -2.0));
300        let after = sim.height(Vec2::new(2.0, 1.0));
301        assert!(before < 0.0 && (before - after).abs() < 1e-5, "{before} {after}");
302        assert_eq!(sim.height(Vec2::new(100.0, 0.0)), 0.0);
303    }
