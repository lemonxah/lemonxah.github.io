1//! `/avatar/parameters` OSC bridge: the address mapping, the declared-only
2//! input filter and the message encoder, with no sockets (the engine owns
3//! those, see `face_input.rs`).
4//!
5//! The convention is VRChat's, so existing tools work: a parameter `Hat` is
6//! `/avatar/parameters/Hat`; tools send to UDP 9000 and listen on 9001; the
7//! bridge announces the worn avatar with `/avatar/change <id>`. Input is
8//! limited to the parameters the avatar *declares*: engine built-ins
9//! (`IsLocal`, `Grounded`, ...) and unknown names are ignored, so a tool cannot
10//! fake what the engine measures. Each parameter accepts at most 60 writes a
11//! second; the newest value that was held back is released on the next poll,
12//! so a sweep always ends on its last value.
13
14use std::collections::HashMap;
15
16use crate::avatar_params::{AvatarParamKind, AvatarParamStore};
17
18/// The address prefix all parameters live under.
19pub const PARAM_PREFIX: &str = "/avatar/parameters/";
20/// The address announcing the worn avatar.
21pub const CHANGE_ADDRESS: &str = "/avatar/change";
22/// Where tools listen by default.
23pub const DEFAULT_OUT_PORT: u16 = 9001;
24/// Most writes per parameter per second.
25pub const MAX_RATE_HZ: f64 = 60.0;
26
27/// `/avatar/parameters/<name>`.
28pub fn param_address(name: &str) -> String {
29    format!("{PARAM_PREFIX}{name}")
30}
31
32/// The parameter name in an address, if it is one (`None` for other
33/// addresses and for an empty name).
34pub fn param_name(address: &str) -> Option<&str> {
35    address.strip_prefix(PARAM_PREFIX).filter(|n| !n.is_empty())
36}
37
38fn pad(out: &mut Vec<u8>, s: &str) {
39    out.extend_from_slice(s.as_bytes());
40    out.push(0);
41    while !out.len().is_multiple_of(4) {
42        out.push(0);
43    }
44}
45
46/// One OSC message setting `name` to `value`, typed by `kind` (Bool `T` / `F`,
47/// Int `i`, Float `f`).
48pub fn encode_param(name: &str, kind: AvatarParamKind, value: f32) -> Vec<u8> {
49    let mut out = Vec::with_capacity(32 + name.len());
50    pad(&mut out, &param_address(name));
51    match kind {
52        AvatarParamKind::Bool => pad(&mut out, if value != 0.0 { ",T" } else { ",F" }),
53        AvatarParamKind::Int => {
54            pad(&mut out, ",i");
55            out.extend_from_slice(&(value.round() as i32).to_be_bytes());
56        }
57        AvatarParamKind::Float => {
58            pad(&mut out, ",f");
59            out.extend_from_slice(&value.to_be_bytes());
60        }
61    }
62    out
63}
64
65/// `/avatar/change <id>`.
66pub fn encode_avatar_change(avatar_id: &str) -> Vec<u8> {
67    let mut out = Vec::with_capacity(32 + avatar_id.len());
68    pad(&mut out, CHANGE_ADDRESS);
69    pad(&mut out, ",s");
70    pad(&mut out, avatar_id);
71    out
72}

74struct Entry {
75    slot: usize,
76    last_accept: f64,
77    /// A value held back by the rate limit, to release later.
78    pending: Option<f32>,
79}
80
81/// Input filter and output tracker for one worn avatar.
82#[derive(Default)]
83pub struct OscBridge {
84    by_name: HashMap<String, usize>,
85    entries: Vec<Entry>,
86    /// Values last sent out, parallel to `entries`.
87    sent: Vec<f32>,
88}
89
90impl OscBridge {
91    /// A bridge for the declared (non-built-in) parameters of `store`.
92    pub fn new(store: &AvatarParamStore) -> Self {
93        let mut b = Self::default();
94        for slot in 0..store.len() {
95            if store.declared(slot).is_some() {
96                b.by_name.insert(store.name(slot).to_owned(), b.entries.len());
97                b.entries.push(Entry { slot, last_accept: f64::NEG_INFINITY, pending: None });
98                b.sent.push(f32::NAN);
99            }
100        }
101        b
102    }
103
104    /// Number of declared parameters in the bridge.
105    pub fn len(&self) -> usize {
106        self.entries.len()
107    }
108    /// True when the bridge declares nothing.
109    pub fn is_empty(&self) -> bool {
110        self.entries.is_empty()
111    }
112    /// True when the avatar declares a parameter of this name.
113    pub fn declares(&self, name: &str) -> bool {
114        self.by_name.contains_key(name)
115    }
116
117    /// An incoming message: the store slot and the value to write, or `None`
118    /// when the address is not a declared parameter or the rate limit held the
119    /// value back (see [`Self::poll_pending`]).
120    pub fn accept(&mut self, store: &AvatarParamStore, address: &str, value: f32, now: f64) -> Option<(usize, f32)> {
121        let name = param_name(address)?;
122        let &i = self.by_name.get(name)?;
123        if !value.is_finite() {
124            return None;
125        }
126        let e = &mut self.entries[i];
127        if now - e.last_accept < 1.0 / MAX_RATE_HZ {
128            e.pending = Some(value);
129            return None;
130        }
131        e.last_accept = now;
132        e.pending = None;
133        Some((e.slot, store.declared(e.slot)?.coerce(value)))
134    }
135
136    /// Values the rate limit held back whose interval has passed.
137    pub fn poll_pending(&mut self, store: &AvatarParamStore, now: f64, out: &mut Vec<(usize, f32)>) {
138        for e in &mut self.entries {
139            if let Some(v) = e.pending
140                && now - e.last_accept >= 1.0 / MAX_RATE_HZ
141            {
142                e.pending = None;
143                e.last_accept = now;
144                if let Some(p) = store.declared(e.slot) {
145                    out.push((e.slot, p.coerce(v)));
146                }
147            }
148        }
149    }
150
151    /// Declared parameters whose value differs from what was last sent, as
152    /// OSC messages. Marks them sent.
153    pub fn changed_messages(&mut self, store: &AvatarParamStore, out: &mut Vec<Vec<u8>>) {
154        for (i, e) in self.entries.iter().enumerate() {
155            let v = store.get(e.slot);
156            if self.sent[i].to_bits() != v.to_bits() {
157                self.sent[i] = v;
158                out.push(encode_param(store.name(e.slot), store.kind(e.slot), v));
159            }
160        }
161    }
162
163    /// Forget what was sent (a new listener, or the avatar changed): the next
164    /// [`Self::changed_messages`] sends everything.
165    pub fn resend_all(&mut self) {
166        self.sent.iter_mut().for_each(|s| *s = f32::NAN);
167    }
168}

171mod tests {
172    use super::*;
173    use crate::avatar_params::{AvatarParam, AvatarParamSync};

175    fn store() -> AvatarParamStore {
176        AvatarParamStore::new(&[
177            AvatarParam::new("Hat", AvatarParamKind::Bool),
178            AvatarParam::new("Outfit", AvatarParamKind::Int),
179            AvatarParam::new("Smile", AvatarParamKind::Float),

184    #[test]
185    fn addresses_map_both_ways() {
186        assert_eq!(param_address("Hat"), "/avatar/parameters/Hat");
187        assert_eq!(param_name("/avatar/parameters/Hat"), Some("Hat"));
188        assert_eq!(param_name("/avatar/parameters/FT/v2/JawOpen"), Some("FT/v2/JawOpen"), "slashes stay in the name");
189        assert_eq!(param_name("/avatar/parameters/"), None);
190        assert_eq!(param_name("/avatar/change"), None);
191        assert_eq!(param_name("/input/Jump"), None);
192    }
193
194    #[test]
195    fn only_declared_parameters_pass() {
196        let s = store();
197        let mut b = OscBridge::new(&s);
198        assert_eq!(b.len(), 4, "built-ins are not part of the bridge");
199        assert!(b.declares("Hat") && !b.declares("IsLocal") && !b.declares("Grounded"));
200        let (slot, v) = b.accept(&s, "/avatar/parameters/Smile", 0.5, 0.0).unwrap();
201        assert_eq!((s.name(slot), v), ("Smile", 0.5));
202        assert!(b.accept(&s, "/avatar/parameters/IsLocal", 0.0, 0.0).is_none(), "built-ins cannot be faked");
203        assert!(b.accept(&s, "/avatar/parameters/Grounded", 1.0, 0.0).is_none());
204        assert!(b.accept(&s, "/avatar/parameters/Nope", 1.0, 0.0).is_none());
205        assert!(b.accept(&s, "/avatar/change", 1.0, 0.0).is_none());
206        assert!(b.accept(&s, "/avatar/parameters/Hat", f32::NAN, 1.0).is_none());
207    }
208
209    #[test]
210    fn values_are_clamped_to_the_synced_ranges() {
211        let s = store();
212        let mut b = OscBridge::new(&s);
213        assert_eq!(b.accept(&s, "/avatar/parameters/Smile", 7.0, 0.0).unwrap().1, 1.0);
214        assert_eq!(b.accept(&s, "/avatar/parameters/Outfit", 999.0, 0.0).unwrap().1, 255.0);
215        assert_eq!(b.accept(&s, "/avatar/parameters/Outfit", 2.4, 1.0).unwrap().1, 2.0);
216        assert_eq!(b.accept(&s, "/avatar/parameters/Hat", 0.3, 0.0).unwrap().1, 1.0, "any non-zero Bool is true");
217        // Local-only floats keep their full range.
218        assert_eq!(b.accept(&s, "/avatar/parameters/Wetness", 7.0, 0.0).unwrap().1, 7.0);
219    }
220
221    #[test]
222    fn writes_are_limited_to_sixty_a_second_and_the_last_one_is_not_lost() {
223        let s = store();
224        let mut b = OscBridge::new(&s);
225        assert!(b.accept(&s, "/avatar/parameters/Smile", 0.1, 0.000).is_some());
226        assert!(b.accept(&s, "/avatar/parameters/Smile", 0.2, 0.005).is_none());
227        assert!(b.accept(&s, "/avatar/parameters/Smile", 0.3, 0.010).is_none());
228        // Another parameter has its own budget.
229        assert!(b.accept(&s, "/avatar/parameters/Outfit", 1.0, 0.010).is_some());
230        let mut out = Vec::new();
231        b.poll_pending(&s, 0.012, &mut out);
232        assert!(out.is_empty(), "too early");
233        b.poll_pending(&s, 0.020, &mut out);
234        assert_eq!(out.len(), 1);
235        assert_eq!(out[0].1, 0.3, "the newest held value wins");
236        out.clear();
237        b.poll_pending(&s, 0.5, &mut out);
238        assert!(out.is_empty(), "released once");
239    }

241    #[test]
242    fn messages_encode_with_the_right_type_tag() {
243        let m = encode_param("Hat", AvatarParamKind::Bool, 1.0);
244        assert_eq!(&m[..24], b"/avatar/parameters/Hat\0\0");
245        assert_eq!(&m[24..28], b",T\0\0");
246        assert_eq!(m.len(), 28);
247        assert_eq!(&encode_param("Hat", AvatarParamKind::Bool, 0.0)[24..28], b",F\0\0");
248        let i = encode_param("Outfit", AvatarParamKind::Int, 3.0);
249        assert_eq!(i.len() % 4, 0);
250        assert_eq!(&i[i.len() - 4..], &3i32.to_be_bytes());
251        let f = encode_param("Smile", AvatarParamKind::Float, 0.25);
252        assert_eq!(&f[f.len() - 4..], &0.25f32.to_be_bytes());
253        let c = encode_avatar_change("abc");
254        assert!(c.starts_with(b"/avatar/change\0\0"));
255        assert_eq!(&c[c.len() - 4..], b"abc\0");
256    }

258    #[test]
259    fn only_changed_parameters_go_out() {
260        let mut s = store();
261        let mut b = OscBridge::new(&s);
262        let mut out = Vec::new();
263        b.changed_messages(&s, &mut out);
264        assert_eq!(out.len(), 4, "the first pass sends every declared parameter");
265        out.clear();
266        b.changed_messages(&s, &mut out);
267        assert!(out.is_empty());
268        let hat = s.index_of("Hat").unwrap();
269        s.set(hat, 1.0);
270        b.changed_messages(&s, &mut out);
271        assert_eq!(out.len(), 1);
272        assert!(out[0].starts_with(b"/avatar/parameters/Hat"));
273        b.resend_all();
274        out.clear();
275        b.changed_messages(&s, &mut out);
276        assert_eq!(out.len(), 4);
277    }
