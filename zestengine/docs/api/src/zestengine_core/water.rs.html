107/// How a reflective surface chooses its reflections: the environment's
108/// global technique (the scene's Reflection setting), the surface's own
109/// planar mirror render (water, or an object with a Planar Reflection
110/// component), baked probes only (no SSR / ray-traced resolve), or none.
111#[derive(Clone, Copy, Debug, PartialEq, Eq, Default, Serialize, Deserialize)]
112pub enum SurfaceReflection {
113    /// The scene's global reflection technique (Environment -> Reflections).
114    #[default]
115    Environment,
116    /// A planar mirror render of the scene. Water renders its own; a mesh
117    /// uses its Planar Reflection component.
118    Planar,
119    /// Baked reflection probes only: the deferred SSR / ray-traced resolve
120    /// skips this surface.
121    Probes,
122    /// No reflections at all (the surface stays matte).
123    Off,
124}
125
126impl SurfaceReflection {
127    /// The u8 the shader's fx.parallax.w carries (0 env, 1 planar, 2 probes, 3 off).
128    pub fn from_u8(v: u8) -> Self {
129        match v {
130            1 => Self::Planar,
131            2 => Self::Probes,
132            3 => Self::Off,
133            _ => Self::Environment,
134        }
135    }

389/// Depth baked where nothing lies under the water (m): open water.
390pub const SHORE_NO_GROUND: f32 = 1.0e4;
391
392/// The depth of water over the ground under a body, on an `n` x `n` grid of
393/// cells spanning local XZ -`half`..`half` (row-major, x fastest; cell
394/// centres at the texel centres the shader samples), clamped to
395/// 0..`SHORE_NO_GROUND`.
396#[derive(Clone, Debug, PartialEq)]
397pub struct ShoreDepth {
398    /// Grid cells per side.
399    pub n: usize,
400    /// Grid half-extent (m).
401    pub half: f32,
402    /// Depth per cell (row-major, x fastest).
403    pub depth: Vec<f32>,
404    /// Changes with every bake (what the renderer re-uploads on).
405    pub version: u64,
406}
407
408impl ShoreDepth {
409    /// Build a grid, clamping NaN cells to open water and bumping the bake
410    /// version so the renderer re-uploads it.
411    pub fn new(n: usize, half: f32, mut depth: Vec<f32>) -> Self {
412        static VERSION: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(1);
413        for d in &mut depth {
414            *d = if d.is_nan() { SHORE_NO_GROUND } else { d.clamp(0.0, SHORE_NO_GROUND) };
415        }
416        Self { n, half, depth, version: VERSION.fetch_add(1, std::sync::atomic::Ordering::Relaxed) }
417    }
418
419    /// The depth under local `xz`, bilinear between cell centres (as the
420    /// GPU filters it); None outside the grid.
421    pub fn depth_at(&self, xz: Vec2) -> Option<f32> {
422        if self.n == 0 || self.half <= 0.0 || self.depth.len() != self.n * self.n {
423            return None;
424        }
425        let uv = xz / (2.0 * self.half) + 0.5;
426        if !(0.0..=1.0).contains(&uv.x) || !(0.0..=1.0).contains(&uv.y) {
427            return None;
428        }
429        let n = self.n;
430        let p = (uv * n as f32 - 0.5).clamp(Vec2::ZERO, Vec2::splat((n - 1) as f32));
431        let (x0, z0) = (p.x.floor() as usize, p.y.floor() as usize);
432        let (x1, z1) = ((x0 + 1).min(n - 1), (z0 + 1).min(n - 1));
433        let f = p - Vec2::new(x0 as f32, z0 as f32);
434        let at = |x: usize, z: usize| self.depth[z * n + x];
435        let a = at(x0, z0) + (at(x1, z0) - at(x0, z0)) * f.x;
436        let b = at(x0, z1) + (at(x1, z1) - at(x0, z1)) * f.x;
437        Some(a + (b - a) * f.y)
438    }
439}

441/// Share of the waves left over water `depth` deep for a body's
442/// `shore_depth`: smoothstep(0, shore_depth, depth); 1 when it is off.
443pub fn shore_damping(depth: f32, shore_depth: f32) -> f32 {
444    if shore_depth <= 0.0 {
445        return 1.0;
446    }
447    let t = (depth / shore_depth).clamp(0.0, 1.0);
448    t * t * (3.0 - 2.0 * t)
449}

457/// Samples per edge of a smooth outline.
458const OUTLINE_SAMPLES: usize = 8;
459
460/// The closed Catmull-Rom curve through `pts`, `OUTLINE_SAMPLES` per edge.
461fn smooth_outline(pts: &[[f32; 2]]) -> Vec<[f32; 2]> {
462    let n = pts.len();
463    let at = |k: usize| Vec2::from(pts[k % n]);
464    let mut out = Vec::with_capacity(n * OUTLINE_SAMPLES);
465    for k in 0..n {
466        let (p0, p1, p2, p3) = (at(k + n - 1), at(k), at(k + 1), at(k + 2));
467        for s in 0..OUTLINE_SAMPLES {
468            let t = s as f32 / OUTLINE_SAMPLES as f32;
469            let (t2, t3) = (t * t, t * t * t);
470            let p = 0.5 * ((2.0 * p1) + (p2 - p0) * t + (2.0 * p0 - 5.0 * p1 + 4.0 * p2 - p3) * t2 + (3.0 * p1 - p0 - 3.0 * p2 + p3) * t3);
471            out.push(p.to_array());
472        }
473    }
474    out
475}

507impl FluidPreset {
508    /// Set `water`'s fluid and look to this template (its shape, waves and

511        water.density = self.density;
512        water.viscosity = self.viscosity;
513        water.steam = self.steam;
514        let l = &mut water.look;
515        l.shallow_color = self.shallow;
516        l.deep_color = self.deep;
517        l.absorption_depth = self.absorption;
518        l.clarity = self.clarity;
519        l.roughness = self.roughness;
520        l.ripple_strength = self.ripples;
521        l.foam = self.foam;
522        l.refraction = self.refraction;
523        l.optics = Some(crate::WaterOptics::from_legacy(l));
524    }

526    /// Whether `water`'s fluid is this one's (density and viscosity).
527    pub fn matches(&self, water: &Water) -> bool {
528        (water.density - self.density).abs() < 0.5 && (water.viscosity - self.viscosity).abs() <= self.viscosity * 0.01
529    }

631/// Centripetal Catmull-Rom through `p1`..`p2` (no cusps or loops).
632fn catmull_rom(p0: Vec3, p1: Vec3, p2: Vec3, p3: Vec3, t: f32) -> Vec3 {
633    let knot = |a: Vec3, b: Vec3| a.distance(b).max(1e-4).sqrt();
634    let (t0, t1) = (0.0, knot(p0, p1));
635    let t2 = t1 + knot(p1, p2);
636    let t3 = t2 + knot(p2, p3);
637    let u = t1 + (t2 - t1) * t;
638    let lerp = |a: Vec3, b: Vec3, ta: f32, tb: f32| a * ((tb - u) / (tb - ta)) + b * ((u - ta) / (tb - ta));
639    let a1 = lerp(p0, p1, t0, t1);
640    let a2 = lerp(p1, p2, t1, t2);
641    let a3 = lerp(p2, p3, t2, t3);
642    let b1 = lerp(a1, a2, t0, t2);
643    let b2 = lerp(a2, a3, t1, t3);
644    lerp(b1, b2, t1, t2)
645}

647/// The river's centre line at `param` (segment index plus fraction).
648fn river_curve(points: &[RiverPoint], param: f32) -> Vec3 {
649    let n = points.len();
650    let at = |i: usize| Vec3::from(points[i].position);
651    if n < 2 {
652        return points.first().map_or(Vec3::ZERO, |p| Vec3::from(p.position));
653    }
654    let k = (param.floor() as usize).min(n - 2);
655    let t = (param - k as f32).clamp(0.0, 1.0);
656    let (p1, p2) = (at(k), at(k + 1));
657    if points[k].linear{return p1.lerp(p2,t);}
658    if t <= 0.0 {
659        return p1;
660    }
661    if t >= 1.0 {
662        return p2;
663    }
664    let p0 = if k == 0 { p1 * 2.0 - p2 } else { at(k - 1) };
665    let p3 = if k + 2 >= n { p2 * 2.0 - p1 } else { at(k + 2) };
666    if points[k].leave_tangent.is_some() || points[k + 1].arrive_tangent.is_some() {
667        let (m1, m2) = river_segment_tangents(points, k);
668        return p1 * (2.0*t*t*t - 3.0*t*t + 1.0) + m1 * (t*t*t - 2.0*t*t + t)
669            + p2 * (-2.0*t*t*t + 3.0*t*t) + m2 * (t*t*t - t*t);
670    }
671    catmull_rom(p0, p1, p2, p3, t)
672}
673
674fn river_segment_tangents(points: &[RiverPoint], k: usize) -> (Vec3, Vec3) {
675    let at = |i: usize| Vec3::from(points[i].position);
676    let (p1, p2) = (at(k), at(k + 1));
677    if points[k].linear{return(p2-p1,p2-p1);}
678    let p0 = if k == 0 { p1 * 2.0 - p2 } else { at(k - 1) };
679    let p3 = if k + 2 >= points.len() { p2 * 2.0 - p1 } else { at(k + 2) };
680    let safe = |v: Option<[f32; 3]>, fallback: Vec3| v.map(Vec3::from).filter(|v| v.is_finite()).unwrap_or(fallback);
681    // Automatic derivatives agree with the existing centripetal curve at
682    // either end, even when only one of the two handles is authored.
683    let knot=|a:Vec3,b:Vec3|a.distance(b).max(1e-4).sqrt();
684    let(dt0,dt1,dt2)=(knot(p0,p1),knot(p1,p2),knot(p2,p3));
685    let a=dt1*((p1-p0)/dt0-(p2-p0)/(dt0+dt1)+(p2-p1)/dt1);
686    let b=dt1*((p2-p1)/dt1-(p3-p1)/(dt1+dt2)+(p3-p2)/dt2);
687    (safe(points[k].leave_tangent, a), safe(points[k + 1].arrive_tangent, b))
688}

841    /// The waves this surface shows: the authored ones plus the wind sea, with
842    /// the depth-aware dispersion (waves slow down in shallow water).
843    pub fn resolved_waves(&self) -> Vec<ResolvedWave> {
844        // (direction, wavelength, amplitude, steepness, speed)
845        let authored = self.profile_waves.as_deref().unwrap_or(&self.waves);
846        let mut list: Vec<(Vec2, f32, f32, f32, f32)> = authored
847            .iter()
848            .map(|w| {
849                let a = w.direction.to_radians();
850                (Vec2::new(a.cos(), a.sin()), w.wavelength, w.amplitude, w.steepness, w.speed)
851            })
852            .collect();
853        let u = self.runtime.wind_speed * self.wind_response.max(0.0);
854        if u > 0.05 && self.runtime.wind_dir.length_squared() > 0.5 {
855            // A developed wind sea: peak wavelength and height grow with the
856            // square of the wind speed, limited by the fetch (the water's
857            // size) and by depth (waves break past ~0.78 of the depth).
858            let mut peak = 0.833 * u * u;
859            let fetch = match (self.size[0] > 0.0, self.size[1] > 0.0) {
860                (true, true) => self.size[0].min(self.size[1]),
861                (true, false) => self.size[0],
862                (false, true) => self.size[1],
863                (false, false) => f32::MAX,
864            };
865            peak = peak.min(fetch * 0.5).max(0.2);
866            let height = (0.0107 * u * u).min(0.39 * self.depth.max(0.05)).min(peak / 14.0);
867            let base = self.runtime.wind_dir.y.atan2(self.runtime.wind_dir.x);
868            for (angle, len, amp) in [(0.0f32, 1.0f32, 1.0f32), (25.0, 0.62, 0.5), (-30.0, 0.41, 0.3), (60.0, 0.27, 0.15)] {
869                let a = base + angle.to_radians();
870                list.push((Vec2::new(a.cos(), a.sin()), peak * len, height * amp, 0.55, 1.0));
871            }
872        }
873        list.truncate(Self::MAX_WAVES);
874        let n = list.len();
875        let depth = self.depth.max(0.05);
876        list.into_iter()
877            .map(|(dir, wavelength, amp, steep, speed)| {
878                let k = std::f32::consts::TAU / wavelength.max(0.05);
879                let omega = (GRAVITY * k * (k * depth).tanh()).sqrt() * speed;
880                let amp = amp * self.wave_share(k);
881                // Steepness is shared out so the summed crests never loop over.
882                let qa = (steep.clamp(0.0, 1.0) / (k * n.max(1) as f32)).min(amp.max(0.0));
883                ResolvedWave { dir, k, omega, qa, amplitude: amp.max(0.0) }
884            })
885            .collect()
886    }

993    /// How readily the fluid splashes (0..1): 1 / (1 + 3 Oh), Oh the
994    /// Ohnesorge number mu / sqrt(rho sigma L) at a 10 cm splash. Water

1013    /// Share of a wave of wave number `k` (rad/m) the fluid keeps (0..1): a
1014    /// wave lives only where wind feeds it (growth about 3% of its angular
1015    /// frequency, Miles' order for a strong wind) faster than viscosity
1016    /// drains it (2 nu k^2 in the body, plus the bed's boundary layer).
1017    /// 1 for water at every size; thick fluids keep only long waves.
1018    pub fn wave_share(&self, k: f32) -> f32 {
1019        let nu = self.kinematic_viscosity();
1020        if nu <= 0.0 {
1021            return 1.0;
1022        }
1023        let depth = self.depth.max(0.05);
1024        let kh = (k * depth).min(10.0);
1025        let omega = ((GRAVITY * k + 7.3e-5 * k * k * k) * kh.tanh()).max(1e-6).sqrt();
1026        let bulk = 2.0 * nu * k * k;
1027        let bed = k * (nu * omega * 0.5).sqrt() / (2.0 * kh).sinh().max(1e-3);
1028        (1.0 - (bulk + bed) / (WAVE_GROWTH * omega)).clamp(0.0, 1.0)
1029    }

1283                for(end,connection)in self.resolved_connections.iter().enumerate(){if let Some(connection)=connection {

1299                let mut connection=*connection;connection.length=connection.length.min(length*0.5).max(0.01);

1308    /// The river segment nearest local `xz`: (segment, unclamped position
1309    /// along it, horizontal distance to it).
1310    pub fn river_nearest_segment(&self, xz: Vec2) -> Option<(usize, f32, f32)> {
1311        let path = self.river_path();
1312        let last = path.len().checked_sub(2)?;
1313        let mut best: Option<(usize, f32, f32)> = None;
1314        for (i, pair) in path.windows(2).enumerate() {
1315            let a = Vec2::new(pair[0].pos.x, pair[0].pos.z);
1316            let seg = Vec2::new(pair[1].pos.x, pair[1].pos.z) - a;
1317            let len2 = seg.length_squared();
1318            let t = if len2 > 1e-8 { (xz - a).dot(seg) / len2 } else { 0.0 };
1319            let dist = (a + seg * t.clamp(0.0, 1.0)).distance(xz);
1320            if best.is_none_or(|(_, _, d)| dist < d) {
1321                best = Some((i, t, dist));
1322            }
1323        }
1324        let (i, t, dist) = best?;
1325        let (pa, pb) = (path[i].param, path[i + 1].param);
1326        // Past either end the fraction runs outside 0..1.
1327        let param = if (i == 0 && t < 0.0) || (i == last && t > 1.0) { pa + (pb - pa) * t } else { pa + (pb - pa) * t.clamp(0.0, 1.0) };
1328        let k = (param.max(0.0).floor() as usize).min(self.river.len().saturating_sub(2));
1329        Some((k, param - k as f32, dist))
1330    }

1361    /// Split a cubic without changing it or its neighbours. Automatic curves
1362    /// become explicit cubics to preserve their shape. Returns the new index.
1363    pub fn split_river_segment(&mut self, segment: usize, t: f32) -> Option<usize> {
1364        if !t.is_finite() { return None; }
1365        let t = t.clamp(0.001, 0.999);
1366        let mut point = self.river_point_on_segment(segment, t)?;
1367        // Freeze neighbouring automatic cubics before inserting another knot;
1368        // otherwise their Catmull-Rom neighbourhood would change as a side effect.
1369        let tangents:Vec<_>=(0..self.river.len().saturating_sub(1)).map(|i|river_segment_tangents(&self.river,i)).collect();
1370        for(i,(a,b))in tangents.into_iter().enumerate(){if !self.river[i].linear{
1371            self.river[i].leave_tangent=Some(a.to_array());self.river[i+1].arrive_tangent=Some(b.to_array());
1372        }}
1373        if !self.river[segment].linear {
1374            let (a, b) = (Vec3::from(self.river[segment].position), Vec3::from(self.river[segment + 1].position));
1375            let (m1, m2) = river_segment_tangents(&self.river, segment);
1376            let derivative = a * (6.0*t*t - 6.0*t) + m1 * (3.0*t*t - 4.0*t + 1.0)
1377                + b * (-6.0*t*t + 6.0*t) + m2 * (3.0*t*t - 2.0*t);
1378            self.river[segment].leave_tangent = Some((m1 * t).to_array());
1379            self.river[segment + 1].arrive_tangent = Some((m2 * (1.0 - t)).to_array());
1380            point.arrive_tangent = Some((derivative * t).to_array());
1381            point.leave_tangent = Some((derivative * (1.0 - t)).to_array());
1382        }
1383        self.river.insert(segment + 1, point);
1384        Some(segment + 1)
1385    }

1387    /// Reverse downstream order, preserving the curve and exchanging endpoint
1388    /// links. Positive flow now follows the reversed direction.
1389    pub fn reverse_river(&mut self){
1390        let linear:Vec<_>=self.river.iter().map(|p|p.linear).collect();
1391        self.river.reverse();let n=self.river.len();
1392        for(i,p)in self.river.iter_mut().enumerate(){
1393            let(a,b)=(p.arrive_tangent,p.leave_tangent);p.arrive_tangent=b.map(|v|(-Vec3::from(v)).to_array());p.leave_tangent=a.map(|v|(-Vec3::from(v)).to_array());
1394            p.linear=if i+1<n{linear[n-2-i]}else{false};
1395        }
1396        self.connections.swap(0,1);self.resolved_connections=[None;2];self.connected_surfaces=[None,None];
1397    }

1399    pub fn river_tangent(&self, point: usize, incoming: bool) -> Vec3 {
1400        if self.river.len() < 2 || point >= self.river.len() { return Vec3::X; }
1401        if incoming && point > 0 { river_segment_tangents(&self.river, point - 1).1 }
1402        else if point + 1 < self.river.len() { river_segment_tangents(&self.river, point).0 }
1403        else { river_segment_tangents(&self.river, point - 1).1 }
1404    }

1406    /// A new river point at `index` (see [`Self::river_insert_index`]) placed at
1407    /// local `position`, its width, depth and flow taken from where it lands.
1408    pub fn insert_river_point(&mut self, index: usize, position: Vec3) {
1409        let n = self.river.len();
1410        let index = index.min(n);
1411        let template = if index == 0 || index == n {
1412            self.river.get(index.min(n.saturating_sub(1))).copied().unwrap_or_default()
1413        } else {
1414            let (a, b) = (self.river[index - 1], self.river[index]);
1415            let (a2, b2) = (Vec2::new(a.position[0], a.position[2]), Vec2::new(b.position[0], b.position[2]));
1416            let seg = b2 - a2;
1417            let t = if seg.length_squared() > 1e-8 { (Vec2::new(position.x, position.z) - a2).dot(seg) / seg.length_squared() } else { 0.5 };
1418            self.river_point_on_segment(index - 1, t).unwrap_or(a)
1419        };
1420        self.river.insert(index, RiverPoint { position: position.to_array(), ..template });
1421    }

1436fn build_river_path(points: &[RiverPoint]) -> Vec<RiverSample> {
1437    let mut out: Vec<RiverSample> = Vec::new();
1438    for (k, pair) in points.windows(2).enumerate() {
1439        let (a, b) = (pair[0], pair[1]);
1440        let chord = Vec2::new(b.position[0] - a.position[0], b.position[2] - a.position[2]).length();
1441        let step = ((a.width + b.width) * 0.25).clamp(0.25, 4.0);
1442        let curve_length = if a.leave_tangent.is_some() || b.arrive_tangent.is_some() {
1443            let (ta, tb) = river_segment_tangents(points, k);
1444            // Cubic Bezier control-polygon length bounds the curved arc, so
1445            // short chords with long tangents still receive enough samples.
1446            let (p0, p3) = (Vec3::from(a.position), Vec3::from(b.position));
1447            let (p1, p2) = (p0 + ta / 3.0, p3 - tb / 3.0);
1448            p0.distance(p1) + p1.distance(p2) + p2.distance(p3)
1449        } else { chord * 1.1 };
1450        let steps = ((curve_length / step).ceil() as usize).clamp(1, 4096);
1451        for s in (if k == 0 { 0 } else { 1 })..=steps {
1452            let t = s as f32 / steps as f32;
1453            let mix = |x: f32, y: f32| x + (y - x) * t;
1454            let param = if s == steps { (k + 1) as f32 } else { k as f32 + t };
1455            let pos = river_curve(points, param);
1456            let dist = out.last().map_or(0.0, |p| p.dist + Vec2::new(pos.x - p.pos.x, pos.z - p.pos.z).length());
1457            out.push(RiverSample { pos, width: mix(a.width, b.width).max(0.01), depth: mix(a.depth, b.depth), speed: mix(a.speed, b.speed), dir: Vec3::X, dist, param, flow_offset:Vec3::ZERO });
1458        }
1459    }
1460    // Tangent from the curve itself, so ends and corners point true.
1461    let end = points.len().saturating_sub(1) as f32;
1462    for p in &mut out {
1463        let (a, b) = ((p.param - 1e-3).max(0.0), (p.param + 1e-3).min(end));
1464        p.dir = (river_curve(points, b) - river_curve(points, a)).normalize_or(Vec3::X);
1465    }
1466    out
1467}

1469/// The river column at `xz` from the path segments `segments` (index of
1470/// each segment's first sample), the nearest one wins. Distances are ranked
1471/// in 1/65536 metre bins; effectively coincident segments keep authoring order
1472/// instead of switching height/flow with CPU/GPU rounding at a crossing.
1473/// Keep this ranking in sync with water_coverage.glsl.
1474fn river_column(path: &[RiverSample], segments: impl Iterator<Item = usize>, xz: Vec2, current: Vec3) -> Option<WaterColumn> {
1475    let mut best: Option<(f32, WaterColumn)> = None;
1476    for k in segments {
1477        let (a, b) = (path[k], path[k + 1]);
1478        let (a2, b2) = (Vec2::new(a.pos.x, a.pos.z), Vec2::new(b.pos.x, b.pos.z));
1479        let seg = b2 - a2;
1480        let len2 = seg.length_squared();
1481        let t = if len2 > 1e-8 { ((xz - a2).dot(seg) / len2).clamp(0.0, 1.0) } else { 0.0 };
1482        let dist = (a2 + seg * t).distance(xz);
1483        let rank = (dist * 65536.0 + 0.5).floor();
1484        let half = (a.width + (b.width - a.width) * t).max(0.01) * 0.5;
1485        if dist > half || best.as_ref().is_some_and(|(d, _)| *d <= rank) {
1486            continue;
1487        }
1488        let mix = |x: f32, y: f32| x + (y - x) * t;
1489        let col = WaterColumn {
1490            surface: mix(a.pos.y, b.pos.y),
1491            depth: mix(a.depth, b.depth).max(0.0),
1492            flow: a.dir.lerp(b.dir, t).normalize_or(Vec3::X) * mix(a.speed, b.speed) + current + a.flow_offset.lerp(b.flow_offset,t),
1493            edge: dist / half,
1494        };
1495        best = Some((rank, col));
1496    }
1497    best.map(|(_, c)| c)
1498}

1500/// Even-odd point in polygon test.
1501fn point_in_polygon(p: Vec2, poly: &[[f32; 2]]) -> bool {
1502    let mut inside = false;
1503    let mut j = poly.len() - 1;
1504    for i in 0..poly.len() {
1505        let (a, b) = (Vec2::from(poly[i]), Vec2::from(poly[j]));
1506        if (a.y > p.y) != (b.y > p.y) && p.x < (b.x - a.x) * (p.y - a.y) / (b.y - a.y) + a.x {
1507            inside = !inside;
1508        }
1509        j = i;
1510    }
1511    inside
1512}
