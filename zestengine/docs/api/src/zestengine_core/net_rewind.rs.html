1//! Lag compensation: the host's short history of networked hitboxes, so a
2//! shot can be tested against where targets were when the shooter saw them.
3//!
4//! The host records every networked object that has a hit shape (Character
5//! Controller capsule, Sphere / Capsule / Box Collider) each frame, stamped
6//! with the session clock. A client firing sends `ctx.net_view_time()` (or
7//! the host derives it from the shooter's latency with
8//! `ctx.ray_cast_rewound`), and the host tests the ray against the poses at
9//! that time with `ctx.ray_cast_at`. Rewinds are capped at [`MAX_REWIND`].
10//! Only the recorded hitboxes are rewound, not world geometry: compare the
11//! hit distance with a normal `ctx.raycast` for walls in the way.
12

19/// How far back a shot may be rewound (seconds).
20pub const MAX_REWIND: f64 = 0.5;
21/// Seconds of history kept per object.
22const KEEP: f64 = 1.0;

24/// A hit volume in the object's local space.
25#[derive(Clone, Copy, Debug, PartialEq)]
26pub enum HitShape {
27    /// A sphere.
28    Sphere {
29        /// Centre in object-local space.
30        center: Vec3,
31        /// Radius.
32        radius: f32,
33    },
34    /// Segment `a`..`b` swept by `radius`.
35    Capsule {
36        /// Segment start in object-local space.
37        a: Vec3,
38        /// Segment end in object-local space.
39        b: Vec3,
40        /// Radius of the sweep.
41        radius: f32,
42    },
43    /// An axis-aligned box.
44    Box {
45        /// Centre in object-local space.
46        center: Vec3,
47        /// Half extents.
48        half: Vec3,
49    },
50}

65/// A rewound hit.
66#[derive(Clone, Copy, Debug, PartialEq)]

84/// Recent poses of every hittable networked object (host side).
85#[derive(Clone, Debug)]
86pub struct HitHistory {
87    tracks: HashMap<ObjectId, Track>,
88    /// Time of the newest record.
89    pub now: f64,
90    /// How far back a rewind may go (never more than [`HARD_MAX_REWIND`](crate::HARD_MAX_REWIND)).
91    max_rewind: f64,
92    /// Seconds a timestamp may be ahead of the clock.
93    future_tolerance: f64,
94    /// Reject out-of-window timestamps instead of clamping them (dedicated servers).
95    strict: bool,
96    /// Rewind queries rejected for a timestamp from the future, older than
97    /// the cap, or not a number (strict mode).
98    rejected: std::cell::Cell<u64>,
99}

101impl Default for HitHistory {
102    fn default() -> Self {
103        Self { tracks: HashMap::new(), now: 0.0, max_rewind: MAX_REWIND, future_tolerance: 0.1, strict: false, rejected: Default::default() }
104    }
105}

107/// Why a rewind timestamp was refused.
108#[derive(Clone, Copy, Debug, PartialEq, Eq)]
109pub enum RewindReject {
110    /// The timestamp was not a finite number.
111    NotANumber,
112    /// The timestamp is ahead of the server clock.
113    Future,
114    /// The timestamp is older than the rewind window.
115    TooOld,
116}
117
118/// Check a client-supplied rewind time against the server clock `now`: it
119/// must be a finite number, not more than `future_tolerance` ahead of `now`
120/// and not older than `max_rewind` (itself capped below one second).
121pub fn check_rewind_time(now: f64, time: f64, max_rewind: f64, future_tolerance: f64) -> Result<f64, RewindReject> {
122    let cap = max_rewind.clamp(0.0, crate::HARD_MAX_REWIND as f64);
123    if !time.is_finite() {
124        Err(RewindReject::NotANumber)
125    } else if time > now + future_tolerance {
126        Err(RewindReject::Future)
127    } else if time < now - cap {
128        Err(RewindReject::TooOld)
129    } else {
130        Ok(time.min(now))
131    }
132}

134impl HitHistory {
135    /// Empty the history and restart the clock at `now` (session clock).
136    /// A host taking over after a migration has none of the old host's
137    /// recorded hitboxes, so its lag-compensation history starts fresh at
138    /// the handover moment; any poses from an earlier stint as host are
139    /// dropped so rewound ray casts never serve stale hitboxes.
140    pub fn restart(&mut self, now: f64) {
141        self.tracks.clear();
142        self.now = now;
143    }
144
145    /// Record an object's pose (call once per frame per object).
146    pub fn record(&mut self, id: ObjectId, time: f64, pos: Vec3, rot: Quat, scale: Vec3, shapes: &[HitShape]) {
147        self.now = self.now.max(time);
148        let t = self.tracks.entry(id).or_default();
149        if t.shapes != shapes {
150            t.shapes = shapes.to_vec();
151        }
152        if t.poses.back().is_some_and(|p| p.time >= time) {
153            t.poses.pop_back();
154        }
155        t.poses.push_back(HitPose { time, pos, rot, scale });
156        while t.poses.front().is_some_and(|p| p.time < time - KEEP) {
157            t.poses.pop_front();
158        }
159    }
160
161    /// Forget objects not recorded for a while (despawned, or no longer hittable).
162    pub fn prune(&mut self, now: f64) {
163        self.tracks.retain(|_, t| t.poses.back().is_some_and(|p| now - p.time < KEEP));
164    }
165
166    /// Set the rewind window (clamped to `0..=HARD_MAX_REWIND`) and whether
167    /// timestamps outside it are rejected (`strict`) or clamped into it.
168    pub fn configure(&mut self, max_rewind: f32, future_tolerance: f32, strict: bool) {
169        let m = if max_rewind.is_finite() { max_rewind } else { MAX_REWIND as f32 };
170        self.max_rewind = (m.clamp(0.0, crate::HARD_MAX_REWIND)) as f64;
171        self.future_tolerance = if future_tolerance.is_finite() { future_tolerance.clamp(0.0, 1.0) as f64 } else { 0.1 };
172        self.strict = strict;
173    }
174
175    /// The rewind window in seconds.
176    pub fn max_rewind(&self) -> f64 {
177        self.max_rewind
178    }
179
180    /// Rewind queries refused so far (strict mode).
181    pub fn rejected(&self) -> u64 {
182        self.rejected.get()
183    }
184
185    /// Objects being tracked.
186    pub fn len(&self) -> usize {
187        self.tracks.len()
188    }
189
190    /// True when nothing is tracked.
191    pub fn is_empty(&self) -> bool {
192        self.tracks.is_empty()
193    }
194
195    /// An object's pose at `time` (interpolated; clamped to the recorded span).
196    pub fn pose_at(&self, id: ObjectId, time: f64) -> Option<HitPose> {
197        sample(&self.tracks.get(&id)?.poses, time)
198    }
199
200    /// First recorded hitbox the ray hits, with every object posed as it was at `time`.
201    /// A `time` outside the rewind window is clamped into it, or (strict mode:
202    /// dedicated servers) refused with no hit, so a client cannot ask for a
203    /// timestamp from the future or from before the cap.
204    pub fn ray_cast_at(&self, time: f64, origin: Vec3, dir: Vec3, max_dist: f32, exclude: &[ObjectId]) -> Option<RewindHit> {
205        let time = if self.strict {
206            match check_rewind_time(self.now, time, self.max_rewind, self.future_tolerance) {
207                Ok(t) => t,
208                Err(_) => {
209                    self.rejected.set(self.rejected.get() + 1);
210                    return None;
211                }
212            }
213        } else if time.is_finite() {
214            time.clamp(self.now - self.max_rewind, self.now)
215        } else {
216            return None;
217        };
218        self.cast(time, origin, dir, max_dist, exclude)
219    }
220
221    /// Like [`ray_cast_at`](Self::ray_cast_at) but always clamps the time into
222    /// the window: for times the server derived itself (round trip and view delay).
223    pub fn ray_cast_clamped(&self, time: f64, origin: Vec3, dir: Vec3, max_dist: f32, exclude: &[ObjectId]) -> Option<RewindHit> {
224        if !time.is_finite() {
225            return None;
226        }
227        self.cast(time.clamp(self.now - self.max_rewind, self.now), origin, dir, max_dist, exclude)
228    }
229
230    fn cast(&self, time: f64, origin: Vec3, dir: Vec3, max_dist: f32, exclude: &[ObjectId]) -> Option<RewindHit> {
231        let dir = dir.normalize_or_zero();
232        if dir == Vec3::ZERO {
233            return None;
234        }
235        let mut best: Option<RewindHit> = None;
236        for (id, track) in &self.tracks {
237            if exclude.contains(id) {
238                continue;
239            }
240            let Some(pose) = sample(&track.poses, time) else {
241                continue;
242            };
243            for shape in &track.shapes {
244                if let Some(d) = ray_shape(origin, dir, shape, &pose)
245                    && d <= max_dist
246                    && best.is_none_or(|b| d < b.distance)
247                {
248                    best = Some(RewindHit { object: *id, point: origin + dir * d, distance: d, time });
249                }
250            }
251        }
252        best
253    }
254}

256fn sample(poses: &VecDeque<HitPose>, time: f64) -> Option<HitPose> {
257    let newest = *poses.back()?;
258    if time >= newest.time {
259        return Some(newest);
260    }
261    let i = poses.iter().position(|p| p.time > time)?;
262    if i == 0 {
263        return Some(poses[0]);
264    }
265    let (a, b) = (poses[i - 1], poses[i]);
266    let t = ((time - a.time) / (b.time - a.time).max(1e-9)) as f32;
267    Some(HitPose { time, pos: a.pos.lerp(b.pos, t), rot: a.rot.slerp(b.rot, t), scale: a.scale.lerp(b.scale, t) })
268}

270fn ray_shape(o: Vec3, d: Vec3, shape: &HitShape, pose: &HitPose) -> Option<f32> {
271    let world = |p: Vec3| pose.pos + pose.rot * (p * pose.scale);
272    let s = pose.scale.abs().max_element();
273    match *shape {
274        HitShape::Sphere { center, radius } => ray_sphere(o, d, world(center), radius * s),
275        HitShape::Capsule { a, b, radius } => ray_capsule(o, d, world(a), world(b), radius * s),
276        HitShape::Box { center, half } => {
277            let inv = pose.rot.inverse();
278            ray_box(inv * (o - world(center)), inv * d, half * pose.scale.abs())
279        }
280    }
281}

283fn ray_sphere(o: Vec3, d: Vec3, c: Vec3, r: f32) -> Option<f32> {
284    let oc = o - c;
285    let b = oc.dot(d);
286    let cc = oc.dot(oc) - r * r;
287    if cc <= 0.0 {
288        return Some(0.0);
289    }
290    let h = b * b - cc;
291    if h < 0.0 {
292        return None;
293    }
294    let t = -b - h.sqrt();
295    (t >= 0.0).then_some(t)
296}
297
298fn ray_capsule(o: Vec3, d: Vec3, pa: Vec3, pb: Vec3, r: f32) -> Option<f32> {
299    let ba = pb - pa;
300    let oa = o - pa;
301    let baba = ba.dot(ba);
302    if baba < 1e-10 {
303        return ray_sphere(o, d, pa, r);
304    }
305    let bard = ba.dot(d);
306    let baoa = ba.dot(oa);
307    let a = baba - bard * bard;
308    let mut best = None;
309    if a.abs() > 1e-8 {
310        let b = baba * d.dot(oa) - baoa * bard;
311        let c = baba * oa.dot(oa) - baoa * baoa - r * r * baba;
312        let h = b * b - a * c;
313        if h >= 0.0 {
314            let t = (-b - h.sqrt()) / a;
315            let y = baoa + t * bard;
316            if y > 0.0 && y < baba && t >= 0.0 {
317                best = Some(t);
318            }
319        }
320    }
321    // End caps (also covers rays parallel to the axis and origins inside).
322    for cap in [ray_sphere(o, d, pa, r), ray_sphere(o, d, pb, r)].into_iter().flatten() {
323        if best.is_none_or(|b| cap < b) {
324            best = Some(cap);
325        }
326    }
327    best
328}

330fn ray_box(o: Vec3, d: Vec3, half: Vec3) -> Option<f32> {
331    let mut tmin = 0.0f32;
332    let mut tmax = f32::INFINITY;
333    for k in 0..3 {
334        if d[k].abs() < 1e-9 {
335            if o[k].abs() > half[k] {
336                return None;
337            }
338            continue;
339        }
340        let inv = 1.0 / d[k];
341        let (t1, t2) = ((-half[k] - o[k]) * inv, (half[k] - o[k]) * inv);
342        tmin = tmin.max(t1.min(t2));
343        tmax = tmax.min(t1.max(t2));
344        if tmin > tmax {
345            return None;
346        }
347    }
348    Some(tmin)
349}

351impl crate::ecs::ObjectContext<'_, '_> {
352    /// The session clock in seconds: the host's time as this peer estimates it.
353    ///
354    /// **Immediate.** Every peer reads nearly the same value at the same moment, so it can time-stamp events across the
355    /// network. It never jumps or runs backwards, not even across a host migration. It is `0.0` offline and on a client
356    /// that has not heard from the host yet.
357    pub fn net_time(&self) -> f64 {
358        self.net_view().clock
359    }
360
361    /// The session-clock time of the world state this peer is showing for remote objects right now. Send it with a shot
362    /// so the host can test the hit against what the shooter actually saw.
363    ///
364    /// Remote objects are drawn behind real time by the one-way transit delay plus the interpolation delay, so this is
365    /// `net_time() - rtt / 2 - view_delay`, in seconds.
366    ///
367    /// **Immediate.** A read of the network view. Outside a session it is `0.0`.
368    ///
369    /// [`NetValue`](crate::NetValue) has no `f64`. Sending this as a `Float` (`f32`) loses precision on long sessions (about 1 ms after
370    /// three hours, 8 ms after a day), so send the eight bytes instead, as in the example. The host reads it back with
371    /// [`ray_cast_at`](Self::ray_cast_at), or ignores it and uses [`ray_cast_rewound`](Self::ray_cast_rewound).
372    ///
373    /// # Examples
374    ///
375    /// ```ignore
376    /// // Client: say where and when the shot was fired from.
377    /// let seen_at = NetValue::Bytes(ctx.net_view_time().to_le_bytes().to_vec());
378    /// ctx.send_rpc(None, "Fire", vec![origin.into(), dir.into(), seen_at], RpcTarget::Server, true);
379    /// ```
380    pub fn net_view_time(&self) -> f64 {
381        self.net_view().clock - self.net_view().rtt_ms as f64 / 2000.0 - self.net_view().view_delay as f64
382    }
383
384    /// Host only: tests a ray against the networked hitboxes posed as they were at session-clock `time` (lag
385    /// compensation), ignoring this object.
386    ///
387    /// **Immediate.** A read of the host's recorded hitbox history. It returns the nearest [`RewindHit`] (object, point,
388    /// distance, the time tested) within `max_dist` metres of `origin` along `dir`. `dir` need not be normalized, but a
389    /// zero vector hits nothing.
390    ///
391    /// # Returns
392    ///
393    /// `None` when nothing is hit, when called offline or on a client (only the host records a history), when `dir` is
394    /// zero, or when `time` is not a finite number. On a dedicated server, or with strict validation on, `None` also
395    /// covers a `time` outside the rewind window (from the future, or too old): it is refused and counted instead of
396    /// being clamped. Elsewhere such a `time` is clamped into the window. The window is the scene's
397    /// [`NetValidation::max_rewind`](crate::NetValidation::max_rewind) (0.25 s by default, never more than
398    /// [`HARD_MAX_REWIND`](crate::HARD_MAX_REWIND)); [`MAX_REWIND`] is only the default of a bare [`HitHistory`].
399    ///
400    /// # What is rewound
401    ///
402    /// Only hitboxes: networked objects ([`NetworkObject`](crate::NetworkObject) with `sync_transform` on) that have a Character Controller
403    /// capsule or a non-trigger Sphere, Capsule or Box Collider. The host records them once per frame, after the frame's
404    /// systems have run, with the pose the other peers are shown. World geometry and physics bodies are not rewound:
405    /// compare `distance` with [`raycast`](Self::raycast) to see whether a wall was in the way.
406    ///
407    /// # Gotchas
408    ///
409    /// Only this object is ignored, not the shooter's avatar. A ray that starts inside a hitbox hits it at (or very
410    /// near) distance 0, so exclude the shooter's own objects with
411    /// [`ray_cast_at_filtered`](Self::ray_cast_at_filtered), or use [`ray_cast_rewound`](Self::ray_cast_rewound), which
412    /// works out the time on the server and excludes everything the shooter owns. `time`, `origin` and `dir` normally
413    /// come from a client: check them against what the shooter's avatar could have done.
414    ///
415    /// # Examples
416    ///
417    /// ```ignore
418    /// // Host: the client sent "Fire" with origin, direction and its `net_view_time` (8 bytes).
419    /// let shots: Vec<(Vec3, Vec3, f64)> = ctx.rpcs_named("Fire").filter_map(|rpc| {
420    ///     let (origin, dir) = (rpc.arg(0)?.as_vec3()?, rpc.arg(1)?.as_vec3()?);
421    ///     let seen_at = f64::from_le_bytes(rpc.arg(2)?.as_bytes()?.try_into().ok()?);
422    ///     Some((origin, dir, seen_at))
423    /// }).collect();
424    /// for (origin, dir, seen_at) in shots {
425    ///     let hit = ctx.ray_cast_at(seen_at, origin, dir, 100.0);
426    ///     let wall = ctx.raycast(origin, dir, 100.0);   // geometry is not rewound
427    ///     if let Some(hit) = hit.filter(|h| wall.is_none_or(|w| w.distance > h.distance)) {
428    ///         self.apply_damage(hit.object, 10);
429    ///     }
430    /// }
431    /// ```
432    pub fn ray_cast_at(&self, time: f64, origin: Vec3, dir: Vec3, max_dist: f32) -> Option<RewindHit> {
433        self.ray_cast_at_filtered(time, origin, dir, max_dist, &[self.self_id()])
434    }
435
436    /// Host only: like [`ray_cast_at`](Self::ray_cast_at), but ignoring exactly the objects in `exclude` instead of just
437    /// this object.
438    ///
439    /// **Immediate.** Use it to leave out the shooter's own hitboxes (its avatar and whatever it carries). `exclude` is
440    /// the whole list: this object is not added to it. The `time` handling (clamped into the rewind window, or refused
441    /// with `None` on a dedicated server), what is rewound, and the `None` cases (offline, not the host, nothing hit) are
442    /// the same as for [`ray_cast_at`](Self::ray_cast_at); the query itself is [`HitHistory::ray_cast_at`].
443    pub fn ray_cast_at_filtered(&self, time: f64, origin: Vec3, dir: Vec3, max_dist: f32, exclude: &[ObjectId]) -> Option<RewindHit> {
444        self.net_view().rewind.as_ref()?.lock().ok()?.ray_cast_at(time, origin, dir, max_dist, exclude)
445    }
446
447    /// Host only: tests a ray against the hitboxes as `shooter` saw the world when it fired, with the time worked out by
448    /// the server.
449    ///
450    /// The time is the session clock now, minus the shooter's measured round trip ([`PeerInfo::rtt_ms`](crate::PeerInfo::rtt_ms)), minus the view
451    /// delay it reported ([`PeerInfo::view_delay`](crate::PeerInfo::view_delay)). It is always clamped into the rewind window and never refused, so a
452    /// client cannot ask for a time of its own. This object and everything the shooter owns (its player and the props it
453    /// holds) are ignored, so a shot cannot hit its own shooter.
454    ///
455    /// **Immediate.** A read of the host's recorded hitbox history. It returns the nearest [`RewindHit`] within `max_dist`
456    /// metres of `origin` along `dir`.
457    ///
458    /// # Returns
459    ///
460    /// `None` when nothing is hit, when called offline or on a client (only the host records a history), when `dir` is
461    /// zero, and when `shooter` is not a peer in the session (its latency is unknown). What is rewound, and what is not,
462    /// is described on [`ray_cast_at`](Self::ray_cast_at): hitboxes only, so compare the distance with
463    /// [`raycast`](Self::raycast) for walls. `origin` and `dir` normally come from the client; on a dedicated server
464    /// check them against the shooter's avatar before trusting them.
465    ///
466    /// # Examples
467    ///
468    /// ```ignore
469    /// // Host: a client asked to fire. `rpc.from` is the connection the RPC arrived on, so it cannot be forged.
470    /// let shots: Vec<(u64, Vec3, Vec3)> = ctx.rpcs_named("Fire")
471    ///     .filter_map(|rpc| Some((rpc.from, rpc.arg(0)?.as_vec3()?, rpc.arg(1)?.as_vec3()?)))
472    ///     .collect();
473    /// for (shooter, origin, dir) in shots {
474    ///     let hit = ctx.ray_cast_rewound(shooter, origin, dir, 100.0);
475    ///     let wall = ctx.raycast(origin, dir, 100.0);   // geometry is not rewound
476    ///     if let Some(hit) = hit.filter(|h| wall.is_none_or(|w| w.distance > h.distance)) {
477    ///         self.apply_damage(hit.object, 10);
478    ///     }
479    /// }
480    /// ```
481    pub fn ray_cast_rewound(&self, shooter: u64, origin: Vec3, dir: Vec3, max_dist: f32) -> Option<RewindHit> {
482        let peer = self.net_view().peer(shooter)?;
483        let time = self.net_view().clock - peer.rtt_ms as f64 / 1000.0 - peer.view_delay as f64;
484        let mut exclude: Vec<ObjectId> = self.net_view().owners.iter().filter(|(_, o)| **o == shooter).map(|(id, _)| *id).collect();
485        exclude.push(self.self_id());
486        // The time is the server's own estimate: clamp it into the window instead of refusing.
487        self.net_view().rewind.as_ref()?.lock().ok()?.ray_cast_clamped(time, origin, dir, max_dist, &exclude)
488    }
489}

495    #[test]
496    fn rewind_times_are_checked_against_the_window() {
497        use crate::RewindReject::*;
498        assert_eq!(check_rewind_time(10.0, 9.9, 0.25, 0.1), Ok(9.9));
499        assert_eq!(check_rewind_time(10.0, 10.05, 0.25, 0.1), Ok(10.0), "a little ahead is clamped to now");
500        assert_eq!(check_rewind_time(10.0, 10.5, 0.25, 0.1), Err(Future));
501        assert_eq!(check_rewind_time(10.0, 9.7, 0.25, 0.1), Err(TooOld));
502        assert_eq!(check_rewind_time(10.0, f64::NAN, 0.25, 0.1), Err(NotANumber));
503        // A cap above the hard limit is cut to it: 0.95 s ago is too old even for a "5 s" window.
504        assert_eq!(check_rewind_time(10.0, 9.05, 5.0, 0.1), Err(TooOld));
505        assert!(check_rewind_time(10.0, 9.2, 5.0, 0.1).is_ok());
506        let mut h = HitHistory::default();
507        h.configure(5.0, 0.1, true);
508        assert!(h.max_rewind() < 1.0, "the history window is capped below a second");
509        h.configure(f32::NAN, 0.1, false);
510        assert_eq!(h.max_rewind(), MAX_REWIND, "a nonsense setting keeps the default");
511    }

513    #[test]
514    fn rewound_ray_hits_where_the_target_was() {
515        let mut h = HitHistory::default();
516        let id = ObjectId::new();
517        let capsule = [HitShape::Capsule { a: Vec3::new(0.0, 0.35, 0.0), b: Vec3::new(0.0, 1.45, 0.0), radius: 0.35 }];
518        // The target runs along +X at 5 m/s, recorded at 60 Hz for a second.
519        for k in 0..=60 {
520            let t = k as f64 / 60.0;
521            h.record(id, t, Vec3::new(5.0 * t as f32, 0.0, 0.0), Quat::IDENTITY, Vec3::ONE, &capsule);
522        }
523        // A shooter at z = 10 aiming at x = 3 (where the target was at t = 0.6).
524        let origin = Vec3::new(3.0, 1.0, 10.0);
525        let hit = h.ray_cast_at(0.6, origin, Vec3::NEG_Z, 100.0, &[]).expect("hit in the past");
526        assert_eq!(hit.object, id);
527        assert!((hit.distance - (10.0 - 0.35)).abs() < 1e-3, "{hit:?}");
528        assert!(h.ray_cast_at(1.0, origin, Vec3::NEG_Z, 100.0, &[]).is_none(), "the target has moved on by now");
529        assert!(h.ray_cast_at(0.6, origin, Vec3::NEG_Z, 100.0, &[id]).is_none(), "excluded");
530        // Rewinds are clamped to MAX_REWIND: t = 0.1 is tested at 0.5.
531        let at = h.ray_cast_at(0.1, Vec3::new(2.5, 1.0, 10.0), Vec3::NEG_Z, 100.0, &[]).expect("clamped time");
532        assert!((at.time - 0.5).abs() < 1e-9);
533        assert!(h.pose_at(id, 0.25).is_some_and(|p| (p.pos.x - 1.25).abs() < 1e-4), "interpolated pose");
534        h.prune(5.0);
535        assert!(h.is_empty());
536    }

539    fn shapes() {
540        let pose = HitPose { time: 0.0, pos: Vec3::new(0.0, 0.0, -5.0), rot: Quat::from_rotation_y(0.7), scale: Vec3::ONE };
541        let b = HitShape::Box { center: Vec3::ZERO, half: Vec3::splat(0.5) };
542        assert!(ray_shape(Vec3::ZERO, Vec3::NEG_Z, &b, &pose).is_some_and(|d| d > 4.2 && d < 4.6));
543        assert!(ray_shape(Vec3::new(2.0, 0.0, 0.0), Vec3::NEG_Z, &b, &pose).is_none());
544        let s = HitShape::Sphere { center: Vec3::ZERO, radius: 1.0 };
545        assert!(ray_shape(Vec3::ZERO, Vec3::NEG_Z, &s, &pose).is_some_and(|d| (d - 4.0).abs() < 1e-4));
546        // A capsule seen end-on (ray along its axis) is hit at its cap.
547        let c = HitShape::Capsule { a: Vec3::new(0.0, 0.0, -1.0), b: Vec3::new(0.0, 0.0, 1.0), radius: 0.5 };
548        let p0 = HitPose { rot: Quat::IDENTITY, ..pose };
549        assert!(ray_shape(Vec3::ZERO, Vec3::NEG_Z, &c, &p0).is_some_and(|d| (d - 3.5).abs() < 1e-4));
550    }
