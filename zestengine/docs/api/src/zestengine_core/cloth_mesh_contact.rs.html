6impl MeshClothSim {
7    /// Body and self constraints can oppose each other. Sweep each body repair
8    /// through self-contact on the next pass, with a bounded convergence budget.
9    pub(super) fn collide_visible_constraints(&mut self,cloth:&MeshCloth,queries:&[ClothColliderQuery<'_>],previous:Option<&[Vec3]>,check_self:bool,dt:f32){
10        if !check_self||!cloth.self_collision{
11            self.collide_visible_surface(cloth,queries,previous,dt);return;
12        }
13        let mut history=previous.map(<[Vec3]>::to_vec);
14        let stationary:Vec<_>=queries.iter().map(|q|ClothColliderMotion::stationary(q.end().clone())).collect();
15        let final_queries:Vec<_>=stationary.iter().map(ClothColliderMotion::prepare).collect();
16        let (_,_,iterations)=cloth.lod.settings(cloth.runtime.lod_level,cloth.particle_spacing,cloth.frequency,cloth.iterations);
17        let tolerance=(cloth.thickness*crate::cloth::max_scale(self.last_world)*0.005).max(1e-6);
18        for pass in 0..iterations.clamp(1,8)*4{
19            let before=self.visible_collision_positions();
20            self.collide_visible_self(cloth,history.is_some(),history.as_deref(),dt);
21            let separated=self.visible_collision_positions();
22            // Only the first pass sweeps the moving body over the frame. Later
23            // passes sweep new self corrections against its final pose; using
24            // the original interval again freezes obsolete contact features.
25            self.collide_visible_surface(cloth,if pass==0{queries}else{&final_queries},if pass==0{previous}else{Some(&before)},dt);
26            let repaired=self.visible_collision_positions();
27            let residual=before.iter().zip(&separated).chain(separated.iter().zip(&repaired)).map(|(a,b)|a.distance_squared(*b)).fold(0.0,f32::max);
28            history=Some(separated);
29            if residual<=tolerance*tolerance{break;}
30        }
31    }
32    pub(super) fn has_collision_detail(&self)->bool{
33        self.particles.len()!=self.vertex_count||!self.transition_offsets.is_empty()
34    }
35    pub(super) fn visible_collision_positions(&self)->Vec<Vec3>{self.collision_render_positions(self.last_world)}
36    pub fn collision_render_positions(&self,world:Mat4)->Vec<Vec3>{
37        let fallback;
38        let anim=if self.anim.len()==self.particles.len(){&self.anim}else{fallback=self.animated(&self.animated_vertices,self.last_world);&fallback};
39        self.animated_vertices.iter().enumerate().map(|(v,&p)|{
40            let owner=self.owner[v]as usize;
41            world.transform_point3(p)+self.particles[owner].pos-anim[owner]+world.transform_vector3(self.transition_offset(v))
42        }).collect()
43    }
44    /// Visible collision topology and the translational driver of each vertex.
45    /// This keeps shared garment contacts faithful at every simulation LOD.
46    pub fn render_collision_geometry(&self,physical:bool,world:Mat4)->(ClothCollisionGeometry,Vec<usize>){
47        let drivers:Vec<_>=self.owner.iter().map(|&i|i as usize).collect();
48        let previous=if self.visible_frame_previous.len()==self.vertex_count{self.visible_frame_previous.clone()}else{drivers.iter().map(|&i|self.frame_previous[i]).collect()};
49        let inverse_mass=drivers.iter().map(|&i|if physical&&self.fabric.weights.len()==self.particles.len(){self.fabric.weights[i]}else{1.0-self.particles[i].pin.clamp(0.0,1.0)}).collect();
50        (ClothCollisionGeometry{positions:self.collision_render_positions(world),previous,inverse_mass,triangles:self.indices.chunks_exact(3).map(|t|[t[0]as usize,t[1]as usize,t[2]as usize]).collect()},drivers)
51    }
52    /// Commit mapped vertex contacts back to the actual solver and reproject
53    /// against body/animation constraints before the next shared contact pass.
54    pub fn apply_render_contact_positions(&mut self,cloth:&MeshCloth,positions:&[Vec3],world:Mat4,colliders:&[ClothShapeWorld],dt:f32){
55        if positions.len()!=self.vertex_count||positions.iter().any(|p|!p.is_finite()){return;}
56        let visible=self.collision_render_positions(world);
57        let positions:Vec<_>=self.members.iter().enumerate().map(|(i,members)|{
58            let delta=members.iter().map(|&v|(positions[v as usize]-visible[v as usize]).as_dvec3()).sum::<glam::DVec3>()/members.len()as f64;
59            self.particles[i].pos+delta.as_vec3()
60        }).collect();
61        self.apply_contact_positions(cloth,&positions,colliders,dt);
62    }
63    /// Project visible vertices and whole faces, feeding corrections back to
64    /// their simulation drivers. Rendering, inertia and shared garment contacts
65    /// consequently agree on the repaired surface.
66    pub(super) fn collide_visible_surface(&mut self,cloth:&MeshCloth,queries:&[ClothColliderQuery<'_>],previous:Option<&[Vec3]>,dt:f32){
67        if !cloth.collide||queries.is_empty()||self.anim.len()!=self.particles.len(){return;}
68        let radius=cloth.radius.max(0.0)*crate::cloth::max_scale(self.last_world);
69        let offsets:Vec<_>=self.animated_vertices.iter().enumerate().map(|(v,&p)|{
70            self.last_world.transform_point3(p)-self.anim[self.owner[v]as usize]+self.last_world.transform_vector3(self.transition_offset(v))
71        }).collect();
72        for _ in 0..3{
73            let mut changed=false;
74            for (v,&offset) in offsets.iter().enumerate(){
75                let owner=self.owner[v]as usize;if self.particles[owner].pin>=1.0||(offset.length_squared()<1e-14&&previous.is_none()){continue;}
76                for query in queries{
77                    let point=self.particles[owner].pos+offset;
78                    let contact=if cloth.continuous_collision&&previous.is_some_and(|p|p.len()==self.vertex_count){
79                        query.resolve(previous.unwrap()[v],point,radius).map(|(out,n,_)|(out-point,n))
80                    }else{penetration(point,radius,query.end())};
81                    if let Some((push,normal))=contact.filter(|(push,_)|push.is_finite()&&push.length_squared()>1e-16){
82                        let p=&mut self.particles[owner];p.pos+=push;p.contact=Some(normal);
83                        if dt>1e-6{p.vel=(p.vel+push/dt).clamp_length_max(30.0);}changed=true;
84                    }
85                }
86            }
87            for t in self.indices.chunks_exact(3){
88                let vertices=[t[0]as usize,t[1]as usize,t[2]as usize];let owners=vertices.map(|v|self.owner[v]as usize);
89                if owners.iter().all(|&i|self.particles[i].pin>=1.0){continue;}
90                for query in queries{
91                    // A linear plane cannot intersect the interior if all
92                    // vertices clear it; their exact checks above suffice.
93                    if matches!(query.end(),ClothShapeWorld::Plane{..}|ClothShapeWorld::Disabled){continue;}
94                    let end=std::array::from_fn(|i|self.particles[owners[i]].pos+offsets[vertices[i]]);
95                    let start=if cloth.continuous_collision{previous.filter(|p|p.len()==self.vertex_count).map_or(end,|p|vertices.map(|v|p[v]))}else{end};
96                    let Some(contact)=query.triangle(start,end,radius)else{continue;};
97                    if !contact.push.is_finite()||contact.push.length_squared()<1e-16{continue;}
98                    let mut weights=[0.0;3];
99                    for k in 0..3{let first=(0..=k).find(|&j|owners[j]==owners[k]).unwrap();weights[first]+=contact.weights[k];}
100                    let mass=owners.map(|i|if self.particles[i].pin>=1.0{0.0}else if cloth.fabric.is_some()&&self.fabric.weights.len()==self.particles.len(){self.fabric.weights[i]}else{1.0-self.particles[i].pin.clamp(0.0,1.0)});
101                    let denominator=(0..3).map(|i|mass[i]*weights[i]*weights[i]).sum::<f32>();if denominator<1e-20{continue;}
102                    for k in 0..3{if weights[k]==0.0||mass[k]==0.0{continue;}
103                        let push=contact.push*(mass[k]*weights[k]/denominator);let p=&mut self.particles[owners[k]];
104                        p.pos+=push;p.contact=Some(contact.normal);if dt>1e-6{p.vel=(p.vel+push/dt).clamp_length_max(30.0);}
105                    }
106                    changed=true;
107                }
108            }
109            if !changed{break;}
110        }
111    }

118    fn artistic_solver_honours_surface_ccd_and_untangling_controls(){
119        let fixed=[Vec3::new(-2.0,-2.0,0.0),Vec3::new(2.0,-2.0,0.0),Vec3::new(0.0,2.0,0.0)];
120        let moving=[Vec3::new(-0.3,-0.3,1.0),Vec3::new(0.3,-0.3,1.0),Vec3::new(0.0,0.3,1.0)];
121        let rest=fixed.into_iter().chain(moving).collect::<Vec<_>>();let pins=[1.0,1.0,1.0,0.0,0.0,0.0];
122        let mut cfg=MeshCloth{fabric:None,self_collision:true,continuous_collision:true,untangle:false,gravity:[0.0;3],damping:0.0,frequency:10.0,iterations:1,stiffness:0.0,bending:0.0,shape_stiffness:0.0,thickness:0.02,..Default::default()};
123        for continuous in [true,false]{
124            cfg.continuous_collision=continuous;let mut sim=MeshClothSim::build(&rest,&[0,1,2,3,4,5],&pins,0.0,Mat4::IDENTITY).unwrap();sim.reset(&rest,Mat4::IDENTITY);
125            for p in &mut sim.particles[3..]{p.vel=Vec3::NEG_Z*25.0;}
126            sim.step(&cfg,&rest,Mat4::IDENTITY,&[],&[],0.0,0.1);
127            for(i,p)in sim.particles[..3].iter().enumerate(){assert_eq!(p.pos,rest[i]);}
128            if continuous{assert!(sim.particles[3..].iter().all(|p|p.pos.z>=0.0199),"artistic cloth must sweep faces, not just particles");}
129            else{assert!(sim.particles[3..].iter().all(|p|p.pos.z< -1.0),"the test must actually cross the sheet without CCD");}
130        }
131        let mut rest=rest;rest[3].z=-0.5;rest[4].z=-0.5;rest[5].z=0.5;
132        cfg.untangle=true;cfg.continuous_collision=true;cfg.damping=1.0;
133        let mut sim=MeshClothSim::build(&rest,&[0,1,2,3,4,5],&pins,0.0,Mat4::IDENTITY).unwrap();sim.reset(&rest,Mat4::IDENTITY);
134        for _ in 0..8{sim.step(&cfg,&rest,Mat4::IDENTITY,&[],&[],0.0,1.0/60.0);}
135        assert!(sim.particles[3..].iter().all(|p|p.pos.z>=0.0199)||sim.particles[3..].iter().all(|p|p.pos.z<=-0.0199),"artistic cloth must repair an initially crossing face");
