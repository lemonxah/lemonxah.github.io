7/// How 3D text turns toward the camera.
8#[derive(Serialize, Deserialize, Clone, Copy, PartialEq, Eq, Debug, Default)]
9pub enum Text3DBillboard {
10    /// The text stays in the object's plane.
11    #[default]
12    None,
13    /// Rotate around the world Y axis only.
14    Y,
15    /// Always face the camera.
16    Full,
17}
18
19impl Text3DBillboard {
20    /// Every mode, in declaration order.
21    pub const ALL: [Text3DBillboard; 3] = [Self::None, Self::Y, Self::Full];
22    /// Inspector label of this mode.
23    pub fn label(self) -> &'static str {
24        match self {
25            Self::None => "None",
26            Self::Y => "Y axis",
27            Self::Full => "Full",
28        }
29    }
30}

32/// Text placed in the world at the object's transform (centred on its origin,
33/// in the object's XY plane facing +Z). Sharp at any distance and scale; rich
34/// text, shaping, bidi, overflow and localization work as for `UiText`.
35#[derive(Serialize, Deserialize)]
36#[serde(default)]
37pub struct Text3D {
38    /// The text to draw.
39    pub text: String,
40    /// Localization key (see `UiText::loc_key`).
41    pub loc_key: String,
42    /// Project-relative .ttf / .otf / .zfont / .zfontstack (empty = built-in).
43    pub font: String,
44    /// Font size in metres (height of one em).
45    pub size: f32,
46    /// Text colour, linear RGBA.
47    pub color: [f32; 4],
48    /// Parse rich-text tags (`<b>`, `<color=...>`, ...).
49    pub rich_text: bool,
50    /// Text box in metres; 0 = unbounded in that direction.
51    pub width: f32,
52    /// Text box height in metres; 0 = unbounded.
53    pub height: f32,
54    /// Horizontal alignment in the box.
55    pub align_x: UiAlign,
56    /// Vertical alignment in the box.
57    pub align_y: UiAlign,
58    /// Wrap at the box width.
59    pub wrap: bool,
60    /// What happens when the text does not fit the box.
61    pub overflow: UiTextOverflow,
62    /// Fit the size within `auto_size_min..=auto_size_max` metres.
63    pub auto_size: bool,
64    /// Lower bound of the auto-size range (m).
65    pub auto_size_min: f32,
66    /// Upper bound of the auto-size range (m).
67    pub auto_size_max: f32,
68    /// Draw the text bold.
69    pub bold: bool,
70    /// Draw the text italic.
71    pub italic: bool,
72    /// Spacing in em/100.
73    pub char_spacing: f32,
74    /// Extra space between lines, in em/100.
75    pub line_spacing: f32,
76    /// Outline width in em/100 (0 = none).
77    pub outline_width: f32,
78    /// Outline colour, linear RGBA (only when `outline_width` is above 0).
79    pub outline_color: [f32; 4],
80    /// How the text turns toward the camera.
81    pub billboard: Text3DBillboard,
82}
83
84impl Default for Text3D {

102            bold: false,
103            italic: false,
104            char_spacing: 0.0,
105            line_spacing: 0.0,
106            outline_width: 0.0,
107            outline_color: [0.0, 0.0, 0.0, 1.0],

111}
