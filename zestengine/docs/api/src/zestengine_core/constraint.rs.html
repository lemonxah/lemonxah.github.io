1//! Transform constraints: an object follows one or more source objects by
2//! position, rotation, scale, a full parent link or by aiming at them.
3//!
4//! The maths ([`solve_world`]) is pure and works on world matrices, so the
5//! same solver evaluates the [`Constraint`] component every frame at runtime
6//! and bakes constraints from a model file into plain animation curves at
7//! import (the importer reads FBX constraint nodes into [`ConstraintSpec`]s).
8//!
9//! Semantics (this engine's own, written from the FBX constraint attributes):
10//!
11//! * `Position`: the world position becomes the weighted mean of the sources'
12//!   positions (each moved by its offset) plus the constraint's own offset.
13//! * `Rotation`: the weighted mean of the sources' rotations, then the own offset.
14//! * `Scale`: the weighted mean of the sources' world scales times the own offset.
15//! * `Parent`: the object keeps the per-source offset it was authored with and
16//!   follows the weighted blend of `source_world * offset` (translation,
17//!   rotation and, if enabled, scale), then the own offset.
18//! * `Aim`: the local aim axis turns towards the weighted mean source
19//!   position; an up vector keeps the roll (scene up, a vector, towards a node
20//!   or aligned with a node).
21//!
22//! Every constraint has a total `weight` (0 = off, 1 = fully constrained: the
23//! result is blended from the unconstrained pose) and per-axis switches for
24//! translation, rotation (as XYZ Euler angles) and scale.
25

31/// What a constraint does.
32#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
33pub enum ConstraintKind {
34    /// Follow the sources as if parented to them (keeps the authored offsets).
35    #[default]
36    Parent,
37    /// The world position becomes the weighted mean of the sources.
38    Position,
39    /// The rotation becomes the weighted mean of the sources.
40    Rotation,
41    /// The scale becomes the weighted mean of the sources.
42    Scale,
43    /// Turn the aim axis towards the sources.
44    Aim,
45}
46
47impl ConstraintKind {
48    /// Every kind, in declaration order.
49    pub const ALL: [ConstraintKind; 5] = [Self::Parent, Self::Position, Self::Rotation, Self::Scale, Self::Aim];
50    /// Inspector label of this kind.
51    pub fn label(self) -> &'static str {
52        match self {
53            Self::Parent => "Parent",
54            Self::Position => "Position",
55            Self::Rotation => "Rotation",
56            Self::Scale => "Scale",
57            Self::Aim => "Aim",
58        }
59    }
60}

62/// How an aim constraint keeps the roll.

64pub enum AimUp {

68    /// The up axis points at the up object.
69    ToObject,
70    /// The up axis copies the up object's own up (Y) axis.

78/// A translation / rotation / scale offset.
79#[derive(Clone, Copy, Debug, PartialEq, Serialize, Deserialize)]
80#[serde(default)]
81pub struct TrsOffset {
82    /// Translation offset.
83    pub translation: [f32; 3],
84    /// Quaternion `[x, y, z, w]`.
85    pub rotation: [f32; 4],
86    /// Scale offset.
87    pub scale: [f32; 3],
88}

90impl Default for TrsOffset {
91    fn default() -> Self {
92        Self { translation: [0.0; 3], rotation: [0.0, 0.0, 0.0, 1.0], scale: [1.0; 3] }
93    }

96impl TrsOffset {
97    /// Build an offset.
98    pub fn new(t: Vec3, r: Quat, s: Vec3) -> Self {
99        Self { translation: t.to_array(), rotation: r.to_array(), scale: s.to_array() }
100    }
101    /// The translation.
102    pub fn t(&self) -> Vec3 {
103        Vec3::from_array(self.translation)
104    }
105    /// The rotation (normalised).
106    pub fn r(&self) -> Quat {
107        Quat::from_array(self.rotation).normalize()
108    }
109    /// The scale.
110    pub fn s(&self) -> Vec3 {
111        Vec3::from_array(self.scale)
112    }
113    /// The offset as a transform matrix.
114    pub fn matrix(&self) -> Mat4 {
115        Mat4::from_scale_rotation_translation(self.s(), self.r(), self.t())
116    }
117    /// True when the offset is the identity.
118    pub fn is_identity(&self) -> bool {
119        *self == Self::default()
120    }
121}

254fn blend_rotations(items: &[(Quat, f32)]) -> Quat {
255    let mut acc = Quat::IDENTITY;
256    let mut total = 0.0;
257    for (q, w) in items {
258        if *w <= 0.0 {
259            continue;
260        }
261        if total == 0.0 {
262            acc = *q;
263            total = *w;
264        } else {
265            // Running weighted slerp keeps neighbours in one hemisphere.
266            let t = w / (total + w);
267            acc = acc.slerp(*q, t);
268            total += w;
269        }
270    }
271    acc.normalize()
272}

274/// `world` with the components `rest` has on the axes the switches turn off,
275/// blended `weight` of the way from `rest`.
276fn mix_axes(rest: &Mat4, target: &Mat4, p: &SolveParams) -> Mat4 {
277    let (rs, rr, rt) = rest.to_scale_rotation_translation();
278    let (ts, tr, tt) = target.to_scale_rotation_translation();
279    let w = p.weight.clamp(0.0, 1.0);
280    let pick = |axes: [bool; 3], rest: Vec3, target: Vec3| Vec3::new(
281        if axes[0] { target.x } else { rest.x },
282        if axes[1] { target.y } else { rest.y },
283        if axes[2] { target.z } else { rest.z },
284    );
285    let t = rt.lerp(pick(p.translation_axes, rt, tt), w);
286    let s = rs.lerp(pick(p.scale_axes, rs, ts), w);
287    let rot = if p.rotation_axes.iter().all(|a| *a) {
288        rr.slerp(if rr.dot(tr) < 0.0 { -tr } else { tr }, w)
289    } else if p.rotation_axes.iter().all(|a| !*a) {
290        rr
291    } else {
292        let (ex, ey, ez) = rr.to_euler(EulerRot::XYZ);
293        let (tx, ty, tz) = tr.to_euler(EulerRot::XYZ);
294        let e = pick(p.rotation_axes, Vec3::new(ex, ey, ez), Vec3::new(tx, ty, tz));
295        let full = Quat::from_euler(EulerRot::XYZ, e.x, e.y, e.z);
296        rr.slerp(if rr.dot(full) < 0.0 { -full } else { full }, w)
297    };
298    Mat4::from_scale_rotation_translation(s, rot.normalize(), t)
299}

301/// The world matrix a constrained object gets: `rest_world` is its own
302/// unconstrained world matrix, `sources` the sources' current worlds, `up` the
303/// aim up object's world (when the mode wants one). With no usable source (or
304/// zero weight) the rest matrix comes back unchanged.
305pub fn solve_world(p: &SolveParams, rest_world: Mat4, sources: &[SourceWorld], up: Option<Mat4>) -> Mat4 {
306    let active: Vec<&SourceWorld> = sources.iter().filter(|s| s.weight > 0.0).collect();
307    let total: f32 = active.iter().map(|s| s.weight).sum();
308    if active.is_empty() || total <= 1e-9 || p.weight <= 0.0 {
309        return rest_world;
310    }
311    let (rest_s, rest_r, rest_t) = rest_world.to_scale_rotation_translation();
312    let wn = |s: &SourceWorld| s.weight / total;
313    let target = match p.kind {
314        ConstraintKind::Position => {
315            let pos: Vec3 = active.iter().map(|s| (s.world * s.offset.t().extend(1.0)).truncate() * wn(s)).sum();
316            Mat4::from_scale_rotation_translation(rest_s, rest_r, pos + p.offset.t())
317        }
318        ConstraintKind::Rotation => {
319            let q = blend_rotations(&active.iter().map(|s| (s.world.to_scale_rotation_translation().1 * s.offset.r(), wn(s))).collect::<Vec<_>>());
320            Mat4::from_scale_rotation_translation(rest_s, (q * p.offset.r()).normalize(), rest_t)
321        }
322        ConstraintKind::Scale => {
323            let sc: Vec3 = active.iter().map(|s| s.world.to_scale_rotation_translation().0 * s.offset.s() * wn(s)).sum();
324            Mat4::from_scale_rotation_translation(sc * p.offset.s(), rest_r, rest_t)
325        }
326        ConstraintKind::Parent => {
327            let mut t = Vec3::ZERO;
328            let mut sc = Vec3::ZERO;
329            let mut rots = Vec::with_capacity(active.len());
330            for s in &active {
331                let m = s.world * s.offset.matrix();
332                let (ms, mr, mt) = m.to_scale_rotation_translation();
333                t += mt * wn(s);
334                sc += ms * wn(s);
335                rots.push((mr, wn(s)));
336            }
337            let blended = Mat4::from_scale_rotation_translation(sc, blend_rotations(&rots), t);
338            blended * p.offset.matrix()
339        }
340        ConstraintKind::Aim => {
341            let aim_pos: Vec3 = active.iter().map(|s| (s.world * s.offset.t().extend(1.0)).truncate() * wn(s)).sum();
342            let dir = (aim_pos - rest_t).normalize_or_zero();
343            if dir == Vec3::ZERO {
344                return rest_world;
345            }
346            let a = p.aim_axis.normalize_or(Vec3::X);
347            let up_world = match p.aim_up {
348                AimUp::Scene => Some(Vec3::Y),
349                AimUp::Vector => Some(p.up_vector.normalize_or(Vec3::Y)),
350                AimUp::ToObject => up.map(|u| (u.w_axis.truncate() - rest_t).normalize_or_zero()).filter(|v| *v != Vec3::ZERO),
351                AimUp::AlignObject => up.map(|u| u.transform_vector3(Vec3::Y).normalize_or(Vec3::Y)),
352                AimUp::None => None,
353            };
354            let rot = aim_rotation(a, p.up_axis.normalize_or(Vec3::Y), dir, up_world) * p.offset.r();
355            Mat4::from_scale_rotation_translation(rest_s, rot.normalize(), rest_t)
356        }
357    };
358    mix_axes(&rest_world, &target, p)
359}

361/// A rotation taking the local `aim` axis to `dir`, and (when given) turning
362/// the local `up` axis as close to `up_world` as the aim allows.
363pub fn aim_rotation(aim: Vec3, up: Vec3, dir: Vec3, up_world: Option<Vec3>) -> Quat {
364    let primary = Quat::from_rotation_arc(aim, dir);
365    let Some(u) = up_world else { return primary };
366    // Where the up axis ends up after the primary turn, against where it should be.
367    let cur = (primary * up) - dir * (primary * up).dot(dir);
368    let want = u - dir * u.dot(dir);
369    if cur.length_squared() < 1e-12 || want.length_squared() < 1e-12 {
370        return primary;
371    }
372    let (cur, want) = (cur.normalize(), want.normalize());
373    let angle = cur.dot(want).clamp(-1.0, 1.0).acos();
374    let sign = if cur.cross(want).dot(dir) < 0.0 { -1.0 } else { 1.0 };
375    Quat::from_axis_angle(dir, angle * sign) * primary
376}

454    /// Sources it cannot find are dropped and listed in `unresolved`.
455    pub fn from_spec(spec: &ConstraintSpec, resolve: impl Fn(&str) -> Option<crate::ObjectId>) -> Self {
456        let mut c = Self {
457            kind: spec.kind,

466            up_vector: spec.up_vector,
467            ..Default::default()
468        };
469        for s in &spec.sources {
470            match resolve(&s.node) {
471                Some(id) => c.sources.push(ConstraintSource { target: ObjectRef::new(id), weight: s.weight, offset: s.offset }),
472                None => c.unresolved.push(s.node.clone()),
473            }
474        }
475        if !spec.up_node.is_empty() {
476            match resolve(&spec.up_node) {
477                Some(id) => c.up_object = ObjectRef::new(id),
478                None => c.unresolved.push(spec.up_node.clone()),
479            }
480        }
481        c
482    }

512fn initialize(world: &crate::ecs::SystemWorld<'_>, context: &mut crate::ecs::SystemContext<'_>) {
513    for (_, (owner, value)) in world.query::<(&crate::ecs::Owner, &mut Constraint)>().iter() {
514        if value.rest.is_some() || !context.is_active(world, owner.0) { continue; }
515        let local = world.get::<crate::Transform>(owner.0).expect("constraint transform");
516        value.rest = Some((local.translation, local.rotation, local.scale));
517    }
518}
519
520fn constrain(world: &crate::ecs::SystemWorld<'_>, context: &mut crate::ecs::SystemContext<'_>) {
521    let mut query = world.query::<(&crate::ecs::Owner, &mut Constraint)>();
522    let mut values: Vec<_> = query.iter().filter(|(_, (owner, _))| context.is_active(world, owner.0)).collect();
523    values.sort_unstable_by_key(|(_, (owner, _))| context.order_key(world, owner.0));
524    for (_, (owner, value)) in values {
525        let Some((t, r, s)) = value.rest else {
526            let local = world.get::<crate::Transform>(owner.0).expect("constraint transform");
527            value.rest = Some((local.translation, local.rotation, local.scale));
528            continue;
529        };
530        let world_of = |id: Option<crate::ObjectId>| id.and_then(|id| world.entity(id))
531            .filter(|&entity| context.is_visible(world, entity)).and_then(|entity| world.world_transform(entity));
532        let sources: Vec<SourceWorld> = value.sources.iter().filter_map(|source|
533            world_of(source.target.id()).map(|matrix| SourceWorld { world: matrix, weight: source.weight, offset: source.offset })).collect();
534        if sources.is_empty() { continue; }
535        let parent = world.get::<crate::ecs::Parent>(owner.0).and_then(|parent| parent.0)
536            .and_then(|entity| world.world_transform(entity)).unwrap_or(Mat4::IDENTITY);
537        let rest = parent * Mat4::from_scale_rotation_translation(s, r, t);
538        let solved = solve_world(&value.params(), rest, &sources, world_of(value.up_object.id()));
539        let (scale, rotation, translation) = (parent.inverse() * solved).to_scale_rotation_translation();
540        if translation.is_finite() && rotation.is_finite() && scale.is_finite() {
541            let mut local = world.get_mut::<crate::Transform>(owner.0).expect("constraint transform");
542            local.translation = translation;
543            local.rotation = rotation;
544            local.scale = scale;
545        }
546    }
547}

642        // Aiming up the sky still leaves a valid rotation.
