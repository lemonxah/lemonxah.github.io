1//! `Mesh Renderer` / `Skinned Mesh Renderer`: the components that own how an object's mesh is
2//! drawn (shadow casting, lightmap scale and ONE material per submesh slot).
3//!
4//! The mesh itself is not here: it stays in the object's `ObjectSource` (the "Mesh Filter" the
5//! inspector shows as a view). Slot `k` draws submesh `k` of that mesh. See
6//! `docs/design/mesh-renderer-split.md`.
7//!
8//! Core has no dependency on assets or render, and an inline material snapshot carries render
9//! types, so [`MaterialSlot::Inline`] keeps the snapshot as opaque RON text that the engine
10//! decodes.
11

16/// Registry name of [`MeshRenderer`].
17pub const MESH_RENDERER: &str = "Mesh Renderer";
18/// Registry name of [`SkinnedMeshRenderer`].
19pub const SKINNED_MESH_RENDERER: &str = "Skinned Mesh Renderer";

21/// One material slot of a renderer. Slot `k` draws submesh `k` of the object's mesh.
22#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize, Default)]
23pub enum MaterialSlot {
24    /// Inherit: the model's own material for this submesh (after the importer's material
25    /// remap table); the default material for a primitive. Costs nothing on disk.
26    #[default]
27    Model,
28    /// A `.material` asset, project-relative. Shared by every user of the path.
29    File(String),
30    /// An edited snapshot of the model's material (RON text of `assets::MaterialRef::Inline`).
31    /// Opaque to core; the engine converts it.
32    Inline(String),
33}
34
35impl MaterialSlot {
36    /// True for the `Model` (inherit) slot.
37    pub fn is_model(&self) -> bool {
38        matches!(self, Self::Model)
39    }
40}
41
42/// What the engine remembers about the slots it last resolved. Never saved, never compared.
43#[derive(Clone, Debug, Default)]
44pub struct RendererRuntime {
45    /// The normalised slot list the object's resolved render cache was built from
46    /// (`None` = not resolved yet, e.g. right after a component reload).
47    pub resolved: Option<Vec<MaterialSlot>>,
48}
49
50impl PartialEq for RendererRuntime {
51    fn eq(&self, _: &Self) -> bool {
52        true
53    }
54}

56fn default_lightmap_scale() -> f32 {
57    1.0
58}
59
60fn is_default_scale(v: &f32) -> bool {
61    (*v - 1.0).abs() < f32::EPSILON
62}

64/// The data both renderer types share.
65pub trait RendererData {
66    /// How the renderer casts shadows.
67    fn cast_shadows(&self) -> ShadowCasting;
68    /// Set how the renderer casts shadows.
69    fn set_cast_shadows(&mut self, value: ShadowCasting);
70    /// Lightmap-resolution multiplier.
71    fn lightmap_scale(&self) -> f32;
72    /// Set the lightmap-resolution multiplier.
73    fn set_lightmap_scale(&mut self, value: f32);
74    /// The material slots.
75    fn materials(&self) -> &[MaterialSlot];
76    /// Mutable material slots.
77    fn materials_mut(&mut self) -> &mut Vec<MaterialSlot>;
78    /// The runtime state.
79    fn runtime(&self) -> &RendererRuntime;
80    /// Mutable runtime state.
81    fn runtime_mut(&mut self) -> &mut RendererRuntime;
82    /// Registry name of the concrete type.
83    fn kind(&self) -> &'static str;
84
85    /// Slot `k`, `Model` past the end of the list (a shorter list inherits from the model).
86    fn slot(&self, k: usize) -> MaterialSlot {
87        self.materials().get(k).cloned().unwrap_or_default()
88    }
89
90    /// The first `count` slots, padded with `Model`. Extra entries (a longer list) are kept in
91    /// the data but not part of this view.
92    fn normalised(&self, count: usize) -> Vec<MaterialSlot> {
93        (0..count).map(|k| self.slot(k)).collect()
94    }
95
96    /// Set slot `k`, padding a short list with `Model`.
97    fn set_slot(&mut self, k: usize, value: MaterialSlot) {
98        let list = self.materials_mut();
99        if list.len() <= k {
100            list.resize(k + 1, MaterialSlot::Model);
101        }
102        list[k] = value;
103    }
104}

106macro_rules! renderer_type {
107    ($(#[$doc:meta])* $name:ident, $label:expr) => {
108        $(#[$doc])*
109        #[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
110        #[serde(default)]
111        pub struct $name {
112            /// "Cast Shadows": On / Off / Two Sided / Shadows Only.
113            pub cast_shadows: ShadowCasting,
114            /// "Scale In Lightmap": lightmap-resolution multiplier (needs Contribute GI).
115            pub lightmap_scale: f32,
116            /// One entry per submesh slot, in the object's mesh order. Shorter = the rest inherit
117            /// from the model; longer = the extras are kept but not drawn.
118            pub materials: Vec<MaterialSlot>,
119            /// Engine-filled state of the last resolved slots (not saved).
120            #[serde(skip)]
121            pub runtime: RendererRuntime,
122        }
123
124        impl Default for $name {
125            fn default() -> Self {
126                Self { cast_shadows: ShadowCasting::On, lightmap_scale: default_lightmap_scale(), materials: Vec::new(), runtime: RendererRuntime::default() }
127            }
128        }
129
130        impl TypedComponent for $name {
131            const NAME: &'static str = $label;
132        }
133
134        impl RendererData for $name {
135            fn cast_shadows(&self) -> ShadowCasting {
136                self.cast_shadows
137            }
138            fn set_cast_shadows(&mut self, value: ShadowCasting) {
139                self.cast_shadows = value;
140            }
141            fn lightmap_scale(&self) -> f32 {
142                self.lightmap_scale
143            }
144            fn set_lightmap_scale(&mut self, value: f32) {
145                self.lightmap_scale = value;
146            }
147            fn materials(&self) -> &[MaterialSlot] {
148                &self.materials
149            }
150            fn materials_mut(&mut self) -> &mut Vec<MaterialSlot> {
151                &mut self.materials
152            }
153            fn runtime(&self) -> &RendererRuntime {
154                &self.runtime
155            }
156            fn runtime_mut(&mut self) -> &mut RendererRuntime {
157                &mut self.runtime
158            }
159            fn kind(&self) -> &'static str {
160                $label
161            }
162        }
163    };
164}

178impl MeshRenderer {
179    /// A renderer whose first `slots` slots all inherit from the model (what the importer's
180    /// parts and a fresh primitive get).
181    pub fn inheriting(slots: usize) -> Self {
182        Self { materials: vec![MaterialSlot::Model; slots], ..Self::default() }
183    }
184
185    /// Settings are the defaults and every slot inherits: nothing worth storing.
186    pub fn is_untouched(&self) -> bool {
187        is_untouched(self)
188    }
189}
190
191/// Every setting is the default and every slot inherits.
192pub fn is_untouched(r: &dyn RendererData) -> bool {
193    r.cast_shadows().is_on() && is_default_scale(&r.lightmap_scale()) && r.materials().iter().all(MaterialSlot::is_model)
194}
195
196/// A renderer component, whether or not it is switched off, as its shared view.
197pub fn renderer_of(c: &dyn Component) -> Option<&(dyn RendererData + 'static)> {
198    let c = component_inner(c);
199    let any = c.as_any();
200    if let Some(r) = any.downcast_ref::<MeshRenderer>() {
201        return Some(r);
202    }
203    any.downcast_ref::<SkinnedMeshRenderer>().map(|r| r as &dyn RendererData)
204}
205
206/// Mutable form of [`renderer_of`].
207pub fn renderer_of_mut(c: &mut Box<dyn Component>) -> Option<&mut (dyn RendererData + 'static)> {
208    let c = component_inner_mut(c);
209    renderer_data_mut(c)
210}
211
212/// Mutable renderer view from a borrowed ECS component.
213pub fn renderer_data_mut(c: &mut dyn Component) -> Option<&mut (dyn RendererData + 'static)> {
214    let c = crate::component_inner_data_mut(c);
215    let is_mesh = c.as_any().is::<MeshRenderer>();
216    let any = c.as_any_mut();
217    if is_mesh {
218        return any.downcast_mut::<MeshRenderer>().map(|r| r as &mut dyn RendererData);
219    }
220    any.downcast_mut::<SkinnedMeshRenderer>().map(|r| r as &mut dyn RendererData)
221}
222
223/// Whether `name` is one of the two renderer component names.
224pub fn is_renderer_kind(name: &str) -> bool {
225    name == MESH_RENDERER || name == SKINNED_MESH_RENDERER
226}

233    #[test]
234    fn defaults_are_written_flat_so_prefab_overrides_can_name_fields_and_round_trip() {
235        let r = MeshRenderer::default();
236        let text = ron::to_string(&r).unwrap();
237        assert_eq!(text, "(cast_shadows:On,lightmap_scale:1.0,materials:[])", "fields are always written: {text}");
238        assert_eq!(ron::from_str::<MeshRenderer>(&text).unwrap(), r);
239        let r = MeshRenderer {
240            cast_shadows: ShadowCasting::ShadowsOnly,
241            lightmap_scale: 2.0,
242            materials: vec![MaterialSlot::Model, MaterialSlot::File("m/a.material".into()), MaterialSlot::Inline("(x)".into())],
243            ..Default::default()
244        };
245        let back: MeshRenderer = ron::from_str(&ron::to_string(&r).unwrap()).unwrap();
246        assert_eq!(back, r);
247    }
248
249    #[test]
250    fn missing_fields_take_defaults() {
251        let r: MeshRenderer = ron::from_str("()").unwrap();
252        assert_eq!(r, MeshRenderer::default());
253        let r: SkinnedMeshRenderer = ron::from_str("(lightmap_scale: 3.0)").unwrap();
254        assert_eq!(r.lightmap_scale, 3.0);
255        assert!(r.cast_shadows.is_on() && r.materials.is_empty());
256    }
257
258    #[test]
259    fn slots_pad_and_trim_without_losing_extras() {
260        let mut r = MeshRenderer::default();
261        r.set_slot(2, MaterialSlot::File("x.material".into()));
262        assert_eq!(r.materials, vec![MaterialSlot::Model, MaterialSlot::Model, MaterialSlot::File("x.material".into())]);
263        assert_eq!(r.normalised(2), vec![MaterialSlot::Model; 2], "a shorter mesh sees only its slots");
264        assert_eq!(r.normalised(4)[3], MaterialSlot::Model, "a longer mesh inherits the tail");
265        assert_eq!(r.materials.len(), 3, "the extras stay in the data");
266        assert!(!r.is_untouched());
267        assert!(MeshRenderer::inheriting(5).is_untouched());
268    }

270    #[test]
271    fn registry_creates_both_and_they_are_client_side() {
272        let reg = ComponentRegistry::with_builtins();
273        for name in [MESH_RENDERER, SKINNED_MESH_RENDERER] {
274            let c = reg.create(name).unwrap_or_else(|| panic!("{name} is registered"));
275            assert_eq!(c.type_name(), name);
276            assert!(renderer_of(c.as_ref()).is_some());
277            assert_eq!(reg.default_role(name), crate::NetRole::Client, "{name} is stripped from a dedicated server");
278            let loaded = reg.load(name, "(materials:[File(\"a.material\")])").unwrap();
279            assert_eq!(renderer_of(loaded.as_ref()).unwrap().slot(0), MaterialSlot::File("a.material".into()));
280        }
281    }
