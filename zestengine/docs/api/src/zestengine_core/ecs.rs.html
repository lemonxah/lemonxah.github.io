1//! Typed ECS storage and the native system API shared by games, the editor,
2//! headless servers and plugins.
3//!
4//! **New here? Read the [ECS Guide](https://lemonxah.github.io/zestengine/docs/ECS.html)**, which explains components,
5//! systems, stages, commands and events with tested examples, and the
6//! [Engine API Guide](https://lemonxah.github.io/zestengine/docs/ENGINE_API.html) for what a system can do (move,
7//! spawn, raycast, play sounds, save, network).
8//!
9//! # The model
10//!
11//! - A scene **object** is a hecs entity. Its saved identity is an
12//!   [`ObjectId`]; the hecs [`Entity`] is only a runtime handle.
13//! - A **component** is a [`TypedComponent`] value in a typed column. The first
14//!   component of each type lives on the object's own entity, so ordinary joined
15//!   queries work. Further instances of the same type live on small auxiliary
16//!   entities carrying an [`Owner`] that points back to the object.
17//! - A **system** is a [`SystemFn`]: it queries a [`SystemWorld`] and receives a
18//!   [`SystemContext`] with time, input, services and a command queue. Systems
19//!   are registered on the [`ComponentRegistry`] and
20//!   run by [`Stage`].
21//! - An **entity template** ([`EntityTemplate`]) is a registered, named, ready-made
22//!   object that the editor lists under Create > Entity and systems spawn by name.
23//!
24//! # Borrowing
25//!
26//! Native systems use hecs' checked column borrows: two borrows of one component
27//! type conflict when either is mutable, and the conflict panics. While a system
28//! runs, the owning [`SceneWorld`] holds exclusive structural access, so
29//! inserting, removing or despawning must be queued as a
30//! [`ComponentCommand`] instead. Editor-style component
31//! views ([`ComponentRef`], [`ComponentMut`]) use checked entity/type leases and
32//! hold a shared structural lock, which is why the two access modes cannot overlap.

88/// The object that a component instance belongs to. Object entities own
89/// themselves; the auxiliary entity holding a *second* instance of a component
90/// type points back at its object.
91///
92/// This is why systems query `(&Owner, &T)`: `Owner` is the way from a component
93/// to its object's [`Transform`], [`Active`] flag and
94/// [`ObjectId`].
95///
96/// ```ignore
97/// for (_, (owner, health)) in world.query::<(&Owner, &mut Health)>().iter() {
98///     if !ctx.is_active(world, owner.0) { continue; }
99///     let id = world.object_id(owner.0);
100/// }
101/// ```
102#[derive(Clone, Copy, Debug, PartialEq, Eq)]
103pub struct Owner(pub Entity);

105/// Why a structural scene operation was refused. Nothing changes when one is
106/// returned.
107#[derive(Clone, Copy, Debug, PartialEq, Eq)]
108pub enum HierarchyError {
109    /// The entity is not a scene object in this world (never spawned, or already removed).
110    MissingEntity,
111    /// Another object already has this saved id.
112    DuplicateId,
113    /// Scene objects need a non-nil saved id.
114    InvalidId,
115    /// Parenting would make an object its own ancestor.
116    Cycle,
117    /// A world transform cannot be preserved under a parent whose matrix cannot be inverted
118    /// (for example a zero scale).
119    SingularParent,
120}
121
122impl fmt::Display for HierarchyError {
123    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
124        f.write_str(match self {
125            Self::MissingEntity => "entity is not a scene object in this world",
126            Self::DuplicateId => "another entity already has this saved object id",
127            Self::InvalidId => "scene objects require a nonzero saved object id",
128            Self::Cycle => "parenting would create a hierarchy cycle",
129            Self::SingularParent => "cannot preserve a world transform under a singular parent transform",
130        })
131    }
132}

136/// Owns a scene's ECS world and the index from saved [`ObjectId`]s to entities.
137///
138/// All **structural** operations (spawn, parent, insert and remove components,
139/// despawn) go through this type, so deleting an object can never leave a saved
140/// reference pointing at a reused slot. It is what editors, loaders and tests
141/// hold; game systems never see it. They receive a [`SystemWorld`] and queue
142/// [`ComponentCommand`]s instead.
143///
144/// ```ignore
145/// let mut scene = SceneWorld::new();
146/// let player = scene.spawn("Player");
147/// scene.insert(player, Health::default())?;
148/// registry.run_systems(Stage::Update, &scene, &mut ctx);
149/// assert_eq!(scene.get::<Health>(player).unwrap().current, 100.0);
150/// ```
151///
152/// See [Testing systems](https://lemonxah.github.io/zestengine/docs/ECS.html#testing-systems) in the ECS guide.
153#[derive(Default)]
154pub struct SceneWorld {
155    world: Arc<storage::Storage>,
156    objects: HashMap<ObjectId, Entity>,
157}

159/// The world as a system sees it: queries and checked component access over the
160/// scene's permanent typed storage.
161///
162/// The owning [`SceneWorld`] holds exclusive structural access for the whole run
163/// of a system, so nothing can be inserted, removed or despawned underneath you
164/// (queue a [`ComponentCommand`] for that) and editor
165/// component views are excluded. Within a system, hecs checks column borrows as
166/// usual: two borrows of one component type conflict if either is mutable.
167///
168/// ```ignore
169/// fn drain(world: &SystemWorld<'_>, ctx: &mut SystemContext<'_>) {
170///     for (_, (owner, tank)) in world.query::<(&Owner, &mut FuelTank)>().iter() {
171///         if !ctx.is_active(world, owner.0) { continue; }
172///         tank.fuel = (tank.fuel - tank.burn_rate * ctx.dt).max(0.0);
173///     }
174/// }
175/// ```
176///
177/// See [Systems](https://lemonxah.github.io/zestengine/docs/ECS.html#systems) and
178/// [Borrowing rules](https://lemonxah.github.io/zestengine/docs/ECS.html#borrowing-rules) in the ECS guide.
179pub struct SystemWorld<'a> {
180    world: &'a hecs::World,
181    objects: &'a HashMap<ObjectId, Entity>,
182}

184impl SystemWorld<'_> {
185    /// Start a hecs query, for example `world.query::<(&Owner, &mut Health)>()`, then
186    /// `.iter()` it. Each yielded item is `(Entity, Q::Item)`. Borrows are checked per
187    /// component column when the query is created: asking for `&mut T` while another
188    /// live borrow of `T` exists panics with "already borrowed".
189    ///
190    /// Query `(&Owner, &T)` rather than `&T` alone when you need the object a
191    /// component belongs to; plain `&T` visits every instance, including duplicates
192    /// on auxiliary entities. Iteration order is archetype order, not authoring
193    /// order; use [`run_ordered`] when order matters.
194    pub fn query<Q: Query>(&self) -> QueryBorrow<'_, Q> { self.world.query::<Q>() }
195
196    /// Borrow component `T` of `entity` for reading. `None` if the entity does not
197    /// carry an (enabled) `T`. Panics if `T` of that entity is already mutably borrowed.
198    pub fn get<T: DataComponent>(&self, entity: Entity) -> Option<Ref<'_, T>> { self.world.get::<&T>(entity).ok() }
199
200    /// Borrow component `T` of `entity` for writing: the way to edit another
201    /// component of the object you are iterating (`world.get_mut::<Transform>(owner.0)`).
202    /// `None` if the entity has no `T`. Panics if `T` of that entity is already
203    /// borrowed, including by an enclosing query over `T`.
204    pub fn get_mut<T: DataComponent>(&self, entity: Entity) -> Option<RefMut<'_, T>> { self.world.get::<&mut T>(entity).ok() }
205
206    /// The runtime entity of the object with this saved id (`None` if it does not exist).
207    pub fn entity(&self, id: ObjectId) -> Option<Entity> { self.objects.get(&id).copied() }
208
209    /// The saved id of an object entity (`None` for an entity that is not a scene object).
210    pub fn object_id(&self, entity: Entity) -> Option<ObjectId> { self.get::<ObjectId>(entity).map(|id| *id) }
211
212    /// The object's world matrix **right now**: local transforms folded through every
213    /// parent, including writes made by earlier systems in this stage (unlike
214    /// [`GlobalTransform`], which is a stage-boundary snapshot). `None` if the entity
215    /// is not an object or its parent links are malformed (a cycle).
216    pub fn world_transform(&self, entity: Entity) -> Option<Mat4> {
217        let mut matrix = self.get::<Transform>(entity)?.matrix();
218        let mut parent = self.get::<Parent>(entity)?.0;
219        for _ in 0..=self.objects.len() {
220            let Some(entity) = parent else { return Some(matrix) };
221            matrix = self.get::<Transform>(entity)?.matrix() * matrix;
222            parent = self.get::<Parent>(entity)?.0;
223        }
224        None // malformed cyclic native parent links
225    }
226    /// Whether the object is active **right now**: its [`Enabled`] flag and every
227    /// ancestor's, read live rather than from the [`Active`] snapshot. Prefer
228    /// [`SystemContext::is_active`] in ordinary systems; this is for code that has
229    /// just changed an enabled flag.
230    pub fn current_active(&self, entity: Entity) -> bool {
231        let mut current = Some(entity);
232        for _ in 0..=self.objects.len() {
233            let Some(entity) = current else { return true };
234            if !self.get::<Enabled>(entity).is_some_and(|value| value.0) { return false; }
235            let Some(parent) = self.get::<Parent>(entity) else { return false };
236            current = parent.0;
237        }
238        false
239    }
240}

262    /// Create an enabled root object with a fresh random [`ObjectId`] and an identity
263    /// transform. Returns its entity. Add components with [`SceneWorld::insert`].
264    pub fn spawn(&mut self, name: impl Into<String>) -> Entity {
265        self.spawn_with_id(ObjectId::new(), name, Transform::default(), true).expect("fresh object id")
266    }

268    /// Create a root object with a given saved id, local transform and enabled state:
269    /// what scene loading does. Fails with [`HierarchyError::InvalidId`] for the nil
270    /// id and [`HierarchyError::DuplicateId`] if the id is taken.
271    pub fn spawn_with_id(
272        &mut self, id: ObjectId, name: impl Into<String>, transform: Transform, enabled: bool,
273    ) -> Result<Entity, HierarchyError> {
274        if id.is_nil() { return Err(HierarchyError::InvalidId); }
275        if self.objects.contains_key(&id) { return Err(HierarchyError::DuplicateId); }
276        let mut world = self.world.exclusive();
277        let entity = world.spawn((id, Name(name.into()), transform, Parent(None), Enabled(enabled), Active(enabled), GlobalTransform(transform.matrix())));
278        world.insert_one(entity, Owner(entity)).expect("new entity");
279        self.objects.insert(id, entity);
280        Ok(entity)
281    }

283    /// Change a saved identity without moving components or invalidating the
284    /// entity. The map and ECS metadata change together, before deletion or
285    /// replication can observe the new identity.
286    pub fn set_object_id(&mut self, entity: Entity, id: ObjectId) -> Result<(), HierarchyError> {
287        if id.is_nil() { return Err(HierarchyError::InvalidId); }
288        let old = self.object_id(entity).ok_or(HierarchyError::MissingEntity)?;
289        if old == id { return Ok(()); }
290        if self.objects.contains_key(&id) { return Err(HierarchyError::DuplicateId); }
291        *self.get_mut::<ObjectId>(entity).expect("object identity") = id;
292        self.objects.remove(&old);
293        self.objects.insert(id, entity);
294        Ok(())
295    }

297    /// Run `run` with a [`SystemWorld`] over this scene: the same view a system
298    /// receives. The world is locked exclusively for the duration, so component
299    /// views ([`SceneWorld::get`]) must be released first and cannot be taken inside
300    /// `run` (both panic). Use it for ad-hoc queries in tools and tests.
301    ///
302    /// ```ignore
303    /// let total: f32 = scene.with_world(|world| {
304    ///     world.query::<&Health>().iter().map(|(_, hp)| hp.current).sum()
305    /// });
306    /// ```
307    pub fn with_world<R>(&self, run: impl FnOnce(&SystemWorld<'_>) -> R) -> R {
308        let world = self.world.exclusive();
309        run(&SystemWorld { world: &world, objects: &self.objects })
310    }

329    /// Attach typed gameplay data. A second instance of the same type gets a
330    /// separate entity, with the same Owner. Metadata types are not accepted.
331    pub fn insert<T: TypedComponent + Send + std::marker::Sync>(&mut self, owner: Entity, value: T) -> Result<Entity, HierarchyError> {
332        if self.object_id(owner).is_none() { return Err(HierarchyError::MissingEntity); }
333        let mut world = self.world.exclusive();
334        if world.satisfies::<&T>(owner).unwrap_or(false) {
335            Ok(world.spawn((Owner(owner), value)))
336        } else {
337            world.insert_one(owner, value).expect("live owner");
338            Ok(owner)
339        }
340    }

342    /// Attach a disabled payload inline. Ordinary T queries continue to select
343    /// enabled instances; `DisabledComponent<T>` queries expose this data directly.
344    pub fn insert_disabled<T: TypedComponent>(&mut self, owner: Entity, value: T) -> Result<Entity, HierarchyError> {
345        if self.object_id(owner).is_none() { return Err(HierarchyError::MissingEntity); }
346        let mut world = self.world.exclusive();
347        let value = DisabledComponent::new(value);
348        if world.satisfies::<&DisabledComponent<T>>(owner).unwrap_or(false) {
349            Ok(world.spawn((Owner(owner), value)))
350        } else {
351            world.insert_one(owner, value).expect("live owner");
352            Ok(owner)
353        }
354    }

356    /// Remove one *disabled* component instance (as inserted by
357    /// [`SceneWorld::insert_disabled`]) and return its value. The owning object stays.
358    pub fn remove_disabled<T: TypedComponent>(&mut self, instance: Entity) -> Option<T> {
359        let owner = self.get::<Owner>(instance)?.0;
360        let mut world = self.world.exclusive();
361        let value = world.remove_one::<DisabledComponent<T>>(instance).ok()?;
362        if owner != instance { world.despawn(instance).expect("live component entity"); }
363        self.world.forget(&[instance]);
364        Some(value.into_value())
365    }

372        if owner != instance { world.despawn(instance).expect("live component entity"); }

377    /// Reparent an object, optionally preserving its world-space TRS. Rejects
378    /// cycles before changing anything. Children keep following their parent.
379    pub fn set_parent(&mut self, child: Entity, parent: Option<Entity>, keep_world: bool) -> Result<(), HierarchyError> {
380        if self.object_id(child).is_none() { return Err(HierarchyError::MissingEntity); }
381        let mut ancestor = parent;
382        let mut visited = HashSet::new();
383        while let Some(entity) = ancestor {
384            if entity == child || !visited.insert(entity) { return Err(HierarchyError::Cycle); }
385            ancestor = self.get::<Parent>(entity).ok_or(HierarchyError::MissingEntity)?.0;
386        }
387        let local = if keep_world {
388            self.propagate()?;
389            let global = self.get::<GlobalTransform>(child).expect("object metadata").0;
390            let parent_global = parent.and_then(|p| self.get::<GlobalTransform>(p).map(|g| g.0)).unwrap_or(Mat4::IDENTITY);
391            let determinant = parent_global.determinant();
392            if !parent_global.is_finite() || !determinant.is_finite() || determinant.abs() < 1e-12 {
393                return Err(HierarchyError::SingularParent);
394            }
395            Some(Transform::from_matrix(parent_global.inverse() * global))
396        } else { None };
397        self.get_mut::<Parent>(child).expect("object metadata").0 = parent;
398        if let Some(local) = local {
399            *self.get_mut::<Transform>(child).expect("object metadata") = local;
400        }
401        self.propagate()
402    }

404    /// Apply an imported hierarchy in one pass. Validation is atomic: on an
405    /// invalid graph, all previous parent links remain intact.
406    pub fn set_parents(&mut self, parents: &[(Entity, Option<Entity>)]) -> Result<(), HierarchyError> {
407        for &(child, parent) in parents {
408            if self.object_id(child).is_none() || parent.is_some_and(|p| self.object_id(p).is_none()) {
409                return Err(HierarchyError::MissingEntity);
410            }
411        }
412        let previous: Vec<_> = parents.iter().map(|&(child, _)| (child, self.get::<Parent>(child).expect("object metadata").0)).collect();
413        for &(child, parent) in parents { self.get_mut::<Parent>(child).expect("object metadata").0 = parent; }
414        if let Err(error) = self.propagate() {
415            for (child, parent) in previous.into_iter().rev() { self.get_mut::<Parent>(child).expect("object metadata").0 = parent; }
416            return Err(error);
417        }
418        Ok(())
419    }

421    /// Resolve transforms and inherited activation in O(objects), without
422    /// recursion or a hierarchy-depth limit. Bad parent graphs are rejected.
423    pub fn propagate(&mut self) -> Result<(), HierarchyError> {
424        let mut world = self.world.exclusive();
425        let nodes: HashMap<Entity, (Mat4, Option<Entity>, bool)> = world
426            .query_mut::<(&Transform, &Parent, &Enabled)>().into_iter()
427            .map(|(entity, (local, parent, enabled))| (entity, (local.matrix(), parent.0, enabled.0))).collect();
428        let mut resolved: HashMap<Entity, (Mat4, bool)> = HashMap::with_capacity(nodes.len());
429        let mut visiting = HashSet::new();
430        let mut chain = Vec::new();
431        for &start in nodes.keys() {
432            if resolved.contains_key(&start) { continue; }
433            chain.clear();
434            let mut cursor = Some(start);
435            while let Some(entity) = cursor {
436                if resolved.contains_key(&entity) { break; }
437                if !visiting.insert(entity) { return Err(HierarchyError::Cycle); }
438                let node = nodes.get(&entity).ok_or(HierarchyError::MissingEntity)?;
439                chain.push(entity);
440                cursor = node.1;
441            }
442            while let Some(entity) = chain.pop() {
443                let &(local, parent, enabled) = &nodes[&entity];
444                let (parent_world, parent_active) = parent.map(|p| resolved[&p]).unwrap_or((Mat4::IDENTITY, true));
445                resolved.insert(entity, (parent_world * local, enabled && parent_active));
446                visiting.remove(&entity);
447            }
448        }
449        for (entity, (global, active)) in world.query_mut::<(&mut GlobalTransform, &mut Active)>() {
450            let (matrix, enabled) = resolved[&entity];
451            global.0 = matrix;
452            active.0 = enabled;
453        }
454        Ok(())
455    }

457    /// Remove a hierarchy, including duplicate component instances. Saved IDs
458    /// are removed too; a reused hecs slot cannot satisfy an old ObjectRef.
459    pub fn despawn_recursive(&mut self, root: Entity) -> Result<(), HierarchyError> {
460        if self.object_id(root).is_none() { return Err(HierarchyError::MissingEntity); }
461        let mut children: HashMap<Entity, Vec<Entity>> = HashMap::new();
462        self.with_world(|world| {
463            for (entity, parent) in world.query::<&Parent>().iter() {
464                if let Some(parent) = parent.0 { children.entry(parent).or_default().push(entity); }
465            }
466        });
467        let mut pending = vec![root];
468        let mut removed = HashSet::new();
469        while let Some(entity) = pending.pop() {
470            if !removed.insert(entity) { continue; }
471            pending.extend(children.get(&entity).into_iter().flatten().copied());
472        }
473        self.remove_objects(&removed);
474        Ok(())
475    }

477    /// Remove one object after its component set has been detached or dropped.
478    /// Retained children become roots; scene editing applies its own parent map.
479    pub fn despawn(&mut self, entity: Entity) -> Result<(), HierarchyError> {
480        if self.object_id(entity).is_none() { return Err(HierarchyError::MissingEntity); }
481        self.remove_objects(&HashSet::from([entity]));
482        Ok(())
483    }
484
485    /// Remove objects missing from an authoring scene during migration. Unlike
486    /// hierarchy deletion, retained children are detached rather than deleted.
487    pub fn retain_objects(&mut self, ids: &HashSet<ObjectId>) {
488        let removed = self.objects.iter().filter_map(|(id, &entity)| (!ids.contains(id)).then_some(entity)).collect();
489        self.remove_objects(&removed);
490    }
491
492    fn remove_objects(&mut self, removed: &HashSet<Entity>) {
493        let mut world = self.world.exclusive();
494        for (_, parent) in world.query_mut::<&mut Parent>() {
495            if parent.0.is_some_and(|p| removed.contains(&p)) { parent.0 = None; }
496        }
497        let entities: Vec<_> = world.query_mut::<&Owner>().into_iter()
498            .filter_map(|(entity, owner)| removed.contains(&owner.0).then_some(entity)).collect();
499        for &entity in &entities { world.despawn(entity).expect("queried entity"); }
500        self.world.forget(&entities);
501        self.objects.retain(|_, entity| !removed.contains(entity));
502    }

505/// When in the frame a system runs. After each stage the engine applies the
506/// [`ComponentCommand`]s that stage queued.
507///
508/// A frame runs: `Start` (repeating while start-up work spawns more objects),
509/// animation markers as `Events`, then `FixedUpdate` zero to eight times at 60 Hz
510/// (each followed by a physics step), physics contacts as `Events`, `Update`,
511/// `Gameplay`, navigation and animation, and finally `LateUpdate`.
512///
513/// See [Stages](https://lemonxah.github.io/zestengine/docs/ECS.html#stages) in the ECS guide.
514#[derive(Clone, Copy, Debug, PartialEq, Eq)]
515pub enum Stage {
516    /// Before each physics step, at a fixed 60 Hz: `dt` is 1/60 s whatever the frame
517    /// rate, and a slow frame runs it up to eight times. Continuous forces belong here.
518    FixedUpdate,
519    /// Once per frame, after physics: movement, controllers, AI steering, general logic.
520    Update,
521    /// After gameplay, navigation and animation: cameras following a target, HUD
522    /// bindings, anything that mirrors the frame's final state.
523    LateUpdate,
524    /// Initialises active new instances, before movement. The engine does not call it
525    /// "once per instance": it runs each frame, up to eight times while start-up
526    /// commands keep spawning objects, so a start-up system tracks completion in its
527    /// own component data (and is best registered with
528    /// [`ComponentRegistry::register_system_if`](crate::ComponentRegistry::register_system_if)).
529    Start,
530    /// After `Update`'s movement commands have been applied, so physics movement has
531    /// resolved before weapons, damage, pickups and health read poses.
532    Gameplay,
533    /// Delivers immediate [`SceneEvent`]s (contacts and animation markers) through
534    /// [`SystemServices::scene_events`]. Runs only when there are such events.
535    Events,
536}
537
538/// The phase of a physics contact between two colliders.
539#[derive(Clone, Copy, Debug, PartialEq, Eq)]
540pub enum ContactPhase {
541    /// The pair started touching (or a trigger was entered) this step.
542    Enter,
543    /// The pair is still touching (reported by 2D physics once per frame).
544    Stay,
545    /// The pair stopped touching (or a trigger was left) this step.
546    Exit,
547}

549/// Events that happen *within* the frame and are delivered to systems in
550/// [`Stage::Events`] through [`SystemServices::scene_events`]. Generic events one
551/// system emits for others are a different channel: they arrive next frame through
552/// [`SystemContext::events`].
553#[derive(Clone, Debug)]
554pub enum SceneEvent {
555    /// Two colliders touched. Each object of a pair receives its own event, with
556    /// itself as `object` and the other as `other`.
557    Contact {
558        /// The receiver of this event.
559        object: ObjectId,
560        /// The object it touched.
561        other: ObjectId,
562        /// A trigger overlap (a collider with *Is Trigger*) rather than a solid collision.
563        trigger: bool,
564        /// Whether the contact began, continues or ended.
565        phase: ContactPhase,
566        /// From the 2D physics world (XY plane) rather than 3D.
567        two_dimensional: bool,
568    },
569    /// An animation clip marker fired on `object`'s animator.
570    Animation {
571        /// The object whose animator passed the marker.
572        object: ObjectId,
573        /// The marker name set on the clip.
574        name: String,
575    },
576}

578/// Everything a system needs besides the world: time, input, the command queue and
579/// the engine services for this frame.
580///
581/// Systems edit component data directly and ask for everything else by pushing a
582/// [`ComponentCommand`] (or calling a helper such as [`SystemContext::emit`]). The
583/// engine applies queued commands, in order, when the stage ends.
584///
585/// Inside [`run_ordered`] you work through an [`EcsMoveHost`], which dereferences to
586/// this type: `host.dt`, `host.input` and `host.emit(..)` all work.
587///
588/// ```ignore
589/// fn tick(world: &SystemWorld<'_>, ctx: &mut SystemContext<'_>) {
590///     let speed = ctx.input.axis2("Move").length() * ctx.dt;       // time and input
591///     ctx.commands.push(ComponentCommand::Quit);                    // a queued command
592///     ctx.emit(LevelComplete);                                      // visible next frame
593/// }
594/// ```
595///
596/// See [`SystemContext` in the ECS guide](https://lemonxah.github.io/zestengine/docs/ECS.html#what-a-system-can-see).
597pub struct SystemContext<'a> {
598    /// Seconds this frame, **scaled by the time scale** (0 while paused). In
599    /// [`Stage::FixedUpdate`] this is the fixed step, 1/60 s. A single frame is clamped
600    /// to 0.25 s so a hitch cannot teleport everything.
601    pub dt: f32,
602    /// Seconds this frame ignoring the time scale: use it for pause menus and UI that
603    /// must keep running while the game is paused.
604    pub unscaled_dt: f32,
605    /// Scaled seconds since Play started.
606    pub time: f32,
607    /// The shared water clock. Networked clients agree on it, which is what makes wave
608    /// riders and floating bodies deterministic.
609    pub water_time: f32,
610    /// This frame's snapshot of the player's actions (`down`, `pressed`, `axis2`, …).
611    pub input: &'a InputState,
612    /// The command queue. Push [`ComponentCommand`]s here; they are applied after the
613    /// stage in the order pushed. A command naming an object that no longer exists is ignored.
614    pub commands: &'a mut Vec<ComponentCommand>,
615    /// Borrowed engine services (physics, navigation, network, screen, …) and the
616    /// event streams.
617    pub services: SystemServices<'a>,
618}

620/// Reads an object's material parameters through checked native render metadata.
621/// The engine implements it over the scene's materials; implementations must read
622/// through the supplied [`SystemWorld`], not through editor component views.
623pub trait MaterialAccess {
624    /// The parameters of material `slot` (0 is the main material) on `object`, or
625    /// `None` if the object or slot does not exist.
626    fn material_params(&self, world: &SystemWorld<'_>, object: ObjectId, slot: usize) -> Option<crate::MaterialParams>;
627}

629/// Engine services lent to systems for one frame, without copying scene data.
630///
631/// Every service is optional: a headless test, a tool or a server without a
632/// renderer simply leaves some of them `None`, and the [`EcsMoveHost`] helpers
633/// built on them degrade gracefully (a raycast returns `None`, the pointer reads as
634/// absent). In tests, set the ones your system uses
635/// (`ctx.services.physics = Some(&mock)`).
636#[derive(Clone, Copy, Default)]
637pub struct SystemServices<'a> {
638    /// Immediate physics queries: raycasts, overlaps, velocities, grounded checks
639    /// (3D and 2D). `None` without a physics world.
640    pub physics: Option<&'a dyn crate::PhysicsQuery>,
641    /// Navmesh queries: paths and nearest walkable point. `None` until the navmesh is baked.
642    pub nav: Option<&'a dyn crate::NavQuery>,
643    /// The network session as seen this frame: connection state, peers, owners, RPCs.
644    pub net: Option<&'a crate::NetView>,
645    /// The game view: size, pointer, camera matrices, UI clicks, VR and control state.
646    pub screen: Option<&'a crate::ScreenInfo>,
647    /// Material parameter reads (see [`MaterialAccess`]).
648    pub materials: Option<&'a dyn MaterialAccess>,
649    /// The current save data (read with `int`, `float`, `bool`, `str`, `vec3`).
650    pub save: Option<&'a crate::SaveGame>,
651    /// Localisation tables for `tr`.
652    pub strings: Option<&'a crate::Localization>,
653    /// The current time scale (1 normal, 0 paused).
654    pub time_scale: Option<f32>,
655    /// Events emitted in the **previous** frame, as type-erased values. Read them by
656    /// type with [`SystemContext::events`]; emitting an event queues it for the next frame.
657    pub events: &'a [crate::AnyBox],
658    /// Authoring slot metadata (which components an object has, in inspector order).
659    /// Metadata only: payload reads go through [`SystemWorld`].
660    pub components: Option<&'a dyn ComponentIndex>,
661    /// Contacts and animation markers of this frame, delivered in [`Stage::Events`].
662    /// Empty in every other stage.
663    pub scene_events: &'a [SceneEvent],
664    /// The scene's authoring order (saved id to position), when hosted by an ordered
665    /// scene. [`run_ordered`] sorts by it so behaviour is deterministic; standalone
666    /// worlds can omit it and fall back to saved ids for ties.
667    pub order: Option<&'a HashMap<ObjectId, usize>>,
668}
669
670impl<'a> SystemContext<'a> {
671    /// A context with zero time and no services, over the given input and command queue.
672    /// The starting point for tests and tools; set `dt`, `time` and the `services` you need.
673    pub fn new(input: &'a InputState, commands: &'a mut Vec<ComponentCommand>) -> Self {
674        Self { dt: 0.0, unscaled_dt: 0.0, time: 0.0, water_time: 0.0, input, commands, services: Default::default() }
675    }
676    /// The events of type `E` that were emitted during the **previous** frame, in
677    /// emission order. Events of other types are skipped. The iterator does not
678    /// borrow the context, so you can call [`SystemContext::emit`] while looping.
679    ///
680    /// ```ignore
681    /// for hit in ctx.events::<Hit>() {
682    ///     total += hit.amount;
683    /// }
684    /// ```
685    pub fn events<E: std::any::Any>(&self) -> impl Iterator<Item = &'a E> + use<'a, E> {
686        self.services.events.iter().filter_map(|event| event.0.downcast_ref::<E>())
687    }
688    /// Broadcast `event` (any `'static` type) to every system. It becomes readable
689    /// through [`SystemContext::events`] in the **next** frame, never in this one, so
690    /// a sender cannot react to its own event within the frame.
691    ///
692    /// ```ignore
693    /// ctx.emit(Died { object: me });
694    /// ```
695    pub fn emit<E: std::any::Any>(&mut self, event: E) {
696        self.commands.push(ComponentCommand::Emit(crate::AnyBox(Box::new(event))));
697    }
698    /// Whether `entity` should run: its [`Active`] flag is set (enabled, and every
699    /// ancestor enabled) and it is visible in the scene. Check this before acting
700    /// for an object; disabled objects must not update.
701    pub fn is_active(&self, world: &SystemWorld<'_>, entity: Entity) -> bool {
702        world.get::<Active>(entity).is_some_and(|active| active.0) && self.is_visible(world, entity)
703    }
704    /// Whether `entity` is an object of the scene being run. Hidden or foreign
705    /// objects (for example, ones not in the scene's authoring order) are not visible,
706    /// and the host helpers treat them as absent.
707    pub fn is_visible(&self, world: &SystemWorld<'_>, entity: Entity) -> bool {
708        world.object_id(entity).is_some_and(|id| self.services.order.is_none_or(|order| order.contains_key(&id)))
709    }
710    /// A sort key that puts objects in scene (authoring) order, ties broken by saved
711    /// id. Sort by it whenever the order in which objects act can change the outcome.
712    pub fn order_key(&self, world: &SystemWorld<'_>, entity: Entity) -> (usize, u128) {
713        let id = world.object_id(entity).unwrap_or(ObjectId::NIL);
714        (self.services.order.and_then(|order| order.get(&id).copied()).unwrap_or(usize::MAX), id.raw())
715    }
716    /// A sort key for one instance of component `T`: object order, then the
717    /// instance's position among the object's components, then entity bits. It keeps
718    /// duplicate instances of a type in inspector order.
719    pub fn instance_order<T: DataComponent>(&self, world: &SystemWorld<'_>, owner: Entity, instance: Entity) -> ((usize, u128), usize, u64) {
720        let slot = self.services.components.and_then(|index|
721            (0..index.component_count(owner)?).find(|&position| index.component_slot(owner, position).is_some_and(|slot|
722                slot.entity() == instance && slot.type_id() == std::any::TypeId::of::<T>())))
723            .unwrap_or(usize::MAX);
724        (self.order_key(world, owner), slot, instance.to_bits().get())
725    }
726}

809impl ComponentRegistry {
810    /// Register data for native ECS systems, retaining the same scene file,
811    /// inspector and gizmo registration as an existing TypedComponent.
812    ///
813    /// Call this for every component type a plugin adds, next to its systems. It is
814    /// the same call as [`ComponentRegistry::register`].
815    pub fn register_ecs<T: TypedComponent + Send + std::marker::Sync>(&mut self) { self.register::<T>(); }
816
817    /// Register an **entity**: a named, ready-made object, described by a mesh shape
818    /// and the components a new instance starts with. The editor lists it under
819    /// Create > Entity, and systems spawn it by name with
820    /// [`EcsMoveHost::spawn_entity`](crate::ecs::EcsMoveHost::spawn_entity).
821    ///
822    /// `build` runs for every new instance, so each object gets its own component
823    /// values. Names are unique: registering an existing name replaces the builder
824    /// in place, so a hot-reloaded plugin supersedes its previous build, and a
825    /// reload drops entities the plugin no longer registers.
826    ///
827    /// ```ignore
828    /// fn crate_entity() -> EntityTemplate {
829    ///     EntityTemplate::new(EntityShape::Cube).with(Lifetime { seconds_left: 10.0 })
830    /// }
831    /// registry.register_entity("Crate", crate_entity);
832    /// ```
833    ///
834    /// See [Entities](https://lemonxah.github.io/zestengine/docs/ECS.html#entities) in the ECS guide.
835    pub fn register_entity(&mut self, name: &'static str, build: EntityFn) {
836        let plugin = self.system_scope.as_ref().map(|(name, _)| name.clone());
837        let generation = self.system_scope.as_ref().map_or(0, |(_, generation)| *generation);
838        let registration = EntityRegistration { name, build, plugin, generation };
839        if let Some(entity) = self.entities.iter_mut().find(|entity| entity.name == name) {
840            *entity = registration;
841        } else { self.entities.push(registration); }
842    }
843
844    /// The names of every registered entity, in registration order (what the
845    /// editor's Create > Entity menu lists).
846    pub fn entity_names(&self) -> impl Iterator<Item = &'static str> + '_ { self.entities.iter().map(|entity| entity.name) }
847
848    /// A fresh template for a new instance of a registered entity. None for an
849    /// unknown name (its plugin is not loaded).
850    pub fn create_entity(&self, name: &str) -> Option<EntityTemplate> {
851        self.entities.iter().find(|entity| entity.name == name).map(|entity| (entity.build)())
852    }

854    /// Register a system that runs every time `stage` runs.
855    ///
856    /// Names are unique: registering an existing name replaces the code and stage in
857    /// place, which keeps execution order deterministic across hot reloads. Systems
858    /// of one stage run in registration order, each seeing the previous one's edits.
859    /// Prefer [`register_system_for`](Self::register_system_for) so unused features cost nothing.
860    ///
861    /// ```ignore
862    /// registry.register_system("game::remove_the_dead", Stage::Gameplay, remove_the_dead);
863    /// ```
864    pub fn register_system(&mut self, name: &'static str, stage: Stage, run: SystemFn) {
865        self.register_system_inner(name, stage, run, None);
866    }

868    /// Register a system that runs only while at least one **enabled** `T` exists.
869    /// Skipping also skips the stage's hierarchy propagation and service set-up when
870    /// no other system needs them, so features nobody uses cost nothing. This is
871    /// the registration to reach for by default.
872    ///
873    /// ```ignore
874    /// registry.register_system_for::<Health>("game::health", Stage::Gameplay, health);
875    /// ```
876    pub fn register_system_for<T: DataComponent>(&mut self, name: &'static str, stage: Stage, run: SystemFn) {
877        self.register_system_if(name, stage, run, |world| world.query::<&T>().iter().next().is_some());
878    }
879
880    /// Register a system that runs only while `condition(world)` is true. The
881    /// condition is evaluated every time the stage is considered, so keep it cheap.
882    /// This is the tool for start-up work: run while any instance still needs
883    /// initialising, then go quiet.
884    ///
885    /// ```ignore
886    /// registry.register_system_if("game::door_start", Stage::Start, start_doors,
887    ///     |world| world.query::<&Door>().iter().any(|(_, door)| door.closed_y.is_none()));
888    /// ```
889    pub fn register_system_if(&mut self, name: &'static str, stage: Stage, run: SystemFn, condition: fn(&SystemWorld<'_>) -> bool) {
890        self.register_system_inner(name, stage, run, Some(condition));
891    }
892
893    fn register_system_inner(&mut self, name: &'static str, stage: Stage, run: SystemFn, condition: Option<fn(&SystemWorld<'_>) -> bool>) {
894        let plugin = self.system_scope.as_ref().map(|(name, _)| name.clone());
895        let generation = self.system_scope.as_ref().map_or(0, |(_, generation)| *generation);
896        let registration = SystemRegistration { name, stage, run, condition, plugin, generation };
897        if let Some(system) = self.systems.iter_mut().find(|system| system.name == name) {
898            *system = registration;
899        } else { self.systems.push(registration); }
900    }
901
902    /// Register one plugin's current systems and entities. Reloading removes
903    /// those that disappeared from its export, while retaining the order of the
904    /// survivors and leaving the engine's and other plugins' registrations alone.
905    pub fn register_plugin(&mut self, name: &str, register: fn(&mut Self)) {
906        use std::panic::{AssertUnwindSafe, catch_unwind, resume_unwind};
907        self.system_generation += 1;
908        let generation = self.system_generation;
909        let prior_scope = self.system_scope.replace((name.to_owned(), generation));
910        let prior_systems = self.systems.clone();
911        let prior_entities = self.entities.clone();
912        let outcome = catch_unwind(AssertUnwindSafe(|| register(self)));
913        self.system_scope = prior_scope;
914        match outcome {
915            Ok(()) => {
916                self.systems.retain(|system| system.plugin.as_deref() != Some(name) || system.generation == generation);
917                self.entities.retain(|entity| entity.plugin.as_deref() != Some(name) || entity.generation == generation);
918            }
919            Err(panic) => { self.systems = prior_systems; self.entities = prior_entities; resume_unwind(panic); }
920        }
921    }

923    /// Whether any system is registered for `stage`, runnable or not.
924    pub fn has_systems(&self, stage: Stage) -> bool { self.systems.iter().any(|system| system.stage == stage) }
925
926    /// Whether any system registered for `stage` would run right now (it has no
927    /// condition, or its condition holds). The engine uses it to skip a stage entirely.
928    pub fn has_runnable_systems(&self, stage: Stage, world: &SceneWorld) -> bool {
929        self.has_systems(stage) && world.with_world(|world| self.systems.iter().filter(|system| system.stage == stage)
930            .any(|system| system.condition.is_none_or(|condition| condition(world))))
931    }
932
933    /// Run every runnable system registered for `stage`, in registration order, over
934    /// `world`. The engine calls this once per stage per frame; tests call it to drive
935    /// systems headlessly. Commands the systems queue accumulate in `context.commands`;
936    /// applying them is the caller's job (the engine does it at the stage boundary).
937    ///
938    /// Panics inside a system propagate after the world's locks have been released.
939    pub fn run_systems(&self, stage: Stage, world: &SceneWorld, context: &mut SystemContext<'_>) {
940        for system in self.systems.iter().filter(|system| system.stage == stage) {
941            world.with_world(|view| {
942                if system.condition.is_none_or(|condition| condition(view)) { (system.run)(view, context); }
943            });
944        }
945    }

948/// Run `update` for every active instance of component `T`, in scene (authoring)
949/// order, handing it the component and an [`EcsMoveHost`] bound to the object.
950///
951/// Compared with a hand-written query loop it skips disabled objects (and anything
952/// under a disabled parent), iterates deterministically (so networked prediction and
953/// object-to-object interactions behave the same on every machine), and gives you the
954/// host helpers for moving, reading other objects, spawning, physics and more.
955/// Duplicate instances of `T` on one object each get a call, in inspector order.
956///
957/// The ordering collects and sorts the instances every call; for thousands of
958/// instances where order does not matter, a bare [`SystemWorld::query`] is cheaper.
959///
960/// ```ignore
961/// impl Lifetime {
962///     fn update(&mut self, host: &mut EcsMoveHost<'_, '_>) {
963///         self.seconds_left -= host.dt;
964///         if self.seconds_left <= 0.0 { host.destroy_self(); }
965///     }
966/// }
967///
968/// registry.register_system_for::<Lifetime>("game::lifetime", Stage::Update, |world, ctx| {
969///     run_ordered(world, ctx, Lifetime::update)
970/// });
971/// ```
972///
973/// See [`run_ordered` in the ECS guide](https://lemonxah.github.io/zestengine/docs/ECS.html#the-comfortable-way-run_ordered-and-ecsmovehost).
974pub fn run_ordered<T: DataComponent>(world: &SystemWorld<'_>, ctx: &mut SystemContext<'_>, update: impl Fn(&mut T, &mut EcsMoveHost<'_, '_>)) {
975    let mut query = world.query::<(&Owner, &mut T)>();
976    let mut actors: Vec<_> = query.iter().filter(|(_, (owner, _))| ctx.is_active(world, owner.0)).collect();
977    actors.sort_unstable_by_key(|(instance, (owner, _))| ctx.instance_order::<T>(world, owner.0, *instance));
978    for (_, (owner, value)) in actors { update(value, &mut EcsMoveHost::new(world, owner.0, ctx)); }
979}

1022fn ownership_controls(world: &SystemWorld<'_>, ctx: &mut SystemContext<'_>) {
1023    let mut query = world.query::<(&Owner, &mut crate::Sync)>();
1024    let mut objects: Vec<_> = query.iter().filter(|(_, (owner, _))| ctx.is_active(world, owner.0)).collect();
1025    objects.sort_unstable_by_key(|(_, (owner, _))| ctx.order_key(world, owner.0));
1026    for (_, (owner, sync)) in objects {
1027        sync.started = true;
1028        let Some(net) = ctx.services.net.filter(|net| net.connected) else { continue };
1029        if !sync.grabbable || sync.grab_action.is_empty() || !ctx.input.pressed(&sync.grab_action) { continue; }
1030        let id = world.object_id(owner.0).expect("component owner");
1031        ctx.commands.push(if net.owns(id) { ComponentCommand::ReleaseOwnership(id) } else { ComponentCommand::RequestOwnership(id) });
1032    }
1033}

1035fn motion(world: &SystemWorld<'_>, ctx: &mut SystemContext<'_>) {
1036    use glam::{Quat, Vec3};
1037    for (_, (owner, spin)) in world.query::<(&Owner, &crate::Spin)>().iter() {
1038        if !world.get::<Active>(owner.0).is_some_and(|active| active.0) { continue; }
1039        let mut transform = world.get_mut::<Transform>(owner.0).expect("component owner");
1040        let axis = Vec3::from(spin.axis).normalize_or(Vec3::Y);
1041        let step = Quat::from_axis_angle(axis, spin.degrees_per_second.to_radians() * ctx.dt);
1042        transform.rotation = (step * transform.rotation).normalize();
1043    }
1044    for (_, (owner, bob)) in world.query::<(&Owner, &mut crate::Bob)>().iter() {
1045        if !world.get::<Active>(owner.0).is_some_and(|active| active.0) { continue; }
1046        let mut transform = world.get_mut::<Transform>(owner.0).expect("component owner");
1047        let phase = ctx.time * bob.frequency * std::f32::consts::TAU;
1048        let offset = Vec3::from(bob.axis) * (bob.amplitude * phase.sin());
1049        transform.translation += offset - bob.applied;
1050        bob.applied = offset;
1051    }
1052}
