9/// One wheel of a wheeled vehicle. Positions are in the vehicle object's
10/// local space (metres), forward = local -Z.
11#[derive(Clone, Copy, Debug, PartialEq, Serialize, Deserialize)]
12#[serde(default)]
13pub struct Wheel {
14    /// Wheel centre in the vehicle's local space.
15    pub anchor: [f32; 3],
16    /// Suspension rest length (metres): how far the wheel hangs below `anchor`.
17    pub rest_length: f32,
18    /// Spring stiffness (N/m).
19    pub spring: f32,
20    /// Damping (N·s/m).
21    pub damper: f32,
22    /// Wheel radius (metres).
23    pub radius: f32,
24    /// How far the suspension can compress from rest (metres).
25    pub travel: f32,
26    /// This wheel steers.
27    pub steer: bool,
28    /// This wheel is driven by the engine and brakes.
29    pub drive: bool,
30    /// Tyre grip (0 = ice, 1 = road tyres).
31    pub grip: f32,
32}
33
34impl Default for Wheel {
35    fn default() -> Self {
36        Self {
37            anchor: [0.0, -0.45, 0.0],
38            rest_length: 0.35,
39            spring: 60_000.0,
40            damper: 2_500.0,
41            radius: 0.35,
42            travel: 0.25,
43            steer: false,
44            drive: true,
45            grip: 1.0,
46        }
47    }
48}

50/// Suspension force (N, up) for a wheel from its compression and compression
51/// velocity: the spring pushes back against compression, and the damper resists
52/// motion (adds force while compressing).
53pub fn suspension_force(wheel: &Wheel, compression: f32, compression_velocity: f32) -> f32 {
54    compression.max(0.0) * wheel.spring + compression_velocity * wheel.damper
55}
56
57/// Net longitudinal drive force (N). Throttle (0..1) accelerates; brake (0..1)
58/// decelerates against the current travel direction, so the vehicle can also
59/// reverse under brake.
60pub fn drive_force(engine: f32, throttle: f32, brake: f32, forward_speed: f32) -> f32 {
61    let accel = throttle.clamp(0.0, 1.0) * engine;
62    let brake_dir = if forward_speed >= 0.0 { -1.0 } else { 1.0 };
63    accel + brake.clamp(0.0, 1.0) * engine * brake_dir
64}

66/// A raycast-suspension wheeled vehicle. Put it on the same object as a
67/// dynamic [`crate::RigidBody`] (and a chassis collider held above the wheel
68/// contact points). Controls default to the First / Third Person Controller's
69/// `Move` / `Jump` / `Sprint` actions, so the same WASD + Space + Shift scheme
70/// drives it: `Move.y` throttles and brakes (reverse), `Move.x` steers, Jump
71/// brakes and Sprint applies the handbrake.
72#[derive(Serialize, Deserialize)]
73#[serde(default)]
74pub struct Vehicle {
75    /// The wheels.
76    pub wheels: Vec<Wheel>,
77    /// Maximum steering angle of the steered wheels, degrees.
78    pub steering_degrees: f32,
79    /// Drive force (Newtons) split across the driven wheels; braking uses the
80    /// same force.
81    pub engine_force: f32,
82    /// Forward speed (m/s) the engine force tapers off at.
83    pub top_speed: f32,
84    /// Downforce at `top_speed`, pushing the chassis into the road (Newtons).
85    pub downforce: f32,
86    /// 2D action: x = steering, y = throttle / brake.
87    pub move_action: String,
88    /// Button action held to brake.
89    pub brake_action: String,
90    /// Button action held to apply the handbrake to the non-steered wheels.
91    pub handbrake_action: String,
92
93    // Transient control state, written by the native fixed-step system and read by the
94    // engine's vehicle step.
95    /// Throttle input 0..1 (runtime, written each fixed step).
96    #[serde(skip)]
97    pub throttle: f32,
98    /// Brake input 0..1 (runtime).
99    #[serde(skip)]
100    pub brake: f32,
101    /// Steering input -1..1 (runtime).
102    #[serde(skip)]
103    pub steer: f32,
104    /// Handbrake on (runtime).
105    #[serde(skip)]
106    pub handbrake: bool,
107}

109fn default_wheels() -> Vec<Wheel> {
110    let mut wheels = Vec::with_capacity(4);
111    for (x, z, steer) in [(-0.8, -1.2, true), (0.8, -1.2, true), (-0.8, 1.2, false), (0.8, 1.2, false)] {
112        wheels.push(Wheel {
113            anchor: [x, -0.45, z],
114            steer,
115            ..Default::default()
116        });
117    }
118    wheels
119}

140impl Vehicle {
141    /// Read the driving controls from an input snapshot into the transient
142    /// control state, so it can be tested without a runtime scene.
143    pub fn read_input(&mut self, input: &InputState) {
144        let mv = input.axis2(&self.move_action);
145        self.steer = mv.x.clamp(-1.0, 1.0);
146        self.throttle = mv.y.clamp(0.0, 1.0);
147        self.brake = ((-mv.y).max(0.0) + input.down(&self.brake_action) as u8 as f32).clamp(0.0, 1.0);
148        self.handbrake = input.down(&self.handbrake_action);
149    }
150}

156/// Native fixed-step controls. Data remains in the same typed Vehicle column
157/// that suspension reads; remote authority and focused UI suppress local input.
158pub(crate) fn controls(world: &SystemWorld<'_>, ctx: &mut SystemContext<'_>) {
159    for (_, (owner, vehicle)) in world.query::<(&Owner, &mut Vehicle)>().iter() {
160        if !ctx.is_active(world, owner.0) { continue; }
161        let remote = ctx.services.net.is_some_and(|net| net.connected
162            && (world.get::<crate::NetworkObject>(owner.0).is_some() || world.get::<crate::Sync>(owner.0).is_some())
163            && !net.has_authority(world.object_id(owner.0).expect("component owner")));
164        if remote || ctx.services.screen.is_some_and(|screen| screen.ui_text_editing) {
165            vehicle.throttle = 0.0;
166            vehicle.brake = 0.0;
167            vehicle.steer = 0.0;
168            vehicle.handbrake = false;
169        } else { vehicle.read_input(ctx.input); }
170    }
171}

173#[cfg(test)]
174mod tests {
175    use super::*;
176    use crate::{ActionValue, InputState};

178    fn input(actions: Vec<(String, ActionValue)>) -> InputState {
179        InputState::from_actions(actions, glam::Vec2::ZERO)
180    }
181
182    #[test]
183    fn suspension_is_zero_at_rest_length() {
184        let w = Wheel::default();
185        assert_eq!(suspension_force(&w, 0.0, 0.0), 0.0);
186        assert_eq!(suspension_force(&w, -0.1, 0.0), 0.0, "negative compression clamps");
187    }
188
189    #[test]
190    fn suspension_spring_and_damper() {
191        let w = Wheel::default();
192        let f = suspension_force(&w, 0.1, 0.5);
193        assert!((f - (0.1 * w.spring + 0.5 * w.damper)).abs() < 1e-3);
194    }
195
196    #[test]
197    fn drive_throttles_and_brakes() {
198        assert!(drive_force(1000.0, 1.0, 0.0, 0.0) > 0.0);
199        assert!(drive_force(1000.0, 0.0, 1.0, 5.0) < 0.0, "brake opposes forward speed");
200        assert!(drive_force(1000.0, 0.0, 1.0, -5.0) > 0.0, "brake opposes reverse speed");
201        assert_eq!(drive_force(1000.0, 0.0, 0.0, 3.0), 0.0);
202    }
203
204    #[test]
205    fn move_axis_steers_throttles_and_brakes() {
206        let mut v = Vehicle::default();
207        let mv = ActionValue {
208            down: true,
209            value: 1.0,
210            axis2: glam::Vec2::new(0.6, 0.8),
211            ..Default::default()
212        };
213        v.read_input(&input(vec![("Move".into(), mv)]));
214        assert!((v.steer - 0.6).abs() < 1e-4);
215        assert!((v.throttle - 0.8).abs() < 1e-4);
216        assert_eq!(v.brake, 0.0);
217
218        let back = ActionValue {
219            down: true,
220            value: 1.0,
221            axis2: glam::Vec2::new(0.0, -1.0),
222            ..Default::default()
223        };
224        v.read_input(&input(vec![("Move".into(), back)]));
225        assert_eq!(v.throttle, 0.0);
226        assert_eq!(v.brake, 1.0);
227    }
