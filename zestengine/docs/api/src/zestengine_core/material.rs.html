1//! Continuously editable material parameters: the single block of data
2//! that describes how a surface shades. Plugins read and write materials
3//! through this type (see `crate::ecs::EcsMoveHost::material_params` and
4//! `crate::ecs::EcsMoveHost::modify_material`), so it lives in core where plugins link.
5
6use serde::{Deserialize, Serialize};
7
8/// Continuously editable material parameters; delivered to the shader as
9/// push constants every draw, so edits are live with zero sync hazards.
10#[derive(Clone, Copy, Debug, Serialize, Deserialize)]
11#[serde(default)]
12pub struct MaterialParams {
13    /// Base albedo, linear RGBA; the alpha is used by transparent materials.
14    pub base_color: [f32; 4],
15    /// Emissive colour, linear RGB (only meaningful when the emission map or intensity makes it visible).
16    pub emission_color: [f32; 3],
17    /// Multiplier on the emission colour / map (0 = off).
18    pub emission_intensity: f32,
19    /// Mean LINEAR RGB of the emission map (1,1,1 = no map). Scales the per-
20    /// instance emission fed to the RT reflection + RT GI gather so a mostly-black
21    /// map (e.g. only a glowing visor) doesn't make the whole mesh reflect/bounce
22    /// as a uniform light. The per-pixel forward emission is unaffected (it
23    /// samples the actual map).
24    pub emission_map_mean: [f32; 3],
25    /// Metalness 0 (dielectric) .. 1 (metal).
26    pub metallic: f32,
27    /// Perceptual roughness 0 (mirror) .. 1 (diffuse).
28    pub roughness: f32,
29    /// Toon cel-shading band count (1 = flat).
30    pub toon_steps: f32,
31    /// Blend between PBR and toon shading, 0 = PBR .. 1 = toon.
32    pub pbr_toon_blend: f32,
33    /// Alpha-test threshold 0..1 for cutout materials.
34    pub alpha_cutoff: f32,
35    /// Normal-map intensity (0 = off).
36    pub normal_strength: f32,
37    /// Ambient-occlusion map strength (0 = off).
38    pub occlusion_strength: f32,
39    // --- extended "FX" params, delivered via a per-material UBO (set 1, binding
40    // 4) so the built-in Standard/Toon shaders can have options beyond the
41    // full 128-byte push block. ---
42    /// Toon rim-light colour.
43    pub rim_color: [f32; 3],
44    /// Toon rim Fresnel exponent (higher = tighter edge).
45    pub rim_power: f32,
46    /// Toon rim strength (0 = off).
47    pub rim_strength: f32,
48    /// Toon cel-ramp edge softness.
49    pub ramp_smoothness: f32,
50    /// Animated UV scroll for the base maps (uv units / second).
51    pub base_scroll: [f32; 2],
52    /// Animated UV scroll for the emission map.
53    pub emission_scroll: [f32; 2],
54    /// Emission pulse speed (0 = steady); brightness oscillates.
55    pub emission_pulse: f32,
56    /// Additive matcap strengths (3 layers; 0 = off). Each layer samples its own
57    /// matcap texture, multiplied by its mask.
58    pub matcap_strength: [f32; 3],
59    /// Per-layer matcap blend mode encoded as a float: 0 = Add, 1 = Multiply,
60    /// 2 = Replace.
61    pub matcap_blend: [f32; 3],
62    // --- Per-texture UV transform for the main maps. `*_tiling` scales the UVs
63    // (Unity tiling; default 1), `*_offset` shifts them (default 0). Mask slots
64    // follow their parent map (opacity→albedo, emission_mask→emission). ---
65    /// UV scale for the albedo map (Unity tiling; default 1).
66    pub albedo_tiling: [f32; 2],
67    /// UV shift for the albedo map (default 0).
68    pub albedo_offset: [f32; 2],
69    /// UV scale for the normal map.
70    pub normal_tiling: [f32; 2],
71    /// UV shift for the normal map.
72    pub normal_offset: [f32; 2],
73    /// UV scale for the ORM map.
74    pub orm_tiling: [f32; 2],
75    /// UV shift for the ORM map.
76    pub orm_offset: [f32; 2],
77    /// UV scale for the emission map.
78    pub emission_tiling: [f32; 2],
79    /// UV shift for the emission map.
80    pub emission_offset: [f32; 2],
81    // Invert a split AO / Roughness / Metallic map (e.g. a smoothness map used in
82    // the roughness slot). Only meaningful when the matching map is assigned.
83    /// Invert the AO channel of a split ORM map.
84    pub ao_invert: bool,
85    /// Invert the roughness channel (e.g. a smoothness map).
86    pub roughness_invert: bool,
87    /// Invert the metallic channel.
88    pub metallic_invert: bool,
89    /// Parallax occlusion mapping strength (uv shift at full height). 0 = off.
90    pub displacement_scale: f32,
91    /// Per-material reflection strength (1 = full, 0 = matte). Scales the env-cube
92    /// probe reflection AND the screen-space/RT reflection for this material.
93    pub reflection_intensity: f32,
94    /// Per-material screen-space/RT reflection toggle (true = on). Off keeps the
95    /// env-cube reflection but skips the deferred SSR/RT resolve for this material.
96    pub screen_reflections: bool,
97    /// Per-material reflection mode: 0 = the environment's global technique,
98    /// 2 = baked probes only (no SSR / ray-traced resolve), 3 = none.
99    /// Mirrors `zestengine_core::ReflectionMode` for the shader (fx.parallax.w).
100    #[serde(default)]
101    pub reflection_mode: u8,
102    /// Detail layer (albedo / normal / ORM tiled over the base maps).
103    pub detail: DetailParams,
104    /// Thin-surface translucency and see-through transmission.
105    pub transmission: TransmissionParams,
106    /// Tessellation + height-map displacement (hardware tessellation stages).
107    pub tess: TessParams,
108}

110/// Displacement through hardware tessellation (UNITY-12): the mesh's triangles are
111/// subdivided by their screen-space edge length and the new vertices pushed along
112/// the (Phong-smoothed) normal by the material's height map (the Displacement
113/// texture slot, white = peak). Off by default; a material without it pays
114/// nothing (its pipelines have no tessellation stages). Needs a device with
115/// `tessellationShader`; elsewhere the material draws flat (use parallax mapping).
116#[derive(Clone, Copy, Debug, PartialEq, Serialize, Deserialize)]
117#[serde(default)]
118pub struct TessParams {
119    /// Whether tessellation displacement is active for this material.
120    pub enabled: bool,
121    /// Displacement at height 1 (above `midlevel`), in world units along the normal.
122    pub scale: f32,
123    /// Height-map value that stays put (0.5: half the map raises, half sinks).
124    pub midlevel: f32,
125    /// Target on-screen edge length of the generated triangles, in pixels.
126    pub edge_px: f32,
127    /// Largest subdivision level (the device caps it at 64).
128    pub max_factor: f32,
129    /// Phong smoothing of the subdivided surface, 0 (flat triangles) to 1.
130    pub phong: f32,
131}
132
133impl Default for TessParams {
134    fn default() -> Self {
135        Self { enabled: false, scale: 0.05, midlevel: 0.5, edge_px: 12.0, max_factor: 32.0, phong: 0.5 }
136    }
137}

139/// Standard-shader detail layer: a second set of albedo, normal and ORM maps
140/// sampled at their own tiling, gated by a mask. Every strength is 0 by default,
141/// which switches the layer off (old materials load unchanged).
142#[derive(Clone, Copy, Debug, PartialEq, Serialize, Deserialize)]
143#[serde(default)]
144pub struct DetailParams {
145    /// How strongly the detail albedo shows (0 = off).
146    pub albedo_strength: f32,
147    /// Detail normal scale (0 = off).
148    pub normal_strength: f32,
149    /// How strongly the detail ORM modulates occlusion / roughness / metallic.
150    pub orm_strength: f32,
151    /// How the detail albedo combines: 0 Multiply x2 (Unity), 1 Overlay, 2 Add, 3 Replace.
152    pub blend: f32,
153    /// UV scale for the detail maps (Unity tiling; default 1).
154    pub tiling: [f32; 2],
155    /// UV shift for the detail maps (default 0).
156    pub offset: [f32; 2],
157}
158
159impl Default for DetailParams {
160    fn default() -> Self {
161        Self {
162            albedo_strength: 0.0,
163            normal_strength: 0.0,
164            orm_strength: 0.0,
165            blend: 0.0,
166            tiling: [1.0, 1.0],
167            offset: [0.0, 0.0],
168        }
169    }
170}
171
172impl DetailParams {
173    /// True when any of the three detail strengths is above 0, i.e. the
174    /// detail layer is active rather than fully switched off.
175    pub fn is_on(&self) -> bool {
176        self.albedo_strength > 0.0 || self.normal_strength > 0.0 || self.orm_strength > 0.0
177    }
178}

180/// Standard-shader translucency (light through a thin surface, leaves / paper /
181/// cloth) and transmission (the scene behind a see-through surface, glass /
182/// thin plastic). Both default to 0 = off.
183#[derive(Clone, Copy, Debug, PartialEq, Serialize, Deserialize)]
184#[serde(default)]
185pub struct TransmissionParams {
186    /// Back-lit translucency strength (0 = off).
187    pub translucency: f32,
188    /// Colour of the light that comes through (linear).
189    pub translucency_color: [f32; 3],
190    /// Sharpness of the transmitted glow (higher = tighter around the light).
191    pub power: f32,
192    /// How much the surface normal bends the transmitted light.
193    pub distortion: f32,
194    /// The translucency map holds thickness (1 = thick, passes less) instead of translucency.
195    pub map_is_thickness: bool,
196    /// See-through amount 0..1 (glTF `KHR_materials_transmission`); needs "Scene Colour Read".
197    pub transmission: f32,
198    /// Colour the scene behind takes on passing through (linear); glTF volume attenuation.
199    pub tint: [f32; 3],
200    /// Screen-space refraction offset, in view-normal units.
201    pub refraction: f32,
202    /// Rough transmission: how strongly roughness blurs the scene seen through
203    /// the surface (0 = always sharp, 1 = the full roughness x thickness blur).
204    pub blur: f32,
205    /// Volume thickness 0..1 that scales the blur (times the translucency /
206    /// thickness map's red channel when `map_is_thickness`). glTF
207    /// `KHR_materials_volume` thickness, normalised by the importer.
208    pub thickness: f32,
209    /// The translucency / thickness map's GREEN channel also scales the
210    /// transmission amount (glTF `transmissionTexture`).
211    pub map_has_transmission: bool,
212}
