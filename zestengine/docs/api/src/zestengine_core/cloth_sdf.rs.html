37impl ClothSdf{
38    pub fn from_samples(dims:[u32;3],origin:Vec3,cell:Vec3,values:Vec<f32>)->Result<Self,String>{
39        if dims.iter().any(|&n|n<3||n>256)||!origin.is_finite()||!cell.is_finite()||cell.min_element()<=1e-8{return Err("Invalid level-set dimensions or cell size".into());}
40        let hi=origin+cell*Vec3::from_array(dims.map(|n|(n-1)as f32));
41        if !hi.is_finite()||!(origin+cell).cmpgt(origin).all(){return Err("Level-set bounds or cell spacing exceed coordinate precision".into());}
42        let dims=dims.map(|n|n as usize);let count=dims.iter().product::<usize>();
43        if count>4_000_000||values.len()!=count||values.iter().any(|v|!v.is_finite()){return Err("Invalid level-set samples (limit: four million)".into());}
44        if !values.iter().any(|v|*v<0.0){return Err("Level set contains no interior samples; increase resolution or use a thicker collision proxy".into());}
45        let index=|x,y,z|x+dims[0]*(y+dims[1]*z);let mut slope=Vec3::ZERO;
46        for z in 0..dims[2]{for y in 0..dims[1]{for x in 0..dims[0]{let i=index(x,y,z);
47            if (x==0||y==0||z==0||x+1==dims[0]||y+1==dims[1]||z+1==dims[2])&&values[i]<=0.0{return Err("Level-set boundary must be outside the body; add padding".into());}
48            for (axis,stride) in [1,dims[0],dims[0]*dims[1]].into_iter().enumerate(){if [x,y,z][axis]+1<dims[axis]{slope[axis]=slope[axis].max((values[i+stride]-values[i]).abs()/cell[axis]);}}
49        }}}
50        // Outside the grid the clamped sample is extended by distance to the
51        // box. This also bounds its gradient, including at box edges.
52        let lipschitz=(slope.length()+1.0).max(1.0);
53        if !lipschitz.is_finite(){return Err("Level-set sample gradient exceeds the supported range".into());}
54        Ok(Self(Arc::new(Field{dims,origin,cell,values,lipschitz,encoded:OnceLock::new()})))
55    }
56    pub fn dimensions(&self)->[usize;3]{self.0.dims}
57    pub fn cell_size(&self)->Vec3{self.0.cell}
58    pub fn bounds(&self)->(Vec3,Vec3){let f=&self.0;(f.origin,f.origin+f.cell*Vec3::from_array(f.dims.map(|n|(n-1)as f32)))}
59    pub fn bytes(&self)->usize{self.0.values.len()*4}
60    /// Interpolated signed scalar and its derivative in field coordinates.
61    /// The derivative is not normalized; it is needed for scaled projection.
62    pub fn sample(&self,p:Vec3)->(f32,Vec3){
63        let f=&self.0;let(lo,hi)=self.bounds();let q=p.clamp(lo,hi);let grid=(q-lo)/f.cell;
64        let base:[usize;3]=std::array::from_fn(|i|(grid[i].floor()as usize).min(f.dims[i]-2));let u=grid-Vec3::from_array(base.map(|n|n as f32));
65        let mut value=0.0;let mut gradient=Vec3::ZERO;
66        for z in 0..2{for y in 0..2{for x in 0..2{
67            let bits=[x,y,z];let w: [f32;3]=std::array::from_fn(|i|if bits[i]==0{1.0-u[i]}else{u[i]});
68            let v=f.values[base[0]+x+f.dims[0]*(base[1]+y+f.dims[1]*(base[2]+z))];value+=v*w[0]*w[1]*w[2];
69            for i in 0..3{gradient[i]+=v*(if bits[i]==0{-1.0}else{1.0})*w[(i+1)%3]*w[(i+2)%3]/f.cell[i];}
70        }}}
71        let outside=p-q;let distance=outside.length();if distance>0.0{
72            value+=distance;for i in 0..3{if p[i]<lo[i]||p[i]>hi[i]{gradient[i]=0.0;}}gradient+=outside/distance;
73        }
74        (value,gradient)
75    }
76    /// Generate from a closed triangle surface. UV/normal splits are welded
77    /// by exact position for manifold validation. Winding may be reversed.
78    pub fn from_mesh(vertices:&[Vec3],indices:&[u32],resolution:u32)->Result<Self,String>{
79        if !(8..=128).contains(&resolution)||vertices.len()<4||vertices.len()>1_000_000||indices.len()<12||indices.len()%3!=0||indices.len()>6_000_000||vertices.iter().any(|p|!p.is_finite())||indices.iter().any(|&i|i as usize>=vertices.len()){return Err("Invalid closed level-set mesh or resolution (8–128)".into());}
80        let mut welded=HashMap::new();let ids:Vec<_>=vertices.iter().map(|p|{let key=p.to_array().map(|x|if x==0.0{0}else{x.to_bits()});let next=welded.len();*welded.entry(key).or_insert(next)}).collect();
81        let mut edges=HashMap::new();let mut triangles=Vec::new();
82        for t in indices.chunks_exact(3){let points:[Vec3;3]=std::array::from_fn(|k|vertices[t[k]as usize]);
83            if (points[1]-points[0]).cross(points[2]-points[0]).length_squared()<1e-20{return Err("Level-set mesh contains a degenerate triangle".into());}
84            for k in 0..3{let(a,b)=(ids[t[k]as usize],ids[t[(k+1)%3]as usize]);*edges.entry((a.min(b),a.max(b))).or_insert(0)+=1;}
85            triangles.push(points);
86        }
87        if edges.values().any(|&n|n!=2){return Err("Level-set collision needs a closed manifold mesh; close holes or use a closed collision proxy".into());}
88        let tree=Node::build(&triangles,(0..triangles.len()).collect());let span=tree.hi-tree.lo;let longest=span.max_element();
89        if span.min_element()<=1e-6{return Err("Level-set mesh has no volume".into());}
90        let target_cell=longest/(resolution-1)as f32;let intervals=(span/target_cell).ceil().max(Vec3::splat(4.0));
91        let cell=span/intervals;let origin=tree.lo-cell*2.0;let dims=intervals.to_array().map(|v|v as u32+5);
92        let directions=[DVec3::new(1.0,0.3713906764,0.52999894).normalize(),DVec3::new(-0.213199,1.0,0.419132).normalize(),DVec3::new(0.319133,-0.271199,1.0).normalize()];
93        let mut values=Vec::with_capacity(dims.iter().map(|&v|v as usize).product());let mut hits=Vec::new();
94        for z in 0..dims[2]{for y in 0..dims[1]{for x in 0..dims[0]{
95            let p=origin+Vec3::new(x as f32,y as f32,z as f32)*cell;let mut distance=f32::INFINITY;tree.nearest(&triangles,p,&mut distance);let distance=distance.sqrt();let mut inside=0;
96            if p.cmpge(tree.lo).all()&&p.cmple(tree.hi).all(){for direction in directions{
97                hits.clear();tree.ray(&triangles,p.as_dvec3(),direction,&mut hits);hits.sort_by(f64::total_cmp);
98                // Shared triangle edges contribute one surface crossing.
99                hits.dedup_by(|a,b|(*a-*b).abs()<longest as f64*1e-9);inside+=usize::from(hits.len()%2==1);
100            }}
101            values.push(if inside>=2{-distance}else{distance});
102        }}}
103        Self::from_samples(dims,origin,cell,values)
104    }
105}

107struct Node{lo:Vec3,hi:Vec3,children:Option<(Box<Node>,Box<Node>)>,triangles:Vec<usize>}
108impl Node{
109    fn build(triangles:&[[Vec3;3]],mut ids:Vec<usize>)->Self{
110        let mut lo=Vec3::splat(f32::INFINITY);let mut hi=Vec3::splat(f32::NEG_INFINITY);for &i in &ids{for p in triangles[i]{lo=lo.min(p);hi=hi.max(p);}}
111        let children=if ids.len()>8{let span=hi-lo;let axis=if span.x>=span.y&&span.x>=span.z{0}else if span.y>=span.z{1}else{2};let center=|i:usize|triangles[i].iter().map(|p|p[axis]).sum::<f32>();ids.sort_by(|&a,&b|center(a).total_cmp(&center(b)));let right=ids.split_off(ids.len()/2);Some((Box::new(Self::build(triangles,std::mem::take(&mut ids))),Box::new(Self::build(triangles,right))))}else{None};Self{lo,hi,children,triangles:ids}
112    }
113    fn distance(&self,p:Vec3)->f32{p.distance_squared(p.clamp(self.lo,self.hi))}
114    fn nearest(&self,triangles:&[[Vec3;3]],p:Vec3,best:&mut f32){
115        if self.distance(p)>*best{return;}
116        if let Some((a,b))=&self.children{let(a,b)=if a.distance(p)<b.distance(p){(a,b)}else{(b,a)};a.nearest(triangles,p,best);b.nearest(triangles,p,best);}else{for &i in &self.triangles{*best=best.min(p.distance_squared(crate::cloth_contact::closest_triangle(p,triangles[i]).0));}}
117    }
118    fn ray(&self,triangles:&[[Vec3;3]],p:DVec3,d:DVec3,hits:&mut Vec<f64>){
119        let a=(self.lo.as_dvec3()-p)/d;let b=(self.hi.as_dvec3()-p)/d;
120        if a.min(b).max_element()>a.max(b).min_element()||a.max(b).min_element()<0.0{return;}
121        if let Some((a,b))=&self.children{a.ray(triangles,p,d,hits);b.ray(triangles,p,d,hits);}else{for &i in &self.triangles{
122            let [a,b,c]=triangles[i].map(|v|v.as_dvec3());let e=b-a;let f=c-a;let h=d.cross(f);let det=e.dot(h);if det.abs()<1e-20{continue;}
123            let u=(p-a).dot(h)/det;let q=(p-a).cross(e);let v=d.dot(q)/det;let t=f.dot(q)/det;
124            if u>=-1e-12&&v>=-1e-12&&u+v<=1.0+1e-12&&t>1e-12{hits.push(t);}
125        }}
126    }
127}

129/// A field in world space. Nonuniform scales and affine transforms use the
130/// inverse-transpose gradient, never an unscaled object-space normal.
131#[derive(Clone,Debug,PartialEq)]
132pub struct ClothSdfShape{pub field:ClothSdf,world:Mat4,inverse:Mat4,normal:Mat3,frame:Frame}
133impl ClothSdfShape{
134    pub fn new(field:ClothSdf,world:Mat4)->Result<Self,String>{
135        if !world.is_finite()||!world.determinant().is_finite()||world.determinant().abs()<1e-10||world.x_axis.w!=0.0||world.y_axis.w!=0.0||world.z_axis.w!=0.0||world.w_axis.w!=1.0{return Err("Level-set collider needs a finite, invertible affine transform".into());}
136        let inverse=world.inverse();Ok(Self{field,world,inverse,normal:Mat3::from_mat4(inverse).transpose(),frame:Frame::new(world)?})
137    }
138    pub fn transform(&self)->Mat4{self.world}
139    pub fn sample(&self,p:Vec3)->(f32,Vec3){
140        let (phi,gradient)=self.field.sample(self.inverse.transform_point3(p));let g=self.normal*gradient;
141        let len=g.length();if len>1e-8{(phi/len,g/len)}else{(phi,self.normal.x_axis.normalize_or(Vec3::X))}
142    }
143    pub fn project(&self,p:Vec3,radius:f32)->Option<(Vec3,Vec3)>{
144        if !p.is_finite()||!radius.is_finite(){return None;}let radius=radius.max(0.0);let mut out=p;let mut normal=Vec3::X;let mut hit=false;
145        for _ in 0..24{let(distance,n)=self.sample(out);if distance>=radius-1e-6{break;}hit=true;normal=n;out+=n*(radius-distance);}
146        hit.then_some((out,normal))
147    }
148}

150/// Affine motion uses a polar decomposition: rotate the orthogonal frame and
151/// interpolate its positive-definite stretch. This supports parent-induced
152/// shear without interpolating a rotation matrix through a singular midpoint.
153#[derive(Clone,Copy,Debug,PartialEq)]
154struct Frame{rotation:Quat,stretch:Mat3,reflection:Mat3,translation:Vec3}
155fn operator_bound(m:Mat3)->f32{
156    let abs=Mat3::from_cols(m.x_axis.abs(),m.y_axis.abs(),m.z_axis.abs());
157    let columns=abs.x_axis.element_sum().max(abs.y_axis.element_sum()).max(abs.z_axis.element_sum());
158    let rows=(abs.x_axis+abs.y_axis+abs.z_axis).max_element();(columns*rows).sqrt().min(matrix_norm(m))
159}
160fn matrix_norm(m:Mat3)->f32{(m.x_axis.length_squared()+m.y_axis.length_squared()+m.z_axis.length_squared()).sqrt()}
161impl Frame{
162    fn new(world:Mat4)->Result<Self,String>{
163        let linear=Mat3::from_mat4(world);let reflection=if linear.determinant()<0.0{Mat3::from_diagonal(Vec3::new(-1.0,1.0,1.0))}else{Mat3::IDENTITY};
164        let positive=linear*reflection;let mut r=positive/matrix_norm(positive);
165        for _ in 0..48{let next=(r+r.inverse().transpose())*0.5;let change=matrix_norm(next-r);r=next;if change<1e-6{break;}}
166        if !r.is_finite(){return Err("Unsupported level-set transform conditioning".into());}
167        let rotation=Quat::from_mat3(&r).normalize();let rotation_matrix=Mat3::from_quat(rotation);let raw=rotation_matrix.transpose()*positive;let stretch=(raw+raw.transpose())*0.5;
168        let minor=stretch.x_axis.x*stretch.y_axis.y-stretch.y_axis.x*stretch.x_axis.y;
169        if stretch.x_axis.x<=0.0||minor<=0.0||stretch.determinant()<=0.0||!stretch.inverse().is_finite()
170            ||matrix_norm(rotation_matrix*stretch-positive)>1e-4*matrix_norm(positive).max(1.0){return Err("Invalid level-set stretch".into());}
171        Ok(Self{rotation,stretch,reflection,translation:world.w_axis.truncate()})
172    }
173}
174/// Continuous query includes translation, rotation, independent scale and shear.
175/// A field-space Lipschitz bound prevents fixed-sample tunnelling.
176pub struct ClothSdfMotion{start:Arc<ClothSdfShape>,end:Arc<ClothSdfShape>,a:Frame,b:Frame,min_scale:f32,stationary:bool}
177impl ClothSdfMotion{
178    pub fn new(start:impl Into<Arc<ClothSdfShape>>,end:impl Into<Arc<ClothSdfShape>>)->Result<Self,String>{
179        let mut start=start.into();let end=end.into();
180        if !Arc::ptr_eq(&start.field.0,&end.field.0){start=end.clone();}
181        let a=start.frame;let b=end.frame;
182        if a.reflection!=b.reflection{return Err("A moving level-set transform cannot change handedness through zero scale".into());}
183        // The smallest eigenvalue of a convex combination of SPD stretches is
184        // at least the smaller endpoint eigenvalue. Inverse matrix norm bounds
185        // give a conservative bound without an iterative eigensolver.
186        let min_scale=1.0/operator_bound(a.stretch.inverse()).max(operator_bound(b.stretch.inverse()));
187        let stationary=start.world==end.world;Ok(Self{start,end,a,b,min_scale,stationary})
188    }
189    pub fn inverse_at(&self,t:f32)->Mat4{if t<=0.0{self.start.inverse}else if self.stationary||t>=1.0{self.end.inverse}else{self.at(t).inverse()}}
190    pub fn at(&self,t:f32)->Mat4{
191        if t<=0.0{return self.start.world;}if self.stationary||t>=1.0{return self.end.world;}
192        let linear=Mat3::from_quat(self.a.rotation.slerp(self.b.rotation,t))*(self.a.stretch*(1.0-t)+self.b.stretch*t)*self.a.reflection;
193        Mat4::from_cols(linear.x_axis.extend(0.0),linear.y_axis.extend(0.0),linear.z_axis.extend(0.0),self.a.translation.lerp(self.b.translation,t).extend(1.0))
194    }
195    /// Certify a swept triangle clear of a moving field.
196    /// Trilinear interpolation reproduces affine functions: the smallest grid
197    /// residual from a tangent plane bounds that residual everywhere inside
198    /// these cells. This avoids subdividing a broad, nearly planar safe region.
199    pub(crate) fn triangle_separated(&self,start:[Vec3;3],end:[Vec3;3],radius:f32)->bool{
200        let p=[start.map(|p|self.start.inverse.transform_point3(p)),end.map(|p|self.end.inverse.transform_point3(p))];
201        if p.iter().flatten().any(|p|!p.is_finite()){return false;}
202        let f=&self.end.field.0;let lo=p.iter().flatten().copied().fold(Vec3::splat(f32::INFINITY),Vec3::min);let hi=p.iter().flatten().copied().fold(Vec3::splat(f32::NEG_INFINITY),Vec3::max);
203        let extent=Vec3::from_array(f.dims.map(|n|(n-1)as f32));
204        let grid_lo=((lo-f.origin)/f.cell).clamp(Vec3::ZERO,extent);let grid_hi=((hi-f.origin)/f.cell).clamp(Vec3::ZERO,extent);
205        if !grid_lo.is_finite()||!grid_hi.is_finite(){return false;}
206        let a=grid_lo.floor().to_array().map(|x|x as usize);let b=grid_hi.ceil().to_array().map(|x|x as usize);
207        if (0..3).map(|i|b[i]-a[i]+1).product::<usize>()>4096{return false;}
208        let center=((lo+hi)*0.5).clamp(f.origin,f.origin+extent*f.cell);let(value,gradient)=self.end.field.sample(center);
209        // Outside the grid the field adds distance from its clamped boundary.
210        // A plane with |gradient| <= 1 grows no faster than that extension,
211        // so the same residual certificate remains valid outside the grid.
212        let gradient=gradient.clamp_length_max(1.0);
213        let plane=|point:Vec3|value+(point-center).dot(gradient);
214        let minimum=p.iter().flatten().map(|&p|plane(p)).fold(f32::INFINITY,f32::min);
215        // In field space a rotating/stretching frame makes the path curved.
216        // Bound its second derivative and the deviation from the endpoint
217        // chord (at most |p''|/8 on [0,1]). For A=S^-1, A'=-A S' A and
218        // A''=2 A S' A S' A. Multiplying the deviation by the field's global
219        // Lipschitz bound makes the endpoint-hull certificate conservative.
220        let inverse_scale=1.0/self.min_scale;let stretch=matrix_norm(self.b.stretch-self.a.stretch);
221        let angle=self.a.rotation.angle_between(self.b.rotation).abs();
222        let curvature=(0..3).map(|i|{
223            let a=start[i]-self.a.translation;let b=end[i]-self.b.translation;let reach=a.length().max(b.length());let speed=(b-a).length();
224            2.0*inverse_scale.powi(3)*stretch*stretch*reach
225                +2.0*inverse_scale*inverse_scale*stretch*(angle*reach+speed)
226                +inverse_scale*(angle*angle*reach+2.0*angle*speed)
227        }).fold(0.0,f32::max);
228        let margin=radius.max(0.0)*inverse_scale+curvature*0.125*self.end.field.0.lipschitz;
229        if !margin.is_finite()||!minimum.is_finite(){return false;}
230        let error=32.0*f32::EPSILON*(1.0+value.abs()+gradient.abs().dot((hi-lo).abs()));
231        for z in a[2]..=b[2]{for y in a[1]..=b[1]{for x in a[0]..=b[0]{
232            let point=f.origin+Vec3::new(x as f32,y as f32,z as f32)*f.cell;
233            let residual=f.values[x+f.dims[0]*(y+f.dims[1]*z)]-plane(point);
234            if minimum+residual<=margin+error{return false;}
235        }}}
236        true
237    }
238    /// First conservative proximity, including time and the impact normal.
239    /// Nonuniform scale uses an enclosing field-space radius; callers must
240    /// retain the conservative flag when resolving multiple/concave contacts.
241    pub fn first_contact(&self,p0:Vec3,p1:Vec3,radius:f32)->Option<crate::cloth_contact::ClothSweep>{
242        if !p0.is_finite()||!p1.is_finite()||!radius.is_finite(){return None;}
243        let radius=radius.max(0.0);let scale=self.min_scale;
244        let reach=(p0-self.a.translation).length().max((p1-self.b.translation).length());let angle=self.a.rotation.angle_between(self.b.rotation).abs();let ds=matrix_norm(self.b.stretch-self.a.stretch);
245        let speed=(((p1-p0)-(self.b.translation-self.a.translation)).length()+angle*reach)/scale+reach*ds/(scale*scale);
246        let speed=speed*self.end.field.0.lipschitz;let margin=radius/scale;let mut t=0.0;let mut exhausted=true;
247        for _ in 0..256{
248            let local=self.inverse_at(t).transform_point3(p0.lerp(p1,t));let gap=self.end.field.sample(local).0-margin;
249            if !gap.is_finite(){return None;}
250            if gap<=1e-6{exhausted=false;break;}if speed<1e-12||gap/speed>1.0-t{return None;}
251            let next=t+0.9*gap/speed;if next<=t{break;}t=next;
252        }
253        let inverse=self.inverse_at(t);let (_,gradient)=self.end.field.sample(inverse.transform_point3(p0.lerp(p1,t)));
254        let normal=(Mat3::from_mat4(inverse).transpose()*gradient).normalize_or(Vec3::X);
255        Some(crate::cloth_contact::ClothSweep{time:t,normal,conservative:exhausted||radius>0.0})
256    }
257    /// Cover the complete segment by adaptively subdivided swept spheres.
258    /// Inflating each midpoint by its half-segment length bounds the interior
259    /// even for nonconvex fields. The spatial budget is conservative, never a
260    /// fixed sample gap through which a thin body can pass.
261    pub fn resolve_segment(&self,start:[Vec3;2],end:[Vec3;2],radii:[f32;2])->Option<(f32,Vec3,Vec3,Vec3)>{
262        if start.iter().chain(&end).any(|p|!p.is_finite())||radii.iter().any(|r|!r.is_finite()){return None;}
263        let mut stack=vec![(0.0f32,1.0f32,0u32)];let mut visited=0;let mut best:Option<(f32,f32,Vec3,Vec3,Vec3)>=None;
264        let tolerance=(self.end.field.cell_size().min_element()*self.min_scale*0.25).max(0.0001);
265        while let Some((a,b,depth))=stack.pop(){
266            visited+=1;let u=(a+b)*0.5;let half=(start[1]-start[0]).length().max((end[1]-end[0]).length())*(b-a)*0.5;
267            let r=(radii[0]+(radii[1]-radii[0])*u).max(0.0)+(radii[1]-radii[0]).abs()*(b-a)*0.5+half;
268            let p0=start[0].lerp(start[1],u);let p1=end[0].lerp(end[1],u);
269            let Some(hit)=self.first_contact(p0,p1,r)else{continue};
270            if best.as_ref().is_some_and(|best|hit.time>best.0){continue;}
271            if half>tolerance&&depth<12&&visited<256{stack.push((u,b,depth+1));stack.push((a,u,depth+1));continue;}
272            if let Some((out,n,carried))=self.respond(p0,p1,r,hit.time){best=Some((hit.time,u,out-p1,n,carried));}
273        }
274        best.map(|(_,u,push,n,carried)|(u,push,n,carried))
275    }
276    /// A single-body response, iterated by the cloth solver alongside material
277    /// constraints and other bodies. Later zero-surface crossings remain
278    /// authoritative when an enclosing radius reports an early proximity.
279    pub fn resolve(&self,p0:Vec3,p1:Vec3,radius:f32)->Option<(Vec3,Vec3,Vec3)>{
280        if !radius.is_finite(){return None;}let radius=radius.max(0.0);let hit=self.first_contact(p0,p1,radius)?;
281        let response=self.respond(p0,p1,radius,hit.time);
282        // An enclosing radius under nonuniform scale can touch a departing
283        // lobe before a different lobe actually crosses the trajectory. A
284        // separating response from that early proximity is not proof that the
285        // rest of the path is clear. Check the proposed path against the zero
286        // surface and keep its first crossing authoritative.
287        if radius>0.0&&self.start.field.sample(self.start.inverse.transform_point3(p0)).0>1e-6{
288            let candidate=response.as_ref().map_or(p1,|r|r.0);
289            if let Some(crossing)=self.first_contact(p0,candidate,0.0){
290                if let Some(blocked)=self.respond(p0,candidate,radius,crossing.time){return Some(blocked);}
291            }
292        }
293        response
294    }
295    fn respond(&self,p0:Vec3,p1:Vec3,radius:f32,t:f32)->Option<(Vec3,Vec3,Vec3)>{
296        // Retain the impact side even when the endpoint has crossed right
297        // through the body. Transport that surface point with the collider.
298        let mut local=self.inverse_at(t).transform_point3(p0.lerp(p1,t));let mut gradient=Vec3::X;
299        for _ in 0..24{let(phi,g)=self.end.field.sample(local);gradient=g.normalize_or(Vec3::X);if phi.abs()<1e-6{break;}local-=gradient*(phi/g.length().max(1.0));}
300        let normal=(self.end.normal*gradient).normalize_or(Vec3::X);
301        let support=self.end.world.transform_point3(local)+normal*radius;
302        let start_normal=(self.start.normal*gradient).normalize_or(normal);
303        let carried=support-(self.start.world.transform_point3(local)+start_normal*radius);
304        let depth=(p1-support).dot(normal);let mut out=p1;if depth<0.0{out-=normal*depth;}
305        if let Some((projected,_))=self.end.project(out,radius){out=projected;}
306        if out.distance_squared(p1)<1e-18&&(p1-p0-carried).dot(normal)>=0.0{return None;}
307        Some((out,normal,carried))
308    }
