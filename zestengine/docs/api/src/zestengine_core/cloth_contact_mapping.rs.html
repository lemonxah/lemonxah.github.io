1//! Translational degrees of freedom for collision vertices attached to a
2//! reduced simulation mesh. A contact must move every vertex of its driver.
3use super::*;

5#[derive(Clone,Debug)]
6pub struct ClothContactMapping{groups:Vec<Vec<usize>>,owner:Vec<usize>}
7impl ClothContactMapping {
8    pub fn new(drivers:&[usize])->Self{
9        let mut ids=std::collections::HashMap::new();let mut groups:Vec<Vec<usize>>=Vec::new();let mut owner=Vec::with_capacity(drivers.len());
10        for (v,&driver) in drivers.iter().enumerate(){let next=ids.len();let group=*ids.entry(driver).or_insert(next);if group==groups.len(){groups.push(Vec::new());}groups[group].push(v);owner.push(group);}
11        Self{groups,owner}
12    }
13    pub(super) fn trivial(&self)->bool{self.groups.len()==self.owner.len()}
14    pub(super) fn valid_weights(&self,weights:&[f32])->bool{
15        self.owner.len()==weights.len()&&self.groups.iter().all(|g|g.iter().all(|&i|weights[i]==weights[g[0]]))
16    }
17    pub(super) fn valid(&self,weights:&[f32],owners:&[usize])->bool{
18        self.owner.len()==weights.len()&&owners.len()==weights.len()&&self.groups.iter().all(|g|g.iter().all(|&i|weights[i]==weights[g[0]]&&owners[i]==owners[g[0]]))
19    }
20    pub(super) fn project(&self,p:&mut[Vec3],w:&[f32],ids:[usize;4],contact:ClothFeatureContact,thickness:f32)->f32{
21        let n=contact.sweep.normal;let gap=(0..4).map(|k|p[ids[k]]*contact.coefficients[k]).sum::<Vec3>().dot(n);
22        if gap>=thickness{return 0.0;}
23        let roots=ids.map(|i|self.owner[i]);let mut coefficients=[0.0;4];
24        for k in 0..4{let first=(0..=k).find(|&j|roots[j]==roots[k]).unwrap();coefficients[first]+=contact.coefficients[k];}
25        let denominator=(0..4).map(|k|w[ids[k]].max(0.0)*coefficients[k]*coefficients[k]).sum::<f32>();
26        if denominator<1e-20{return 0.0;}
27        for k in 0..4{if coefficients[k]==0.0{continue;}
28            let delta=n*((thickness-gap)*w[ids[k]].max(0.0)*coefficients[k]/denominator);
29            for &i in &self.groups[roots[k]]{p[i]+=delta;}
30        }
31        thickness-gap
32    }
33    /// Geometric recovery and fixed-face cleanup can propose individual vertex
34    /// motion. Fit it back to the available translations without deforming a
35    /// cluster. Subsequent mapped contact passes resolve remaining constraints.
36    pub(super) fn cohere(&self,before:&[Vec3],after:&mut[Vec3]){
37        for group in &self.groups{
38            let delta=(group.iter().map(|&i|(after[i]-before[i]).as_dvec3()).sum::<glam::DVec3>()/group.len()as f64).as_vec3();
39            for &i in group{after[i]=before[i]+delta;}
40        }
41    }
42}

49    fn mapped_contacts_keep_clusters_rigid_and_use_driver_mass_once(){
50        let topology=ClothContactTopology::new(vec![[0,1,2],[3,4,5]]);let map=ClothContactMapping::new(&[17,17,17,usize::MAX,usize::MAX,usize::MAX]);
51        for pinned in [false,true]{
52            let mut p=layers();let before=p.clone();let weights=[if pinned{0.0}else{1.0},if pinned{0.0}else{1.0},if pinned{0.0}else{1.0},2.0,2.0,2.0];
53            let center=|p:&[Vec3]|(p[..3].iter().copied().sum::<Vec3>()+p[3..].iter().copied().sum::<Vec3>()*0.5)/3.0;
54            let mass_center=center(&p);
55            let report=topology.solve_between_mapped(&mut p,&before,&weights,&[0,0,0,1,1,1],&[0.01;6],&map,true,true);
56            assert!(report.vertex_face+report.edge_edge>0);assert!(p[3].z-p[0].z>=0.01999);
57            for base in [0,3]{for i in 1..3{assert!((p[base+i]-p[base]).distance(before[base+i]-before[base])<1e-6,"contacts cannot deform a rigid simulation cluster");}}
58            if pinned{assert_eq!(&p[..3],&before[..3]);}else{assert!(center(&p).distance(mass_center)<1e-6,"driver masses must not be counted separately for every face vertex");}
59        }
60    }
