1//! Resolution reduction respects connectivity inside each spatial cell. Nearby
2//! layers of a fold must not become one rigid driver merely because the full
3//! garment is connected elsewhere.
4use super::*;
5struct Regions{parent:Vec<usize>}
6impl Regions{
7    fn root(&mut self,mut i:usize)->usize{while self.parent[i]!=i{self.parent[i]=self.parent[self.parent[i]];i=self.parent[i];}i}
8    fn join(&mut self,a:usize,b:usize){let a=self.root(a);let b=self.root(b);self.parent[a.max(b)]=a.min(b);}
9}
10pub(super) fn build(positions:&[Vec3],indices:&[u32],pins:&[f32],spacing:f32)->(Vec<u32>,Vec<Vec<u32>>){
11    let cell=if spacing>0.0{spacing}else{1e-4};
12    let key=|p:Vec3,cell:f32|((p.x/cell).round()as i64,(p.y/cell).round()as i64,(p.z/cell).round()as i64);
13    let keys=positions.iter().enumerate().map(|(i,&p)|(key(p,cell),pins.get(i).copied().unwrap_or(0.0).clamp(0.0,1.0).to_bits())).collect::<Vec<_>>();
14    let mut regions=Regions{parent:(0..positions.len()).collect()};let mut welds=HashMap::new();
15    for (i,&p)in positions.iter().enumerate(){
16        // Preserve UV-seam welding, but never weld different pin classes or
17        // use a coarse cell as evidence that two disconnected layers touch.
18        let weld=(key(p,1e-4),keys[i]);if let Some(&other)=welds.get(&weld){regions.join(i,other);}else{welds.insert(weld,i);}
19    }
20    if spacing>0.0{for triangle in indices.chunks_exact(3){for k in 0..3{let(a,b)=(triangle[k]as usize,triangle[(k+1)%3]as usize);if keys[a]==keys[b]{regions.join(a,b);}}}}
21    let mut roots=HashMap::new();let mut owner=Vec::with_capacity(positions.len());let mut members:Vec<Vec<u32>>=Vec::new();
22    for i in 0..positions.len(){let root=regions.root(i);let next=members.len()as u32;let group=*roots.entry(root).or_insert(next);
23        if group==next{members.push(Vec::new());}members[group as usize].push(i as u32);owner.push(group);
24    }
25    (owner,members)
26}
