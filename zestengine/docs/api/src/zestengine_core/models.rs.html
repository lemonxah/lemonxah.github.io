1//! Egui-free data models shared by the runtime (scene/shader handling) and the
2//! editor inspector. These describe material and shader state as plain data;
3//! the editor builds its egui UI on top of them, the runtime reads them when
4//! loading scenes. Neither needs the other's crate.

6/// How a material's transparency is treated when rendering (mirrors the
7/// render crate's alpha mode).
8#[derive(Clone, Copy, PartialEq, Eq, Debug)]
9pub enum AlphaModeModel {
10    /// Fully opaque; alpha is ignored.
11    Opaque,
12    /// Alpha-tested: pixels with alpha below `alpha_cutoff` are discarded.
13    Cutout,
14    /// Clipped then blended: fragments are alpha-tested first, survivors blend.
15    Transclipping,
16    /// Full alpha blending.
17    Blend,
18}

20/// How a transparent material combines with the scene behind it (mirrors the

50/// Stencil comparison (mirrors the render crate's `StencilCompare`).
51#[derive(Clone, Copy, PartialEq, Eq, Debug, Default)]
52pub enum StencilCompareModel {
53    /// Test always fails.
54    Never,
55    /// Passes when the reference value is less than the stored stencil value.
56    Less,
57    /// Passes when both values are equal.
58    Equal,
59    /// Passes when the reference value is less than or equal to the stored value.
60    LessEqual,
61    /// Passes when the reference value is greater than the stored value.
62    Greater,
63    /// Passes when both values differ.
64    NotEqual,
65    /// Passes when the reference value is greater than or equal to the stored value.
66    GreaterEqual,
67    /// Test always passes.
68    #[default]
69    Always,
70}
71
72impl StencilCompareModel {
73    /// Every comparison, in declaration order.
74    pub const ALL: [StencilCompareModel; 8] =
75        [Self::Always, Self::Never, Self::Less, Self::Equal, Self::LessEqual, Self::Greater, Self::NotEqual, Self::GreaterEqual];
76
77    /// Inspector label of this comparison.
78    pub fn label(self) -> &'static str {
79        match self {
80            Self::Never => "Never",
81            Self::Less => "Less",
82            Self::Equal => "Equal",
83            Self::LessEqual => "Less or equal",
84            Self::Greater => "Greater",
85            Self::NotEqual => "Not equal",
86            Self::GreaterEqual => "Greater or equal",
87            Self::Always => "Always",
88        }
89    }
90}

92/// Stencil outcome op (mirrors the render crate's `StencilOp`).
93#[derive(Clone, Copy, PartialEq, Eq, Debug, Default)]
94pub enum StencilOpModel {
95    /// Leave the stored stencil value unchanged.
96    #[default]
97    Keep,
98    /// Set the stored value to 0.
99    Zero,
100    /// Replace the stored value with the reference value.
101    Replace,
102    /// Add 1, clamped at the maximum stencil value.
103    IncrementClamp,
104    /// Subtract 1, clamped at 0.
105    DecrementClamp,
106    /// Bitwise-invert the stored value.
107    Invert,
108    /// Add 1, wrapping from the maximum to 0.
109    IncrementWrap,
110    /// Subtract 1, wrapping from 0 to the maximum.
111    DecrementWrap,
112}
113

115    /// Every operation, in declaration order.
116    pub const ALL: [StencilOpModel; 8] =
117        [Self::Keep, Self::Zero, Self::Replace, Self::IncrementClamp, Self::DecrementClamp, Self::Invert, Self::IncrementWrap, Self::DecrementWrap];
118
119    /// Inspector label of this operation.
120    pub fn label(self) -> &'static str {
121        match self {
122            Self::Keep => "Keep",
123            Self::Zero => "Zero",
124            Self::Replace => "Replace",
125            Self::IncrementClamp => "Increment (clamp)",
126            Self::DecrementClamp => "Decrement (clamp)",
127            Self::Invert => "Invert",
128            Self::IncrementWrap => "Increment (wrap)",
129            Self::DecrementWrap => "Decrement (wrap)",
130        }
131    }

148/// A material's stencil state as plain data (mirrors the render crate's
149/// `StencilState`; the engine converts). Inactive by default.
150#[derive(Clone, Copy, PartialEq, Eq, Debug)]
151pub struct StencilModel {
152    /// Value compared with the stored stencil value (bits masked by `read_mask`).
153    pub reference: u8,
154    /// Bits of both values that take part in the comparison (255 = all).
155    pub read_mask: u8,
156    /// Bits of the stored value that can be modified (255 = all).
157    pub write_mask: u8,
158    /// Operations for front-facing fragments.
159    pub front: StencilFaceModel,
160    /// Operations for back-facing fragments (mirrors `front` while `two_sided` is off).
161    pub back: StencilFaceModel,
162    /// Unity `ColorMask` bits (R 1, G 2, B 4, A 8).
163    pub color_mask: u8,
164    /// Unity `ZWrite`: off for stencil writers that must not occlude later draws.
165    pub depth_write: bool,
166    /// Separate back-face state in the inspector (off = back mirrors front).
167    pub two_sided: bool,
168}
169
170impl Default for StencilModel {
171    fn default() -> Self {
172        Self {
173            reference: 0,
174            read_mask: 255,
175            write_mask: 255,
176            front: StencilFaceModel::default(),
177            back: StencilFaceModel::default(),
178            color_mask: 15,
179            depth_write: true,
180            two_sided: false,
181        }
182    }
183}

207/// Project-relative texture paths for a material's 12 sampler slots. Mirrors
208/// `zestengine_assets::MaterialTextures`; kept here (egui-free) so the inspector can
209/// edit texture assignments as plain data and the engine can resolve them to
210/// GPU handles at apply time. `None` = use the material's import-embedded
211/// texture for that slot (or the 1×1 default if there is none).
212#[derive(Clone, PartialEq, Eq, Default, Debug)]
213pub struct MaterialTexturePaths {

215    pub albedo: Option<std::path::PathBuf>,

218    /// Combined occlusion / roughness / metallic map (slot 2).
219    pub orm: Option<std::path::PathBuf>,
220    /// Emission map (slot 3).
221    pub emission: Option<std::path::PathBuf>,
222    /// Opacity map (slot 4).
223    pub opacity: Option<std::path::PathBuf>,
224    /// Mask for the emission map (slot 5).
225    pub emission_mask: Option<std::path::PathBuf>,
226    /// Three additive matcap textures (slots 6, 8, 10).
227    pub matcap: [Option<std::path::PathBuf>; 3],
228    /// Masks for the three matcaps (slots 7, 9, 11).
229    pub matcap_mask: [Option<std::path::PathBuf>; 3],
230    /// Split AO / Roughness / Metallic maps (slots 12/13/14).
231    pub ao: Option<std::path::PathBuf>,
232    /// Split roughness map (slot 13).
233    pub roughness: Option<std::path::PathBuf>,
234    /// Split metallic map (slot 14).
235    pub metallic: Option<std::path::PathBuf>,
236    /// Height / displacement map (slot 15) for parallax occlusion mapping.
237    pub displacement: Option<std::path::PathBuf>,
238    /// Detail layer maps (pseudo-slots 1000..1004 in [`Self::slot_mut`]).
239    pub detail_albedo: Option<std::path::PathBuf>,
240    /// Detail-layer normal map (pseudo-slot 1001).
241    pub detail_normal: Option<std::path::PathBuf>,
242    /// Detail-layer ORM map (pseudo-slot 1002).
243    pub detail_orm: Option<std::path::PathBuf>,
244    /// Detail-layer mask (pseudo-slot 1003).
245    pub detail_mask: Option<std::path::PathBuf>,
246    /// Translucency / thickness map (pseudo-slot 1004).
247    pub translucency: Option<std::path::PathBuf>,
248    /// Custom-shader texture properties (slots 16..), in the shader's texture
249    /// declaration order: (property name, path). Synced to the shader layout
250    /// by the engine; saved by name.
251    pub shader: Vec<(String, Option<std::path::PathBuf>)>,
252}

357pub const TEXTURE_SLOT_LABELS: [&str; 16] = [
358    "Albedo",
359    "Normal Map",
360    "ORM (Occl/Rough/Metal)",
361    "Emission",
362    "Opacity",

370    "Ambient Occlusion",
371    "Roughness",
372    "Metallic",
373    "Displacement",

375
376impl MaterialTexturePaths {
377    /// Mutable access to slot `i` (0..15) in binding order; `None` if out of range.
378    pub fn slot_mut(&mut self, i: usize) -> Option<&mut Option<std::path::PathBuf>> {
379        if i >= DETAIL_SLOT_BASE {
380            return Some(match i - DETAIL_SLOT_BASE {
381                0 => &mut self.detail_albedo,
382                1 => &mut self.detail_normal,
383                2 => &mut self.detail_orm,
384                3 => &mut self.detail_mask,
385                4 => &mut self.translucency,
386                _ => return None,
387            });
388        }
389        if i >= 16 {
390            return self.shader.get_mut(i - 16).map(|(_, p)| p);
391        }
392        Some(match i {
393            0 => &mut self.albedo,
394            1 => &mut self.normal,
395            2 => &mut self.orm,
396            3 => &mut self.emission,
397            4 => &mut self.opacity,
398            5 => &mut self.emission_mask,
399            6 => &mut self.matcap[0],
400            7 => &mut self.matcap_mask[0],
401            8 => &mut self.matcap[1],
402            9 => &mut self.matcap_mask[1],
403            10 => &mut self.matcap[2],
404            11 => &mut self.matcap_mask[2],
405            12 => &mut self.ao,
406            13 => &mut self.roughness,
407            14 => &mut self.metallic,
408            15 => &mut self.displacement,
409            _ => return None,
410        })
411    }
412}

414/// How a matcap layer combines with the shaded colour beneath it.
415#[derive(Clone, Copy, PartialEq, Eq, Debug, Default)]
416pub enum MatcapBlend {
417    /// Add the matcap on top (classic sphere-map highlight).
418    #[default]
419    Add,
420    /// Multiply with the base colour (tint / shading).
421    Multiply,
422    /// Replace the base colour (masked overlay).
423    Replace,
424}
425
426impl MatcapBlend {
427    /// Inspector label of this blend mode.
428    pub fn label(self) -> &'static str {
429        match self {
430            Self::Add => "Add",
431            Self::Multiply => "Multiply",
432            Self::Replace => "Replace",
433        }
434    }
435
436    /// Every blend mode, in declaration order.
437    pub const ALL: [MatcapBlend; 3] = [Self::Add, Self::Multiply, Self::Replace];
438
439    /// Shader encoding of the mode: 0 Add, 1 Multiply, 2 Replace.
440    pub fn to_f32(self) -> f32 {
441        match self {
442            Self::Add => 0.0,
443            Self::Multiply => 1.0,
444            Self::Replace => 2.0,
445        }
446    }
447
448    /// Decode a shader value; anything other than 1 (Multiply) or 2 (Replace)
449    /// becomes Add.
450    pub fn from_f32(v: f32) -> Self {
451        match v as i32 {
452            1 => Self::Multiply,
453            2 => Self::Replace,
454            _ => Self::Add,
455        }
456    }
457}

627/// Value-comparison operators used by shader property visibility conditions.
628#[derive(Clone, Copy, Debug, PartialEq, Eq)]
629pub enum ShaderCmpUi {
630    /// Equal.
631    Eq,
632    /// Not equal.
633    Ne,
634    /// Less than.
635    Lt,
636    /// Less than or equal.
637    Le,
638    /// Greater than.
639    Gt,
640    /// Greater than or equal.
641    Ge,
642}
643
644/// Visibility condition: OR of AND-groups of (value offset, op, value).
645#[derive(Clone, Debug, Default, PartialEq)]
646pub struct ShaderCondUi(pub Vec<Vec<(usize, ShaderCmpUi, f32)>>);
647
648impl ShaderCondUi {
649    /// True when the condition holds for the given property values: an empty
650    /// condition is always visible, otherwise any AND-group in which every
651    /// comparison passes. Equality compares with a 0.5 epsilon.
652    pub fn eval(&self, values: &[f32]) -> bool {
653        self.0.is_empty()
654            || self.0.iter().any(|all| {
655                all.iter().all(|&(at, op, b)| {
656                    let a = values.get(at).copied().unwrap_or(0.0);
657                    match op {
658                        ShaderCmpUi::Eq => (a - b).abs() < 0.5,
659                        ShaderCmpUi::Ne => (a - b).abs() >= 0.5,
660                        ShaderCmpUi::Lt => a < b,
661                        ShaderCmpUi::Le => a <= b,
662                        ShaderCmpUi::Gt => a > b,
663                        ShaderCmpUi::Ge => a >= b,
664                    }
665                })
666            })
667    }
668}

714/// Everything the inspector needs to draw a custom shader's material UI.
715#[derive(Clone, Debug, Default)]
716pub struct ShaderUiInfo {
717    /// Shader name shown in the inspector.
718    pub display_name: String,
719    /// "GLSL" or "HLSL".
720    pub language: &'static str,
721    /// Editable properties, in shader declaration order.
722    pub props: Vec<ShaderPropUi>,
723    /// Property groups for the group headers.
724    pub groups: Vec<ShaderGroupUi>,
725    /// Value presets offered by the shader.
726    pub presets: Vec<ShaderPresetUi>,
727    /// Compile/parse error; shown instead of properties.
728    pub error: Option<String>,
729}
730
731impl ShaderUiInfo {
732    /// Apply preset `i` to a material's values.
733    pub fn apply_preset(&self, i: usize, values: &mut [f32]) {
734        if let Some(p) = self.presets.get(i) {
735            for (at, v) in &p.values {
736                if at + v.len() <= values.len() {
737                    values[*at..at + v.len()].copy_from_slice(v);
738                }
739            }
740        }
741    }
742
743    /// Index of the preset whose assignments all match `values`.
744    pub fn matching_preset(&self, values: &[f32]) -> Option<usize> {
745        self.presets.iter().position(|p| {
746            p.values
747                .iter()
748                .all(|(at, v)| values.get(*at..at + v.len()).is_some_and(|cur| cur == v.as_slice()))
749        })
750    }
751
752    /// Enabled keyword count (variant summary).
753    pub fn active_keywords(&self, values: &[f32]) -> usize {
754        self.props
755            .iter()
756            .filter(|p| p.keyword && values.get(p.offset).is_some_and(|v| *v != 0.0))
757            .count()
758    }
759}
