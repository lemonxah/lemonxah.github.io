1//! State transfer through the common render topology. Simulation tiers never
2//! replace UVs, materials or render triangles.
3use super::*;
4impl MeshClothSim {
5    pub(super) fn transition_offset(&self,vertex:usize)->Vec3{
6        let t=if self.transition_duration>0.0{(self.transition_remaining/self.transition_duration).clamp(0.0,1.0)}else{0.0};
7        // Smoothstep has zero slope at both ends of a transition.
8        self.transition_offsets.get(vertex).copied().unwrap_or(Vec3::ZERO)*(t*t*(3.0-2.0*t))
9    }
10    /// Import displacement and velocity from another resolution of the SAME
11    /// render mesh. State stays in the previous frame, so normal inertia,
12    /// animation and teleport handling still happen exactly once on the step.
13    /// Rejects incompatible topology. Callers must also check mesh source identity.
14    pub fn transfer_resolution(&mut self,old:&Self,seconds:f32)->bool{
15        if self.vertex_count!=old.vertex_count||self.indices!=old.indices||old.anim.len()!=old.particles.len(){return false;}
16        self.reset(&old.animated_vertices,old.last_world);
17        let inv=old.last_world.inverse();
18        let previous_offsets:Vec<_>=old.owner.iter().map(|&owner|{let i=owner as usize;old.particles[i].pos-old.anim[i]}).collect();
19        for (i,members) in self.members.iter().enumerate(){
20            let p=&mut self.particles[i];
21            if p.pin>=1.0{continue;}
22            p.pos+=members.iter().map(|&v|previous_offsets[v as usize]).sum::<Vec3>()/members.len() as f32;
23            p.vel=members.iter().map(|&v|old.particles[old.owner[v as usize] as usize].vel).sum::<Vec3>()/members.len() as f32;
24        }
25        self.frame_previous=self.particles.iter().map(|p|p.pos).collect();
26        self.transition_duration=if seconds.is_finite(){seconds.clamp(0.0,10.0)}else{0.0};
27        self.transition_remaining=self.transition_duration;
28        if self.transition_duration>0.0 {
29            self.transition_offsets=(0..self.vertex_count).map(|v|{
30                let i=self.owner[v] as usize;
31                if self.particles[i].pin>=1.0{return Vec3::ZERO;}
32                inv.transform_vector3(previous_offsets[v]-(self.particles[i].pos-self.anim[i]))+old.transition_offset(v)
33            }).collect();
34        }
35        self.visible_frame_previous=self.visible_collision_positions();
36        true
37    }
38}

44    fn grid_size(n:u32)->(Vec<Vec3>,Vec<u32>,Vec<f32>){
45        let mut p=Vec::new();let mut t=Vec::new();let mut pins=Vec::new();
46        for y in 0..=n{for x in 0..=n{p.push(Vec3::new(x as f32/n as f32,y as f32/n as f32,0.0));pins.push(if y==n{1.0}else{0.0});}}
47        for y in 0..n{for x in 0..n{let a=y*(n+1)+x;t.extend([a,a+1,a+n+1,a+1,a+n+2,a+n+1]);}}
48        (p,t,pins)
49    }
50    #[test]
51    fn cloth_lod_transfer_keeps_render_pose_velocity_and_pins_then_fades(){
52        let (p,t,pins)=grid();let world=Mat4::from_scale_rotation_translation(Vec3::splat(2.0),glam::Quat::from_rotation_y(0.4),Vec3::new(3.0,2.0,1.0));
53        let mut fine=MeshClothSim::build(&p,&t,&pins,0.0,world).unwrap();fine.reset(&p,world);
54        for (i,point) in fine.particles.iter_mut().enumerate().filter(|(_,p)|p.pin<1.0){point.pos+=Vec3::Z*(0.05*(i as f32).sin());point.vel=Vec3::new(0.2,0.1,0.3);}
55        let mut before=p.clone();let mut normal=vec![Vec3::Z;p.len()];fine.apply(&mut before,&mut normal,world);
56        let mut coarse=MeshClothSim::build(&p,&t,&pins,0.3,world).unwrap();assert!(coarse.particles.len()<fine.particles.len()/2);
57        assert!(coarse.transfer_resolution(&fine,0.25));
58        let mut after=p.clone();let mut after_normal=vec![Vec3::Z;p.len()];coarse.apply(&mut after,&mut after_normal,world);
59        for ((a,b),(an,bn)) in before.iter().zip(&after).zip(normal.iter().zip(&after_normal)){assert!((*a-*b).length()<1e-5);assert!((*an-*bn).length()<1e-4);}
60        for (i,point) in coarse.particles.iter().enumerate(){
61            if point.pin>=1.0{assert!((point.pos-coarse.anim[i]).length()<1e-6);}else{assert!((point.vel-Vec3::new(0.2,0.1,0.3)).length()<1e-6);}
62        }
63        // A second switch midway includes the first switch's remaining detail.
64        coarse.transition_remaining=0.125;
65        let mut mid=p.clone();coarse.apply(&mut mid,&mut normal,world);
66        let mut finer=MeshClothSim::build(&p,&t,&pins,0.0,world).unwrap();assert!(finer.transfer_resolution(&coarse,0.25));
67        let mut switched=p.clone();finer.apply(&mut switched,&mut normal,world);
68        assert!(mid.iter().zip(switched).all(|(a,b)|(*a-b).length()<1e-5));
69        let cloth=MeshCloth{gravity:[0.0;3],fabric:None,shape_stiffness:0.0,stiffness:0.0,bending:0.0,tether_slack:100.0,..Default::default()};
70        for _ in 0..30{finer.step(&cloth,&p,world,&[],&[],0.0,1.0/60.0);}
71        assert!(finer.transition_offsets.is_empty());assert!(finer.particles.iter().all(|p|p.pos.is_finite()));
72        // Resetting after teleport/cull discards residual detail.
73        finer.transfer_resolution(&coarse,0.25);finer.reset(&p,world);assert!(finer.transition_offsets.is_empty());
74    }
