1//! VR play-space rig: the transform from the headset's stage/tracking space into
2//! world space, plus the locomotion that moves it — fly, grab-drag the world,
3//! scale yourself, and snap/smooth turn. Headset-agnostic pure math so the editor
4//! and a shipped game share it; the OpenXR layer (`zestengine-xr`) supplies raw
5//! stage-space poses, this maps them to world and drives movement.
6//!
7//! Model: a point in the player's tracking space maps to world as
8//!   world = origin + Rot_y(yaw) · (stage · scale)
9//! so moving `origin` flies, rotating `yaw` turns, and `scale` resizes the player
10//! relative to the world (scale > 1 = you're a giant, the world feels small).
11
12use glam::{Quat, Vec3};
13
14/// The play-space → world transform. Locomotion mutates this; pose mapping reads
15/// it to place the headset/controllers/trackers in the world.
16#[derive(Clone, Copy, Debug)]
17pub struct VrRig {
18    /// World position of the tracking-space origin (floor under the player).
19    pub origin: Vec3,
20    /// Yaw (radians) of the tracking space about world +Y.
21    pub yaw: f32,
22    /// Player scale. >1 makes the player large (covers ground faster, world looks
23    /// small); <1 shrinks the player. Clamped to a sane range by the setters.
24    pub scale: f32,
25}
26
27impl Default for VrRig {
28    fn default() -> Self {
29        Self {
30            origin: Vec3::ZERO,
31            yaw: 0.0,
32            scale: 1.0,
33        }
34    }
35}
36
37impl VrRig {
38    const MIN_SCALE: f32 = 0.01;
39    const MAX_SCALE: f32 = 1000.0;
40
41    fn rot(&self) -> Quat {
42        Quat::from_rotation_y(self.yaw)
43    }
44
45    /// The stage→world matrix (translate(origin) · rotY(yaw) · scale). Compose its
46    /// inverse with an XR eye's stage-space view to get the world-space view:
47    /// `world_view = eye.view · rig.stage_to_world_mat().inverse()`.
48    pub fn stage_to_world_mat(&self) -> glam::Mat4 {
49        glam::Mat4::from_scale_rotation_translation(
50            Vec3::splat(self.scale),
51            self.rot(),
52            self.origin,
53        )
54    }
55
56    /// Map a stage-space position to world space.
57    pub fn stage_to_world(&self, p: Vec3) -> Vec3 {
58        self.origin + self.rot() * (p * self.scale)
59    }
60
61    /// Map a stage-space orientation to world space (just the yaw rotation).
62    pub fn stage_rot_to_world(&self, r: Quat) -> Quat {
63        self.rot() * r
64    }
65
66    /// Map a full stage-space pose (position + orientation) to world space.
67    pub fn stage_pose_to_world(&self, pose: (Vec3, Quat)) -> (Vec3, Quat) {
68        (self.stage_to_world(pose.0), self.stage_rot_to_world(pose.1))
69    }
70
71    /// Fly: translate the play space by a world-space delta (so the player moves
72    /// the opposite way through the world). `dir` need not be normalized.
73    pub fn fly(&mut self, world_delta: Vec3) {
74        self.origin += world_delta;
75    }
76
77    /// Smooth/snap turn about a world pivot (usually the headset position so the
78    /// player rotates in place rather than orbiting the origin).
79    pub fn turn_about(&mut self, pivot: Vec3, delta_yaw: f32) {
80        // Keep `pivot` fixed in world while rotating the play space about it.
81        let rel = self.origin - pivot;
82        let rot = Quat::from_rotation_y(delta_yaw);
83        self.origin = pivot + rot * rel;
84        self.yaw += delta_yaw;
85    }
86
87    /// Scale the player about a world pivot (e.g. the controller, or the midpoint
88    /// between two controllers), keeping that pivot fixed in the world.
89    pub fn scale_about(&mut self, pivot: Vec3, factor: f32) {
90        let new_scale = (self.scale * factor).clamp(Self::MIN_SCALE, Self::MAX_SCALE);
91        let applied = new_scale / self.scale; // re-derive after clamping
92        // origin' = pivot - (pivot - origin) * applied keeps `pivot` fixed.
93        self.origin = pivot - (pivot - self.origin) * applied;
94        self.scale = new_scale;
95    }
96
97    /// Grab-drag the world: given the world point that was under the grabbing
98    /// controller when the grab started (`anchor_world`) and the controller's
99    /// CURRENT stage-space position, move `origin` so the anchor stays glued to
100    /// the controller. Call every frame while the grab is held.
101    pub fn drag_to(&mut self, anchor_world: Vec3, controller_stage: Vec3) {
102        // We need stage_to_world(controller_stage) == anchor_world:
103        //   origin = anchor_world - Rot · (controller_stage · scale)
104        self.origin = anchor_world - self.rot() * (controller_stage * self.scale);
105    }
106
107    /// The world point currently under a controller (its stage position mapped to
108    /// world). Capture this at grab start to feed `drag_to`.
109    pub fn grab_anchor(&self, controller_stage: Vec3) -> Vec3 {
110        self.stage_to_world(controller_stage)
111    }
112}

114/// Headset + controller state for gameplay this frame (`ctx.screen.vr`).
115/// Poses are in the tracking (stage) space; a `VrPlayer` maps them to world
116/// through its own transform. `active` is false without a running headset.
117#[derive(Clone, Copy, Debug, Default)]
118pub struct VrFrame {
119    /// A headset is running and the poses are live.
120    pub active: bool,
121    /// Head pose (stage space).
122    pub head: Option<(Vec3, Quat)>,
123    /// Left-hand pose (stage space).
124    pub left_hand: Option<(Vec3, Quat)>,
125    /// Right-hand pose (stage space).
126    pub right_hand: Option<(Vec3, Quat)>,
127    /// Left thumbstick, -1..1 per axis.
128    pub left_stick: glam::Vec2,
129    /// Right thumbstick, -1..1 per axis.
130    pub right_stick: glam::Vec2,
131    /// Left grip pull, 0..1.
132    pub left_grip: f32,
133    /// Right grip pull, 0..1.
134    pub right_grip: f32,
135    /// Left trigger pressed (thresholded).
136    pub left_trigger: bool,
137    /// Right trigger pressed (thresholded).
138    pub right_trigger: bool,
139    /// The menu button is pressed.
140    pub menu: bool,
141    /// Analog trigger pull (0..1).
142    pub left_trigger_value: f32,
143    /// Analog trigger pull (0..1).
144    pub right_trigger_value: f32,
145    /// Face buttons: `X` / `Y` on the left controller, `A` / `B` on the right.
146    pub a: bool,
147    /// The B button (right controller).
148    pub b: bool,
149    /// The X button (left controller).
150    pub x: bool,
151    /// The Y button (left controller).
152    pub y: bool,
153    /// The left stick is clicked.
154    pub left_stick_click: bool,
155    /// The right stick is clicked.
156    pub right_stick_click: bool,
157}

159/// A world-space ray (origin + unit direction), e.g. a controller's pointer.
160#[derive(Clone, Copy, Debug)]
161pub struct Ray {
162    /// Ray origin (world space).
163    pub origin: Vec3,
164    /// Unit direction.
165    pub dir: Vec3,
166}
167
168impl Ray {
169    /// The point `t` units along the ray.
170    pub fn point_at(&self, t: f32) -> Vec3 {
171        self.origin + self.dir * t
172    }
173}
174
175/// Pointer ray from a controller's WORLD pose, aiming along the controller's
176/// forward (-Z, the OpenXR aim convention). Use for selecting / placing /
177/// pointing at the hand menu.
178pub fn pointer_ray(controller_world_pos: Vec3, controller_world_rot: Quat) -> Ray {
179    Ray {
180        origin: controller_world_pos,
181        dir: (controller_world_rot * Vec3::NEG_Z).normalize_or(Vec3::NEG_Z),
182    }
183}
184
185/// Intersect a ray with an infinite plane (point + normal). Returns the world hit
186/// point if the ray crosses the plane in front of its origin. Handy for laser-
187/// pointer interaction with a hand-menu quad.
188pub fn ray_plane(ray: Ray, plane_point: Vec3, plane_normal: Vec3) -> Option<Vec3> {
189    let denom = ray.dir.dot(plane_normal);
190    if denom.abs() < 1e-6 {
191        return None;
192    }
193    let t = (plane_point - ray.origin).dot(plane_normal) / denom;
194    (t >= 0.0).then(|| ray.point_at(t))
195}

197#[cfg(test)]
198mod tests {
199    use super::*;
200
201    #[test]
202    fn fly_moves_player_through_world() {
203        let mut rig = VrRig::default();
204        rig.fly(Vec3::new(0.0, 0.0, -5.0));
205        // A stage point at the origin now sits 5 units forward in the world.
206        assert!((rig.stage_to_world(Vec3::ZERO) - Vec3::new(0.0, 0.0, -5.0)).length() < 1e-5);
207    }
208
209    #[test]
210    fn scale_about_keeps_pivot_fixed() {
211        let mut rig = VrRig::default();
212        let pivot = Vec3::new(2.0, 0.0, 1.0);
213        rig.scale_about(pivot, 3.0);
214        assert!((rig.scale - 3.0).abs() < 1e-5);
215        // The pivot maps to the same world point before/after (it's its own preimage).
216        let s = (pivot - rig.origin) / rig.scale; // stage preimage of pivot
217        let _ = s;
218        assert!((rig.stage_to_world((pivot - rig.origin) / rig.scale * 1.0) - pivot).length() < 1e-3);
219    }
220
221    #[test]
222    fn drag_glues_anchor_to_controller() {
223        let mut rig = VrRig::default();
224        let controller_stage_start = Vec3::new(0.3, 1.2, -0.4);
225        let anchor = rig.grab_anchor(controller_stage_start);
226        // Controller moves; dragging keeps the same world anchor under it.
227        let controller_stage_now = Vec3::new(0.6, 1.2, -0.1);
228        rig.drag_to(anchor, controller_stage_now);
229        assert!((rig.stage_to_world(controller_stage_now) - anchor).length() < 1e-4);
230    }
231
232    #[test]
233    fn turn_about_keeps_pivot_and_adds_yaw() {
234        let mut rig = VrRig::default();
235        let pivot = Vec3::new(0.0, 0.0, -2.0);
236        rig.turn_about(pivot, std::f32::consts::FRAC_PI_2);
237        assert!((rig.yaw - std::f32::consts::FRAC_PI_2).abs() < 1e-5);
238        // The pivot is unchanged in world: map its stage preimage back.
239        // Stage preimage of pivot: rot⁻¹·(pivot-origin)/scale.
240        let pre = rig.rot().inverse() * ((pivot - rig.origin) / rig.scale);
241        assert!((rig.stage_to_world(pre) - pivot).length() < 1e-3);
242    }
243
244    #[test]
245    fn pointer_ray_aims_forward() {
246        let r = pointer_ray(Vec3::ZERO, Quat::IDENTITY);
247        assert!((r.dir - Vec3::NEG_Z).length() < 1e-5);
248        let hit = ray_plane(r, Vec3::new(0.0, 0.0, -3.0), Vec3::Z).unwrap();
249        assert!((hit - Vec3::new(0.0, 0.0, -3.0)).length() < 1e-4);
250    }
