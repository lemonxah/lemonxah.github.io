1//! Authored river endpoints and their engine-resolved receiving surfaces.
2use serde::{Deserialize,Serialize};
3#[derive(Clone,Copy,Debug,PartialEq,Serialize,Deserialize)]
4#[serde(default)]
5pub struct WaterConnection {pub target:crate::ObjectRef,pub length:f32}
6impl Default for WaterConnection {fn default()->Self{Self{target:crate::ObjectRef::NONE,length:5.0}}}
7#[derive(Clone,Copy,Debug,PartialEq)]
8pub struct ResolvedWaterConnection {
9    /// Receiving surface plane in river-local coordinates, normal.xyz + d.
10    pub plane:[f32;4],
11    pub flow:glam::Vec3,
12    pub depth:f32,
13    pub length:f32,
14}
15impl ResolvedWaterConnection {
16    pub fn weight(&self,distance:f32)->f32 {let x=(distance/self.length.max(0.01)).clamp(0.0,1.0);1.0-x*x*(3.0-2.0*x)}
17    pub fn height(&self,p:glam::Vec3)->f32 {if self.plane[1].abs()<1e-6{p.y}else{-(self.plane[0]*p.x+self.plane[2]*p.z+self.plane[3])/self.plane[1]}}
18}

28/// Higher-ranked intersecting water; an immutable geometry snapshot without
29/// its own overlap list avoids recursive ownership tests.
30#[derive(Clone,Debug)]
31pub struct WaterOverlap {pub to_target:glam::Mat4,pub water:std::sync::Arc<crate::Water>,pub wave_padding:f32}
32impl WaterOverlap {
33    pub fn contains(&self,p:glam::Vec3)->bool {
34        let p=self.to_target.transform_point3(p);
35        if self.water.excluded(p){return false;}
36        self.water.column(glam::Vec2::new(p.x,p.z)).is_some_and(|c|p.y>=c.surface-c.depth && p.y<=c.surface+self.wave_padding+0.05)
37    }
38}
