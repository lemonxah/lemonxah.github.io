1//! Continuous body contacts across cloth faces, including their interiors.
2use super::*;

4pub struct ClothSurfaceContact {
5    pub weights:[f32;3],
6    pub push:Vec3,
7    pub normal:Vec3,
8    pub carried:Vec3,
9}
10
11impl ClothColliderQuery<'_> {
12    /// Spheres/capsules use closest features of the complete triangle.
13    /// Concave fields use a covering hierarchy of swept spheres; unresolved
14    /// leaves stay conservative instead of leaving gaps between point probes.
15    /// Returns one constraint; the solver iterates for simultaneous contacts.
16    pub fn triangle(&self,start:[Vec3;3],end:[Vec3;3],radius:f32)->Option<ClothSurfaceContact>{
17        if start.iter().chain(&end).any(|p|!p.is_finite())||!radius.is_finite(){return None;}
18        let radius=radius.max(0.0);
19        if self.field.is_some(){return self.field_triangle(start,end,radius);}
20        if matches!(self.motion.end,ClothShapeWorld::Disabled){return None;}
21        let sample=|time:f32|{
22            let triangle=std::array::from_fn(|i|start[i].lerp(end[i],time));
23            match self.motion.at(time){
24                ClothShapeWorld::Sphere{c,r}=>{let(q,w)=closest_triangle(c,triangle);(q.distance(c)-r.max(0.0),w)},
25                ClothShapeWorld::Capsule{a,b,ra,rb}=>{
26                    let edge=b-a;let dr=rb-ra;
27                    let derivative=|u:f32|{let center=a+edge*u;let(q,_)=closest_triangle(center,triangle);edge.dot((center-q).normalize_or_zero())-dr};
28                    let u=if derivative(0.0)>=0.0{0.0}else if derivative(1.0)<=0.0{1.0}else{
29                        let(mut lo,mut hi)=(0.0,1.0);for _ in 0..32{let m=(lo+hi)*0.5;if derivative(m)<0.0{lo=m}else{hi=m}}(lo+hi)*0.5
30                    };
31                    let center=a+edge*u;let(q,w)=closest_triangle(center,triangle);(q.distance(center)-(ra+dr*u).max(0.0),w)
32                },
33                ClothShapeWorld::Plane{p,n}=>{
34                    let n=n.normalize_or(Vec3::Y);let i=(0..3).min_by(|&i,&j|(triangle[i]-p).dot(n).total_cmp(&(triangle[j]-p).dot(n))).unwrap();
35                    let mut w=[0.0;3];w[i]=1.0;((triangle[i]-p).dot(n),w)
36                },
37                _=>(f32::INFINITY,[0.0;3]),
38            }
39        };
40        let speed=(0..3).map(|i|self.motion.speed_bound(start[i],end[i])).fold(0.0,f32::max);
41        let(time,conservative)=advance(|t|sample(t).0,speed,radius)?;let(_,weights)=sample(time);
42        let p0=(0..3).map(|i|start[i]*weights[i]).sum::<Vec3>();let p1=(0..3).map(|i|end[i]*weights[i]).sum::<Vec3>();
43        let normal=shape_sample(p0.lerp(p1,time),self.motion.at(time)).1;
44        let(out,normal,carried)=respond_to_collider(p0,p1,radius,self.motion.clone(),ClothSweep{time,normal,conservative})?;
45        Some(ClothSurfaceContact{weights,push:out-p1,normal,carried})
46    }

48    fn field_triangle(&self,start:[Vec3;3],end:[Vec3;3],radius:f32)->Option<ClothSurfaceContact>{
49        let field=self.field.as_ref().unwrap();let point=|p:&[Vec3;3],w:Vec3|p[0]*w.x+p[1]*w.y+p[2]*w.z;
50        if field.triangle_separated(start,end,radius){return None;}
51        // Refine the earliest possible impact first. A depth-first walk can
52        // spend its budget on one harmless patch, then report a very large
53        // unrefined neighbour as an artificial early collision.
54        let bound=|patch:[Vec3;3],depth:u32|{
55            if depth>0&&field.triangle_separated(patch.map(|w|point(&start,w)),patch.map(|w|point(&end,w)),radius){return None;}
56            let weights=(patch[0]+patch[1]+patch[2])/3.0;
57            let p0=point(&start,weights);let p1=point(&end,weights);
58            let half=patch.iter().map(|&w|point(&start,w).distance(p0).max(point(&end,w).distance(p1))).fold(0.0,f32::max);
59            field.first_contact(p0,p1,radius+half).map(|hit|(hit.time,half,patch,depth))
60        };
61        let mut pending=vec![bound([Vec3::X,Vec3::Y,Vec3::Z],0)?];let mut visited=0;
62        let tolerance=(radius*0.125).clamp(0.00025,0.001);
63        while !pending.is_empty(){
64            let first=(0..pending.len()).min_by(|&a,&b|pending[a].0.total_cmp(&pending[b].0).then(pending[a].1.total_cmp(&pending[b].1))).unwrap();
65            let(_,half,patch,depth)=pending.swap_remove(first);visited+=1;
66            let weights=(patch[0]+patch[1]+patch[2])/3.0;let p0=point(&start,weights);let p1=point(&end,weights);
67            // Prefer a real contact witness to the enclosing patch. Subsequent
68            // iterations revisit the remaining surface after this constraint
69            // is satisfied. This also avoids exhaustively refining a broad,
70            // already contacting flat region down to submillimetre patches.
71            if let Some((out,normal,carried))=field.resolve(p0,p1,radius).filter(|(out,_,_)|out.distance_squared(p1)>1e-16){
72                return Some(ClothSurfaceContact{weights:weights.to_array(),push:out-p1,normal,carried});
73            }
74            if half>tolerance&&depth<20&&visited<512{
75                let (i,j)=[(0,1),(1,2),(2,0)].into_iter().max_by(|&(i,j),&(k,l)|{
76                    let length=|i,j|point(&start,patch[i]).distance_squared(point(&start,patch[j])).max(point(&end,patch[i]).distance_squared(point(&end,patch[j])));
77                    length(i,j).total_cmp(&length(k,l))
78                }).unwrap();let middle=(patch[i]+patch[j])*0.5;
79                let mut a=patch;let mut b=patch;a[j]=middle;b[i]=middle;
80                if let Some(node)=bound(a,depth+1){pending.push(node);}if let Some(node)=bound(b,depth+1){pending.push(node);}continue;
81            }
82            if let Some((out,normal,carried))=field.resolve(p0,p1,radius+half){
83                return Some(ClothSurfaceContact{weights:weights.to_array(),push:out-p1,normal,carried});
84            }
85        }
86        None
87    }
88}

94    fn moving_bodies_hit_face_interiors_when_every_vertex_and_edge_misses(){
95        let face=[Vec3::new(-1.0,-1.0,0.0),Vec3::new(1.0,-1.0,0.0),Vec3::new(0.0,1.0,0.0)];
96        for capsule in [false,true]{
97            let shape=|z|if capsule{ClothShapeWorld::Capsule{a:Vec3::new(-0.05,0.0,z),b:Vec3::new(0.05,0.0,z),ra:0.03,rb:0.05}}else{ClothShapeWorld::Sphere{c:Vec3::Z*z,r:0.05}};
98            let motion=ClothColliderMotion{start:shape(-1.0),end:shape(1.0)};let query=motion.prepare();
99            for &p in &face{assert!(query.resolve(p,p,0.01).is_none());}
100            for(i,j)in [(0,1),(1,2),(2,0)]{assert!(query.segment([face[i],face[j]],[face[i],face[j]],[0.01;2]).is_none());}
101            let c=query.triangle(face,face,0.01).expect("the body crosses the face interior");
102            assert!(c.weights.iter().all(|&w|w>0.0));assert!(c.push.z>=1.0599&&c.normal.z>0.99);assert!(c.carried.z>=1.9999);
103        }

111        }}}
112        let field=ClothSdf::from_samples([5;3],Vec3::splat(-0.1),Vec3::splat(0.05),samples).unwrap();
113        let shape=|z|ClothShapeWorld::LevelSet(std::sync::Arc::new(ClothSdfShape::new(field.clone(),glam::Mat4::from_translation(Vec3::new(0.137,0.231,z))).unwrap()));
114        let face=[Vec3::new(-1.0,-1.0,0.0),Vec3::new(1.0,-1.0,0.0),Vec3::new(0.0,1.0,0.0)];
