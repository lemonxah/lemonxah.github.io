1//! Shared mod contracts. Filesystem/package handling belongs to assets; loading
2//! and multiplayer admission belong to the engine. Profiles are frozen for a session.
3use std::collections::{BTreeMap,BTreeSet};
4use serde::{Serialize,Deserialize};
5
6#[derive(Clone,Debug,PartialEq,Eq,Serialize,Deserialize)]
7#[serde(default)]
8pub struct ModSettings {
9    pub enabled:bool,
10    /// Stable identifier chosen by the game, independent of its display name.
11    pub game_id:String,
12    pub game_api:u32,
13    pub profile:String,
14    /// Exact asset paths or directory prefixes ending in `/` exposed by the game.
15    pub override_paths:Vec<String>,
16    /// Component.Action names scripts may invoke, explicitly exposed by the game.
17    pub actions:Vec<String>,
18}
19impl Default for ModSettings {fn default()->Self{Self{enabled:true,game_id:String::new(),game_api:1,profile:"default".into(),override_paths:Vec::new(),actions:Vec::new()}}}
20#[derive(Clone,Debug,PartialEq,Eq,Serialize,Deserialize)]
21pub struct ModDependency {pub id:String,#[serde(default)]pub version:Option<String>}
22#[derive(Clone,Debug,Default,PartialEq,Eq,Serialize,Deserialize)]
23#[serde(default)]
24pub struct ModManifest {
25    /// Canonical UUID v4, generated once by `new` and retained across releases.
26    pub id:String,pub name:String,pub author:String,pub version:String,
27    pub game_id:String,pub game_api:u32,
28    pub dependencies:Vec<ModDependency>,pub conflicts:Vec<String>,pub load_after:Vec<String>,
29    /// Base-game path -> packaged replacement path. Later enabled mods win.
30    pub overrides:BTreeMap<String,String>,
31    /// Base-game scene -> packaged additive scene. Entries receive stable namespaced ids.
32    pub additions:BTreeMap<String,String>,
33    pub actions:Vec<String>,
34}

36#[serde(default)]
37pub struct ModProfile {pub enabled:Vec<String>}
38#[derive(Clone,Debug,PartialEq,Eq,Serialize,Deserialize)]
39pub struct LockedMod {pub id:String,pub version:String,pub content_hash:String}
40#[derive(Clone,Debug,Default,PartialEq,Eq,Serialize,Deserialize)]
41#[serde(default)]
42pub struct ModLock {pub game_id:String,pub game_api:u32,pub mods:Vec<LockedMod>}
43impl ModLock {
44    pub fn check(&self,required:&Self)->Result<(),String>{
45        if self==required{return Ok(());}
46        if self.game_id!=required.game_id||self.game_api!=required.game_api{return Err("The save/session targets a different game or mod API".into());}
47        let wanted=required.mods.iter().map(|m|format!("{}@{} [{}]",m.id,m.version,m.content_hash.chars().take(12).collect::<String>())).collect::<Vec<_>>().join(", ");
48        Err(format!("Mod profile mismatch (versions, contents and order must match). Required: [{}]",wanted))
49    }
50}
51pub fn valid_id(s:&str)->bool{!s.is_empty()&&s.len()<=96&&s.bytes().all(|c|c.is_ascii_alphanumeric()||matches!(c,b'.'|b'_'|b'-'))&&s!="."&&s!=".."}
52pub fn valid_mod_id(s:&str)->bool{
53    s.parse::<crate::ObjectId>().is_ok_and(|id|{let b=id.raw().to_be_bytes();id.to_string()==s&&(b[6]>>4)==4&&(b[8]&0xc0)==0x80})
54}
55pub fn relative_asset(s:&str)->bool{
56    !s.is_empty()&&s.len()<=1024&&!s.contains(['\\',':','\0'])&&!s.starts_with('/')&&s.split('/').all(|p|!p.is_empty()&&p!="."&&p!="..")
57}
58impl ModManifest {
59    /// Authoring only: deserialization and packaging must never regenerate identity.
60    pub fn new(game_id:impl Into<String>,game_api:u32)->Self{
61        Self{id:crate::ObjectId::new().to_string(),name:"New mod".into(),version:"1.0.0".into(),game_id:game_id.into(),game_api,..Default::default()}
62    }
63    pub fn validate(&self)->Result<(),String>{
64        if !valid_mod_id(&self.id){return Err("Mod ID must be a canonical UUID v4, created once and retained across updates".into());}
65        if !valid_id(&self.game_id){return Err("Game ID must use letters, digits, dot, dash or underscore".into());}
66        if self.game_api==0||self.version.trim().is_empty()||self.version.len()>64{return Err("A mod needs a version and a nonzero game API".into());}
67        if self.dependencies.len()+self.conflicts.len()+self.load_after.len()>256||self.overrides.len()+self.additions.len()>4096||self.actions.len()>256{return Err("Mod manifest exceeds its entry limit".into());}
68        for id in self.dependencies.iter().map(|d|&d.id).chain(&self.conflicts).chain(&self.load_after){if !valid_mod_id(id)||id==&self.id{return Err(format!("Invalid/self dependency or conflict: {id}"));}}
69        for (target,source)in self.overrides.iter().chain(&self.additions){if !relative_asset(target)||!relative_asset(source){return Err(format!("Invalid mod asset path: {target} -> {source}"));}}
70        if self.additions.iter().any(|(a,b)|!a.ends_with(".scene")||!b.ends_with(".scene")){return Err("Scene additions must reference .scene files".into());}
71        if self.actions.iter().any(|a|a.is_empty()||a.len()>256||!a.contains('.')){return Err("Actions must be Component.Action names".into());}
72        Ok(())
73    }
74    pub fn check_game(&self,settings:&ModSettings)->Result<(),String>{
75        self.validate()?;
76        if !settings.enabled{return Err("This game disables mods".into());}
77        if self.game_id!=settings.game_id||self.game_api!=settings.game_api{return Err(format!("{} requires game {} API {}",self.id,self.game_id,self.game_api));}
78        for path in self.overrides.keys().chain(self.additions.keys()){
79            if !settings.override_paths.iter().any(|p|p==path||(p.ends_with('/')&&path.starts_with(p))){return Err(format!("{}: game does not expose {path}",self.id));}
80        }
81        for action in &self.actions{if !settings.actions.contains(action){return Err(format!("{}: game does not expose action {action}",self.id));}}
82        Ok(())
83    }
84}
85/// Stable topological sort. Profile order decides ties and conflicting overrides.
86/// Dependencies must be explicitly enabled; missing mods never silently disappear.
87pub fn resolve<'a>(profile:&ModProfile,installed:&'a BTreeMap<String,ModManifest>,settings:&ModSettings)->Result<Vec<&'a ModManifest>,String>{
88    if profile.enabled.len()>256{return Err("At most 256 mods may be enabled".into());}
89    let selected:BTreeSet<_>=profile.enabled.iter().cloned().collect();
90    if selected.len()!=profile.enabled.len(){return Err("A mod occurs more than once in the profile".into());}
91    for id in &profile.enabled{
92        let m=installed.get(id).ok_or_else(||format!("Missing installed mod: {id}"))?;m.check_game(settings)?;
93        if &m.id!=id{return Err("Installed mod ID does not match its manifest".into());}
94        for d in &m.dependencies{
95            let dep=installed.get(&d.id).filter(|_|selected.contains(&d.id)).ok_or_else(||format!("{id} requires enabled mod {}",d.id))?;
96            if d.version.as_ref().is_some_and(|v|v!=&dep.version){return Err(format!("{id} requires {} version {}",d.id,d.version.as_deref().unwrap()));}
97        }
98        if let Some(conflict)=m.conflicts.iter().find(|c|selected.contains(*c)){return Err(format!("{id} conflicts with {conflict}"));}
99    }
100    let mut ordered=Vec::new();let mut done=BTreeSet::new();
101    while ordered.len()<selected.len(){
102        let next=profile.enabled.iter().filter(|id|!done.contains(*id)).find(|id|{
103            let m=&installed[*id];m.dependencies.iter().all(|d|done.contains(&d.id))&&m.load_after.iter().filter(|id|selected.contains(*id)).all(|id|done.contains(id))
104        }).ok_or("Mod dependency/load-order cycle")?;
105        ordered.push(&installed[next]);done.insert(next.clone());
106    }
107    Ok(ordered)
108}
