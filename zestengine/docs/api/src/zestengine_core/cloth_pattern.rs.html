7pub enum ClothPanelSide{#[default]Left,Right,Bottom,Top}
8impl ClothPanelSide{
9    pub const ALL:[Self;4]=[Self::Left,Self::Right,Self::Bottom,Self::Top];
10    pub fn label(self)->&'static str{match self{Self::Left=>"Left",Self::Right=>"Right",Self::Bottom=>"Bottom",Self::Top=>"Top"}}
11}

14pub struct ClothPanel{
15    pub name:String,
16    pub size:[f32;2],
17    pub segments:[u32;2],
18    pub translation:[f32;3],

25#[derive(Clone,Debug,Serialize,Deserialize)]

28pub vertices:Vec<u32>}
29impl Default for ClothPanelEdge{fn default()->Self{Self{panel:0,side:Default::default(),vertices:Vec::new()}}}
30#[derive(Clone,Debug,Serialize,Deserialize)]

41impl ClothPanel{
42    pub fn mesh(&self)->Result<(Vec<Vec2>,Vec<u32>),String>{
43        if self.translation.iter().chain(&self.rotation).any(|v|!v.is_finite()){return Err("Non-finite panel transform".into());}
44        if !self.vertices.is_empty(){
45            if self.vertices.len()>250_000||self.triangles.len()>1_500_000||self.vertices.iter().flatten().any(|v|!v.is_finite())||self.triangles.len()%3!=0||self.triangles.len()<3||self.triangles.iter().any(|&v|v as usize>=self.vertices.len()){return Err("Invalid custom panel geometry".into());}
46            return Ok((self.vertices.iter().copied().map(Vec2::from).collect(),self.triangles.clone()));
47        }
48        let [x,y]=self.segments;if x==0||y==0||x>256||y>256||self.size.iter().any(|v|!v.is_finite()||*v<=0.0){return Err("Panel size must be positive; subdivisions must be 1–256".into());}
49        let mut p=Vec::new();let mut t=Vec::new();
50        for j in 0..=y{for i in 0..=x{p.push(Vec2::new(i as f32/x as f32*self.size[0],j as f32/y as f32*self.size[1]));}}
51        for j in 0..y{for i in 0..x{let a=j*(x+1)+i;t.extend([a,a+1,a+x+1,a+1,a+x+2,a+x+1]);}}
52        Ok((p,t))
53    }
54    fn edge(&self,edge:&ClothPanelEdge,vertices:usize,triangles:&[u32])->Result<Vec<u32>,String>{
55        let chain=if !edge.vertices.is_empty(){edge.vertices.clone()}else{
56            if !self.vertices.is_empty(){return Err("Custom panels need an ordered seam vertex chain".into());}
57            let [x,y]=self.segments;match edge.side{ClothPanelSide::Left=>(0..=y).map(|j|j*(x+1)).collect(),ClothPanelSide::Right=>(0..=y).map(|j|j*(x+1)+x).collect(),ClothPanelSide::Bottom=>(0..=x).collect(),ClothPanelSide::Top=>(0..=x).map(|i|y*(x+1)+i).collect()}
58        };
59        if chain.len()<2||chain.iter().any(|&i|i as usize>=vertices){return Err("Invalid seam edge vertices".into());}
60        let mut counts=std::collections::HashMap::new();for t in triangles.chunks_exact(3){for i in 0..3{let(a,b)=(t[i],t[(i+1)%3]);*counts.entry((a.min(b),a.max(b))).or_insert(0)+=1;}}
61        if chain.windows(2).any(|e|counts.get(&(e[0].min(e[1]),e[0].max(e[1])))!=Some(&1)){return Err("Seams must follow consecutive boundary edges".into());}
62        Ok(chain)
63    }
64}
65impl ClothPattern{
66    pub fn assemble(&self)->Result<AssembledCloth,String>{
67        if self.panels.is_empty()||self.panels.len()>128||self.seams.len()>512{return Err("Clothing needs 1–128 panels and at most 512 seams".into());}
68        let mut out=AssembledCloth{positions:Vec::new(),uv:Vec::new(),indices:Vec::new(),stitches:Vec::new()};let mut meshes=Vec::new();let mut starts=Vec::new();
69        for panel in &self.panels{
70            let(p,t)=panel.mesh()?;starts.push(out.positions.len() as u32);let offset=out.positions.len() as u32;
71            let r=panel.rotation.map(f32::to_radians);let rotation=Quat::from_euler(EulerRot::XYZ,r[0],r[1],r[2]);let translation=Vec3::from(panel.translation);
72            out.positions.extend(p.iter().map(|v|rotation*v.extend(0.0)+translation));out.uv.extend(p.iter().copied());out.indices.extend(t.iter().map(|&i|i+offset));meshes.push((p,t));
73            if out.positions.len()>250_000{return Err("Clothing exceeds 250,000 vertices".into());}
74        }
75        for seam in &self.seams{
76            if !seam.compliance.is_finite()||seam.compliance<0.0{return Err("Seam compliance must be finite and nonnegative".into());}
77            let mut chains=Vec::new();for edge in [&seam.a,&seam.b]{let panel=self.panels.get(edge.panel).ok_or("Missing seam panel")?;let(p,t)=&meshes[edge.panel];chains.push(panel.edge(edge,p.len(),t)?.iter().map(|v|v+starts[edge.panel]).collect::<Vec<_>>());}
78            if seam.reverse{chains[1].reverse();}
79            if chains[0].iter().any(|v|chains[1].contains(v)){return Err("A seam cannot stitch an edge to itself".into());}
80            for side in 0..2{
81                let a=&chains[side];let b=&chains[1-side];
82                let distances=|chain:&[u32]|{let mut total=vec![0.0];for e in chain.windows(2){total.push(total.last().unwrap()+out.positions[e[0]as usize].distance(out.positions[e[1]as usize]));}total};
83                let da=distances(a);let db=distances(b);let la=*da.last().unwrap();let lb=*db.last().unwrap();if la<1e-6||lb<1e-6{return Err("Zero-length seam".into());}
84                for (i,&vertex) in a.iter().enumerate(){let distance=da[i]/la*lb;let j=db.partition_point(|&d|d<distance).saturating_sub(1).min(b.len()-2);let span=db[j+1]-db[j];if span<1e-8{return Err("Zero-length seam segment".into());}
85                    out.stitches.push(ClothStitch{vertex,edge:[b[j],b[j+1]],weight:((distance-db[j])/span).clamp(0.0,1.0),compliance:seam.compliance});
86                }
87            }
88        }
89        Ok(out)
90    }

102        let cloth=MeshCloth{gravity:[0.0;3],iterations:12,stitches:g.stitches.clone(),self_collision:true,fabric:Some(ClothFabric{drag:0.0,lift:0.0,..Default::default()}),..Default::default()};
103        for spacing in [0.0,0.3]{
104            let mut sim=MeshClothSim::build(&g.positions,&g.indices,&pins,spacing,Mat4::IDENTITY).unwrap();sim.set_texcoords(&g.uv);sim.set_stitches(&g.stitches).unwrap();
105            for _ in 0..240{sim.step(&cloth,&g.positions,Mat4::IDENTITY,&[],&[],0.0,1.0/60.0);}
106            let mut rendered=g.positions.clone();let mut normals=vec![Vec3::Z;rendered.len()];sim.apply(&mut rendered,&mut normals,Mat4::IDENTITY);
107            for (p,want) in rendered[..20].iter().zip(&g.positions){assert!(p.distance(*want)<1e-5);}
108            let error=g.stitches.iter().map(|s|rendered[s.vertex as usize].distance(rendered[s.edge[0]as usize].lerp(rendered[s.edge[1]as usize],s.weight))).fold(0.0f32,f32::max);
109            eprintln!("cloth seam spacing {spacing}: max render separation {error}");assert!(error<0.01,"spacing {spacing}, seam error {error}");assert!(sim.particles.iter().all(|p|p.pos.is_finite()));
