1//! History-free recovery of existing intersections. Cutting intersected edges
2//! exposes connected cloth regions; all contacts between a pair of regions
3//! agree on which side should move. This prevents independent endpoint choices
4//! from repeatedly undoing one another along a tessellated intersection.
5use super::*;
6use std::collections::BTreeMap;
7
8struct Crossing {edge:usize,face:usize,normal:Vec3,d:[f32;2],bary:[f32;3],gap:f32}
9fn face(top:&ClothContactTopology,index:usize)->[usize;3]{let mut t=top.triangles[index];t.sort_unstable();t}
10struct Regions {parent:Vec<usize>}
11impl Regions {
12    fn new(n:usize)->Self{Self{parent:(0..n).collect()}}
13    fn root(&mut self,mut i:usize)->usize{
14        while self.parent[i]!=i{self.parent[i]=self.parent[self.parent[i]];i=self.parent[i];}i
15    }
16    fn join(&mut self,a:usize,b:usize){let a=self.root(a);let b=self.root(b);self.parent[a.max(b)]=a.min(b);}
17}

19// Choose the same symbolic side for exactly-on-surface vertices even when
20// imported triangles have inconsistent winding. Nonzero crossings are unaffected.
21fn normal(t:[glam::DVec3;3])->glam::DVec3 {
22    let n=(t[1]-t[0]).cross(t[2]-t[0]).normalize_or_zero();
23    let a=n.abs();let axis=if a.x>=a.y&&a.x>=a.z{0}else if a.y>=a.z{1}else{2};
24    if n[axis]<0.0{-n}else{n}
25}
26fn crossing(e:[Vec3;2],t:[Vec3;3])->Option<(Vec3,[f32;2],[f32;3],f64)>{
27    // Thin triangles are common while an intersection is being removed.
28    // Double precision and oriented-area barycentrics avoid cancellation in
29    // the Gram determinant used by a generic closest-point query.
30    let e=e.map(|v|v.as_dvec3());let t=t.map(|v|v.as_dvec3());
31    let n=normal(t);if n==glam::DVec3::ZERO||!n.is_finite(){return None;}
32    let extent=(t[1]-t[0]).length().max((t[2]-t[0]).length()).max((e[1]-e[0]).length());
33    let coordinate=e.iter().chain(t.iter()).map(|p|p.abs().max_element()).fold(0.0,f64::max);
34    let eps=extent*1e-6+coordinate*f32::EPSILON as f64*4.0;
35    let d=e.map(|p|{let d=(p-t[0]).dot(n);if d.abs()<=eps{0.0}else{d}});
36    if (d[0]>=0.0)==(d[1]>=0.0){return None;}
37    let at=e[0].lerp(e[1],d[0]/(d[0]-d[1]));let cross=(t[1]-t[0]).cross(t[2]-t[0]);let area=cross.length_squared();
38    if area<1e-30{return None;}
39    let a=(t[1]-at).cross(t[2]-at).dot(cross)/area;let b=(t[2]-at).cross(t[0]-at).dot(cross)/area;let c=1.0-a-b;
40    if [a,b,c].iter().any(|v|*v< -1e-6||*v>1.0+1e-6){return None;}
41    let bary=[a.max(0.0),b.max(0.0),c.max(0.0)];let sum=bary.iter().sum::<f64>();
42    Some((n.as_vec3(),d.map(|d|d as f32),bary.map(|v|(v/sum)as f32),eps))
43}
44fn separation_tangent(surface:Vec3,normal:Vec3)->Vec3{
45    let tangent=surface-normal*surface.dot(normal);
46    if tangent.length_squared()>1e-6{tangent.normalize()}else{Vec3::ZERO}
47}

49pub(super) fn recover(top:&ClothContactTopology,p:&mut[Vec3],weights:&[f32],thickness:f32,owners:Option<&[usize]>,radii:Option<&[f32]>)->usize {
50    let mut resolved=recover_pass(top,p,weights,thickness,owners,radii,false);
51    if resolved>0&&weights.iter().any(|w|*w<=0.0){
52        // A self-intersection repair must not leave cloth inside a fixed
53        // garment that was resolved earlier in the same pass.
54        for _ in 0..2{let n=recover_pass(top,p,weights,thickness,owners,radii,true);resolved+=n;if n==0{break;}}
55    }
56    if resolved>0{project_fixed(top,p,weights,thickness,owners,radii);}
57    resolved
58}
59fn recover_pass(top:&ClothContactTopology,p:&mut[Vec3],weights:&[f32],thickness:f32,owners:Option<&[usize]>,radii:Option<&[f32]>,fixed_only:bool)->usize {
60    if p.len()!=weights.len()||!thickness.is_finite()||thickness<=0.0||p.iter().any(|p|!p.is_finite())
61        ||weights.iter().any(|w|!w.is_finite())||top.triangles.iter().flatten().chain(top.edges.iter().flatten()).any(|&i|i>=p.len()){return 0;}
62    let bounds:Vec<_>=top.triangles.iter().map(|t|Bounds::points(&t.map(|i|p[i])).expanded(thickness)).collect();
63    let tree=Tree::new(&bounds);let mut candidates=Vec::new();let mut crossings=Vec::new();let mut cut=vec![false;top.edges.len()];
64    let mut neighbours=vec![Vec::new();p.len()];for e in &top.edges{neighbours[e[0]].push(e[1]);neighbours[e[1]].push(e[0]);}
65    let mut visited=vec![0usize;p.len()];let mut stamp=0;let mut stack=Vec::new();
66    for (ei,e) in top.edges.iter().enumerate(){
67        candidates.clear();tree.query(Bounds::points(&e.map(|i|p[i])).expanded(thickness),&mut candidates);
68        for &ti in &candidates {
69            let t=face(top,ti);
70            if fixed_only&&t.iter().any(|&i|weights[i]>0.0){continue;}
71            // Proximity repulsion skips the one-ring to avoid swelling a rest
72            // mesh. True intersections in that ring still require recovery.
73            if owners.is_some_and(|o|o[e[0]]==o[t[0]])||e.iter().any(|&i|t.iter().any(|&j|i==j||top.excluded.contains(&(i.min(j),i.max(j))))){continue;}
74            let Some((normal,d,bary,epsilon))=crossing(e.map(|i|p[i]),t.map(|i|p[i]))else{continue};
75            // An endpoint merely touching a face is not an intersection to
76            // reverse. A contour through an exact mesh vertex is real only
77            // when its incident cloth continues onto the other side too.
78            let mut touches=false;
79            for k in 0..2{if d[k]==0.0{
80                stamp+=1;stack.clear();stack.push(e[k]);visited[e[k]]=stamp;let mut opposite=false;
81                while let Some(i)=stack.pop(){
82                    for &j in &neighbours[i]{
83                        if visited[j]==stamp{continue;}visited[j]=stamp;
84                        let distance=(p[j].as_dvec3()-p[t[0]].as_dvec3()).dot(normal.as_dvec3());
85                        if distance>epsilon{opposite=true;break;}
86                        if distance>=-epsilon{stack.push(j);}
87                    }
88                    if opposite{break;}
89                }
90                touches|=!opposite;
91            }}
92            if touches{continue;}
93            let gap=radii.map_or(thickness,|r|r[e[0]].max(r[e[1]])+t.iter().map(|&j|r[j]).fold(0.0,f32::max));
94            cut[ei]=true;crossings.push(Crossing{edge:ei,face:ti,normal,d,bary,gap});
95        }
96    }
97    if crossings.is_empty(){return 0;}
98    let mut regions=Regions::new(p.len());
99    for (ei,e) in top.edges.iter().enumerate(){if !cut[ei]{regions.join(e[0],e[1]);}}
100    let region:Vec<_>=(0..p.len()).map(|i|regions.root(i)).collect();
101    let mut mass=vec![0.0f64;p.len()];
102    for (i,&w) in weights.iter().enumerate(){mass[region[i]]+=if w>0.0{1.0/w as f64}else{f64::INFINITY};}
103    // Pick the smaller movable region, not the currently cheapest endpoint.
104    // Boundary depth would prefer a freshly repaired row over the next trapped
105    // row and immediately reverse the recovery on the following pass.
106    let mut choices:BTreeMap<(usize,usize),usize>=BTreeMap::new();
107    for c in &crossings {
108        let r=top.edges[c.edge].map(|i|region[i]);if r[0]==r[1]{continue;}
109        let pair=(r[0].min(r[1]),r[0].max(r[1]));
110        if mass[pair.0].is_finite()||mass[pair.1].is_finite(){choices.insert(pair,if mass[pair.0]<=mass[pair.1]{pair.0}else{pair.1});}
111    }
112    let mut handled=vec![false;p.len()];for &r in choices.values(){handled[r]=true;}
113    let original=p.to_vec();
114    let mut surface=vec![Vec3::ZERO;p.len()];
115    for i in 0..top.triangles.len(){let t=face(top,i);
116        let n=normal(t.map(|i|original[i].as_dvec3())).as_vec3();
117        for i in t{let oriented=if surface[i].dot(n)<0.0{-n}else{n};surface[i]+=oriented;}
118    }
119    for n in &mut surface{*n=n.normalize_or_zero();}
120    let mut resolved=0;let mut members=vec![Vec::new();p.len()];let mut constraints=vec![Vec::new();p.len()];
121    for i in 0..p.len(){if handled[region[i]]{members[region[i]].push(i);}}
122    for (ci,c) in crossings.iter().enumerate() {
123        let e=top.edges[c.edge];let r=e.map(|i|region[i]);let pair=(r[0].min(r[1]),r[0].max(r[1]));
124        let Some(&chosen)=choices.get(&pair)else{continue};
125        let k=usize::from(r[1]==chosen);let side=if c.d[1-k]>=0.0{1.0}else{-1.0};
126        constraints[chosen].push((ci,c.normal*side));
127    }
128    // Move a trapped region coherently. Repairing one boundary row at a time
129    // collapses successive rows onto one another and creates new self-crossings.
130    let mut order:Vec<_>=(0..p.len()).filter(|&r|handled[r]).collect();
131    order.sort_by(|&a,&b|mass[b].total_cmp(&mass[a]).then(b.cmp(&a)));
132    let mut offsets=vec![Vec3::ZERO;p.len()];
133    for r in order{
134        let mut planes=Vec::new();let mut tangent=Vec3::ZERO;let mut gap=0.0f32;
135        let source=members[r].iter().map(|&i|surface[i]).sum::<Vec3>().normalize_or_zero();
136        for &(ci,n) in &constraints[r]{
137            let c=&crossings[ci];let t=face(top,c.face);
138            let own=(0..3).filter(|&j|region[t[j]]==r).map(|j|c.bary[j]).sum::<f32>();
139            if own>0.9999{continue;}
140            let target=(0..3).map(|j|(original[t[j]]+offsets[region[t[j]]])*c.bary[j]).sum::<Vec3>();
141            let depth=members[r].iter().map(|&i|(original[i]-target).dot(n)).fold(f32::INFINITY,f32::min);
142            planes.push((n,(c.gap-depth)/(1.0-own)));tangent+=separation_tangent(source,n);gap=gap.max(c.gap);
143        }
144        let mut offset=tangent.normalize_or_zero()*gap;
145        for _ in 0..8{for &(n,depth) in &planes{let missing=depth-n.dot(offset);if missing>0.0{offset+=n*missing;}}}
146        let points:Vec<_>=members[r].iter().map(|&i|original[i]).collect();let bound=Bounds::points(&points);
147        offset=offset.clamp_length_max((bound.hi-bound.lo).length()*2.0+gap*4.0);
148        if offset.is_finite()&&offset.length_squared()>1e-20{
149            offsets[r]=offset;for &i in &members[r]{p[i]+=offset;resolved+=1;}
150        }else{handled[r]=false;}
151    }
152    // A contour ending inside a sheet does not partition the mesh. Extend the
153    // local face plane through its connected source patch for a consistent
154    // side decision. Excluding the opposing triangle keeps its own vertices
155    // from becoming a bridge between those source regions.
156    // Process one target face at a time: storing a full partition for every
157    // intersected face would require quadratic memory on dense tangled cloth.
158    let mut fallback:Vec<_>=crossings.iter().filter(|c|{
159        let e=top.edges[c.edge];let t=face(top,c.face);
160        !e.iter().chain(t.iter()).any(|&i|handled[region[i]])
161    }).collect();
162    fallback.sort_by_key(|c|c.face);
163    let mut extended=None;
164    for c in fallback {
165        let e=top.edges[c.edge];let t=face(top,c.face);
166        if extended.as_ref().is_none_or(|(id,_,_,_)|*id!=c.face){
167            let side:Vec<_>=original.iter().map(|p|(*p-original[t[0]]).dot(c.normal)>=0.0).collect();
168            let mut parts=Regions::new(p.len());
169            for edge in &top.edges{if !edge.iter().any(|i|t.contains(i))&&side[edge[0]]==side[edge[1]]{parts.join(edge[0],edge[1]);}}
170            let labels:Vec<_>=(0..p.len()).map(|i|parts.root(i)).collect();let mut masses=vec![0.0;p.len()];let mut pinned=vec![false;p.len()];
171            for i in 0..p.len(){if weights[i]<=0.0{pinned[labels[i]]=true;}else{masses[labels[i]]+=1.0/weights[i]as f64;}}
172            extended=Some((c.face,labels,masses,pinned));
173        }
174        let (_,labels,masses,pinned)=extended.as_ref().unwrap();
175        let r=e.map(|i|labels[i]);
176        let cost=|k:usize|if weights[e[k]]<=0.0||(pinned[r[k]]&&!pinned[r[1-k]]){f64::INFINITY}else{masses[r[k]]};
177        let k=usize::from(cost(1)<cost(0));if !cost(k).is_finite(){continue;}
178        let side=if c.d[1-k]>=0.0{1.0}else{-1.0};let n=c.normal*side;
179        let target=(0..3).map(|j|original[t[j]]*c.bary[j]).sum::<Vec3>();let gap=(p[e[k]]-target).dot(n);
180        if gap<c.gap{
181            let tangent=separation_tangent(surface[e[k]],n);
182            p[e[k]]+=n*(c.gap-gap)+tangent*c.gap;resolved+=1;
183        }
184    }
185    if resolved>0{
186        // Remove net translation from each entirely free interacting group.
187        // A uniform shift preserves the repaired relative geometry and the
188        // mass-weighted centre. Anchored groups transmit repair to their pins;
189        // those pins, including unrelated fixed objects, must never be moved.
190        let mut groups=Regions::new(p.len());
191        for e in &top.edges{groups.join(e[0],e[1]);}
192        for c in &crossings{groups.join(top.edges[c.edge][0],top.triangles[c.face][0]);}
193        let mut total=vec![0.0f64;p.len()];let mut delta=vec![glam::DVec3::ZERO;p.len()];
194        for i in 0..p.len(){let root=groups.root(i);if weights[i]<=0.0{total[root]=f64::INFINITY;}else{
195            let m=1.0/weights[i]as f64;total[root]+=m;delta[root]+=(p[i]-original[i]).as_dvec3()*m;
196        }}
197        for i in 0..p.len(){let root=groups.root(i);if total[root].is_finite()&&total[root]>0.0{p[i]-=(delta[root]/total[root]).as_vec3();}}
198    }
199    resolved
200}

202/// True geometric crossings, including nearby topology that proximity
203/// repulsion deliberately excludes. Coplanar contact is not a crossing.
204fn intersects(top:&ClothContactTopology,p:&[Vec3],owners:Option<&[usize]>)->bool{
205    let bounds:Vec<_>=top.triangles.iter().map(|t|Bounds::points(&t.map(|i|p[i]))).collect();
206    let tree=Tree::new(&bounds);let mut candidates=Vec::new();
207    for e in &top.edges{
208        candidates.clear();tree.query(Bounds::points(&e.map(|i|p[i])),&mut candidates);
209        for &ti in &candidates{
210            let t=face(top,ti);
211            if owners.is_some_and(|o|o[e[0]]==o[t[0]])||e.iter().any(|&i|t.iter().any(|&j|i==j||top.excluded.contains(&(i.min(j),i.max(j))))){continue;}
212            if let Some((_,d,_,_))=crossing(e.map(|i|p[i]),t.map(|i|p[i])){if d[0]!=0.0&&d[1]!=0.0{return true;}}
213        }
214    }
215    false
216}

218/// Constraint projection can introduce a new crossing even after swept
219/// contacts were solved. Backtrack such a correction toward the previous
220/// valid pose. Pins retain their current animation targets throughout. If
221/// those targets already make the old pose intersect, geometric recovery
222/// remains responsible; we must not restore an invalid previous pose.
223pub(super) fn preserve_valid_pose(top:&ClothContactTopology,p:&mut[Vec3],previous:&[Vec3],weights:&[f32],owners:Option<&[usize]>)->bool{
224    if !intersects(top,p,owners){return false;}
225    let origin:Vec<_>=(0..p.len()).map(|i|if weights[i]>0.0{previous[i]}else{p[i]}).collect();
226    if intersects(top,&origin,owners){return false;}
227    let proposed=p.to_vec();let mut fraction=0.5;
228    for _ in 0..12{
229        for i in 0..p.len(){p[i]=origin[i].lerp(proposed[i],fraction);}
230        if !intersects(top,p,owners){return true;}
231        fraction*=0.5;
232    }
233    p.copy_from_slice(&origin);true
234}

236/// Fixed faces take priority over corrections between movable cloth layers.
237/// Their final thickness pass cannot displace another part of the fixed mesh.
238pub(super) fn project_fixed(top:&ClothContactTopology,p:&mut[Vec3],weights:&[f32],thickness:f32,owners:Option<&[usize]>,radii:Option<&[f32]>){
239    let fixed:Vec<_>=(0..top.triangles.len()).map(|i|face(top,i)).filter(|t|t.iter().all(|&i|weights[i]<=0.0)).collect();
240    if fixed.is_empty(){return;}
241    let bounds:Vec<_>=fixed.iter().map(|t|Bounds::points(&t.map(|i|p[i]))).collect();let tree=Tree::new(&bounds);
242    let mut neighbours=vec![Vec::new();p.len()];for e in &top.edges{neighbours[e[0]].push(e[1]);neighbours[e[1]].push(e[0]);}
243    let mut candidates=Vec::new();let mut visited=vec![0usize;p.len()];let mut stamp=0;let mut stack=Vec::new();
244    for i in 0..p.len(){if weights[i]<=0.0{continue;}
245        candidates.clear();tree.query(Bounds::points(&[p[i]]).expanded(thickness),&mut candidates);
246        for &face in &candidates{
247            let t=fixed[face];if owners.is_some_and(|o|o[i]==o[t[0]])||t.iter().any(|&j|top.neighbours(i,j)){continue;}
248            let gap=radii.map_or(thickness,|r|r[i]+t.iter().map(|&j|r[j]).fold(0.0,f32::max));
249            let(q,_)=closest_triangle(p[i],t.map(|i|p[i]));let delta=p[i]-q;let distance=delta.length();
250            if distance>=gap{continue;}
251            let n=if distance>1e-6{delta/distance}else{
252                let n=normal(t.map(|j|p[j].as_dvec3())).as_vec3();
253                stamp+=1;stack.clear();stack.push(i);visited[i]=stamp;let mut side=1.0;
254                'search:while let Some(j)=stack.pop(){for &k in &neighbours[j]{
255                    if visited[k]==stamp{continue;}visited[k]=stamp;
256                    let d=(p[k]-q).dot(n);if d.abs()>1e-6{side=d.signum();break 'search;}stack.push(k);
257                }}
258                n*side
259            };
260            p[i]=q+n*gap;
261        }
262    }
