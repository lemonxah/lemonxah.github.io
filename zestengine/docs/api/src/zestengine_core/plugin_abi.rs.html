1//! The ABI stamp of a component plugin (NET-3).
2//!
3//! A plugin is a `cdylib` whose `zestengine_register(&mut ComponentRegistry)` takes Rust-ABI types.
4//! That is sound only when the plugin and its host were built by the same rustc, from the same
5//! engine version and dependency features, with the same `debug_assertions` and pointer width. A plugin proves it by
6//! exporting `zestengine_plugin_abi` (add `zestengine_core::export_plugin_abi!();` to the crate);
7//! the host compares the stamp with its own before calling anything else. The stamp is plain
8//! `#[repr(C)]` data returned by an `extern "C"` function, so reading it is safe even when the
9//! other side is a different compiler.
10
11/// Bump when the `Component` / `ComponentRegistry` contract changes incompatibly.
12///
13/// Pre-release the contract is still moving and every plugin is rebuilt together
14/// with the engine (the stamp also checks rustc and shared type layouts), so the
15/// version stays at 1. From the first public release, increment it on every
16/// incompatible change to the plugin boundary.
17pub const PLUGIN_ABI_VERSION: u32 = 1;
18const MAGIC: u32 = 0x5A50_4C47; // "ZPLG"
19
20/// What a plugin or a host was built with.
21#[repr(C)]
22#[derive(Clone, Copy, Debug, PartialEq, Eq)]
23pub struct PluginAbi {
24    /// Magic number identifying a plugin ABI stamp.
25    pub magic: u32,
26    /// The ABI version.
27    pub abi: u32,
28    /// Core major version.
29    pub core_major: u16,
30    /// Core minor version.
31    pub core_minor: u16,
32    /// Core patch version.
33    pub core_patch: u16,
34    /// Built with debug assertions.
35    pub debug_assertions: u8,
36    /// Pointer width in bits (64).
37    pub pointer_width: u8,
38    /// Toolchain and shared Rust type identity/layout fingerprint. The historical
39    /// field name stays unchanged to preserve the C stamp's wire layout.
40    pub rustc_hash: u64,
41}

43fn shared_type_hash() -> u64 {
44    use std::any::TypeId;
45    use std::hash::{Hash, Hasher};
46    let mut hash = std::collections::hash_map::DefaultHasher::new();
47    // Cargo feature unification can give identical source different TypeIds.
48    // Such copies cannot share typed ECS storage across a cdylib boundary.
49    fn shared<T: 'static>(hash: &mut impl Hasher) {
50        TypeId::of::<T>().hash(hash);
51        std::mem::size_of::<T>().hash(hash);
52        std::mem::align_of::<T>().hash(hash);
53    }
54    shared::<crate::ComponentRegistry>(&mut hash);
55    shared::<crate::Transform>(&mut hash);
56    shared::<hecs::World>(&mut hash);
57    shared::<crate::ecs::ComponentSet>(&mut hash);
58    shared::<crate::ecs::ComponentType>(&mut hash);
59    shared::<crate::ecs::ComponentSlot>(&mut hash);
60    shared::<crate::actions::BindingState>(&mut hash);
61    shared::<crate::ComponentCommand>(&mut hash);
62    shared::<crate::ecs::Parent>(&mut hash);
63    shared::<crate::ecs::Owner>(&mut hash);
64    shared::<crate::ObjectId>(&mut hash);
65    shared::<crate::ecs::DisabledComponent<crate::Spin>>(&mut hash);
66    shared::<crate::ecs::SystemContext<'static>>(&mut hash);
67    shared::<crate::ecs::SystemServices<'static>>(&mut hash);
68    shared::<crate::AnyBox>(&mut hash);
69    shared::<crate::ecs::SystemRegistration>(&mut hash);
70    shared::<crate::ecs::EntityRegistration>(&mut hash);
71    shared::<crate::ecs::Stage>(&mut hash);
72    shared::<crate::ecs::SceneEvent>(&mut hash);
73    shared::<crate::ecs::NativeActionCall<'static>>(&mut hash);
74    shared::<crate::ecs::NativeProperties<'static, 'static>>(&mut hash);
75    shared::<crate::ClickToMove>(&mut hash);
76    shared::<crate::RtsSelection>(&mut hash);
77    shared::<crate::ControlsMenu>(&mut hash);
78    shared::<crate::NetworkAvatar>(&mut hash);
79    shared::<crate::AvatarHands>(&mut hash);
80    shared::<crate::Script>(&mut hash);
81    shared::<crate::WasmScript>(&mut hash);
82    hash.finish()
83}

85fn fnv(s: &str) -> u64 {
86    s.bytes().fold(0xcbf29ce484222325u64, |h, b| (h ^ b as u64).wrapping_mul(0x100000001b3))
87}

89impl PluginAbi {
90    /// The stamp of the code being compiled right now (inside a plugin: the plugin's copy of core).
91    pub fn current() -> Self {
92        let mut v = env!("CARGO_PKG_VERSION").split('.').map(|p| p.parse::<u16>().unwrap_or(0));
93        Self {
94            magic: MAGIC,
95            abi: PLUGIN_ABI_VERSION,
96            core_major: v.next().unwrap_or(0),
97            core_minor: v.next().unwrap_or(0),
98            core_patch: v.next().unwrap_or(0),
99            debug_assertions: cfg!(debug_assertions) as u8,
100            pointer_width: usize::BITS as u8,
101            rustc_hash: fnv(env!("ZESTENGINE_RUSTC_VERSION")) ^ shared_type_hash(),
102        }
103    }
104
105    /// `Ok` when a plugin with stamp `self` may be loaded by `host`; otherwise a sentence naming
106    /// the first mismatch and what to do about it.
107    pub fn check_against(&self, host: &PluginAbi) -> Result<(), String> {
108        if self.magic != MAGIC {
109            return Err("the plugin's `zestengine_plugin_abi` export is not a ZestEngine ABI stamp".into());
110        }
111        if self.abi != host.abi {
112            return Err(format!("plugin ABI version {} but this build expects {}: rebuild the plugin against this engine", self.abi, host.abi));
113        }
114        if (self.core_major, self.core_minor, self.core_patch) != (host.core_major, host.core_minor, host.core_patch) {
115            return Err(format!(
116                "plugin built against zestengine-core {}.{}.{} but this build is {}.{}.{}: rebuild the plugin",
117                self.core_major, self.core_minor, self.core_patch, host.core_major, host.core_minor, host.core_patch
118            ));
119        }
120        if self.rustc_hash != host.rustc_hash {
121            return Err("plugin built with a different rustc, dependency features or shared type layout: shared Rust type identities and layouts must match; rebuild the host and plugin together with the same toolchain, profile and features".into());
122        }
123        if self.debug_assertions != host.debug_assertions {
124            let mode = |d: u8| if d != 0 { "debug" } else { "release" };
125            return Err(format!("plugin built in {} mode but this build is {}: build the plugin with the same profile", mode(self.debug_assertions), mode(host.debug_assertions)));
126        }
127        if self.pointer_width != host.pointer_width {
128            return Err(format!("plugin is {}-bit but this build is {}-bit", self.pointer_width, host.pointer_width));
129        }
130        Ok(())
131    }
132}

134/// Export the ABI stamp from a plugin crate: `zestengine_core::export_plugin_abi!();`
135#[macro_export]
136macro_rules! export_plugin_abi {
137    () => {
138        #[unsafe(no_mangle)]
139        pub extern "C" fn zestengine_plugin_abi() -> $crate::PluginAbi {
140            $crate::PluginAbi::current()
141        }
142    };
143}

145#[cfg(test)]
146mod tests {
147    use super::*;
148
149    #[test]
150    fn a_matching_stamp_passes_and_each_mismatch_is_named() {
151        let host = PluginAbi::current();
152        assert!(host.check_against(&host).is_ok());
153        let bad = |f: &dyn Fn(&mut PluginAbi)| {
154            let mut p = host;
155            f(&mut p);
156            p.check_against(&host).unwrap_err()
157        };
158        assert!(bad(&|p| p.magic = 1).contains("not a ZestEngine ABI stamp"));
159        assert!(bad(&|p| p.abi += 1).contains("ABI version"));
160        assert!(bad(&|p| p.core_minor += 1).contains("zestengine-core"));
161        assert!(bad(&|p| p.rustc_hash ^= 1).contains("rustc"));
162        assert!(bad(&|p| p.debug_assertions ^= 1).contains("mode"));
163        assert!(bad(&|p| p.pointer_width = 32).contains("bit"));
164    }
