1//! Avatar authoring components and the finished expression menu.
2//!
3//! These components describe intent, not mechanism: a [`MenuToggle`] says "this
4//! menu entry shows or hides these things"; the avatar build pass
5//! (`zestengine_assets::avatar_build`) expands it into a parameter, a menu
6//! entry, an animation clip and a controller layer, on a clone of the avatar,
7//! and strips the authoring components from the result. Nothing here runs per
8//! frame.
9//!
10//! Targets of actions are object paths below the avatar root ("" = the root
11//! itself, "Body/Hat" = a descendant), the same addressing animation clips
12//! use, so they survive renames done through the editor's rename helpers and
13//! travel with prefabs.
14//!
15//! [`AvatarMenu`] is the radial menu's data: a tree of [`AvatarMenuItem`]s
16//! with at most [`MAX_MENU_CONTROLS`] entries per page. The build pass
17//! assembles it from the hierarchy's `Avatar Menu Item`s and the generated
18//! toggles and stores it on the avatar root as [`AvatarMenuData`].
19

32/// One thing a toggle changes while it is on. `target` is a path below the
33/// avatar root.
34#[derive(Serialize, Deserialize, Clone, Debug, PartialEq)]
35pub enum ToggleAction {
36    /// The object is active (`on`) or inactive while the toggle is on.
37    Active {
38        /// Object path below the avatar root ("" = the root).
39        target: String,
40        /// The object's active state while the toggle is on.
41        on: bool,
42    },
43    /// A component (by type name, "Light") is switched on or off.
44    ComponentEnabled {
45        /// Object path below the avatar root.
46        target: String,
47        /// Component type name.
48        component: String,
49        /// The component's enabled state while the toggle is on.
50        on: bool,
51    },
52    /// A blendshape weight on the object's skinned mesh.
53    BlendShape {
54        /// Object path below the avatar root.
55        target: String,
56        /// Blendshape name.
57        shape: String,
58        /// Weight while the toggle is on (0..1).
59        value: f32,
60    },
61    /// A material scalar of a render slot (`MaterialProp` names, "metallic").
62    /// `off` is the value while the toggle is off.
63    MaterialFloat {
64        /// Object path below the avatar root.
65        target: String,
66        /// Render slot.
67        slot: u8,
68        /// Material property name.
69        prop: String,
70        /// Value while the toggle is on.
71        value: f32,
72        /// Value while the toggle is off.
73        off: f32,
74    },
75    /// A material colour of a render slot ("base_color", "emission_color").
76    MaterialColor {
77        /// Object path below the avatar root.
78        target: String,
79        /// Render slot.
80        slot: u8,
81        /// Material property name.
82        prop: String,
83        /// Colour while the toggle is on (linear RGBA).
84        value: [f32; 4],
85        /// Colour while the toggle is off (linear RGBA).
86        off: [f32; 4],
87    },
88    /// Every track of an authored `.anim` clip plays while the toggle is on.
89    Clip {
90        /// Project-relative `.anim` path.
91        path: String,
92    },
93    /// A parameter is set when the toggle turns on.
94    SetParam {
95        /// Parameter name.
96        name: String,
97        /// Value written while the toggle is on.
98        value: f32,
99    },
100}

102impl ToggleAction {
103    /// Inspector label of this action.
104    pub fn label(&self) -> &'static str {
105        match self {
106            Self::Active { .. } => "Object",
107            Self::ComponentEnabled { .. } => "Component",
108            Self::BlendShape { .. } => "Blend Shape",
109            Self::MaterialFloat { .. } => "Material Value",
110            Self::MaterialColor { .. } => "Material Colour",
111            Self::Clip { .. } => "Clip",
112            Self::SetParam { .. } => "Set Parameter",
113        }
114    }
115    /// The object path this action touches (`None` for clips and parameters).
116    pub fn target(&self) -> Option<&str> {
117        match self {
118            Self::Active { target, .. }
119            | Self::ComponentEnabled { target, .. }
120            | Self::BlendShape { target, .. }
121            | Self::MaterialFloat { target, .. }
122            | Self::MaterialColor { target, .. } => Some(target),
123            Self::Clip { .. } | Self::SetParam { .. } => None,
124        }
125    }
126    /// Mutable access to the object path, when the action touches one.
127    pub fn target_mut(&mut self) -> Option<&mut String> {
128        match self {
129            Self::Active { target, .. }
130            | Self::ComponentEnabled { target, .. }
131            | Self::BlendShape { target, .. }
132            | Self::MaterialFloat { target, .. }
133            | Self::MaterialColor { target, .. } => Some(target),
134            Self::Clip { .. } | Self::SetParam { .. } => None,
135        }
136    }
137}

165/// A menu entry that shows or hides things. Put it on any object under the
166/// avatar root; one component builds the whole chain (see module docs).
167#[derive(Serialize, Deserialize, Clone, Debug, PartialEq)]
168#[serde(default)]
169pub struct MenuToggle {
170    /// Where it sits in the menu: slashes make sub menus (`Outfit/Hat`),
171    /// `\/` is a literal slash. Empty uses the object's name.
172    pub menu_path: String,
173    /// Project-relative icon shown in the menu.
174    pub icon: String,
175    /// What the toggle changes while on.
176    pub actions: Vec<ToggleAction>,
177    /// How the toggle behaves when used.
178    pub mode: ToggleMode,
179    /// Remember the setting between sessions.
180    pub saved: bool,
181    /// Send the setting to other players (costs 1 bit, a slider 8).
182    pub synced: bool,
183    /// Starts on.
184    pub default_on: bool,
185    /// Toggles sharing a tag switch each other off.
186    pub exclusive_tags: Vec<String>,
187    /// With a tag: this toggle turns itself on when every toggle it shares a
188    /// tag with is off (a default outfit that cannot be deselected).
189    pub exclusive_off_state: bool,
190    /// Use this existing parameter instead of a generated one (a built-in
191    /// name makes the toggle local only).
192    pub global_param: String,
193    /// Bool parameters set true while on and false while off.
194    pub drive_globals: Vec<String>,
195    /// Count the actions' effect as the avatar's resting look when the toggle
196    /// is off (default) or on.
197    pub invert_rest: bool,
198    /// Toggles with the same non-empty selector are the options of one Int
199    /// parameter (a radio group): option `n` (1, 2, ... in hierarchy order)
200    /// is on while the parameter equals `n`, 0 is none. One Int costs 8 synced
201    /// bits however many options it has, and exclusivity is built in.
202    pub selector: String,
203    /// Seconds the on / off transitions cross-fade (0 = instant).
204    pub transition_in: f32,
205    /// Seconds the off transition cross-fades (0 = instant).
206    pub transition_out: f32,
207    /// Actions applied only on the local player's own avatar (hide your own
208    /// head, show a first-person arm). They get their own layer, gated on
209    /// `IsLocal`; remote copies never run it.
210    pub local_actions: Vec<ToggleAction>,
211}

240impl MenuToggle {
241    /// The menu path split into segments (escapes resolved), falling back to
242    /// `fallback` (the object's name) when empty.
243    pub fn path_segments(&self, fallback: &str) -> Vec<String> {
244        let p = if self.menu_path.trim().is_empty() { fallback } else { self.menu_path.as_str() };
245        split_menu_path(p)
246    }
247    /// The parameter kind this toggle needs.
248    pub fn param_kind(&self) -> crate::AvatarParamKind {
249        if self.mode.is_slider() { crate::AvatarParamKind::Float } else { crate::AvatarParamKind::Bool }
250    }
251    /// Synced bits this toggle adds when it declares its own parameter.
252    pub fn synced_bit_cost(&self) -> u32 {
253        if !self.synced || !self.global_param.is_empty() || !self.selector.trim().is_empty() {
254            0
255        } else {
256            self.param_kind().bits()
257        }
258    }

279/// Split a menu path at unescaped slashes. Empty segments are dropped.
280pub fn split_menu_path(path: &str) -> Vec<String> {
281    let mut out = Vec::new();
282    let mut cur = String::new();
283    let mut chars = path.chars().peekable();
284    while let Some(c) = chars.next() {
285        match c {
286            '\\' if chars.peek() == Some(&'/') => {
287                cur.push('/');
288                chars.next();
289            }
290            '/' => {
291                if !cur.trim().is_empty() {
292                    out.push(cur.trim().to_owned());
293                }
294                cur.clear();
295            }
296            c => cur.push(c),
297        }
298    }
299    if !cur.trim().is_empty() {
300        out.push(cur.trim().to_owned());
301    }
302    out
303}

510/// The finished expression menu: the root page and everything below it.
511#[derive(Serialize, Deserialize, Clone, Debug, PartialEq, Default)]
512#[serde(default)]
513pub struct AvatarMenu {
514    /// The root page's entries.
515    pub root: Vec<AvatarMenuNode>,
516}
517
518/// Text normalised for matching menu labels (case, markup tags, whitespace).
519pub fn normalise_label(s: &str) -> String {
520    let mut out = String::new();
521    let mut in_tag = false;
522    for c in s.chars() {
523        match c {
524            '<' => in_tag = true,
525            '>' if in_tag => in_tag = false,
526            _ if in_tag => {}
527            c if c.is_whitespace() => {}
528            c => out.extend(c.to_lowercase()),
529        }
530    }
531    out
532}
533
534impl AvatarMenu {
535    /// True when the root page is empty.
536    pub fn is_empty(&self) -> bool {
537        self.root.is_empty()
538    }
539
540    /// The page reached by following `path` (child indices) from the root.
541    pub fn page(&self, path: &[usize]) -> Option<&[AvatarMenuNode]> {
542        let mut page: &[AvatarMenuNode] = &self.root;
543        for &i in path {
544            page = &page.get(i)?.children;
545        }
546        Some(page)
547    }
548
549    /// The page at the sub menu path `segments`, created on the way, as a
550    /// mutable list. An empty path is the root page.
551    pub fn page_at_mut(&mut self, segments: &[String]) -> &mut Vec<AvatarMenuNode> {
552        let mut path: Vec<usize> = Vec::new();
553        for d in segments {
554            let want = normalise_label(d);
555            let page = self.page_mut(&path).expect("path built from this tree");
556            let at = match page.iter().position(|n| n.item.kind == MenuItemKind::SubMenu && normalise_label(&n.item.label) == want) {
557                Some(i) => i,
558                None => {
559                    page.push(AvatarMenuNode::new(AvatarMenuItem { label: d.clone(), kind: MenuItemKind::SubMenu, parameter: String::new(), ..Default::default() }));
560                    page.len() - 1
561                }
562            };
563            path.push(at);
564        }
565        self.page_mut(&path).expect("path built from this tree")
566    }
567
568    fn page_mut(&mut self, path: &[usize]) -> Option<&mut Vec<AvatarMenuNode>> {
569        let mut page = &mut self.root;
570        for &i in path {
571            page = &mut page.get_mut(i)?.children;
572        }
573        Some(page)
574    }
575
576    /// Add `item` at `segments` (sub menus created on the way, matched by
577    /// normalised label); the last segment is the item's own label.
578    pub fn insert_at(&mut self, segments: &[String], mut item: AvatarMenuItem) {
579        let mut path: Vec<usize> = Vec::new();
580        if let Some((leaf, dirs)) = segments.split_last() {
581            for d in dirs {
582                let want = normalise_label(d);
583                let page = self.page_mut(&path).expect("path built from this tree");
584                let at = match page.iter().position(|n| n.item.kind == MenuItemKind::SubMenu && normalise_label(&n.item.label) == want) {
585                    Some(i) => i,
586                    None => {
587                        page.push(AvatarMenuNode::new(AvatarMenuItem {
588                            label: d.clone(),
589                            kind: MenuItemKind::SubMenu,
590                            parameter: String::new(),
591                            ..Default::default()
592                        }));
593                        page.len() - 1
594                    }
595                };
596                path.push(at);
597            }
598            item.label = leaf.clone();
599        }
600        self.page_mut(&path).expect("path built from this tree").push(AvatarMenuNode::new(item));
601    }
602
603    /// Pages longer than [`MAX_MENU_CONTROLS`] keep their first seven entries
604    /// and move the rest into a "More" sub menu, repeatedly. Empty sub menus
605    /// (no children and no asset) are dropped.
606    pub fn finalize(&mut self) {
607        fn fix(page: &mut Vec<AvatarMenuNode>) {
608            for n in page.iter_mut() {
609                fix(&mut n.children);
610            }
611            page.retain(|n| n.item.kind != MenuItemKind::SubMenu || !n.children.is_empty() || !n.item.sub_menu_asset.is_empty());
612            if page.len() > MAX_MENU_CONTROLS {
613                let mut rest = page.split_off(MAX_MENU_CONTROLS - 1);
614                fix_overflow(&mut rest);
615                page.push(AvatarMenuNode {
616                    item: AvatarMenuItem { label: "More".into(), kind: MenuItemKind::SubMenu, parameter: String::new(), ..Default::default() },
617                    children: rest,
618                });
619            }
620        }
621        fn fix_overflow(page: &mut Vec<AvatarMenuNode>) {
622            if page.len() > MAX_MENU_CONTROLS {
623                let mut rest = page.split_off(MAX_MENU_CONTROLS - 1);
624                fix_overflow(&mut rest);
625                page.push(AvatarMenuNode {
626                    item: AvatarMenuItem { label: "More".into(), kind: MenuItemKind::SubMenu, parameter: String::new(), ..Default::default() },
627                    children: rest,
628                });
629            }
630        }
631        fix(&mut self.root);
632    }
633
634    /// Visit every node depth first with its page depth.
635    pub fn visit(&self, f: &mut dyn FnMut(&AvatarMenuNode, usize)) {
636        fn go(page: &[AvatarMenuNode], depth: usize, f: &mut dyn FnMut(&AvatarMenuNode, usize)) {
637            for n in page {
638                f(n, depth);
639                go(&n.children, depth + 1, f);
640            }
641        }
642        go(&self.root, 0, f);
643    }
644
645    /// Total nodes, pages included.
646    pub fn count(&self) -> usize {
647        let mut n = 0;
648        self.visit(&mut |_, _| n += 1);
649        n
650    }
651
652    /// Every parameter name the menu reads or writes (items and puppet axes).
653    pub fn parameters(&self) -> Vec<String> {
654        let mut out: Vec<String> = Vec::new();
655        self.visit(&mut |n, _| {
656            for p in std::iter::once(&n.item.parameter).chain(&n.item.sub_parameters) {
657                if !p.is_empty() && !out.contains(p) {
658                    out.push(p.clone());
659                }
660            }
661        });
662        out
663    }
664
665    /// Rename a parameter everywhere in the menu.
666    pub fn rename_parameter(&mut self, old: &str, new: &str) {
667        fn go(page: &mut [AvatarMenuNode], old: &str, new: &str) {
668            for n in page {
669                for p in std::iter::once(&mut n.item.parameter).chain(n.item.sub_parameters.iter_mut()) {
670                    if p == old {
671                        *p = new.to_owned();
672                    }
673                }
674                go(&mut n.children, old, new);
675            }
676        }
677        go(&mut self.root, old, new);
678    }
679
680    /// Problems with the finished menu: undeclared parameters, shape errors.
681    pub fn problems(&self, known: &dyn Fn(&str) -> bool) -> Vec<String> {
682        let mut out = Vec::new();
683        self.visit(&mut |n, _| out.extend(n.item.problems(known)));
684        out
685    }
686
687    /// Serialize to RON.
688    pub fn to_ron(&self) -> String {
689        ron::ser::to_string_pretty(self, ron::ser::PrettyConfig::default()).unwrap_or_default()
690    }
691    /// Parse RON.
692    pub fn from_ron(text: &str) -> Result<Self, String> {
693        ron::from_str(text).map_err(|e| e.to_string())
694    }
695    /// Read and parse from `path`.
696    pub fn load(path: &Path) -> Result<Self, String> {
697        let text = std::fs::read_to_string(path).map_err(|e| format!("reading {}: {e}", path.display()))?;
698        Self::from_ron(&text)
699    }
700    /// Write to `path` (creating parent directories).
701    pub fn save(&self, path: &Path) -> std::io::Result<()> {
702        if let Some(parent) = path.parent() {
703            std::fs::create_dir_all(parent)?;
704        }
705        std::fs::write(path, self.to_ron())
706    }
707}
