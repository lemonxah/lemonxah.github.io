1//! zestengine-core: the engine-runtime API shared by the engine, editor, and
2//! plugins. Egui-free by design so a shipped game links it (through the engine)
3//! without pulling in any editor code.
4//!
5//! Holds serializable [`Component`]/[`TypedComponent`] data, native ECS
6//! systems registered on [`ComponentRegistry`], [`Transform`], deferred
7//! [`ComponentCommand`]s, and the built-in component data. Editor-only concerns
8//! (inspector UI, viewport gizmos) are separate traits in `zestengine-editor`.
9//! Typed ECS storage, hierarchy and plugin system registration live in [`ecs`].
10//!
11//! # Guides
12//!
13//! This reference documents every item; the guides teach how they fit together,
14//! with examples that are compiled and run by the engine's tests.
15//!
16//! - **[ECS Guide](https://lemonxah.github.io/zestengine/docs/ECS.html)**: components,
17//!   systems, stages, commands, events and entities, with recipes (movement, damage,
18//!   pickups, doors, spawners, shooting).
19//! - **[Engine API Guide](https://lemonxah.github.io/zestengine/docs/ENGINE_API.html)**:
20//!   how to move things, spawn things, raycast, play sounds, read input, save and
21//!   network, organised by task.
22//! - **[WASM Guide](https://lemonxah.github.io/zestengine/docs/WASM.html)**: sandboxed
23//!   WebAssembly scripts for user-generated content and mods.
24//! - [Documentation home](https://lemonxah.github.io/zestengine/docs/)
25//!
26//! # Where to start in this reference
27//!
28//! - [`ecs`]: the ECS model, [`ecs::Stage`], [`ecs::SystemWorld`], [`ecs::SystemContext`],
29//!   [`ecs::run_ordered`] and entity templates.
30//! - [`ecs::EcsMoveHost`]: everything a system can do to the world.
31//! - [`TypedComponent`]: write a component. [`ComponentRegistry`]: register components,
32//!   systems and entities. [`ComponentCommand`]: every request a system can queue.

290/// A stable, globally-unique object identity (UUID v4). Assigned when an object
291/// is created (editor spawn / import / scene load) and serialized in `.scene`,
292/// so cross-references survive reloads, reordering, and networking. Names stay
293/// purely cosmetic. Serialized as the canonical hyphenated string.
294#[derive(Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
295#[serde(into = "String", try_from = "String")]
296pub struct ObjectId(u128);
297
298impl ObjectId {
299    /// The "unassigned" id (all-zero). Used as a placeholder until a real id is
300    /// generated (e.g. loading a legacy scene without ids).
301    pub const NIL: ObjectId = ObjectId(0);
302
303    /// Generate a fresh random v4 UUID.
304    pub fn new() -> Self {
305        let mut b = [0u8; 16];
306        getrandom::fill(&mut b).expect("getrandom for object id");
307        b[6] = (b[6] & 0x0f) | 0x40; // version 4
308        b[8] = (b[8] & 0x3f) | 0x80; // RFC 4122 variant
309        ObjectId(u128::from_be_bytes(b))
310    }
311
312    /// Whether this is the unassigned [`ObjectId::NIL`] placeholder (no real
313    /// id has been generated or resolved yet).
314    pub fn is_nil(&self) -> bool {
315        self.0 == 0
316    }
317
318    /// Raw 128-bit value for wire encoding (networking).
319    pub fn raw(&self) -> u128 {
320        self.0
321    }
322
323    /// Reconstruct from a raw 128-bit value (networking).
324    pub fn from_raw(v: u128) -> Self {
325        ObjectId(v)
326    }
327}
328
329impl Default for ObjectId {
330    fn default() -> Self {
331        ObjectId::NIL
332    }
333}
334
335impl std::fmt::Display for ObjectId {
336    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
337        let b = self.0.to_be_bytes();
338        write!(
339            f,
340            "{:02x}{:02x}{:02x}{:02x}-{:02x}{:02x}-{:02x}{:02x}-{:02x}{:02x}-{:02x}{:02x}{:02x}{:02x}{:02x}{:02x}",
341            b[0], b[1], b[2], b[3], b[4], b[5], b[6], b[7], b[8], b[9], b[10], b[11], b[12], b[13],
342            b[14], b[15]
343        )
344    }
345}
346
347impl std::fmt::Debug for ObjectId {
348    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
349        write!(f, "ObjectId({self})")
350    }
351}
352
353impl std::str::FromStr for ObjectId {
354    type Err = String;
355    fn from_str(s: &str) -> Result<Self, Self::Err> {
356        let hex: String = s.chars().filter(|c| *c != '-').collect();
357        if hex.len() != 32 {
358            return Err(format!("invalid object id {s:?}"));
359        }
360        u128::from_str_radix(&hex, 16)
361            .map(ObjectId)
362            .map_err(|e| format!("invalid object id {s:?}: {e}"))
363    }
364}
365
366impl From<ObjectId> for String {
367    fn from(id: ObjectId) -> String {
368        id.to_string()
369    }
370}
371
372impl TryFrom<String> for ObjectId {
373    type Error = String;
374    fn try_from(s: String) -> Result<Self, Self::Error> {
375        s.parse()
376    }
377}

379/// A reference to another object, settable in the editor and resolved at

399    }

410/// A decomposed transform (translation / rotation / scale). The gameplay-side
411/// view of any object. The in-game API resolves object references into this.
412#[derive(Clone, Copy, Debug, PartialEq)]
413pub struct Transform {
414    /// Position in meters (world space when read through [`crate::ecs::EcsMoveHost::transform_of`]).
415    pub translation: Vec3,
416    /// Orientation as a unit quaternion (world space when read through the
417    /// context; local space when held by a component).
418    pub rotation: Quat,
419    /// Per-axis scale multiplier (1 = unscaled).
420    pub scale: Vec3,
421}
422
423impl Default for Transform {
424    fn default() -> Self {
425        Self {
426            translation: Vec3::ZERO,
427            rotation: Quat::IDENTITY,
428            scale: Vec3::ONE,
429        }
430    }
431}
432
433impl Transform {
434    /// Decompose a matrix into translation / rotation / scale (TRS order:
435    /// scale, then rotation, then translation).
436    pub fn from_matrix(m: Mat4) -> Self {
437        let (scale, rotation, translation) = m.to_scale_rotation_translation();
438        Self {
439            translation,
440            rotation,
441            scale,
442        }
443    }
444    /// Recompose the transform into a `Mat4` (TRS order).
445    pub fn matrix(&self) -> Mat4 {
446        Mat4::from_scale_rotation_translation(self.scale, self.rotation, self.translation)
447    }
448    /// Local -Z in world space (the "looking" direction).
449    pub fn forward(&self) -> Vec3 {
450        self.rotation * Vec3::NEG_Z
451    }
452    /// Local +X rotated into the transform's frame.
453    pub fn right(&self) -> Vec3 {
454        self.rotation * Vec3::X
455    }
456    /// Local +Y rotated into the transform's frame.
457    pub fn up(&self) -> Vec3 {
458        self.rotation * Vec3::Y
459    }
460}

464/// A request a system queues for the engine: spawn, destroy, push a body, play a
465/// sound, load a scene, save, talk over the network.
466///
467/// Systems edit component data directly; anything that reaches beyond it is queued
468/// here by pushing onto [`ecs::SystemContext::commands`] (or through the helpers on
469/// [`ecs::EcsMoveHost`], which are usually nicer). The engine applies the queue, in
470/// order, when the stage that queued it finishes, after all checked ECS borrows have
471/// been released. A command that names an object that no longer exists is ignored.
472/// The play session applies most commands itself; session, window and content
473/// commands are handed on to the host application.
474///
475/// See [Commands](https://lemonxah.github.io/zestengine/docs/ECS.html#commands) in the
476/// ECS guide, and the
477/// [command reference](https://lemonxah.github.io/zestengine/docs/ENGINE_API.html#command-reference)
478/// in the Engine API guide for which helper queues which command.
479#[derive(Debug)]
480pub enum ComponentCommand {
481    /// Switch to another scene (level change, menu -> game). Path is
482    /// project-relative, e.g. "scenes/level2.scene". Replaces the current scene.
483    LoadScene(String),
484    /// Start loading a scene in the background; a later `LoadScene` /
485    /// `LoadSceneAdditive` of it switches instantly once ready.
486    PreloadScene(String),
487    /// Add another scene's objects to the current one (streaming rooms).
488    LoadSceneAdditive(String),
489    /// Remove the objects an additive load brought in.
490    UnloadScene(String),
491    /// Carry an object (and its children) into the next scene on `LoadScene`.
492    KeepOnSceneChange(ObjectId, bool),
493    /// Make this camera object the active render camera (2A).
494    SetActiveCamera(ObjectId),
495    /// Write another object's LOCAL transform (each field optional). The play
496    /// session still applies it, but native systems do not need it: they edit the
497    /// `Transform` in place, immediately, with
498    /// [`ecs::EcsMoveHost::set_local_transform`], which is visible to later systems
499    /// in the same stage. Use this command only when you cannot reach the world
500    /// (external tools).
501    SetLocalTransform {
502        /// Target object.
503        id: ObjectId,
504        /// New local position in meters (`None` = keep).
505        translation: Option<Vec3>,
506        /// New local rotation (`None` = keep).
507        rotation: Option<Quat>,
508        /// New local scale (`None` = keep).
509        scale: Option<Vec3>,
510    },
511    /// Teleport: write another object's WORLD position/rotation (each field
512    /// optional). The engine converts to local using the object's parent, so
513    /// this is correct whether the target is a root or nested. Local-space
514    /// writes use `SetLocalTransform`. As with that command, native systems
515    /// should call [`ecs::EcsMoveHost::set_world_transform_of`] instead, which
516    /// edits the transform in place and takes effect immediately.
517    SetWorldTransform {
518        /// Target object.

820/// Object-safe component, as stored on scene objects. Implement
821/// [`TypedComponent`] instead of this directly. Runtime-only: editor inspector
822/// UI and viewport gizmos are separate traits in `zestengine-editor`.
823pub trait Component: Send + std::marker::Sync + 'static {
824    /// Move a component into typed ECS storage without serializing its live state.
825    fn into_any(self: Box<Self>) -> Box<dyn std::any::Any>;
826    /// Normalize temporary construction wrappers before permanent ECS insertion.
827    fn into_storage(self: Box<Self>) -> Box<dyn Component>;
828    /// Move live state between enabled and typed disabled representations.
829    fn into_enabled(self: Box<Self>) -> Box<dyn Component>;
830    fn into_disabled(self: Box<Self>) -> Box<dyn Component>;
831    fn inner_data(&self) -> &dyn Component;
832    fn inner_data_mut(&mut self) -> &mut dyn Component;
833    fn is_enabled(&self) -> bool { true }
834    /// Typed operations for permanent ECS storage, including plugin-defined types.
835    fn storage_type(&self) -> ecs::ComponentType;
836    /// The registered component type name (the `NAME` const of its
837    /// [`TypedComponent`]), used for serialization and by-name lookups.
838    fn type_name(&self) -> &'static str;
839    /// Serialize to RON.
840    fn save(&self) -> String;
841    /// Downcasting for engine + editor systems that read typed component data.
842    fn as_any(&self) -> &dyn std::any::Any;
843    /// Mutable downcasting (assigning ids, editor inspector/gizmo dispatch).
844    fn as_any_mut(&mut self) -> &mut dyn std::any::Any;
845    /// The component as a movement controller networked prediction can
846    /// rewind and replay, if it is one.
847    fn predictable(&self) -> Option<&dyn Predictable> {
848        None
849    }
850    /// Mutable version of [`Component::predictable`].
851    fn predictable_mut(&mut self) -> Option<&mut dyn Predictable> {
852        None
853    }
854    /// Functions this component exposes to event bindings (see [`actions`]).
855    fn actions(&self) -> &'static [ActionDesc] {
856        &[]
857    }
858    /// Run one of [`Component::actions`]; `args` are already coerced to the
859    /// declared parameter types and as long as the parameter list.
860    fn invoke_action(&mut self, _ctx: &mut ActionCtx, name: &str, _args: &[ArgValue]) -> Result<(), String> {
861        Err(actions::unknown_action(self.type_name(), name))
862    }
863    /// Values other bindings may read from this component.
864    fn properties(&self) -> &'static [PropertyDesc] {
865        &[]
866    }
867    /// Read one of the values declared in [`Component::properties`] by name.
868    fn read_property(&self, _name: &str) -> Option<ArgValue> {
869        None
870    }
871    /// Short text telling several components of one type on an object apart
872    /// (a Wasm Script's module name); bindings store it next to the index.
873    fn instance_label(&self) -> Option<String> {
874        None
875    }
876}

959/// Serializable ECS component data; blanket-implements [`Component`]. Native
960/// systems registered on [`ComponentRegistry`] implement runtime behavior.
961/// Inspector UI and gizmos live in `zestengine-editor`. Data must be Send + Sync;
962/// thread-confined runtime state belongs outside component data.
963///
964/// A component is a struct that is `Serialize + Deserialize + Default`, with a stable
965/// [`NAME`](TypedComponent::NAME). It holds data only: behaviour lives in systems that
966/// query the type. Fields not marked `#[serde(skip)]` are saved in scenes and shown in
967/// the inspector; `#[serde(skip)]` fields are runtime state.
968///
969/// ```ignore
970/// #[derive(Serialize, Deserialize)]
971/// pub struct Rotator { pub degrees_per_second: f32 }
972///
973/// impl Default for Rotator {
974///     fn default() -> Self { Self { degrees_per_second: 45.0 } }
975/// }
976///
977/// impl TypedComponent for Rotator {
978///     const NAME: &'static str = "Rotator";
979/// }
980///
981/// registry.register_ecs::<Rotator>();
982/// ```
983///
984/// Optionally expose [`actions`](TypedComponent::actions) and
985/// [`properties`](TypedComponent::properties) so the editor's event bindings (and
986/// WASM scripts) can call and read the component.
987///
988/// See [Components](https://lemonxah.github.io/zestengine/docs/ECS.html#components) in
989/// the ECS guide.
990pub trait TypedComponent: Serialize + DeserializeOwned + Default + Send + std::marker::Sync + 'static {
991    const NAME: &'static str;
992    /// Previous serialized names, accepted on load but hidden in Add menus.
993    const ALIASES: &'static [&'static str] = &[];
994    /// Where this type exists by default (NET-3): `Client` types are stripped
995    /// from a dedicated server, `Server` types from players. A scene can
996    /// override it per component. The built-ins are classified in
997    /// [`net_role::builtin_default_role`] (used while this stays `Both`).
998    const NET_ROLE: NetRole = NetRole::Both;
999    /// A movement component that implements [`Predictable`] returns
1000    /// `Some(self)` from this and [`Self::predictable_mut`]: an owner-predicted
1001    /// `NetworkObject` carrying it then moves at once on its owner and is
1002    /// reconciled against the host, like the built-in controllers.
1003    fn predictable(&self) -> Option<&dyn Predictable> {
1004        None
1005    }
1006    fn predictable_mut(&mut self) -> Option<&mut dyn Predictable> {
1007        None
1008    }
1009    /// Functions exposed to event bindings; declare them as a `static` slice.
1010    fn actions(&self) -> &'static [ActionDesc] {
1011        &[]
1012    }
1013    /// Run an action. `args` match the declared parameters in count and type.
1014    fn invoke_action(&mut self, _ctx: &mut ActionCtx, name: &str, _args: &[ArgValue]) -> Result<(), String> {
1015        Err(actions::unknown_action(Self::NAME, name))
1016    }
1017    /// Values other bindings may read from this component.
1018    fn properties(&self) -> &'static [PropertyDesc] {
1019        &[]
1020    }
1021    fn read_property(&self, _name: &str) -> Option<ArgValue> {
1022        None
1023    }
1024    /// See [`Component::instance_label`].
1025    fn instance_label(&self) -> Option<String> {
1026        None
1027    }
1028}

1246    /// Register a component type. Re-registering a name replaces the entry;
1247    /// that's how hot-reloaded plugin components supersede their old build.
1248    /// Data is available to ECS queries; frame callbacks remain enabled. Use
1249    /// `register_ecs` when systems supply the type's frame behaviour.
1250    pub fn register<T: TypedComponent>(&mut self) {
1251        let info = ComponentInfo {
1252            aliases: T::ALIASES,
1253            name: T::NAME,
1254            net_role: if T::NET_ROLE == NetRole::Both { net_role::builtin_default_role(T::NAME) } else { T::NET_ROLE },
1255            create: || Box::new(T::default()),
1256            load: |data| Ok(Box::new(ron::from_str::<T>(data)?) as Box<dyn Component>),
1257        };
1258        match self.entries.iter_mut().find(|e| e.name == T::NAME) {
1259            Some(entry) => {
1260                tracing::debug!("component {:?} re-registered (reload)", T::NAME);
1261                *entry = info;
1262            }
1263            None => self.entries.push(info),
1264        }
1265    }

1267    /// The default [`NetRole`] of a registered type; `Both` for unknown names
1268    /// (a plugin that is not loaded keeps its components on both sides).
1269    pub fn default_role(&self, name: &str) -> NetRole {
1270        let entry = self.entries.iter().find(|e| e.name == name || e.aliases.contains(&name));
1271        let override_role = entry.and_then(|e| self.role_overrides.get(e.name)
1272            .or_else(|| e.aliases.iter().find_map(|alias| self.role_overrides.get(*alias))))
1273            .or_else(|| self.role_overrides.get(name));
1274        if let Some(r) = override_role {
1275            return *r;
1276        }
1277        self.entries.iter().find(|e| e.name == name || e.aliases.contains(&name)).map_or(NetRole::Both, |e| e.net_role)
1278    }

1298                previous = previous.or(legacy);
1299            }
1300        }
1301        if let Some(role) = role { self.role_overrides.insert(name, role); }

1452/// Right-handed orthographic projection (0..1 depth, like `perspective_rh`)
1453/// for a view `half_height` tall (world units) at `aspect` (width / height).
1454/// Unlike a perspective matrix, `near` may be negative to put the camera
1455/// inside the visible volume.
1456pub fn orthographic_projection(half_height: f32, aspect: f32, near: f32, far: f32) -> Mat4 {
1457    let h = half_height.max(1e-4);
1458    let w = h * aspect.max(0.01);
1459    Mat4::orthographic_rh(-w, w, -h, h, near, far)
1460}

1462/// True when `proj` is an orthographic matrix (the last row of a perspective
1463/// matrix is `(0, 0, -1, 0)`, of an orthographic one `(0, 0, 0, 1)`).
1464pub fn is_orthographic(proj: &Mat4) -> bool {
1465    proj.w_axis.w.abs() > 0.5
1466}

1468/// World-space ray `(origin, direction)` through normalised device coords
1469/// `ndc` for a view/projection pair. Perspective rays start at `eye`;
1470/// orthographic rays are parallel and start on the near plane at the pointer.
1471pub fn ndc_ray(view: Mat4, proj: Mat4, ndc: glam::Vec2, eye: Vec3) -> Option<(Vec3, Vec3)> {
1472    let inv = (proj * view).inverse();
1473    let near = inv.project_point3(ndc.extend(0.0));
1474    let far = inv.project_point3(ndc.extend(1.0));
1475    let dir = (far - near).try_normalize()?;
1476    Some((if is_orthographic(&proj) { near } else { eye }, dir))
1477}

1481    pub fn projection_matrix(&self, aspect: f32) -> Mat4 {
1482        let near = self.near.max(0.001);
1483        let far = self.far.max(near + 0.001);
1484        match self.projection {
1485            CameraProjection::Perspective => Mat4::perspective_rh(self.fov_deg.to_radians(), aspect.max(0.01), near, far),
1486            CameraProjection::Orthographic => orthographic_projection(self.ortho_size, aspect, near, far),
1487        }
1488    }
1489
1490    /// Half-width and half-height of the view volume at distance `d` in front
1491    /// of the camera (constant for orthographic cameras).
1492    pub fn half_extents_at(&self, aspect: f32, d: f32) -> (f32, f32) {
1493        let h = match self.projection {
1494            CameraProjection::Perspective => (self.fov_deg.to_radians() * 0.5).tan() * d,
1495            CameraProjection::Orthographic => self.ortho_size.max(1e-4),
1496        };
1497        (h * aspect, h)
1498    }

1525/// Box-projected (parallax-corrected) reflection direction — the **reference
1526/// implementation** the `standard.frag` reflection-probe block must match. Given a
1527/// surface at `world_pos` reflecting along `refl_dir`, and a probe whose box is
1528/// centred at `center` with half-extents `half` (and whose cubemap was captured
1529/// from `center`), it intersects the reflection ray with the box and re-vectors
1530/// from the capture centre, so a flat surface lines the reflection up with the
1531/// captured environment instead of treating the cube as infinitely distant.
1532///
1533/// Kept here (egui-free, glam-only) so the math is **unit-tested** independently of
1534/// the GPU — `cargo test -p zestengine-core box_projection` verifies the algorithm, and
1535/// any divergence between this and the shader is a bug in one of them.
1536pub fn box_project_reflection(world_pos: Vec3, refl_dir: Vec3, center: Vec3, half: Vec3) -> Vec3 {
1537    let bmin = center - half;
1538    let bmax = center + half;
1539    let inv_r = refl_dir.recip();
1540    let t1 = (bmax - world_pos) * inv_r;
1541    let t2 = (bmin - world_pos) * inv_r;
1542    let tmax = t1.max(t2);
1543    let dist = tmax.x.min(tmax.y).min(tmax.z);
1544    let hit = world_pos + refl_dir * dist;
1545    hit - center
1546}

2907fn foliage_height(heights: &[f32], n: usize, gx: f32, gz: f32) -> f32 {
2908    let x0 = gx.floor().clamp(0.0, (n - 1) as f32) as usize;
2909    let z0 = gz.floor().clamp(0.0, (n - 1) as f32) as usize;
2910    let x1 = (x0 + 1).min(n - 1);
2911    let z1 = (z0 + 1).min(n - 1);
2912    let (fx, fz) = (gx - x0 as f32, gz - z0 as f32);
2913    let h00 = heights[z0 * n + x0];
2914    let h10 = heights[z0 * n + x1];
2915    let h01 = heights[z1 * n + x0];
2916    let h11 = heights[z1 * n + x1];
2917    (h00 * (1.0 - fx) + h10 * fx) * (1.0 - fz) + (h01 * (1.0 - fx) + h11 * fx) * fz
2918}

2920/// Surface normal at the nearest grid point, matching `heightfield_mesh`'s
2921/// clamped central difference.
2922fn foliage_normal(heights: &[f32], n: usize, res: usize, xi: i64, zi: i64, step: f32) -> Vec3 {
2923    let sample = |x: i64, z: i64| {
2924        let x = x.clamp(0, res as i64) as usize;
2925        let z = z.clamp(0, res as i64) as usize;
2926        heights[z * n + x]
2927    };
2928    let dh_dx = sample(xi + 1, zi) - sample(xi - 1, zi);
2929    let dh_dz = sample(xi, zi + 1) - sample(xi, zi - 1);
2930    Vec3::new(-dh_dx, 2.0 * step, -dh_dz).normalize_or(Vec3::Y)
2931}

2933/// Scatter one foliage layer over `terrain`, deterministically from `seed`.
2934/// Returns the placed instances (fewer than the density target where the slope
2935/// limit rejects spots).
2936pub fn scatter_foliage(terrain: &Terrain, layer: &FoliageLayer, seed: u32) -> Vec<FoliageInstance> {
2937    let res = terrain.resolution as usize;
2938    let n = res + 1;
2939    if terrain.heights.len() != n * n || layer.density <= 0.0 {
2940        return Vec::new();
2941    }
2942    let step = terrain.size / terrain.resolution as f32;
2943    let half = terrain.size * 0.5;
2944    let count = ((layer.density * terrain.size * terrain.size).round() as usize).min(MAX_FOLIAGE_INSTANCES);
2945    let mut rng = FoliageRng::new(seed as u64);
2946    let mut out = Vec::with_capacity(count);
2947    for _ in 0..count {
2948        let wx = rng.range(-half, half);
2949        let wz = rng.range(-half, half);
2950        let gx = (wx + half) / step;
2951        let gz = (wz + half) / step;
2952        let height = foliage_height(terrain.effective_heights(), n, gx, gz);
2953        let normal = foliage_normal(terrain.effective_heights(), n, res, gx.round() as i64, gz.round() as i64, step);
2954        if layer.max_slope > 0.0 && normal.y.acos() > layer.max_slope {
2955            continue;
2956        }
2957        let scale = rng.range(layer.scale_min, layer.scale_max);
2958        let yaw = rng.range(-layer.yaw, layer.yaw);
2959        let up = if layer.align_to_normal { normal } else { Vec3::Y };
2960        // Yaw about the local up first, then tilt the up axis onto the surface,
2961        // so the instance keeps its yaw about the aligned normal.
2962        let rotation = Quat::from_rotation_arc(Vec3::Y, up) * Quat::from_rotation_y(yaw);
2963        out.push(FoliageInstance {
2964            position: [wx, height, wz],
2965            rotation: rotation.normalize().to_array(),
2966            scale,
2967        });
2968    }
2969    out
2970}

3061/// Deterministic 3D value noise in [0,1].
3062fn cloud_noise(p: Vec3, seed: u32) -> f32 {
3063    let (xi, yi, zi) = (p.x.floor() as i32, p.y.floor() as i32, p.z.floor() as i32);
3064    let (xf, yf, zf) = (p.x - xi as f32, p.y - yi as f32, p.z - zi as f32);
3065    let smooth = |t: f32| t * t * (3.0 - 2.0 * t);
3066    let (u, v, w) = (smooth(xf), smooth(yf), smooth(zf));
3067    let hash = |x: i32, y: i32, z: i32| -> f32 {
3068        let mut n = (seed as i32)
3069            .wrapping_add(x.wrapping_mul(1619))
3070            .wrapping_add(y.wrapping_mul(31337))
3071            .wrapping_add(z.wrapping_mul(6971));
3072        n = (n ^ (n >> 13)).wrapping_mul(1274126177);
3073        n ^= n >> 16;
3074        (n & 0x7FFFFF) as f32 / 0x7FFFFF as f32
3075    };
3076    let lerp = |a: f32, b: f32, t: f32| a + (b - a) * t;
3077    let c000 = hash(xi, yi, zi);
3078    let c100 = hash(xi + 1, yi, zi);
3079    let c010 = hash(xi, yi + 1, zi);
3080    let c110 = hash(xi + 1, yi + 1, zi);
3081    let c001 = hash(xi, yi, zi + 1);
3082    let c101 = hash(xi + 1, yi, zi + 1);
3083    let c011 = hash(xi, yi + 1, zi + 1);
3084    let c111 = hash(xi + 1, yi + 1, zi + 1);
3085    let x00 = lerp(c000, c100, u);
3086    let x10 = lerp(c010, c110, u);
3087    let x01 = lerp(c001, c101, u);
3088    let x11 = lerp(c011, c111, u);
3089    lerp(lerp(x00, x10, v), lerp(x01, x11, v), w)
3090}

3092/// Four octaves of value noise for puffy cloud structure, roughly in [0,1].
3093fn cloud_fbm(p: Vec3, seed: u32) -> f32 {
3094    let mut sum = 0.0;
3095    let mut amp = 0.5;
3096    let mut q = p;
3097    for _ in 0..4 {
3098        sum += amp * cloud_noise(q, seed);
3099        amp *= 0.5;
3100        q *= 2.0;
3101    }
3102    sum / 0.9375
3103}

3105/// Sample the volumetric cloud density at a world point, in \[0,1\]. This is the
3106/// deterministic CPU reference the raymarch shader reproduces: a height-graded
3107/// layer shaped by coverage-modulated fBm noise, advected by wind.
3108pub fn cloud_density(point: Vec3, clouds: &VolumetricClouds, time: f32) -> f32 {
3109    let p = point - Vec3::from(clouds.wind) * time;
3110    let t = (p.y - clouds.base) / clouds.thickness.max(1.0);
3111    if t <= 0.0 || t >= 1.0 {
3112        return 0.0;
3113    }
3114    let height = 4.0 * t * (1.0 - t);
3115    let n = cloud_fbm(p * clouds.scale, clouds.seed);
3116    let lo = (1.0 - clouds.coverage).clamp(0.0, 1.0);
3117    let s = ((n - lo) / 0.06).clamp(0.0, 1.0);
3118    let shape = s * s * (3.0 - 2.0 * s);
3119    (shape * height * clouds.density).clamp(0.0, 1.0)
3120}

4153/// Deepest nesting walked (a guard, trees are finite).
4154const BLEND_TREE_MAX_DEPTH: usize = 16;
4155const BT_EPS: f32 = 1e-5;
4156
4157fn bt_angle(a: [f32; 2], b: [f32; 2]) -> f32 {
4158    let cross = a[0] * b[1] - a[1] * b[0];
4159    let dot = a[0] * b[0] + a[1] * b[1];
4160    cross.atan2(dot)
4161}
4162
4163fn bt_len(a: [f32; 2]) -> f32 {
4164    (a[0] * a[0] + a[1] * a[1]).sqrt()
4165}

4167/// 1D weights: a tent per child between its neighbours' thresholds.
4168pub fn blend_weights_1d(thresholds: &[f32], x: f32) -> Vec<f32> {
4169    let n = thresholds.len();
4170    let mut w = vec![0.0; n];
4171    if n == 0 {
4172        return w;
4173    }
4174    let mut order: Vec<usize> = (0..n).collect();
4175    order.sort_by(|&a, &b| thresholds[a].total_cmp(&thresholds[b]));
4176    let (first, last) = (order[0], order[n - 1]);
4177    let x = x.clamp(thresholds[first], thresholds[last]);
4178    if n == 1 {
4179        w[first] = 1.0;
4180        return w;
4181    }
4182    let last_window = n - 2;
4183    for (k, pair) in order.windows(2).enumerate() {
4184        let (a, b) = (pair[0], pair[1]);
4185        let (xa, xb) = (thresholds[a], thresholds[b]);
4186        if x < xb || k == last_window {
4187            // Equal thresholds: the upper child takes the full weight.
4188            let t = if xb - xa > 1e-6 { ((x - xa) / (xb - xa)).clamp(0.0, 1.0) } else { 1.0 };
4189            w[a] = 1.0 - t;
4190            w[b] = t;
4191            break;
4192        }
4193    }
4194    w
4195}

4197/// 2D Simple Directional: blend the two children angularly next to `p`
4198/// (largest dot product on each side of it) and give the remainder to the
4199/// child at the origin, or spread it over all children when there is none.
4200pub fn blend_weights_simple_directional(pos: &[[f32; 2]], p: [f32; 2]) -> Vec<f32> {
4201    let n = pos.len();
4202    let mut w = vec![0.0; n];
4203    if n == 0 {
4204        return w;
4205    }
4206    if n < 2 {
4207        w[0] = 1.0;
4208        return w;
4209    }
4210    let at_origin: Vec<usize> = (0..n).filter(|&i| bt_len(pos[i]) < BT_EPS).collect();
4211    if at_origin.len() > 1 {
4212        return w;
4213    }
4214    let centre = at_origin.first().copied();
4215    let pl = bt_len(p);
4216    if pl < BT_EPS {
4217        match centre {
4218            Some(c) => w[c] = 1.0,
4219            None => w.iter_mut().for_each(|v| *v = 1.0 / n as f32),
4220        }
4221        return w;
4222    }
4223    // Angular neighbours: the best-aligned child on each side of p.
4224    type Side = Option<(usize, f32)>;
4225    let (mut left, mut right): (Side, Side) = (None, None);
4226    for (i, pos_i) in pos.iter().enumerate() {
4227        let l = bt_len(*pos_i);
4228        if l < BT_EPS {
4229            continue;
4230        }
4231        let d = [pos_i[0] / l, pos_i[1] / l];
4232        let cross = d[0] * p[1] - d[1] * p[0];
4233        let dot = d[0] * p[0] + d[1] * p[1];
4234        let slot = if cross >= 0.0 { &mut left } else { &mut right };
4235        if slot.is_none_or(|(_, best)| dot > best) {
4236            *slot = Some((i, dot));
4237        }
4238    }
4239    let (mut a, mut b, mut c) = (0.0, 0.0, 1.0);
4240    let mut sides = None;
4241    if let (Some((l, _)), Some((r, _))) = (left, right) {
4242        let (pl_, pr) = (pos[l], pos[r]);
4243        let det = pl_[0] * pr[1] - pl_[1] * pr[0];
4244        if det.abs() > 1e-9 {
4245            let ca = (p[0] * pr[1] - p[1] * pr[0]) / det;
4246            let cb = (pl_[0] * p[1] - pl_[1] * p[0]) / det;
4247            sides = Some((l, r));
4248            a = ca.max(0.0);
4249            b = cb.max(0.0);
4250            c = 1.0 - a - b;
4251            if c < 0.0 {
4252                let s = a + b;
4253                if s > BT_EPS {
4254                    a /= s;
4255                    b /= s;
4256                }
4257                c = 0.0;
4258            } else if c > 1.0 {
4259                a = 0.0;
4260                b = 0.0;
4261                c = 1.0;
4262            }
4263        }
4264    }
4265    if let Some((l, r)) = sides {
4266        w[l] += a;
4267        w[r] += b;
4268    }
4269    match centre {
4270        Some(ci) => w[ci] += c,
4271        None => w.iter_mut().for_each(|v| *v += c / n as f32),
4272    }
4273    w
4274}

4276fn bt_normalize(w: &mut [f32], fallback: impl FnOnce() -> usize) {
4277    let sum: f32 = w.iter().sum();
4278    if sum > 1e-9 {
4279        w.iter_mut().for_each(|v| *v /= sum);
4280    } else if !w.is_empty() {
4281        w.iter_mut().for_each(|v| *v = 0.0);
4282        let k = fallback();
4283        w[k] = 1.0;
4284    }
4285}

4287fn nearest_child(pos: &[[f32; 2]], p: [f32; 2]) -> usize {
4288    let d = |i: usize| (pos[i][0] - p[0]).powi(2) + (pos[i][1] - p[1]).powi(2);
4289    (0..pos.len()).min_by(|&a, &b| d(a).total_cmp(&d(b))).unwrap_or(0)
4290}

4292/// 2D Freeform Cartesian: each child falls off along the line to every other
4293/// child (gradient bands), the minimum wins, and the weights are normalised.
4294pub fn blend_weights_freeform_cartesian(pos: &[[f32; 2]], p: [f32; 2]) -> Vec<f32> {
4295    let n = pos.len();
4296    let mut w = vec![0.0; n];
4297    if n < 2 {
4298        if n == 1 {
4299            w[0] = 1.0;
4300        }
4301        return w;
4302    }
4303    for i in 0..n {
4304        let mut wi = 1.0f32;
4305        for j in 0..n {
4306            if i == j {
4307                continue;
4308            }
4309            let d = [pos[j][0] - pos[i][0], pos[j][1] - pos[i][1]];
4310            let dd = d[0] * d[0] + d[1] * d[1];
4311            if dd < 1e-12 {
4312                continue;
4313            }
4314            let dp = [p[0] - pos[i][0], p[1] - pos[i][1]];
4315            let h = 1.0 - (dp[0] * d[0] + dp[1] * d[1]) / dd;
4316            if h <= 0.0 {
4317                wi = 0.0;
4318                break;
4319            }
4320            wi = wi.min(h);
4321        }
4322        w[i] = wi;
4323    }
4324    bt_normalize(&mut w, || nearest_child(pos, p));
4325    w
4326}

4328/// 2D Freeform Directional: like Cartesian, but in polar terms (angle and
4329/// radius, the radius scaled by each pair's mean radius so they compare).
4330pub fn blend_weights_freeform_directional(pos: &[[f32; 2]], p: [f32; 2]) -> Vec<f32> {
4331    let n = pos.len();
4332    let mut w = vec![0.0; n];
4333    if n < 2 {
4334        if n == 1 {
4335            w[0] = 1.0;
4336        }
4337        return w;
4338    }
4339    let pr = bt_len(p);
4340    let rad: Vec<f32> = pos.iter().map(|q| bt_len(*q)).collect();
4341    let origin: Vec<bool> = rad.iter().map(|r| *r < BT_EPS).collect();
4342    // Offset of p from child i: (angle, radius).
4343    let alpha = |i: usize, via: Option<usize>| -> f32 {
4344        if pr < BT_EPS {
4345            return 0.0;
4346        }
4347        if !origin[i] {
4348            bt_angle(pos[i], p)
4349        } else if let Some(j) = via.filter(|&j| !origin[j]) {
4350            bt_angle(pos[j], p)
4351        } else {
4352            0.0
4353        }
4354    };
4355    for i in 0..n {
4356        if origin[i] && origin.iter().filter(|o| **o).count() > 1 {
4357            continue;
4358        }
4359        let cap_alpha = if origin[i] || pr < BT_EPS { 0.0 } else { bt_angle(pos[i], p) };
4360        let mut wi = 1.0 - cap_alpha.abs() / std::f32::consts::PI;
4361        let rho = pr - rad[i];
4362        for j in 0..n {
4363            if i == j || (origin[i] && origin[j]) {
4364                continue;
4365            }
4366            let s = 2.0 / (rad[i] + rad[j]).max(1e-9);
4367            let d_rho = (rad[j] - rad[i]) * s;
4368            let d_alpha = if origin[i] || origin[j] { 0.0 } else { bt_angle(pos[i], pos[j]) };
4369            let den = d_rho * d_rho + d_alpha * d_alpha;
4370            if den < 1e-12 {
4371                continue;
4372            }
4373            let h = 1.0 - (s * rho * d_rho + alpha(i, Some(j)) * d_alpha) / den;
4374            if h <= 0.0 {
4375                wi = 0.0;
4376                break;
4377            }
4378            wi = wi.min(h);
4379        }
4380        w[i] = wi.max(0.0);
4381    }
4382    bt_normalize(&mut w, || nearest_child(pos, p));
4383    w
4384}

4386/// Direct weights: each child's own (non-negative) parameter value, divided
4387/// by the sum of the raw values when `normalize` is on and the sum exceeds 1.
4388pub fn blend_weights_direct(values: &[f32], normalize: bool) -> Vec<f32> {
4389    let sum: f32 = values.iter().sum();
4390    let mut w: Vec<f32> = values.iter().map(|v| v.max(0.0)).collect();
4391    if normalize && sum > 1.0 {
4392        w.iter_mut().for_each(|v| *v /= sum);
4393    }
4394    w
4395}

4973    /// One movement step from an input frame (`dt` is the frame's scaled
4974    /// delta). `view` replaces the look input with a recorded (yaw, pitch).
4975    /// `physical` is `Some(grounded)` when a Character Controller moves the
4976    /// pawn: the returned world delta goes to `move_character`. Otherwise the
4977    /// pawn walks a flat floor and `pos` (local translation) moves in place.
4978    pub fn step(&mut self, input: &InputState, dt: f32, pos: &mut Vec3, physical: Option<bool>, view: Option<(f32, f32)>) -> Option<Vec3> {
4979        let dt = dt.min(0.1);
4980        let mv = input.axis2("Move");
4981
4982        match view {
4983            Some((yaw, pitch)) => {
4984                self.yaw = yaw;
4985                // A networked view is the owner's claim: held to the pawn's pitch limits.
4986                self.pitch = pitch.clamp(self.min_pitch, self.max_pitch);
4987            }
4988            None => {
4989                // Look: mouse delta is per-frame (no dt); analog sticks pre-scaled.
4990                let look = input.axis2("Look");
4991                self.yaw -= look.x * self.look_sensitivity;
4992                if matches!(self.mode, ControlMode::FirstPerson | ControlMode::ThirdPerson) {
4993                    self.pitch = (self.pitch - look.y * self.look_sensitivity)
4994                        .clamp(self.min_pitch, self.max_pitch);
4995                }
4996            }
4997        }
4998
4999        // Movement direction per mode.
5000        let yaw_rot = Quat::from_rotation_y(self.yaw.to_radians());
5001        let (dir, face_move) = match self.mode {
5002            ControlMode::FirstPerson | ControlMode::ThirdPerson => {
5003                let fwd = yaw_rot * Vec3::NEG_Z;
5004                let right = yaw_rot * Vec3::X;
5005                (right * mv.x + fwd * mv.y, false)
5006            }
5007            // World-plane movement; body faces the move direction.
5008            ControlMode::TopDown | ControlMode::Strategy => {
5009                (Vec3::new(mv.x, 0.0, -mv.y), true)
5010            }
5011        };
5012        let dir = dir.normalize_or_zero();
5013
5014        let horizontal = if dir.length_squared() > 1e-4 {
5015            let target = dir * self.move_speed;
5016            let cur = Vec3::new(self.vel.x, 0.0, self.vel.z);
5017            let next = cur.lerp(target, (self.accel * dt).min(1.0));
5018            if face_move && self.mode == ControlMode::TopDown {
5019                self.yaw = (-dir.x).atan2(-dir.z).to_degrees();
5020            }
5021            next
5022        } else {
5023            let cur = Vec3::new(self.vel.x, 0.0, self.vel.z);
5024            cur.lerp(Vec3::ZERO, (self.decel * dt).min(1.0))
5025        };
5026        self.vel.x = horizontal.x;
5027        self.vel.z = horizontal.z;
5028
5029        let walking = matches!(self.mode, ControlMode::FirstPerson | ControlMode::ThirdPerson);
5030        if let Some(grounded) = physical {
5031            self.grounded = grounded;
5032            if self.grounded && self.vel.y < 0.0 {
5033                self.vel.y = 0.0;
5034            }
5035        }
5036        if walking {
5037            if self.grounded && input.pressed("Jump") {
5038                self.vel.y = self.jump_power;
5039                self.grounded = false;
5040            }
5041            self.vel.y -= self.gravity * dt;
5042        } else {
5043            self.vel.y = 0.0;
5044        }
5045
5046        if physical.is_some() {
5047            return Some(self.vel * dt);
5048        }
5049        *pos += self.vel * dt;
5050        if walking && pos.y <= self.floor_y {
5051            pos.y = self.floor_y;
5052            self.vel.y = 0.0;
5053            self.grounded = true;
5054        }
5055        None
5056    }

5059    /// so do networked prediction's replays.

5064            Some(delta) => host.move_character(delta),
