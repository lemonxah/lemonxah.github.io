4//! covers everything else.

13/// How two touching materials combine a coefficient. When they disagree the
14/// stronger rule wins (Maximum > Multiply > Minimum > Average).
15#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
16pub enum CombineMode {
17    /// The mean of the two values.
18    #[default]
19    Average,
20    /// The smaller of the two values.
21    Minimum,
22    /// The product of the two values.
23    Multiply,
24    /// The larger of the two values.
25    Maximum,
26}
27
28impl CombineMode {
29    /// Every mode, in declaration order.
30    pub const ALL: [CombineMode; 4] = [Self::Average, Self::Minimum, Self::Multiply, Self::Maximum];
31
32    /// Inspector label of this mode.
33    pub fn label(self) -> &'static str {
34        match self {
35            Self::Average => "Average",
36            Self::Minimum => "Minimum",
37            Self::Multiply => "Multiply",
38            Self::Maximum => "Maximum",
39        }
40    }
41}

43/// How a surface behaves in contact and in water.
44#[derive(Clone, Copy, Debug, PartialEq, Serialize, Deserialize)]
45#[serde(default)]
46pub struct PhysicsMaterial {
47    /// Coulomb friction coefficient while sliding (0 = ice, ~0.5 wood, 1+
48    /// rubber).
49    pub friction: f32,
50    /// Friction holding things at rest, usually a little above `friction`
51    /// (it takes more to start something sliding than to keep it going).
52    /// None = the same as `friction`.
53    pub static_friction: Option<f32>,
54    /// How this material's friction combines with the other surface's.
55    pub friction_combine: CombineMode,
56    /// Restitution 0..1 (1 = keeps all its speed on a bounce).
57    pub bounciness: f32,
58    /// How this material's bounciness combines with the other surface's.
59    pub bounce_combine: CombineMode,
60    /// kg/m^3, used for the mass of bodies whose Rigid Body mass is 0 (water
61    /// is 1000, wood ~600, steel ~7850).
62    pub density: f32,
63    /// Scales the water's lift on the body (Archimedes, or a Buoyancy
64    /// component's override).
65    pub buoyancy: f32,
66    /// Scales the water's drag on this body.
67    pub water_drag: f32,
68    /// Extra contact distance (m) so thin or fast shapes touch before they
69    /// overlap.
70    pub contact_offset: f32,
71}
72
73impl Default for PhysicsMaterial {
74    fn default() -> Self {
75        Self {
76            friction: 0.5,
77            static_friction: None,
78            friction_combine: CombineMode::Average,
79            bounciness: 0.0,
80            bounce_combine: CombineMode::Average,
81            density: 1000.0,
82            buoyancy: 1.0,
83            water_drag: 1.0,
84            contact_offset: 0.0,
85        }
86    }
87}

89impl PhysicsMaterial {
90    /// Static friction: `static_friction`, or `friction` when unset.
91    pub fn static_mu(&self) -> f32 {
92        self.static_friction.unwrap_or(self.friction).max(0.0)
93    }
94
95    /// A few starting points for the editor.
96    pub const PRESETS: [(&'static str, PhysicsMaterial); 6] = [
97        ("Wood", Self::preset(0.5, 0.6, 0.2, 600.0)),
98        ("Metal", Self::preset(0.4, 0.55, 0.1, 7850.0)),
99        ("Rubber", Self::preset(0.9, 1.1, 0.8, 1100.0)),
100        ("Ice", Self::preset(0.02, 0.08, 0.05, 917.0)),
101        ("Stone", Self::preset(0.6, 0.75, 0.05, 2600.0)),
102        ("Foam", Self::preset(0.6, 0.8, 0.3, 50.0)),
103    ];
104
105    const fn preset(friction: f32, static_friction: f32, bounciness: f32, density: f32) -> Self {
106        Self {
107            friction,
108            static_friction: Some(static_friction),
109            friction_combine: CombineMode::Average,
110            bounciness,
111            bounce_combine: CombineMode::Average,
112            density,
113            buoyancy: 1.0,
114            water_drag: 1.0,
115            contact_offset: 0.0,
116        }
117    }
118}

120/// Read and parse a `.physmat`.
121pub fn load_physmat(path: impl AsRef<Path>) -> Result<PhysicsMaterial, String> {
122    let path = path.as_ref();
123    let text = std::fs::read_to_string(path).map_err(|e| format!("reading {}: {e}", path.display()))?;
124    ron::from_str(&text).map_err(|e| format!("parsing {}: {e}", path.display()))
125}
126
127/// Serialize and write a `.physmat`.
128pub fn save_physmat(path: impl AsRef<Path>, mat: &PhysicsMaterial) -> Result<(), String> {
129    let path = path.as_ref();
130    let text = ron::ser::to_string_pretty(mat, ron::ser::PrettyConfig::default()).map_err(|e| e.to_string())?;
131    if let Some(parent) = path.parent() {
132        std::fs::create_dir_all(parent).map_err(|e| e.to_string())?;
133    }
134    std::fs::write(path, text).map_err(|e| format!("writing {}: {e}", path.display()))
135}

137/// Project-wide physics (Project Settings > Physics). Contact stiffness is a
138/// solver setting rather than a per-material one: every contact uses it.
139#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
140#[serde(default)]
141pub struct PhysicsSettings {
142    /// Gravity (m/s^2, usually negative Y).
143    pub gravity: [f32; 3],
144    /// Material for bodies and colliders that don't name one ("" = built-in
145    /// defaults).
146    pub default_material: String,
147    /// How hard contacts push overlapping shapes apart (Hz). Higher is
148    /// stiffer; lower lets things sink in and settle softly.
149    pub contact_stiffness: f32,
150    /// Contact damping ratio: higher absorbs the push-back, lower springs.
151    pub contact_damping: f32,
152    /// Solver iterations per step: more is steadier stacks and joints.
153    pub solver_iterations: u32,
154}
155
156impl Default for PhysicsSettings {
157    fn default() -> Self {
158        Self { gravity: [0.0, -9.81, 0.0], default_material: String::new(), contact_stiffness: 30.0, contact_damping: 5.0, solver_iterations: 4 }
159    }
160}

162#[cfg(test)]
163mod tests {
164    use super::*;
165
166    #[test]
167    fn round_trips_and_fills_missing_fields() {
168        let m = PhysicsMaterial { friction: 0.9, bounce_combine: CombineMode::Maximum, ..PhysicsMaterial::PRESETS[2].1 };
169        let text = ron::ser::to_string(&m).unwrap();
170        assert_eq!(ron::from_str::<PhysicsMaterial>(&text).unwrap(), m);
171        let partial: PhysicsMaterial = ron::from_str("(bounciness: 0.7)").unwrap();
172        assert_eq!(partial.bounciness, 0.7);
173        assert_eq!(partial.density, 1000.0);
174    }
