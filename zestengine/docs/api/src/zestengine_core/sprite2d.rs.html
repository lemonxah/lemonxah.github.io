1//! 2D sprite components (2D-3): the authoring data for the sprite render pass
2//! (`zestengine-render/src/sprite2d.rs`). Plain serde data, egui-free; the
3//! inspectors and the scene gather come with the engine integration.
4//!
5//! A sprite is referenced by its text form `"<project-relative path>#<16 hex
6//! digits of the sprite's SubAssetId>"` (the same string the assets crate's
7//! `sprite_ref_string` / `SpriteRef::from_ref_string` read and write; core cannot
8//! name that type, assets depends on core). An empty string draws nothing.
9//!
10//! # Sorting
11//!
12//! Draw order is decided by, in this order: the sorting layer's **value**, the
13//! **order in layer**, the **sorting group** (a group's renderers sort as one
14//! unit against everything outside it), the material / texture, then depth.
15//! Renderers store a stable [`SortingLayerId`], never the layer's position, so
16//! reordering the project's [`SortingLayers`] never rewrites scene data. The
17//! "Default" layer has id 0 and value 0; layers listed before it have negative
18//! values and layers after it positive ones (`value = index - index_of(Default)`).
19
20use serde::{Deserialize, Serialize};

24/// Stable id of a sorting layer (random-looking for new layers, 0 = Default).
25#[derive(Clone, Copy, PartialEq, Eq, Hash, Debug, Default, PartialOrd, Ord, Serialize, Deserialize)]
26#[serde(transparent)]
27pub struct SortingLayerId(pub u32);
28
29/// The id of the built-in "Default" sorting layer.
30pub const DEFAULT_SORTING_LAYER: SortingLayerId = SortingLayerId(0);
31
32/// One project sorting layer.
33#[derive(Clone, PartialEq, Eq, Debug, Serialize, Deserialize)]
34#[serde(default)]
35pub struct SortingLayer {
36    /// Stable id (0 = Default).
37    pub id: SortingLayerId,
38    /// Layer name.
39    pub name: String,
40    /// Editor picking aid only: a locked layer's sprites cannot be picked.
41    pub locked: bool,
42    /// Draw this layer's sprites (and every layer below it) before the 3D
43    /// transparent queue, so transparent 3D objects composite over them.
44    pub behind_3d: bool,
45}
46
47impl Default for SortingLayer {
48    fn default() -> Self {
49        Self { id: DEFAULT_SORTING_LAYER, name: "Default".to_string(), locked: false, behind_3d: false }
50    }
51}

53/// The project's ordered sorting layers (back to front). Stored next to the
54/// object-layer names in [`crate::LayerSettings`] (`sorting`), serde default =
55/// one "Default" layer.
56#[derive(Clone, PartialEq, Eq, Debug, Serialize, Deserialize)]
57#[serde(default)]
58pub struct SortingLayers {
59    /// Layers back to front.
60    pub layers: Vec<SortingLayer>,
61}
62
63impl Default for SortingLayers {
64    fn default() -> Self {
65        Self { layers: vec![SortingLayer::default()] }
66    }
67}
68
69impl SortingLayers {
70    /// Index of the Default layer (a repaired list always has one).
71    pub fn default_index(&self) -> usize {
72        self.layers.iter().position(|l| l.id == DEFAULT_SORTING_LAYER).unwrap_or(0)
73    }
74
75    /// The layer's position in the list.
76    pub fn index_of(&self, id: SortingLayerId) -> Option<usize> {
77        self.layers.iter().position(|l| l.id == id)
78    }
79
80    /// The layer with this id.
81    pub fn get(&self, id: SortingLayerId) -> Option<&SortingLayer> {
82        self.layers.iter().find(|l| l.id == id)
83    }
84
85    /// The layer's draw-order value: `index - index_of(Default)`, clamped to
86    /// `i16`. An unknown id (a deleted layer) sorts as Default (0).
87    pub fn value_of(&self, id: SortingLayerId) -> i16 {
88        match self.index_of(id) {
89            Some(i) => (i as i64 - self.default_index() as i64).clamp(i16::MIN as i64, i16::MAX as i64) as i16,
90            None => 0,
91        }
92    }
93
94    /// The layer's name ("Default" for an unknown id).
95    pub fn name_of(&self, id: SortingLayerId) -> &str {
96        self.get(id).map(|l| l.name.as_str()).unwrap_or("Default")
97    }
98
99    /// The highest layer value flagged `behind_3d` (layers at or below it draw
100    /// behind the 3D transparents), if any is.
101    pub fn behind_max(&self) -> Option<i16> {
102        self.layers.iter().filter(|l| l.behind_3d).map(|l| self.value_of(l.id)).max()
103    }
104
105    /// True when the layer exists and is locked (unpickable).
106    pub fn is_locked(&self, id: SortingLayerId) -> bool {
107        self.get(id).is_some_and(|l| l.locked)
108    }
109
110    /// Append a layer named `name` (front-most) with a fresh stable id.
111    pub fn add(&mut self, name: &str) -> SortingLayerId {
112        // FNV-1a over the name, then probe until unused and non-zero: stable
113        // enough to look random, deterministic for tests.
114        let mut h: u32 = 0x811c_9dc5;
115        for b in name.bytes().chain((self.layers.len() as u32).to_le_bytes()) {
116            h = (h ^ b as u32).wrapping_mul(0x0100_0193);
117        }
118        while h == 0 || self.index_of(SortingLayerId(h)).is_some() {
119            h = h.wrapping_add(1);
120        }
121        let id = SortingLayerId(h);
122        self.layers.push(SortingLayer { id, name: name.to_string(), locked: false, behind_3d: false });
123        id
124    }
125
126    /// Remove a layer; the Default layer cannot be removed. Returns whether
127    /// anything was removed.
128    pub fn remove(&mut self, id: SortingLayerId) -> bool {
129        if id == DEFAULT_SORTING_LAYER {
130            return false;
131        }
132        let before = self.layers.len();
133        self.layers.retain(|l| l.id != id);
134        self.layers.len() != before
135    }
136
137    /// Move the layer at `from` to position `to` (both clamped).
138    pub fn move_layer(&mut self, from: usize, to: usize) {
139        if self.layers.is_empty() {
140            return;
141        }
142        let from = from.min(self.layers.len() - 1);
143        let to = to.min(self.layers.len() - 1);
144        let l = self.layers.remove(from);
145        self.layers.insert(to, l);
146    }
147
148    /// Repair a deserialized value: a Default layer exists, ids are unique.
149    pub fn normalized(mut self) -> Self {
150        let mut seen = std::collections::HashSet::new();
151        self.layers.retain(|l| seen.insert(l.id));
152        if !self.layers.iter().any(|l| l.id == DEFAULT_SORTING_LAYER) {
153            self.layers.insert(0, SortingLayer::default());
154        }
155        self
156    }
157}

159/// How the sprite fills its rect.
160#[derive(Clone, Copy, PartialEq, Eq, Debug, Default, Serialize, Deserialize)]
161pub enum SpriteDrawMode {
162    /// The sprite mesh at the object's scale.
163    #[default]
164    Simple,
165    /// 9-slice: corners keep their size, edges and centre stretch to `size`.
166    Sliced,
167    /// 9-slice with the edges and centre repeated instead of stretched.
168    Tiled,
169}

172    /// Every mode, in declaration order.
173    pub const ALL: [SpriteDrawMode; 3] = [Self::Simple, Self::Sliced, Self::Tiled];
174    /// Inspector label of this mode.
175    pub fn label(self) -> &'static str {
176        match self {
177            Self::Simple => "Simple",
178            Self::Sliced => "Sliced",
179            Self::Tiled => "Tiled",
180        }
181    }
182}

266/// Draws one sprite in the XY plane of the scene (rotation about Z, scale from
267/// the transform). In `Sliced` / `Tiled` mode `size` is the drawn rect in world
268/// units (the transform's scale applies on top). Needs a Full Rect sprite for
269/// those two modes (a tight mesh has no rect to slice).
270#[derive(Clone, PartialEq, Debug, Serialize, Deserialize)]
271#[serde(default)]
272pub struct SpriteRenderer {
273    /// `"<path>#<sprite id hex>"`; empty draws nothing.
274    pub sprite: String,
275    /// Tint, linear RGBA (alpha multiplies the sprite's).
276    pub color: [f32; 4],
277    /// Mirrored in the vertex shader (never a negative scale: flipping does not
278    /// break a batch).
279    pub flip_x: bool,
280    /// Mirrored vertically.
281    pub flip_y: bool,
282    /// How the sprite fills its rect.
283    pub draw_mode: SpriteDrawMode,
284    /// Drawn rect in world units (Sliced / Tiled).
285    pub size: [f32; 2],
286    /// Tiled mode's filling rule.
287    pub tile_mode: SpriteTileMode,
288    /// Adaptive tiling: the stretch ratio past which the middle tiles instead
289    /// of stretching (Unity's default 0.5).
290    pub adaptive_threshold: f32,
291    /// Material override (`.mat` path); empty = the default sprite material.
292    pub material: String,
293    /// Sorting layer.
294    pub sorting_layer: SortingLayerId,
295    /// Sort key inside the layer.
296    pub order_in_layer: i16,
297    /// How sprite masks affect this renderer.
298    pub mask_interaction: MaskInteraction2D,
299    /// Which point depth tie-breaks use.
300    pub sort_point: SpriteSortPoint,
301}
302
303impl Default for SpriteRenderer {

331/// Parse the `"<path>#<16 hex>"` sprite reference text (id must be non-zero).
332pub fn split_sprite_ref(text: &str) -> Option<(&str, u64)> {
333    let (path, id) = text.rsplit_once('#')?;
334    if path.is_empty() || id.len() != 16 || !id.bytes().all(|b| b.is_ascii_hexdigit()) {
335        return None;
336    }
337    let id = u64::from_str_radix(id, 16).ok()?;
338    (id != 0).then_some((path, id))
339}

345/// Renders its descendants' sprites (and nested groups) as one unit: they sort
346/// against each other inside the group, and the group sorts against everything
347/// else by its own layer and order. Nested groups flatten into the outermost
348/// enabled ancestor unless a group has `sort_at_root`.
349#[derive(Clone, PartialEq, Debug, Serialize, Deserialize)]
350#[serde(default)]
351pub struct SortingGroup {
352    /// The group's sorting layer.
353    pub sorting_layer: SortingLayerId,
354    /// The group's sort key inside the layer.
355    pub order_in_layer: i16,
356    /// This group is a root even when nested in another group.
357    pub sort_at_root: bool,
358}
359
360impl Default for SortingGroup {
361    fn default() -> Self {
362        Self { sorting_layer: DEFAULT_SORTING_LAYER, order_in_layer: 0, sort_at_root: false }
363    }
364}

367    const NAME: &'static str = "Sorting Group";

370/// Which sorting range a [`SpriteMask`] affects.
371#[derive(Clone, Copy, PartialEq, Eq, Debug, Default, Serialize, Deserialize)]
372pub enum SpriteMaskRange {
373    /// From the mask's own sort position to the next mask's.
374    #[default]
375    Auto,
376    /// From (`back_layer`, `back_order`) to (`front_layer`, `front_order`),
377    /// inclusive.
378    Custom {
379        /// First layer the mask affects.
380        back_layer: SortingLayerId,
381        /// First order in the first layer.
382        back_order: i16,
383        /// Last layer the mask affects.
384        front_layer: SortingLayerId,
385        /// Last order in the last layer.
386        front_order: i16,
387    },
388}

394/// Writes its sprite's shape into the stencil buffer; renderers with a
395/// [`MaskInteraction2D`] other than `None` inside its range are then shown only
396/// inside / outside that shape.
397#[derive(Clone, PartialEq, Debug, Serialize, Deserialize)]
398#[serde(default)]
399pub struct SpriteMask {
400    /// `"<path>#<sprite id hex>"` like [`SpriteRenderer::sprite`].
401    pub sprite: String,
402    /// Texels with alpha below this do not write the mask.
403    pub alpha_cutoff: f32,
404    /// Which sorting range the mask affects.
405    pub range: SpriteMaskRange,
406    /// Sorting layer.
407    pub sorting_layer: SortingLayerId,
408    /// Sort key inside the layer.
409    pub order_in_layer: i16,
410}

432    #[test]
433    fn behind_max_is_the_highest_flagged_layer_value() {
434        let mut s = SortingLayers::default();
435        assert_eq!(s.behind_max(), None);
436        let bg = s.add("Background");
437        let fg = s.add("Foreground");
438        s.move_layer(1, 0); // Background below Default
439        s.move_layer(2, 2);
440        assert_eq!(s.value_of(bg), -1);
441        s.layers.iter_mut().find(|l| l.id == bg).unwrap().behind_3d = true;
442        assert_eq!(s.behind_max(), Some(-1));
443        s.layers.iter_mut().find(|l| l.id == DEFAULT_SORTING_LAYER).unwrap().behind_3d = true;
444        assert_eq!(s.behind_max(), Some(0));
445        assert_eq!(s.value_of(fg), 1);
446        let back: SortingLayers = ron::from_str(&ron::to_string(&s).unwrap()).unwrap();
447        assert_eq!(back, s);
448    }
