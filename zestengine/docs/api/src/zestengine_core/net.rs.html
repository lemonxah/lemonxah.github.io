1//! Networking API (2G): the data components see and the built-in networking
2//! components. The engine's `net` module owns the transport and session; this
3//! side is egui-free and transport-free so plugins and game code only touch
4//! [`NetView`], the [`crate::ecs::ObjectContext`] helpers below and these components.
5//!
6//! Model: one peer hosts (peer id 1, the server), others join it. Every
7//! networked object has an owner peer; owner 0 means "unowned", and the host
8//! is then its authority. The authority simulates the object and replicates
9//! it; everyone else interpolates what it sends. When the host leaves, the
10//! next member in its succession list takes over (host migration): peer ids
11//! stay, `NetView::host_peer` names the new host and `HostChanged` fires.
12

27/// A small typed value carried by synced variables and RPC arguments.
28#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
29pub enum NetValue {
30    /// A boolean.
31    Bool(bool),
32    /// A signed 64-bit integer.
33    Int(i64),
34    /// A 32-bit float.
35    Float(f32),
36    /// A 3D vector (xyz).
37    Vec3([f32; 3]),
38    /// A rotation quaternion (xyzw).
39    Quat([f32; 4]),
40    /// A string.
41    Str(String),
42    /// Raw bytes.
43    Bytes(Vec<u8>),
44}
45
46impl Default for NetValue {
47    fn default() -> Self {
48        NetValue::Int(0)
49    }
50}

52impl NetValue {
53    /// The value as a bool: booleans pass through, non-zero integers become
54    /// true, anything else is None.
55    pub fn as_bool(&self) -> Option<bool> {
56        match self {
57            NetValue::Bool(b) => Some(*b),
58            NetValue::Int(i) => Some(*i != 0),
59            _ => None,
60        }
61    }
62    /// The value as an i64: integers pass through, floats truncate, booleans
63    /// become 0 / 1.
64    pub fn as_int(&self) -> Option<i64> {
65        match self {
66            NetValue::Int(i) => Some(*i),
67            NetValue::Float(f) => Some(*f as i64),
68            NetValue::Bool(b) => Some(*b as i64),
69            _ => None,
70        }
71    }
72    /// The value as an f32: floats pass through, integers are widened.
73    pub fn as_float(&self) -> Option<f32> {
74        match self {
75            NetValue::Float(f) => Some(*f),
76            NetValue::Int(i) => Some(*i as f32),
77            _ => None,
78        }
79    }
80    /// The value as a Vec3 when it holds one.
81    pub fn as_vec3(&self) -> Option<Vec3> {
82        match self {
83            NetValue::Vec3(v) => Some(Vec3::from(*v)),
84            _ => None,
85        }
86    }
87    /// The value as a Quat when it holds one.
88    pub fn as_quat(&self) -> Option<Quat> {
89        match self {
90            NetValue::Quat(q) => Some(Quat::from_array(*q)),
91            _ => None,
92        }
93    }
94    /// The string when the value holds one.
95    pub fn as_str(&self) -> Option<&str> {
96        match self {
97            NetValue::Str(s) => Some(s),
98            _ => None,
99        }
100    }
101    /// The raw bytes when the value holds them.
102    pub fn as_bytes(&self) -> Option<&[u8]> {
103        match self {
104            NetValue::Bytes(b) => Some(b),
105            _ => None,
106        }
107    }
108    /// Short type name for inspectors.
109    pub fn kind_label(&self) -> &'static str {
110        match self {
111            NetValue::Bool(_) => "Bool",
112            NetValue::Int(_) => "Int",
113            NetValue::Float(_) => "Float",
114            NetValue::Vec3(_) => "Vec3",
115            NetValue::Quat(_) => "Quat",
116            NetValue::Str(_) => "String",
117            NetValue::Bytes(_) => "Bytes",
118        }
119    }
120}

415impl NetView {
416    /// Current owner of an object, if any peer holds authority.
417    pub fn owner(&self, id: ObjectId) -> Option<u64> {
418        self.owners.get(&id).copied()
419    }
420    /// Whether the local peer holds ownership of an object.
421    pub fn owns(&self, id: ObjectId) -> bool {
422        self.connected && self.owner(id) == Some(self.local_peer)
423    }
424    /// Whether the local peer simulates an object: its owner, the host for
425    /// unowned objects, and always true offline.
426    pub fn has_authority(&self, id: ObjectId) -> bool {
427        if !self.connected {
428            return true;
429        }
430        match self.owner(id) {
431            Some(o) => o == self.local_peer,
432            None => self.is_server,
433        }
434    }
435    /// The peer with this id, if present in the session.
436    pub fn peer(&self, id: u64) -> Option<&PeerInfo> {
437        self.peers.iter().find(|p| p.id == id)
438    }
439}

443/// How a [`NetworkObject`]'s ownership may change hands.
444#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
445pub enum OwnershipMode {
446    /// Any peer can take it at any time (grab-to-own; last claim wins).
447    Takeover,
448    /// Granted only when unowned; otherwise the owner gets an
449    /// `OwnershipRequested` event and decides.
450    Request,
451    /// Only the host assigns ownership.
452    Fixed,
453    /// Like `Takeover`, for props meant to be picked up: on a dedicated
454    /// server (or with strict validation) it is granted only while the
455    /// object is free (or its owner is gone), the requester's avatar is
456    /// within reach and the per-object cooldown has passed. `Takeover` keeps
457    /// the any-peer-any-time rule on a listen host.
458    Grabbable,
459}
460
461impl OwnershipMode {
462    /// Every mode, in declaration order.
463    pub const ALL: [OwnershipMode; 4] = [Self::Takeover, Self::Request, Self::Fixed, Self::Grabbable];
464    /// Inspector label of this mode.
465    pub fn label(self) -> &'static str {
466        match self {
467            Self::Takeover => "Takeover",
468            Self::Request => "Request",
469            Self::Fixed => "Host only",
470            Self::Grabbable => "Grabbable",
471        }
472    }
473    /// Stable wire encoding of the mode.
474    pub fn to_u8(self) -> u8 {
475        self as u8
476    }
477}

618/// Typed key/value state replicated with owner-write semantics: the object's

632impl NetworkVariables {
633    /// The variable's current value, if present.
634    pub fn get(&self, key: &str) -> Option<&NetValue> {
635        self.vars.iter().find(|v| v.name == key).map(|v| &v.value)
636    }
637    /// Set (or create) a variable. On the authority (or with `anyone_can_write`)
638    /// the change replicates to every peer.
639    pub fn set(&mut self, key: &str, value: NetValue) {
640        match self.vars.iter_mut().find(|v| v.name == key) {
641            Some(v) => v.value = value,
642            None => self.vars.push(NetVar { name: key.to_owned(), value }),
643        }
644    }
645}

758impl WornAvatar {
759    /// Wire encoding: the id then the url, newline-separated.
760    pub fn encode(&self) -> String {
761        format!("{}\n{}", self.id, self.url)
762    }
763
764    /// Parse the wire encoding; an empty id decodes to None.
765    pub fn decode(s: &str) -> Option<Self> {
766        let (id, url) = s.split_once('\n').unwrap_or((s, ""));
767        (!id.is_empty()).then(|| Self { id: id.to_owned(), url: url.to_owned() })
768    }

770    /// What a peer passes to the content loader: the URL, else the id (works
771    /// when the package is already in that peer's cache).
772    pub fn source(&self) -> &str {
773        if self.url.is_empty() { &self.id } else { &self.url }
774    }

804impl NetworkAvatar {
805    /// Part references in wire order.
806    pub fn parts(&self) -> [ObjectRef; 6] {
807        [self.head, self.left_hand, self.right_hand, self.hips, self.left_foot, self.right_foot]
808    }
809
810    /// The scene's VR Player rig (cached by id, rescanned when it goes away).
811    fn local_player(&mut self, ctx: &crate::ecs::ObjectContext<'_, '_>) -> Option<ObjectId> {
812        use crate::MotionContext;
813        if let Some(id) = self.player.filter(|id| ctx.entity_of(*id).is_some()) { return Some(id); }
814        let id = ctx.objects_with::<crate::VrCharacter>().into_iter().find(|id| *id != ctx.self_id())?;
815        self.player = Some(id);
816        Some(id)
817    }
818
819    /// Owner side of `follow_local_player`: sit on the rig root and copy the
820    /// rig's head / hand poses (relative to its root) onto our parts.
821    fn follow_player(&mut self, ctx: &mut crate::ecs::ObjectContext<'_, '_>) {
822        use crate::MotionContext;
823        if !self.hidden_applied {
824            self.hidden_applied = true;
825            for &r in &self.owner_hidden {
826                ctx.set_active(r, false);
827            }
828        }
829        let Some(pi) = self.local_player(ctx) else {
830            return;
831        };
832        let Some(root) = ctx.matrix_of(pi) else {
833            return;
834        };
835        let rig = ctx.get::<crate::VrCharacter>(pi).map(|p| [p.head, p.left_hand, p.right_hand]);
836        let Some(rig) = rig else {
837            return;
838        };
839        let (scale, rot, pos) = root.to_scale_rotation_translation();
840        ctx.set_world_position(pos);
841        ctx.set_world_rotation(rot);
842        ctx.edit::<crate::Transform>(ctx.self_id(), |transform| transform.scale = scale);
843        let inv = root.inverse();
844        for (part, src) in [self.head, self.left_hand, self.right_hand].into_iter().zip(rig) {
845            let Some(world) = ctx.matrix_of(src) else {
846                continue;
847            };
848            if part.is_set() {
849                let (_, r, t) = (inv * world).to_scale_rotation_translation();
850                ctx.set_local_transform(part, Some(t), Some(r), None);
851            }
852        }
853    }
854}

860impl NetworkAvatar {
861    fn update_native(&mut self, ctx: &mut crate::ecs::ObjectContext<'_, '_>) {
862        use crate::MotionContext;
863        if !ctx.has_authority(ctx.self_id()) {
864            return;
865        }
866        if self.follow_local_player {
867            self.follow_player(ctx);
868            return;
869        }
870        if !self.drive_from_vr || !ctx.vr_frame().active {
871            return;
872        }
873        // VR poses are stage-space, i.e. local to the rig root this sits on.
874        let vr = ctx.vr_frame();
875        for (part, pose) in [(self.head, vr.head), (self.left_hand, vr.left_hand), (self.right_hand, vr.right_hand)] {
876            if let (true, Some((p, r))) = (part.is_set(), pose) {
877                ctx.set_local_transform(part, Some(p), Some(r), None);
878            }
879        }
880    }

1023impl crate::ecs::ObjectContext<'_, '_> {
1024    /// Whether this peer is in a network session, as host or as a joined client.
1025    ///
1026    /// **Immediate.** A read of the network view the engine built at the start of the frame. It is `false` offline
1027    /// (no session), while a join is still handshaking, and after the session has ended. It turns `true` on the frame a
1028    /// [`NetEvent::Connected`] is delivered. Identical to [`connected`](Self::connected).
1029    pub fn net_connected(&self) -> bool {
1030        self.net_view().connected
1031    }
1032    /// The peer id of the session's current host: [`HOST_PEER`] (1) for the first host, another id after a host migration.
1033    ///
1034    /// **Immediate.** Returns `0` offline (no session). To ask whether you are that host use
1035    /// [`i_am_host`](Self::i_am_host); this peer's own id is [`local_peer`](Self::local_peer).
1036    pub fn net_host_peer(&self) -> u64 {
1037        if self.net_view().connected { self.net_view().host_peer } else { 0 }
1038    }
1039    /// Whether this peer hosts the session, including on the frame it takes over after a host migration.
1040    ///
1041    /// **Immediate.** The same as [`is_server`](Self::is_server). It is `false` offline (no session), so do not write
1042    /// `!i_am_host()` to mean "client or single-player". When the question is "do I simulate this object?" ask
1043    /// [`has_authority_of`](Self::has_authority_of), which is `true` offline.
1044    pub fn i_am_host(&self) -> bool {
1045        self.is_server()
1046    }
1047    /// The new host's peer id on the frame a host migration completes, otherwise `None`.
1048    ///
1049    /// **Immediate.** Reads the [`NetEvent::HostChanged`] events delivered this frame (the latest one if there are
1050    /// several). The id can be your own, in which case [`i_am_host`](Self::i_am_host) is now `true`. The event is
1051    /// delivered once, so react in the frame you see it. Always `None` offline.
1052    pub fn host_changed(&self) -> Option<u64> {
1053        self.net_view().events.iter().rev().find_map(|e| match e {
1054            NetEvent::HostChanged { host, .. } => Some(*host),
1055            _ => None,
1056        })
1057    }
1058    /// Whether the host was lost and a successor is being found.
1059    ///
1060    /// **Immediate.** While it is `true`, objects that have no owner (the host is their authority) pause until a
1061    /// [`NetEvent::HostChanged`] arrives; see [`host_changed`](Self::host_changed). It is `false` offline and whenever no
1062    /// migration is under way. Migration only happens when the Network Manager's `host_migration` is on; with it off the
1063    /// session ends with its host instead ([`NetEvent::Disconnected`]).
1064    pub fn net_migrating(&self) -> bool {
1065        self.net_view().connected && self.net_view().migrating
1066    }
1067    /// A peer's verified player id: its Ed25519 public key, checked by the host during the handshake.
1068    ///
1069    /// **Immediate.** Reads [`PeerInfo::identity`]. `None` offline, for an id that is not in the session, and for a peer
1070    /// whose key this peer does not know. The peer id is assigned per session, but the key identifies the same install
1071    /// across sessions, so key persistent player data on it (or on its fingerprint).
1072    pub fn peer_identity(&self, id: u64) -> Option<[u8; 32]> {
1073        self.net_view().peer(id).and_then(|p| p.identity)
1074    }
1075    /// A peer's player-id fingerprint (8 groups of 4 hex digits): the form the Network Manager's allow and ban lists hold.
1076    ///
1077    /// **Immediate.** Reads [`PeerInfo::fingerprint`]. `None` offline, for an unknown peer id, and for a peer whose
1078    /// identity is not known (empty fingerprint).
1079    pub fn peer_fingerprint(&self, id: u64) -> Option<&str> {
1080        self.net_view().peer(id).map(|p| p.fingerprint.as_str()).filter(|f| !f.is_empty())
1081    }
1082    /// Removes a player from the session (host only); with `ban` its player id is also refused from now on.
1083    ///
1084    /// **Queued (host).** Pushes [`ComponentCommand::NetKick`]. The play session does not handle it, so the host
1085    /// application applies it after the frame's last stage. The kicked player is told `reason`.
1086    ///
1087    /// Only the host can kick: on a client, offline, or aimed at the host's own peer id the command does nothing (no
1088    /// event, no error). `ban` adds the peer's fingerprint (see [`peer_fingerprint`](Self::peer_fingerprint)) to the
1089    /// session's ban list before kicking; a peer whose identity is unknown is only kicked. The ban lasts for this
1090    /// session, and across restarts only on a server that keeps a ban file.
1091    pub fn net_kick(&mut self, peer: u64, reason: impl Into<String>, ban: bool) {
1092        self.commands.push(ComponentCommand::NetKick { peer, reason: reason.into(), ban });
1093    }
1094    /// A peer's player name.
1095    ///
1096    /// **Immediate.** Reads [`PeerInfo::name`]. `None` offline or for an id that is not in the session. The local peer is
1097    /// in the session list too, so while connected `peer_name(local_peer())` is your own name.
1098    pub fn peer_name(&self, id: u64) -> Option<&str> {
1099        self.net_view().peer(id).map(|p| p.name.as_str())
1100    }
1101    /// The session and replication events delivered this frame: connects and disconnects, players joining and leaving,
1102    /// ownership changes, spawns and despawns, synced-variable changes and host migration ([`NetEvent`]).
1103    ///
1104    /// **Immediate.** Each event is delivered once. Every system sees the same slice during the frame and it is gone the
1105    /// next, so look for what you need in the frame it arrives. Things you caused yourself (your own networked spawn or
1106    /// variable write) show up too, one frame after you queued them. Empty offline.
1107    ///
1108    /// The slice borrows the host. To call a `&mut self` method for an event (for example
1109    /// [`transfer_ownership`](Self::transfer_ownership)), copy what you need out of the slice first.
1110    pub fn net_events(&self) -> &[NetEvent] {
1111        &self.net_view().events
1112    }
1113    /// Round-trip time to the host, in milliseconds.
1114    ///
1115    /// **Immediate.** `0.0` on the host itself and offline. Round trips to other peers are in [`PeerInfo::rtt_ms`] (see
1116    /// [`peers`](Self::peers)); the host measures every peer, a client only its own link to the host.
1117    pub fn net_rtt_ms(&self) -> f32 {
1118        self.net_view().rtt_ms
1119    }
1120    /// Round-trip time to the host, in seconds: [`net_rtt_ms`](Self::net_rtt_ms) divided by 1000.
1121    ///
1122    /// **Immediate.** `0.0` on the host itself and offline.
1123    pub fn net_rtt(&self) -> f32 {
1124        self.net_view().rtt_ms / 1000.0
1125    }
1126    /// Quality of the link to the host: round trip, jitter, packet loss in each direction, and whether traffic goes
1127    /// through a relay ([`NetQuality`]).
1128    ///
1129    /// **Immediate.** All zero (and `relayed: false`) on the host itself and offline.
1130    pub fn net_quality(&self) -> NetQuality {
1131        self.net_view().quality
1132    }
1133
1134    /// Whether this peer is the one that simulates `target`, and so may move it, change its state and despawn it.
1135    ///
1136    /// **Immediate.** A read of the network view. An object's authority is its owner peer; an object with no owner
1137    /// belongs to the host. So:
1138    ///
1139    /// * offline (no session): `true` for every set reference;
1140    /// * on the host: `true` for objects it owns and for unowned ones;
1141    /// * on a client: `true` only for objects that client owns. Ordinary scene objects that are not networked have no
1142    ///   owner, so on a client this is `false` for them too.
1143    ///
1144    /// An unset [`ObjectRef`] gives `false` whether or not there is a session, and the object is not checked for
1145    /// existence. [`has_authority`](Self::has_authority) is the same test for an [`ObjectId`]. Ownership moves with
1146    /// [`request_ownership_of`](Self::request_ownership_of), [`release_ownership_of`](Self::release_ownership_of) and
1147    /// [`transfer_ownership`](Self::transfer_ownership); the answer changes on the frame the
1148    /// [`NetEvent::OwnershipChanged`] arrives, not when you ask.
1149    ///
1150    /// # Examples
1151    ///
1152    /// ```ignore
1153    /// // Only the peer that simulates the crate moves it; everyone else just receives its replicated transform.
1154    /// if ctx.has_authority_of(self.crate_ref) {
1155    ///     ctx.set_world_position_of(self.crate_ref, next_position);
1156    /// }
1157    /// ```
1158    pub fn has_authority_of(&self, target: impl Into<ObjectRef>) -> bool {
1159        target.into().id().is_some_and(|id| self.net_view().has_authority(id))
1160    }
1161    /// The peer that owns `target`, or `None` when it has no owner.
1162    ///
1163    /// **Immediate.** `None` means unowned (the host is then its authority, see
1164    /// [`has_authority_of`](Self::has_authority_of)). It is also what you get offline, for an unset reference and for
1165    /// objects that are not networked. The id `0` never appears: unowned objects are simply absent.
1166    pub fn owner_of(&self, target: impl Into<ObjectRef>) -> Option<u64> {
1167        self.net_view().owner(target.into().id()?)
1168    }
1169    /// Gives up ownership of `target`: it becomes unowned and the host becomes its authority.
1170    ///
1171    /// **Queued (host).** Pushes [`ComponentCommand::ReleaseOwnership`] (nothing is queued for an unset reference). The
1172    /// play session does not handle it, so the host application applies it after the frame's last stage.
1173    ///
1174    /// Only the current owner can release. From any other peer, for an unowned object, or offline, it does nothing (no
1175    /// event, no error). On success every peer receives a [`NetEvent::OwnershipChanged`] with `owner: 0`. To pass an
1176    /// object to a particular peer use [`transfer_ownership`](Self::transfer_ownership); to take one use
1177    /// [`request_ownership_of`](Self::request_ownership_of).
1178    pub fn release_ownership_of(&mut self, target: impl Into<ObjectRef>) {
1179        if let Some(id) = target.into().id() {
1180            self.commands.push(ComponentCommand::ReleaseOwnership(id));
1181        }
1182    }
1183    /// Hands `target` to another peer, or to nobody with `peer` = `0`. Only the object's owner or the host may do this.
1184    ///
1185    /// **Queued (host).** Pushes [`ComponentCommand::TransferOwnership`] (nothing is queued for an unset reference). The
1186    /// play session does not handle it, so the host application applies it after the frame's last stage.
1187    ///
1188    /// Nothing is checked at the call. From a peer that is neither the owner nor the host the request is ignored, and
1189    /// offline it does nothing. Pass the id of a peer that is in the session (see [`peers`](Self::peers)); a client
1190    /// naming a peer that is not there is refused by the host and counted against it as a violation. The object's
1191    /// [`OwnershipMode`] does not restrict a transfer. On success every peer, the old and the new owner included,
1192    /// receives a [`NetEvent::OwnershipChanged`], and [`has_authority_of`](Self::has_authority_of) flips when that
1193    /// arrives.
1194    ///

1216    /// Removes a networked object from the world on every peer. Only the object's authority may do it.
1217    ///
1218    /// **Queued, then queued (host).** The authority check happens at the call: from a peer that does not have
1219    /// [`has_authority_of`](Self::has_authority_of) it silently does nothing and queues nothing. When it passes,
1220    /// [`ComponentCommand::NetDespawn`] removes the object locally as the current stage ends (later stages no longer see
1221    /// it), and the host application replicates the despawn after the frame's last stage. The other peers receive a
1222    /// [`NetEvent::Despawned`].
1223    ///
1224    /// Offline every object passes the check, so this removes any object, like [`destroy`](Self::destroy). Online, an
1225    /// object that has no owner counts as the host's: a client cannot despawn it, and not even the host can despawn an
1226    /// object a client owns (transfer it first with [`transfer_ownership`](Self::transfer_ownership)). Prefer this over
1227    /// `destroy` for objects made with [`net_spawn`](Self::net_spawn). A plain `destroy` of an owned runtime spawn is also
1228    /// noticed and despawned everywhere, but about a quarter of a second later.
1229    ///
1230    /// # Examples
1231    ///
1232    /// ```ignore
1233    /// // A bullet this peer owns removes itself on every peer when it hits something.
1234    /// if self.hit {
1235    ///     ctx.net_despawn(ctx.self_id());
1236    /// }
1237    /// ```
1238    pub fn net_despawn(&mut self, target: impl Into<ObjectRef>) {
1239        if let Some(id) = target.into().id()
1240            && self.net_view().has_authority(id)
1241        {
1242            self.commands.push(ComponentCommand::NetDespawn(id));
1243        }
1244    }

1246    /// The current value of a synced variable on `target`'s [`NetworkVariables`] component, or `None`.
1247    ///
1248    /// **Immediate.** A plain read of the local copy of the component, so it works offline too. `None` when the
1249    /// reference is unset, the object is gone or hidden, it has no enabled [`NetworkVariables`], or the key was never
1250    /// set. The object's authority writes variables with [`set_net_var`](Self::set_net_var), which updates the local copy
1251    /// at once. A value written by another peer reaches the local copy when the frame's network sync runs, so in the
1252    /// very frame its [`NetEvent::VarChanged`] is delivered (see [`net_var_changes`](Self::net_var_changes)) this can
1253    /// still return the previous value. React to the value the event carries.
1254    ///
1255    /// # Panics
1256    ///
1257    /// The result is a checked borrow of the component. It panics if that object's [`NetworkVariables`] is already
1258    /// mutably borrowed, for example when called on the object a system is running on from inside a system that iterates
1259    /// `NetworkVariables` (see the borrow rules on [`ObjectContext`](crate::ecs::ObjectContext)). While the returned guard is
1260    /// alive, [`edit`](Self::edit) of that component panics as well. Copy the value out (`as_int`, `clone`, ...) and drop
1261    /// the guard.
1262    ///
1263    /// # Examples
1264    ///
1265    /// ```ignore
1266    /// // The health everyone sees on `target`; its authority writes it with `set_net_var`.
1267    /// let hp = ctx.net_var(target, "hp").and_then(|v| v.as_int()).unwrap_or(100);
1268    /// ```
1269    pub fn net_var(&self, target: impl Into<ObjectRef>, key: &str) -> Option<crate::ecs::Ref<'_, NetValue>> {
1270        let vars = self.get::<NetworkVariables>(target)?;
1271        vars.get(key)?;
1272        Some(crate::ecs::Ref::map(vars, |vars| vars.get(key).expect("borrowed variable")))
1273    }

1275    ///
1276    /// **Immediate.** Same results and the same borrow rules as [`net_var`](Self::net_var): `None` when this object has
1277    /// no enabled [`NetworkVariables`] or the key is unset, and a panic when called from a system that is iterating
1278    /// `NetworkVariables`. Write with [`set_net_var_self`](Self::set_net_var_self).
1279    pub fn net_var_self(&self, key: &str) -> Option<crate::ecs::Ref<'_, NetValue>> { self.net_var(self.self_id(), key) }
1280    /// The synced-variable changes delivered this frame, as `(object, key, new value, writer peer)`.
1281    ///
1282    /// **Immediate.** Walks this frame's [`NetEvent::VarChanged`] events for every object. Each change is delivered once,
1283    /// to every peer including the one that wrote it (one frame after the write, with `writer` = your own peer id; filter
1284    /// with [`local_peer`](Self::local_peer) to see only other peers' writes). Use it to react to a change. The value in
1285    /// the event is already the new one even when [`net_var`](Self::net_var) has not caught up yet. Offline there are no
1286    /// events: a local [`set_net_var`](Self::set_net_var) still changes the data but reports nothing.
1287    ///
1288    /// The iterator borrows the host. Collect what you need before calling a `&mut self` method such as

1291        self.net_view().events.iter().filter_map(|e| match e {
1292            NetEvent::VarChanged { object, key, value, from } => Some((*object, key.as_str(), value, *from)),
1293            _ => None,
1294        })

1297    /// Sends a reliable RPC (a named call with arguments) that is not tied to an object.
1298    ///
1299    /// **Queued (host).** Online this pushes [`ComponentCommand::NetRpc`], exactly [`send_rpc`](Self::send_rpc) with
1300    /// `object: None` and `reliable: true`. The play session does not handle it, so the host application sends it after
1301    /// the frame's last stage. Reliable means it is retransmitted until acknowledged, delivered in order with the other
1302    /// reliable messages, and never delivered twice: use it for things that must not be missed (round start, damage, a
1303    /// purchase). For frequent cosmetic events use [`rpc_unreliable`](Self::rpc_unreliable).
1304    ///
1305    /// `target` ([`RpcTarget`]) decides who receives it: `All` every peer including the sender, `Others` everyone but
1306    /// the sender, `Server` the host only, `Peer(id)` that one peer, `Owner` the owner of the RPC's object (with no
1307    /// object, as here, that is the host). Everything goes through the host, so a client-to-client RPC is relayed by it.
1308    /// Receivers read it with [`rpcs`](Self::rpcs) or [`rpcs_named`](Self::rpcs_named). Offline behaviour, delivery
1309    /// timing and the host's limits on client RPCs are described on [`send_rpc`](Self::send_rpc).
1310    ///
1311    /// # Examples
1312    ///
1313    /// ```ignore
1314    /// // Any peer: ask the host to open a door. `Server` reaches only the host (and loops back when offline).
1315    /// ctx.rpc("OpenDoor", vec![NetValue::from(door_number)], RpcTarget::Server);
1316    ///
1317    /// // Host (or offline), on a later frame: collect the requests, then act (the borrow of `rpcs_named` has to end
1318    /// // before a `&mut` call such as `ctx.rpc`).
1319    /// let asked: Vec<(u64, i64)> = ctx.rpcs_named("OpenDoor")
1320    ///     .filter_map(|rpc| Some((rpc.from, rpc.arg(0)?.as_int()?)))
1321    ///     .collect();
1322    /// for (_from, door) in asked {
1323    ///     self.open_door(door);
1324    ///     ctx.rpc("DoorOpened", vec![NetValue::from(door)], RpcTarget::All);
1325    /// }
1326    /// ```
1327    pub fn rpc(&mut self, name: &str, args: Vec<NetValue>, target: RpcTarget) {
1328        self.send_rpc(None, name, args, target, true);
1329    }
1330    /// Sends an unreliable RPC that is not tied to an object: one datagram that may be lost or arrive out of order and is
1331    /// never retransmitted.
1332    ///
1333    /// **Queued (host).** The same as [`rpc`](Self::rpc), but with `reliable: false` ([`send_rpc`](Self::send_rpc)).
1334    /// Good for frequent cosmetic events where the next one makes a lost one irrelevant (footsteps, muzzle flashes);
1335    /// keep the payload small. Never use it for something that must arrive. Targets, offline loopback, timing and the
1336    /// host's limits are the same as for [`send_rpc`](Self::send_rpc). Offline the reliability flag makes no difference.
1337    pub fn rpc_unreliable(&mut self, name: &str, args: Vec<NetValue>, target: RpcTarget) {
1338        self.send_rpc(None, name, args, target, false);
1339    }
1340    /// Sends an RPC with an explicit object, target and reliability: the general form behind [`rpc`](Self::rpc),
1341    /// [`rpc_unreliable`](Self::rpc_unreliable) and [`rpc_self`](Self::rpc_self).
1342    ///
1343    /// `object` ties the RPC to an object (`Some(id)`) or leaves it global (`None`). Receivers see it as
1344    /// [`NetRpc::object`], and [`rpcs_for_self`](Self::rpcs_for_self) lists the ones sent on the object a system runs on.
1345    /// `RpcTarget::Owner` goes to that object's owner (the host when it is unowned or `object` is `None`). `name` is the
1346    /// string receivers match with [`rpcs_named`](Self::rpcs_named); `args` are [`NetValue`]s, read back in order with
1347    /// [`NetRpc::arg`]. `reliable` picks guaranteed in-order delivery or a single lossy datagram.
1348    ///
1349    /// **Queued (host).** Online this pushes [`ComponentCommand::NetRpc`]. The play session does not handle it, so the
1350    /// host application sends it after the frame's last stage. Receivers get it on the frame its packet arrives. When
1351    /// `target` includes the sending peer (`All`, `Server` on the host, `Owner` when you are the object's owner, or the
1352    /// host for an unowned or absent object, `Peer` with your own id) a copy with `from` = your own peer id is delivered to
1353    /// you on the frame after you queued it.
1354    ///
1355    /// **Offline** (no session, which includes a join that is still handshaking) nothing is sent. When `target` includes
1356    /// the local player (`All`, `Server`, `Owner`, or `Peer` with the id `0`) a copy with `from` = `0` is queued as a
1357    /// local event and is readable through [`rpcs`](Self::rpcs) on the next frame; `Others` is dropped. This keeps
1358    /// single-player code paths working.
1359    ///
1360    /// **Limits.** The host checks every RPC a client sends before relaying or delivering it. An RPC is dropped, and
1361    /// counted against the sender as a violation, when it exceeds the rate limit or the size limits (by default 16
1362    /// arguments, 1024 bytes, a 256-byte name) or when its name has a permission rule ([`RpcRule`](crate::RpcRule)) the
1363    /// sender does not satisfy. With the server's `strict_rpc_permissions` on, a dedicated server treats unregistered
1364    /// names as owner-only, which a global RPC can never satisfy because it has no owner. A dropped RPC produces no
1365    /// error on the sender.
1366    ///
1367    /// # Examples
1368    ///
1369    /// ```ignore
1370    /// // Tell the owner of a door (the host if nobody owns it) to open it, reliably, with the opener's id.
1371    /// ctx.send_rpc(Some(door), "Open", vec![NetValue::from(ctx.local_peer() as i64)], RpcTarget::Owner, true);
1372    /// ```
1373    pub fn send_rpc(&mut self, object: Option<ObjectId>, name: &str, args: Vec<NetValue>, target: RpcTarget, reliable: bool) {
1374        if !self.net_view().connected {
1375            // Offline: targets that include the local player loop back as a local event.
1376            let local = match target {
1377                RpcTarget::All | RpcTarget::Server | RpcTarget::Owner => true,
1378                RpcTarget::Others => false,
1379                RpcTarget::Peer(p) => p == 0 || p == self.net_view().local_peer,
1380            };
1381            if local {
1382                let rpc = NetRpc { from: 0, object, name: name.to_owned(), args };
1383                self.commands.push(ComponentCommand::Emit(crate::AnyBox(Box::new(rpc))));
1384            }
1385            return;
1386        }
1387        self.commands.push(ComponentCommand::NetRpc { object, name: name.to_owned(), args, target, reliable });
1388    }
1389    /// Every RPC delivered to this peer this frame: the ones other peers sent, plus the loopback copies of ones you sent
1390    /// to yourself.
1391    ///
1392    /// **Immediate.** Each RPC is delivered exactly once. It is in this list for one frame, for every system, and then
1393    /// gone; nothing is kept for later, so handle it in the frame you see it. An RPC from another peer appears on the
1394    /// frame its packet is received. One you sent to yourself ([`RpcTarget::All`] and friends) appears on the frame after
1395    /// you queued it, offline as well (see [`send_rpc`](Self::send_rpc)).
1396    ///
1397    /// The list is not filtered by object: it holds global RPCs and RPCs sent on any object. Use
1398    /// [`rpcs_named`](Self::rpcs_named) to match a name or [`rpcs_for_self`](Self::rpcs_for_self) for the ones sent on
1399    /// this object. [`NetRpc::from`] is the sending peer (`0` for an offline loopback). On the host it is the connection
1400    /// the RPC arrived on, so a client cannot forge it and it is safe to base authority decisions on.
1401    ///
1402    /// The iterator borrows the host. To call a `&mut self` method such as [`rpc`](Self::rpc) or
1403    /// [`set_net_var`](Self::set_net_var) for each RPC, collect what you need first.
1404    pub fn rpcs(&self) -> impl Iterator<Item = &NetRpc> {
1405        self.net_view().rpcs.iter().chain(self.events::<NetRpc>())
1406    }

1411    /// name is used on several objects. Collect before calling a `&mut self` method inside the loop (see
