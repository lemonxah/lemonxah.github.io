1//! Marks scene content a dedicated server keeps across restarts (NET-6).
2//!
3//! The server saves the flagged objects of the running scene under its state
4//! folder (`<project>/.server/`) on an interval and at shutdown, and puts the
5//! values back when that scene loads again. Only scene objects have a stable id;
6//! objects spawned at runtime are not saved.
7
8use serde::{Deserialize, Serialize};
9
10use crate::TypedComponent;
11
12/// What of this object a dedicated server persists.
13#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
14#[serde(default)]
15pub struct PersistentState {
16    /// World: save and restore the object's position, rotation and scale (a pushed crate, an opened door).
17    pub transform: bool,
18    /// World: save and restore the values of the object's Network Variables (a score, a lever state).
19    pub variables: bool,
20    /// Per player: on a player object, save its Network Variables (only them) under the owner's
21    /// player id and give them back when that player joins again (inventory, progress, currency).
22    pub per_player: bool,
23}
24
25impl Default for PersistentState {
26    fn default() -> Self {
27        Self { transform: true, variables: true, per_player: false }
28    }
29}
30
31impl TypedComponent for PersistentState {
32    const NAME: &'static str = "Persistent State";
33}
