1//! Tile assets (2D-5): `.tile` ([`TileAsset`], optionally animated), `.ruletile`
2//! ([`RuleTile`]) and `.palette` ([`TilePalette`]), plus the rule precompiler.
3//! RON files, every field defaulted so old files keep loading.
4//!
5//! # Rule tiles
6//!
7//! A rule lists, per neighbour slot of the ring (8 around a rectangle /
8//! isometric cell, 6 around a hexagon, see [`crate::neighbor_offset`]), whether
9//! that neighbour must be this tile ([`NeighborReq::This`]), must not
10//! ([`NeighborReq::NotThis`]) or does not matter. The first rule (in list
11//! order) that matches wins; a rule can also match rotated or mirrored
12//! ([`RuleTransform`]), which draws its sprite with that transform.
13//!
14//! [`RuleTile::compile`] turns the rule list into a lookup table once: the
15//! "same tile" state of the ring is a bitmask (2^8 or 2^6 values), and for each
16//! mask the table holds the winning rule and transform. At runtime a cell
17//! costs one mask build (8 neighbour reads) and one table read, independent of
18//! the number of rules. Editing a cell only changes the masks of its 8
19//! neighbours.
20//!
21//! "Same tile" compares the neighbour's tile asset path with the cell's. The
22//! random output picks a sprite by value noise of the cell position (stable
23//! across refreshes, not Perlin); animated outputs use up to four speed buckets
24//! between the min and max speed so equal animations still batch.
25

302/// How ring slots move under rotation and mirroring.
303#[derive(Clone, Copy, Debug)]
304pub struct RingSpec {
305    /// Slots in the ring.
306    pub n: usize,
307    /// Distinct rotations (4 for rectangles, 6 for hexagons).
308    pub rotations: usize,
309    /// Ring slots one rotation step moves a neighbour.
310    pub slots_per_rotation: usize,
311    /// [`TileXform::rot`] units of one rotation step.
312    pub rot15: u8,
313    /// `slot -> (mirror_x - slot) mod n` is the x mirror.
314    pub mirror_x: usize,
315}
316
317/// The ring geometry of a layout (slot count, rotations, mirror axis).
318pub fn ring_spec(layout: CellLayout) -> RingSpec {
319    match layout {
320        CellLayout::HexPointy => RingSpec { n: 6, rotations: 6, slots_per_rotation: 1, rot15: ROT_STEPS / 6, mirror_x: 3 },
321        CellLayout::HexFlat => RingSpec { n: 6, rotations: 6, slots_per_rotation: 1, rot15: ROT_STEPS / 6, mirror_x: 2 },
322        _ => RingSpec { n: 8, rotations: 4, slots_per_rotation: 2, rot15: ROT_STEPS / 4, mirror_x: 4 },
323    }
324}
325
326impl RingSpec {
327    /// Where ring slot `i` lands after mirroring (x -> -x) and then rotating
328    /// counter clockwise by `k` steps.
329    pub fn permute(&self, k: usize, mirror: bool, i: usize) -> usize {
330        let j = if mirror { (self.mirror_x + self.n - i % self.n) % self.n } else { i % self.n };
331        (j + k * self.slots_per_rotation) % self.n
332    }
333
334    /// The `(rotation steps, mirror)` orientations a rule transform tries, in
335    /// priority order (the rule as written first).
336    pub fn variants(&self, t: RuleTransform) -> Vec<(usize, bool)> {
337        let half = self.rotations / 2;
338        match t {
339            RuleTransform::Fixed => vec![(0, false)],
340            RuleTransform::Rotated => (0..self.rotations).map(|k| (k, false)).collect(),
341            RuleTransform::MirrorX => vec![(0, false), (0, true)],
342            // Mirror Y = mirror X then a half turn.
343            RuleTransform::MirrorY => vec![(0, false), (half, true)],
344            RuleTransform::MirrorXY => vec![(0, false), (0, true), (half, true), (half, false)],
345            RuleTransform::RotatedMirror => (0..self.rotations).flat_map(|k| [(k, false), (k, true)]).collect(),
346        }
347    }
348
349    /// The tile transform that draws a sprite designed for the rule as written
350    /// when the variant `(k, mirror)` matched.
351    pub fn xform(&self, k: usize, mirror: bool) -> TileXform {
352        TileXform::new((k as u8) * self.rot15, mirror)
353    }
354}

374/// A rule tile compiled to a lookup table indexed by the neighbour mask.
375#[derive(Clone, Debug, PartialEq)]
376pub struct CompiledRules {
377    /// The layout the table was compiled for.
378    pub layout: CellLayout,
379    /// Ring slots (mask bits).
380    pub n: usize,
381    table: Vec<RuleHit>,
382}
383
384impl CompiledRules {
385    /// The winner for a mask (bit `i` set = ring slot `i` holds the same tile).
386    #[inline]
387    pub fn lookup(&self, mask: u32) -> RuleHit {
388        self.table[(mask as usize) & (self.table.len() - 1)]
389    }
390    /// Number of entries in the lookup table.
391    pub fn table_len(&self) -> usize {
392        self.table.len()
393    }
394}

396impl RuleTile {
397    /// Compile the rules into the bitmask table (2^8 or 2^6 entries).
398    pub fn compile(&self) -> CompiledRules {
399        let spec = ring_spec(self.layout);
400        // Per rule and variant: the slots that must be set / clear.
401        let mut variants: Vec<(u16, u32, u32, TileXform)> = Vec::new();
402        for (ri, r) in self.rules.iter().enumerate() {
403            for (k, mirror) in spec.variants(r.transform) {
404                let (mut set, mut clear) = (0u32, 0u32);
405                for i in 0..spec.n {
406                    let to = spec.permute(k, mirror, i);
407                    match r.req(i) {
408                        NeighborReq::This => set |= 1 << to,
409                        NeighborReq::NotThis => clear |= 1 << to,
410                        NeighborReq::Any => {}
411                    }
412                }
413                variants.push((ri as u16, set, clear, spec.xform(k, mirror)));
414            }
415        }
416        let none = RuleHit { rule: RuleHit::NONE, xform: TileXform::IDENTITY };
417        let table = (0..1u32 << spec.n)
418            .map(|mask| {
419                variants
420                    .iter()
421                    .find(|(_, set, clear, _)| mask & set == *set && mask & clear == 0)
422                    .map_or(none, |&(rule, _, _, xform)| RuleHit { rule, xform })
423            })
424            .collect();
425        CompiledRules { layout: self.layout, n: spec.n, table }
426    }
427
428    /// Every sprite any output of this tile can show.
429    pub fn all_sprites(&self) -> Vec<&str> {
430        let mut v: Vec<&str> = Vec::new();
431        if !self.default_sprite.is_empty() {
432            v.push(&self.default_sprite);
433        }
434        for r in &self.rules {
435            for s in &r.sprites {
436                if !s.is_empty() && !v.contains(&s.as_str()) {
437                    v.push(s);
438                }
439            }
440        }
441        v
442    }
443}

445/// The neighbour mask of `cell`: bit `i` set when ring slot `i` satisfies
446/// `same` (it holds the same tile). Reads 8 (or 6) neighbours.
447pub fn neighbor_mask(layout: CellLayout, cell: IVec2, same: impl Fn(IVec2) -> bool) -> u32 {
448    let n = layout.neighbor_count();
449    let mut mask = 0u32;
450    for i in 0..n {
451        if same(cell + neighbor_offset(layout, cell, i)) {
452            mask |= 1 << i;
453        }
454    }
455    mask
456}

458/// The cells whose appearance can change when `cell` changes: itself and its
459/// ring (8 or 6 neighbours).
460pub fn refresh_cells(layout: CellLayout, cell: IVec2) -> Vec<IVec2> {
461    let mut v = vec![cell];
462    v.extend((0..layout.neighbor_count()).map(|i| cell + neighbor_offset(layout, cell, i)));
463    v
464}

466/// Smooth value noise in 0..1 at a lattice position (bilinear over hashed corners).
467pub fn value_noise(x: f32, y: f32) -> f32 {
468    let (x0, y0) = (x.floor(), y.floor());
469    let (fx, fy) = (x - x0, y - y0);
470    let h = |ix: i32, iy: i32| -> f32 {
471        let mut v = (ix as u32).wrapping_mul(0x9E37_79B1) ^ (iy as u32).wrapping_mul(0x85EB_CA6B) ^ 0x27D4_EB2F;
472        v ^= v >> 15;
473        v = v.wrapping_mul(0x2C1B_3C6D);
474        v ^= v >> 12;
475        (v >> 8) as f32 / (1u32 << 24) as f32
476    };
477    let (ix, iy) = (x0 as i32, y0 as i32);
478    let s = |t: f32| t * t * (3.0 - 2.0 * t);
479    let (sx, sy) = (s(fx), s(fy));
480    let top = h(ix, iy) + (h(ix + 1, iy) - h(ix, iy)) * sx;
481    let bottom = h(ix, iy + 1) + (h(ix + 1, iy + 1) - h(ix, iy + 1)) * sx;
482    top + (bottom - top) * sy
483}

485/// A hash of a cell in 0..1 (per-cell variation that is not smooth).
486pub fn cell_hash(cell: IVec2, salt: u32) -> f32 {
487    let mut v = (cell.x as u32).wrapping_mul(0x9E37_79B1) ^ (cell.y as u32).wrapping_mul(0x85EB_CA6B) ^ salt.wrapping_mul(0xC2B2_AE35);
488    v ^= v >> 16;
489    v = v.wrapping_mul(0x7FEB_352D);
490    v ^= v >> 15;
491    v = v.wrapping_mul(0x846C_A68B);
492    v ^= v >> 16;
493    (v >> 8) as f32 / (1u32 << 24) as f32
494}

529fn speed_bucket(cell: IVec2, min: f32, max: f32) -> f32 {
530    if (max - min).abs() < 1e-6 {
531        return min;
532    }
533    let b = (cell_hash(cell, 77) * 4.0).floor().min(3.0);
534    min + (max - min) * (b / 3.0)
535}

561impl RuleTile {
562    /// The output for `cell` given the table hit of its neighbour mask.
563    pub fn select(&self, hit: RuleHit, cell: IVec2) -> TileSel<'_> {
564        let rule = self.rules.get(hit.rule as usize).filter(|_| hit.matched());
565        let Some(r) = rule else {
566            return TileSel {
567                sprites: std::slice::from_ref(&self.default_sprite),
568                xform: TileXform::IDENTITY,
569                color: self.color,
570                collider: self.default_collider,
571                flags: crate::tilemap::FLAG_LOCK_TRANSFORM,
572                anim: None,
573            };
574        };
575        let sprites: &[String] = if r.sprites.is_empty() { std::slice::from_ref(&self.default_sprite) } else { &r.sprites };
576        let mut xform = hit.xform;
577        let (sprites, anim) = match r.output {
578            RuleOutput::Single => (&sprites[..1], None),
579            RuleOutput::Random => {
580                let n = value_noise((cell.x as f32 + 100_000.0) * r.random_scale, (cell.y as f32 + 100_000.0) * r.random_scale);
581                let i = ((n * sprites.len() as f32) as usize).min(sprites.len() - 1);
582                if r.random_rotation {
583                    let spec = ring_spec(self.layout);
584                    let k = (cell_hash(cell, 5) * spec.rotations as f32) as usize % spec.rotations;
585                    xform = xform.rotated_ccw(k as u8 * spec.rot15);
586                }
587                (&sprites[i..=i], None)
588            }
589            RuleOutput::Animation => (sprites, (sprites.len() > 1).then(|| AnimSel { speed: speed_bucket(cell, r.min_speed, r.max_speed), start_time: 0.0, loop_once: false })),
590        };
591        TileSel { sprites, xform, color: self.color, collider: r.collider, flags: crate::tilemap::FLAG_LOCK_TRANSFORM, anim }
592    }
593}

613        Self { layout: CellLayout::Rectangle, cell_size: [1.0; 3], cells: Vec::new() }
614    }
615}
616
617impl TilePalette {
618    /// The tile asset path at (x, y).
619    pub fn tile_at(&self, x: i32, y: i32) -> Option<&str> {
620        self.cells.iter().find(|(p, _)| *p == (x, y)).map(|(_, t)| t.as_str())
621    }
622    /// Set a cell (an empty path removes it).
623    pub fn set(&mut self, x: i32, y: i32, tile: &str) {
624        self.cells.retain(|(p, _)| *p != (x, y));
625        if !tile.is_empty() {
626            self.cells.push(((x, y), tile.to_string()));
627        }
628    }
629    /// Lay `tiles` out row by row, `columns` per row, top row first (y falls).
630    pub fn from_tiles(tiles: &[String], columns: usize) -> Self {
631        let columns = columns.max(1);
632        let rows = tiles.len().div_ceil(columns) as i32;
633        let cells = tiles.iter().enumerate().map(|(i, t)| (((i % columns) as i32, rows - 1 - (i / columns) as i32), t.clone())).collect();
634        Self { cells, ..Default::default() }
635    }
636    /// The cells' inclusive bounds (None for an empty palette).
637    pub fn bounds(&self) -> Option<(IVec2, IVec2)> {
638        self.cells.iter().fold(None, |b, ((x, y), _)| {
639            let c = IVec2::new(*x, *y);
640            Some(b.map_or((c, c), |(lo, hi): (IVec2, IVec2)| (lo.min(c), hi.max(c))))
641        })
642    }
643}

649/// A loaded tile file.
650#[derive(Clone, PartialEq, Debug)]
651pub enum TileFile {
652    /// A `.tile`.
653    Tile(TileAsset),
654    /// A `.ruletile`.
655    Rule(RuleTile),
656}
657
658impl TileFile {
659    /// Load by extension (`.tile` or `.ruletile`).
660    pub fn load(path: &Path) -> Result<TileFile, String> {
661        let text = std::fs::read_to_string(path).map_err(|e| format!("{}: {e}", path.display()))?;
662        let ext = path.extension().map(|e| e.to_string_lossy().to_lowercase()).unwrap_or_default();
663        match ext.as_str() {
664            RULETILE_EXTENSION => ron::from_str(&text).map(TileFile::Rule).map_err(|e| format!("{}: {e}", path.display())),
665            _ => ron::from_str(&text).map(TileFile::Tile).map_err(|e| format!("{}: {e}", path.display())),
666        }
667    }
668
669    /// Serialize and write, by extension (`.tile` or `.ruletile`).
670    pub fn save(&self, path: &Path) -> Result<(), String> {
671        let cfg = ron::ser::PrettyConfig::default();
672        let text = match self {
673            TileFile::Tile(t) => ron::ser::to_string_pretty(t, cfg),
674            TileFile::Rule(r) => ron::ser::to_string_pretty(r, cfg),
675        }
676        .map_err(|e| e.to_string())?;
677        std::fs::write(path, text).map_err(|e| format!("{}: {e}", path.display()))
678    }
679}
680
681/// Read and parse a `.palette`.
682pub fn load_palette(path: &Path) -> Result<TilePalette, String> {
683    let text = std::fs::read_to_string(path).map_err(|e| format!("{}: {e}", path.display()))?;
684    ron::from_str(&text).map_err(|e| format!("{}: {e}", path.display()))
685}
686
687/// Serialize and write a `.palette`.
688pub fn save_palette(path: &Path, p: &TilePalette) -> Result<(), String> {
689    let text = ron::ser::to_string_pretty(p, ron::ser::PrettyConfig::default()).map_err(|e| e.to_string())?;
690    std::fs::write(path, text).map_err(|e| format!("{}: {e}", path.display()))
691}

747    /// An independent matcher: no bitmask, no ring permutation. The rule's
748    /// offsets are transformed geometrically and compared against a real tilemap.
749    fn brute(rules: &[TilingRule], map: &Tilemap, cell: IVec2) -> RuleHit {
750        let me = map.tile_at(cell).unwrap().to_string();
751        let same = |c: IVec2| map.tile_at(c) == Some(me.as_str());
752        let rot = |o: IVec2, k: usize| (0..k).fold(o, |o, _| IVec2::new(-o.y, o.x));
753        for (ri, r) in rules.iter().enumerate() {
754            let variants: Vec<(usize, bool)> = match r.transform {
755                RuleTransform::Fixed => vec![(0, false)],
756                RuleTransform::Rotated => (0..4).map(|k| (k, false)).collect(),
757                RuleTransform::MirrorX => vec![(0, false), (0, true)],
758                RuleTransform::MirrorY => vec![(0, false), (2, true)],
759                RuleTransform::MirrorXY => vec![(0, false), (0, true), (2, true), (2, false)],
760                RuleTransform::RotatedMirror => (0..4).flat_map(|k| [(k, false), (k, true)]).collect(),
761            };
762            for (k, mirror) in variants {
763                let ok = (0..8).all(|i| {
764                    let o = neighbor_offset(CellLayout::Rectangle, cell, i);
765                    let o = if mirror { IVec2::new(-o.x, o.y) } else { o };
766                    let to = rot(o, k);
767                    match r.req(i) {
768                        NeighborReq::Any => true,
769                        NeighborReq::This => same(cell + to),
770                        NeighborReq::NotThis => !same(cell + to),
771                    }
772                });
773                if ok {
774                    return RuleHit { rule: ri as u16, xform: TileXform::new(k as u8 * 6, mirror) };
775                }
776            }
777        }
778        RuleHit { rule: RuleHit::NONE, xform: TileXform::IDENTITY }
779    }
