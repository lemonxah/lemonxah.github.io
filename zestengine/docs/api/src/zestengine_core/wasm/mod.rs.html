1//! WASM components that expose callable functions (EVT-2): the
2//! [`WasmScript`] component hosts a sandboxed `.wasm` module whose declared
3//! actions and properties appear in the same event-binding dropdowns as the
4//! built-in Rust components (see [`crate::actions`]).
5//!
6//! The older [`crate::Script`] component (`env` imports, `start` / `update`
7//! hooks) is a separate prototype and keeps working; this is the component
8//! ABI that bindings call into.
9//!
10//! # Authoring a module
11//!
12//! Compile for `wasm32-unknown-unknown`. The `guest/zest-guest` crate in the
13//! repository wraps everything below (manifest section, value codec, host
14//! imports, export glue) for Rust authors; the rest of this section is the
15//! exact contract it implements.
16//!
17//! ## Manifest
18//!
19//! A custom section named `zest_manifest` holding UTF-8 RON (at most 64 KiB):
20//!
21//! ```ron
22//! (
23//!     name: "Door",                       // inspector title
24//!     api: 1,                             // manifest version
25//!     capabilities: ["state", "events"],  // host imports the module uses
26//!     actions: [
27//!         (name: "Open", label: "Open door", ugc_safe: true, params: [
28//!             (name: "speed", ty: Float, default: Some(Float(2.0))),
29//!         ]),
30//!     ],
31//!     properties: [(name: "state", ty: Int)],
32//! )
33//! ```
34//!
35//! `ty` is `Bool`, `Int` (i64), `Float` (f32), `String`, `Vec3` or `Object`.
36//! A custom section was chosen over an exported function because the host,
37//! the package builder and the extractor can read it without running any guest
38//! code. The position of an action in `actions` is the `action_index` the
39//! host passes to `zest_invoke`; the position of a property is the `index` of
40//! `zest_read_property`. Names are what scenes store, so never rename a
41//! shipped action; indices may change freely between builds.
42//!
43//! ## Value encoding
44//!
45//! All integers are little-endian. A value is a one-byte tag and a payload:
46//!
47//! | tag | type   | payload                                          |
48//! |-----|--------|--------------------------------------------------|
49//! | 0   | Bool   | 1 byte, 0 or 1                                   |
50//! | 1   | Int    | i64, 8 bytes                                     |
51//! | 2   | Float  | f32, 4 bytes                                     |
52//! | 3   | String | u32 byte length (at most 16384), UTF-8 bytes     |
53//! | 4   | Vec3   | 3 x f32, 12 bytes                                |
54//! | 5   | Object | u128 object id, 16 bytes, all zero = none        |
55//!
56//! An argument block is a `u32` count (at most 32) followed by that many
57//! values, in parameter order, at most 64 KiB in all. The host has already
58//! coerced each value to the declared parameter type and filled in defaults,
59//! so a guest can decode by position without checking tags. [`abi`] has the
60//! encoder and decoder.
61//!
62//! ## Exports
63//!
64//! | export | signature | notes |
65//! |--------|-----------|-------|
66//! | `memory` | memory | exactly one, 32-bit, within the memory cap |
67//! | `zest_alloc` | `(len: i32) -> i32` | returns a pointer to at least `len` writable bytes the host fills with an argument block; the same scratch buffer may be returned every time (the host uses it once per call). Needed when actions are declared |
68//! | `zest_invoke` | `(action_index: i32, args_ptr: i32, args_len: i32) -> i32` | 0 = ok; anything else is a failure reported to the binding |
69//! | `zest_last_error` | `() -> i64` | optional; after a non-zero status, `(len << 32) \| ptr` of a UTF-8 message |
70//! | `zest_read_property` | `(index: i32) -> i64` | `(len << 32) \| ptr` of one encoded value in guest memory (valid until the next call), or 0 for "no value". Needed when properties are declared |
71//! | `zest_update` | `(dt: f32)` | optional; once per frame while playing |
72//!
73//! ## Host imports (module `host`, nothing else)
74//!
75//! Pointers are `i32` offsets into guest memory; strings are UTF-8 `(ptr, len)`
76//! pairs of at most 256 bytes. A bad pointer traps the call.
77//!
78//! | import | capability | notes |
79//! |--------|------------|-------|
80//! | `log(ptr, len)` | none | 32 lines per call, 512 bytes each |
81//! | `get_dt() -> f32`, `get_time() -> f32` | none | frame time and game time; the only clock |
82//! | `get_self(out_ptr)` | none | writes the owner's 16-byte object id |
83//! | `get_instigator(out_ptr) -> i32` | none | 16-byte id of whoever caused the event; 0 if none |
84//! | `get_float(ptr, len) -> f32`, `set_float(ptr, len, v)` | `state` | per-component named floats (64 keys) |
85//! | `emit_event(ptr, len)` | `events` | broadcasts an `ActionEvent`, 16 per call |
86//! | `get_property(obj_ptr, comp_ptr, comp_len, prop_ptr, prop_len, out_ptr, out_cap) -> i32` | `read` | encoded value written to `out_ptr`, returns its length; an all-zero object means this one. Inside `zest_update`, and inside `zest_invoke` when the engine passes a world view (always in Play; -2 elsewhere, e.g. a test calling an action without one); -1 not found, -4 buffer too small. In `zest_invoke` the module's own component is not readable, its sibling components and other objects are |
87//! | `invoke(obj_ptr, comp_ptr, comp_len, act_ptr, act_len, args_ptr, args_len) -> i32` | `invoke` | queues a call of any component's action (same type checks as a binding); it runs after the guest call returns, in the frame's next command pass. 0 queued, -5 bad arguments, -4 queue full |
88//!
89//! A module may import only these. A host function whose capability the
90//! manifest doesn't declare makes the module fail to load. There is no WASI,
91//! wall clock or randomness.
92//!
93//! # Sandbox
94//!
95//! - The `.wasm` file is capped ([`SandboxLimits`]), its memory is capped in
96//!   pages (declared and grown; growth past the cap fails with -1), and so is
97//!   the guest call stack (deep recursion traps).
98//! - Every entry-point call, and instantiation, gets a fuel budget; running
99//!   out terminates the call. The instance is reused across calls.
100//! - Any trap (fuel, stack, `unreachable`, bad pointer) disables the
101//!   component: one error is logged, later calls do nothing, and the engine
102//!   keeps running. Editing the file (hot reload) or changing the `module`
103//!   field loads a fresh instance.
104//! - Imports are checked against the allowlist before instantiation.
105//!
106//! # Trust
107//!
108//! Modules inside an extracted content package are untrusted
109//! ([`register_untrusted_root`]). They get tighter limits, may not use the
110//! `invoke` capability, and expose an action to bindings only if its manifest
111//! flag `ugc_safe` is set and the package's content limits allow module
112//! actions (`ContentLimits::allow_wasm_actions`). Project modules are trusted.
113//! See `docs/user-content.md`.
114//!
115//! # Dynamic descriptors
116//!
117//! `Component::actions` returns `&'static [ActionDesc]`, so a module's
118//! manifest is turned into such tables once per distinct manifest text and
119//! leaked ([`component`] interns them; at most 512 distinct manifests per
120//! process, then loading fails). Reloading an unchanged manifest reuses the
121//! tables.

139/// How modules under an untrusted root behave.
140#[derive(Clone, Copy, Debug, PartialEq)]
141pub struct UntrustedPolicy {
142    /// Resource limits for modules under the root.
143    pub limits: SandboxLimits,
144    /// Whether the module's actions can be bound at all (the content limits').
145    pub allow_actions: bool,
146}
147
148static UNTRUSTED: RwLock<Vec<(PathBuf, UntrustedPolicy)>> = RwLock::new(Vec::new());
149
150/// Mark everything under `root` (an extracted package's folder) as untrusted
151/// content. The engine's content manager calls this when it extracts a package.
152pub fn register_untrusted_root(root: &Path, policy: UntrustedPolicy) {
153    let root = std::fs::canonicalize(root).unwrap_or_else(|_| root.to_path_buf());
154    let mut roots = UNTRUSTED.write().unwrap_or_else(|e| e.into_inner());
155    match roots.iter_mut().find(|(r, _)| *r == root) {
156        Some(entry) => entry.1 = policy,
157        None => roots.push((root, policy)),
158    }
159}
160
161/// The policy for a module file, if it sits in an untrusted root.
162pub fn untrusted_policy(module: &Path) -> Option<UntrustedPolicy> {
163    let path = std::fs::canonicalize(module).unwrap_or_else(|_| module.to_path_buf());
164    let roots = UNTRUSTED.read().unwrap_or_else(|e| e.into_inner());
165    roots.iter().find(|(r, _)| path.starts_with(r)).map(|(_, p)| *p)
166}

168// Approved mod scripts retain the UGC resource limits; only named game actions
169// are added to their host API. A package cannot grant itself this policy.
170static MOD_ACTIONS: RwLock<Vec<(PathBuf, Vec<String>)>> = RwLock::new(Vec::new());
171pub fn register_mod_root(root:&Path,actions:Vec<String>){
172    register_untrusted_root(root,UntrustedPolicy{limits:SandboxLimits::untrusted(),allow_actions:true});
173    let root=std::fs::canonicalize(root).unwrap_or_else(|_|root.to_owned());
174    let mut roots=MOD_ACTIONS.write().unwrap_or_else(|e|e.into_inner());
175    if let Some(entry)=roots.iter_mut().find(|(r,_)|r==&root){entry.1=actions;}else{roots.push((root,actions));}
176}
177pub fn mod_actions(module:&Path)->Option<Vec<String>>{
178    let path=std::fs::canonicalize(module).unwrap_or_else(|_|module.to_owned());
179    MOD_ACTIONS.read().unwrap_or_else(|e|e.into_inner()).iter().filter(|(r,_)|path.starts_with(r)).max_by_key(|(r,_)|r.components().count()).map(|(_,a)|a.clone())
180}
181pub fn unregister_mod_root(root:&Path){
182    let root=std::fs::canonicalize(root).unwrap_or_else(|_|root.to_owned());
183    MOD_ACTIONS.write().unwrap_or_else(|e|e.into_inner()).retain(|(r,_)|r!=&root);
184    UNTRUSTED.write().unwrap_or_else(|e|e.into_inner()).retain(|(r,_)|r!=&root);
185}
