41/// What happens to the object when its [`Health`] reaches zero.

43pub enum DeathAction {

46    /// Deactivate the object.

60            Self::Destroy => "Destroy object",

65/// Hit points. Native systems damage it through a checked mutable ECS borrow.
66#[derive(Serialize, Deserialize)]
67#[serde(default)]
68pub struct Health {

70    pub max: f32,

87    /// Initialization completed for this component instance (runtime).
88    #[serde(skip)]
89    pub started: bool,
90}

108impl Health {
109    /// Subtract hit points. Returns true when this hit killed it.
110    pub fn apply_damage(&mut self, amount: f32) -> bool {
111        if self.invulnerable || self.is_dead() || amount <= 0.0 {
112            return false;
113        }
114        self.current = (self.current - amount).max(0.0);
115        self.since_damage = 0.0;
116        self.is_dead()
117    }
118
119    /// Add hit points (capped at `max`).
120    pub fn heal(&mut self, amount: f32) {
121        if !self.is_dead() {
122            self.current = (self.current + amount.max(0.0)).min(self.max);
123        }
124    }
125
126    /// True when `current` is zero.
127    pub fn is_dead(&self) -> bool {
128        self.current <= 0.0
129    }
130
131    /// Back to full health (respawns, dummies standing back up).
132    pub fn revive(&mut self) {
133        self.current = self.max;
134        self.death_handled = false;
135        self.since_damage = 0.0;
136    }
137
138    /// Current / max, clamped to 0..1.
139    pub fn fraction(&self) -> f32 {
140        if self.max <= 0.0 { 0.0 } else { (self.current / self.max).clamp(0.0, 1.0) }
141    }
142}

157impl TypedComponent for Health {

159
160    fn actions(&self) -> &'static [ActionDesc] {
161        HEALTH_ACTIONS
162    }
163    fn invoke_action(&mut self, _ctx: &mut ActionCtx, name: &str, args: &[ArgValue]) -> Result<(), String> {
164        match name {
165            "Damage" => {
166                self.apply_damage(args[0].as_f32());
167            }
168            "Heal" => self.heal(args[0].as_f32()),
169            "Revive" => self.revive(),
170            "SetInvulnerable" => self.invulnerable = args[0].as_bool(),
171            _ => return Err(crate::actions::unknown_action(Self::NAME, name)),
172        }
173        Ok(())
174    }
175    fn properties(&self) -> &'static [PropertyDesc] {
176        HEALTH_PROPERTIES
177    }
178    fn read_property(&self, name: &str) -> Option<ArgValue> {
179        match name {
180            "current" => Some(ArgValue::Float(self.current)),
181            "max" => Some(ArgValue::Float(self.max)),
182            "fraction" => Some(ArgValue::Float(self.fraction())),
183            "dead" => Some(ArgValue::Bool(self.is_dead())),
184            _ => None,
185        }
186    }

189impl Health {
190    fn start_native(&mut self, _ctx: &mut ObjectContext<'_, '_>) {
191        if self.current <= 0.0 || self.current > self.max {
192            self.current = self.max;
193        }
194        self.started = true;
195    }
196
197    fn update_native(&mut self, ctx: &mut ObjectContext<'_, '_>) {
198        if !self.started { self.start_native(ctx); }
199        if self.is_dead() {
200            if !self.death_handled {
201                self.death_handled = true;
202                let object = ctx.self_id();
203                ctx.emit(DeathEvent { object });
204                match self.on_death {
205                    DeathAction::Nothing => {}
206                    DeathAction::Disable => ctx.set_active(ctx.self_id(), false),
207                    DeathAction::Destroy => ctx.destroy_self(),
208                }
209            }
210            return;
211        }
212        self.since_damage += ctx.dt;
213        if self.regen_per_sec > 0.0 && self.since_damage >= self.regen_delay {
214            self.current = (self.current + self.regen_per_sec * ctx.dt).min(self.max);
215        }
216    }
217}

353/// `forward` rotated by a pseudo-random angle inside a cone of `spread_deg`.
354/// `seed` advances on every call so successive shots differ.
355pub fn spread_direction(forward: Vec3, spread_deg: f32, seed: &mut u32) -> Vec3 {
356    let forward = forward.normalize_or(Vec3::NEG_Z);
357    if spread_deg <= 0.0 {
358        return forward;
359    }
360    let mut next = || {
361        *seed ^= *seed << 13;
362        *seed ^= *seed >> 17;
363        *seed ^= *seed << 5;
364        (*seed as f32) / (u32::MAX as f32)
365    };
366    let angle = spread_deg.to_radians() * next().sqrt();
367    let around = next() * std::f32::consts::TAU;
368    let side = forward.any_orthonormal_vector();
369    let axis = Quat::from_axis_angle(forward, around) * side;
370    (Quat::from_axis_angle(axis, angle) * forward).normalize()
371}

373impl HitscanWeapon {
374    /// True while a reload is running.
375    pub fn reloading(&self) -> bool {
376        self.reload_timer > 0.0
377    }
378
379    /// True briefly after a shot, so the held weapon keeps pointing at the
380    /// crosshair while firing from the hip.
381    pub fn firing(&self) -> bool {
382        self.fire_flash > 0.0
383    }
384
385    fn start_reload(&mut self) {
386        if self.reloading() || self.ammo >= self.magazine_size || (!self.infinite_reserve && self.reserve == 0) {
387            return;
388        }
389        self.reload_timer = self.reload_time.max(0.01);
390    }
391
392    fn finish_reload(&mut self) {
393        let need = self.magazine_size.saturating_sub(self.ammo);
394        let take = if self.infinite_reserve { need } else { need.min(self.reserve) };
395        self.ammo += take;
396        if !self.infinite_reserve {
397            self.reserve -= take;
398        }
399    }
400
401    fn trigger_state(&self, ctx: &ObjectContext<'_, '_>) -> (bool, bool) {
402        match self.trigger {
403            WeaponTrigger::Action => (ctx.input.down(&self.fire_action), ctx.input.pressed(&self.fire_action)),
404            WeaponTrigger::Grabbed => {
405                let down = ctx.world.get::<crate::Grabbable>(ctx.owner).is_some_and(|g| g.held && g.trigger_down);
406                (down, down && !self.trigger_prev)
407            }
408        }
409    }
410
411    fn fire(&mut self, ctx: &mut ObjectContext<'_, '_>) {
412        let aim = ctx
413            .transform_of(self.aim_from)
414            .or_else(|| ctx.transform_of(self.muzzle))
415            .unwrap_or_else(|| ctx.self_transform());
416        let dir = spread_direction(aim.forward(), self.spread_deg, &mut self.rng);
417        self.ammo = self.ammo.saturating_sub(1);
418        self.shots_fired += 1;
419        self.cooldown = 1.0 / self.fire_rate.max(0.01);
420        self.fire_flash = 0.12;
421        if self.muzzle.is_set() {
422            ctx.emit_particles(self.muzzle, self.muzzle_burst);
423        }
424        if !self.fire_sound.is_empty() {
425            ctx.play_sound_at(self.fire_sound.clone(), aim.translation, 1.0);
426        }
427        let filter = RayFilter { layer_mask: self.layer_mask, include_triggers: false, exclude: Some(ctx.self_id()) };
428        let Some(hit) = ctx.raycast_filtered(aim.translation, dir, self.range, &filter) else {
429            self.last_hit = None;
430            return;
431        };
432        self.last_hit = Some(hit.point);
433        let damage = self.damage;
434        ctx.edit::<Health>(hit.object, move |h| {
435            h.apply_damage(damage);
436        });
437        let source = ctx.self_id();
438        ctx.emit(DamageEvent { target: hit.object, source, amount: damage, point: hit.point });
439        if self.impact_force > 0.0 {
440            ctx.add_impulse(hit.object, dir * self.impact_force);
441        }
442        if self.hit_effect.is_set() {
443            let rot = crate::kit::look_rotation(hit.normal);
444            ctx.set_world_transform_of(self.hit_effect, Some(hit.point + hit.normal * 0.02), Some(rot));
445            ctx.emit_particles(self.hit_effect, self.hit_burst);
446        }
447        if !self.hit_sound.is_empty() {
448            ctx.play_sound_at(self.hit_sound.clone(), hit.point, 0.8);
449        }
450    }
451}

457impl HitscanWeapon {
458    fn start_native(&mut self, ctx: &mut ObjectContext<'_, '_>) {
459        self.ammo = self.ammo.min(self.magazine_size);
460        self.rng ^= (ctx.self_id().raw() as u32) | 1;
461        self.started = true;
462    }
463
464    fn update_native(&mut self, ctx: &mut ObjectContext<'_, '_>) {
465        if !self.started { self.start_native(ctx); }
466        let (down, pressed) = self.trigger_state(ctx);
467        self.trigger_prev = down;
468        if ctx.dt <= 0.0 || ctx.ui_text_editing() {
469            return;
470        }
471        self.cooldown = (self.cooldown - ctx.dt).max(0.0);
472        self.fire_flash = (self.fire_flash - ctx.dt).max(0.0);
473        if self.reloading() {
474            self.reload_timer -= ctx.dt;
475            if self.reload_timer <= 0.0 {
476                self.reload_timer = 0.0;
477                self.finish_reload();
478            }
479            return;
480        }
481        if !self.reload_action.is_empty() && ctx.input.pressed(&self.reload_action) {
482            self.start_reload();
483            return;
484        }
485        let wants = if self.automatic { down } else { pressed };
486        if !wants || self.cooldown > 0.0 {
487            return;
488        }
489        if self.ammo == 0 {
490            self.start_reload();
491            return;
492        }
493        self.fire(ctx);
494    }
495}

497/// Shooting-range target: when its [`Health`] hits zero it tips over, scores,

539impl TargetDummy {
540    fn start_native(&mut self, ctx: &mut ObjectContext<'_, '_>) {
541        self.base = Some(ctx.rotation());
542    }
543
544    fn update_native(&mut self, ctx: &mut ObjectContext<'_, '_>) {
545        let base = *self.base.get_or_insert(ctx.rotation());
546        let dead = ctx.world.get::<Health>(ctx.owner).is_some_and(|h| h.is_dead());
547        if dead && !self.knocked {
548            self.knocked = true;
549            self.timer = self.respawn_delay;
550            ctx.emit(ScoreEvent { points: self.points });
551        }
552        if self.knocked {
553            self.timer -= ctx.dt;
554            if self.timer <= 0.0 {
555                self.knocked = false;
556                ctx.edit::<Health>(ctx.self_id(), |h| h.revive());
557            }
558        }
559        let goal = if self.knocked { 85.0 } else { 0.0 };
560        let step = self.fall_speed * ctx.dt;
561        self.angle += (goal - self.angle).clamp(-step, step);
562        ctx.set_rotation(base * Quat::from_rotation_x(-self.angle.to_radians()));
563    }
564}

566pub(crate) fn register_systems(registry: &mut crate::ComponentRegistry) {
567    registry.register_ecs::<Health>();
568    registry.register_ecs::<HitscanWeapon>();
569    registry.register_ecs::<TargetDummy>();
570    registry.register_system_if("zestengine::combat_start", Stage::Start, start, |world|
571        world.query::<&Health>().iter().any(|(_, value)| !value.started)
572        || world.query::<&HitscanWeapon>().iter().any(|(_, value)| !value.started)
573        || world.query::<&TargetDummy>().iter().any(|(_, value)| value.base.is_none()));
574    // Movement/tracking precedes shots, shots precede death handling, and
575    // dummies see the resulting health before deciding whether to stand up.
576    registry.register_system_for::<HitscanWeapon>("zestengine::hitscan", Stage::Gameplay, weapons);
577    registry.register_system_for::<Health>("zestengine::health", Stage::Gameplay, health);
578    registry.register_system_for::<TargetDummy>("zestengine::target_dummies", Stage::Gameplay, dummies);
579}

581fn start(world: &SystemWorld<'_>, context: &mut SystemContext<'_>) {
582    macro_rules! initialize { ($t:ty, $pending:expr) => {
583        let mut query = world.query::<(&Owner, &mut $t)>();
584        let mut actors: Vec<_> = query.iter().filter(|(_, (owner, value))| ($pending)(value) && context.is_active(world, owner.0)).collect();
585        actors.sort_unstable_by_key(|(_, (owner, _))| context.order_key(world, owner.0));
586        for (_, (owner, value)) in actors { value.start_native(&mut ObjectContext::new(world, owner.0, context)); }
587    }; }
588    initialize!(Health, |value: &Health| !value.started);
589    initialize!(HitscanWeapon, |value: &HitscanWeapon| !value.started);
590    initialize!(TargetDummy, |value: &TargetDummy| value.base.is_none());
591}

635        let parent = scene.spawn("Parent");
636        *scene.get_mut::<Transform>(parent).unwrap() = Transform { translation: Vec3::new(8.0, 0.0, -1.0), rotation: Quat::from_rotation_y(0.7), scale: Vec3::splat(1.5) };

640        scene.set_parent(hand, Some(rig), false).unwrap();

642        scene.insert(rig, VrCharacter { right_hand: hand_id.into(), locomotion: VrLocomotion::Disabled, turning: VrTurning::Disabled, ..Default::default() }).unwrap();

647        scene.insert(impact, ParticleEmitter::default()).unwrap();

651            muzzle: scene.object_id(muzzle).unwrap().into(), hit_effect: scene.object_id(impact).unwrap().into(), ..Default::default() }).unwrap();
652        let target = scene.spawn("Target");

654        scene.insert(target, TargetDummy { points: 7, respawn_delay: 0.3, ..Default::default() }).unwrap();
655        let hud = scene.spawn("Score");

657        scene.insert(hud, UiText::default()).unwrap();
658        scene.propagate().unwrap();
659        let hits = Hits { target: scene.object_id(target).unwrap(), rays: RefCell::new(Vec::new()) };
660        let screen = ScreenInfo { vr: VrFrame { active: true, right_hand: Some((Vec3::new(0.3, 1.3, -0.4), Quat::from_rotation_y(0.4))), ..Default::default() }, ..Default::default() };

665            context.dt = 0.1;
666            context.services.physics = Some(&hits);
667            context.services.screen = Some(&screen);
668            for stage in [Stage::Start, Stage::Update, Stage::Gameplay, Stage::LateUpdate] { registry.run_systems(stage, &scene, &mut context); }
669        }
670        let actual = hits.rays.borrow();
671        assert_eq!(actual.len(), 1);
672        scene.with_world(|world| {
673            let hand = world.world_transform(hand).unwrap();

675            assert!(actual[0].1.abs_diff_eq(hand.to_scale_rotation_translation().1 * Vec3::NEG_Z, 1e-5));

682        assert_eq!(scene.get::<ParticleEmitter>(impact).unwrap().control.emit, 16);
683        assert_eq!(scene.get::<UiText>(hud).unwrap().text, "0", "events are delivered next frame");
684        assert!(!commands.iter().any(|command| matches!(command, ComponentCommand::Modify { .. } | ComponentCommand::SetLocalTransform { .. } | ComponentCommand::SetWorldTransform { .. })));
685        let events: Vec<AnyBox> = commands.drain(..).filter_map(|command| match command { ComponentCommand::Emit(event) => Some(event), _ => None }).collect();

687        assert_eq!(events.iter().filter(|event| event.0.is::<DamageEvent>()).count(), 1);

691        assert_eq!(scene.get::<UiText>(hud).unwrap().text, "7");
692        assert_eq!(scene.get::<HudBinding>(hud).unwrap().score, 7);
693        for _ in 0..3 {
694            let mut context = SystemContext::new(&input, &mut commands);
695            context.dt = 0.1;
696            registry.run_systems(Stage::Gameplay, &scene, &mut context);
697        }
698        assert!(!scene.get::<TargetDummy>(target).unwrap().knocked());
699        assert_eq!(scene.get::<Health>(target).unwrap().current, 25.0);
700        assert!(!commands.iter().any(|command| matches!(command, ComponentCommand::Emit(event) if event.0.is::<DeathEvent>())));
701    }

704    fn native_health_initializes_once_and_queues_death_actions_once() {
705        use crate::{ComponentCommand, ComponentRegistry, InputState};
706        use crate::ecs::SceneWorld;
707        let registry = ComponentRegistry::with_builtins();
708        for action in [DeathAction::Disable, DeathAction::Destroy] {
709            let mut scene = SceneWorld::new();
710            let entity = scene.spawn("Health");
711            let id = scene.object_id(entity).unwrap();
712            scene.insert(entity, Health { max: 40.0, current: 0.0, on_death: action, ..Default::default() }).unwrap();
713            let input = InputState::default();
714            let mut commands = Vec::new();
715            let mut context = SystemContext::new(&input, &mut commands);
716            registry.run_systems(Stage::Start, &scene, &mut context);
717            scene.get_mut::<Health>(entity).unwrap().apply_damage(100.0);
718            registry.run_systems(Stage::Start, &scene, &mut context);
719            for _ in 0..2 { registry.run_systems(Stage::Gameplay, &scene, &mut context); }
720            assert!(scene.get::<Health>(entity).unwrap().is_dead(), "repeated startup must not revive a dead instance");
721            assert_eq!(commands.iter().filter(|command| matches!(command, ComponentCommand::Emit(event) if event.0.is::<DeathEvent>())).count(), 1);
722            assert_eq!(commands.iter().filter(|command| match command {
723                ComponentCommand::SetActive(target, false) => *target == id && action == DeathAction::Disable,
724                ComponentCommand::Destroy(target) => *target == id && action == DeathAction::Destroy,
725                _ => false,
726            }).count(), 1);
727            scene.remove::<Health>(entity).unwrap();
728            scene.insert(entity, Health { max: 9.0, current: 0.0, ..Default::default() }).unwrap();
729            registry.run_systems(Stage::Start, &scene, &mut SystemContext::new(&input, &mut commands));
730            assert_eq!(scene.get::<Health>(entity).unwrap().current, 9.0);
731        }
732    }
