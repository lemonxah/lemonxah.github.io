9/// Runs actions when something enters or leaves a trigger collider on this
10/// object (a `Box`/`Sphere`/`Capsule Collider` with `Is Trigger` on). The
11/// object that entered is the instigator, and `other` in the event payload.
12#[derive(Serialize, Deserialize)]
13#[serde(default)]
14#[derive(Default)]
15pub struct TriggerEvents {
16    /// Only objects that carry this component (by type name, e.g. "Pawn")
17    /// count. Empty = anything.
18    pub require_component: String,
19    /// Run `on_enter` / `on_exit` for every object, or only when the first
20    /// object enters / the last one leaves.
21    pub first_and_last_only: bool,
22    /// Fire `on_enter` at most once.
23    pub once: bool,
24    /// Actions run when an object enters.
25    pub on_enter: EventActions,
26    /// Actions run when an object leaves.
27    pub on_exit: EventActions,
28    /// Objects currently inside (counted, not saved).
29    #[serde(skip)]
30    pub inside: u32,
31    /// Whether the trigger has entered its active state (runtime).
32    #[serde(skip)]
33    pub entered: bool,
34}
35
36
37impl TriggerEvents {
38    fn accepts(&self, world: &SystemWorld<'_>, context: &SystemContext<'_>, other: ObjectId, instance: Entity) -> bool {
39        if self.require_component.is_empty() {
40            return true;
41        }
42        let Some(owner) = world.entity(other) else { return false };
43        let Some(index) = context.services.components else { return false };
44        (0..index.component_count(owner).unwrap_or(0)).any(|position| index.component_slot(owner, position).is_some_and(|slot|
45            slot.name() == self.require_component && !(slot.entity() == instance && slot.type_id() == std::any::TypeId::of::<Self>())))
46    }
47
48    fn run(&self, list: &EventActions, context: &mut SystemContext<'_>, me: ObjectId, other: ObjectId) {
49        let payload = [("other", ArgValue::Object(other.into())), ("object", ArgValue::Object(me.into()))];
50        queue_actions(list, &FireSource { source: me, instigator: Some(other), payload: &payload }, context.commands);
51    }
52}
53
54impl TypedComponent for TriggerEvents {
55    const NAME: &'static str = "Trigger Events";
56}
57
58impl TriggerEvents {
59    fn enter(&mut self, world: &SystemWorld<'_>, context: &mut SystemContext<'_>, instance: Entity, me: ObjectId, other: ObjectId) {
60        if !self.accepts(world, context, other, instance) {
61            return;
62        }
63        self.inside += 1;
64        if (self.once && self.entered) || (self.first_and_last_only && self.inside > 1) {
65            return;
66        }
67        self.entered = true;
68        self.run(&self.on_enter, context, me, other);
69    }
70
71    fn exit(&mut self, world: &SystemWorld<'_>, context: &mut SystemContext<'_>, instance: Entity, me: ObjectId, other: ObjectId) {
72        if !self.accepts(world, context, other, instance) || self.inside == 0 {
73            return;
74        }
75        self.inside -= 1;
76        if self.first_and_last_only && self.inside > 0 {
77            return;
78        }
79        self.run(&self.on_exit, context, me, other);
80    }
81}

83/// One animation event name and what it runs.

86pub struct AnimationEventBinding {

93/// Runs actions when the Animator on this object passes a named animation
94/// event marker.
95#[derive(Default, Serialize, Deserialize)]
96#[serde(default)]
97pub struct AnimationEventBindings {
98    /// The bindings, in declaration order.
99    pub events: Vec<AnimationEventBinding>,
100}
101
102impl TypedComponent for AnimationEventBindings {
103    const NAME: &'static str = "Animation Event Bindings";
104}
105
106impl AnimationEventBindings {
107    fn fire(&self, context: &mut SystemContext<'_>, me: ObjectId, name: &str) {
108        let payload = [("name", ArgValue::String(name.to_owned())), ("object", ArgValue::Object(me.into()))];
109        for binding in self.events.iter().filter(|b| b.event == name) {
110            queue_actions(&binding.actions, &FireSource { source: me, instigator: None, payload: &payload }, context.commands);
111        }
112    }
113}

122fn order<T: crate::Component>(world: &SystemWorld<'_>, context: &SystemContext<'_>, owner: Entity, instance: Entity) -> ((usize, u128), usize) {
123    let slot = context.services.components.and_then(|index|
124        (0..index.component_count(owner)?).find(|&position| index.component_slot(owner, position).is_some_and(|slot|
125            slot.entity() == instance && slot.type_id() == std::any::TypeId::of::<T>())))
126        .unwrap_or(usize::MAX);
127    (context.order_key(world, owner), slot)
128}

130fn triggers(world: &SystemWorld<'_>, context: &mut SystemContext<'_>) {
131    let mut query = world.query::<(&Owner, &mut TriggerEvents)>();
132    let mut actors: Vec<_> = query.iter().filter(|(_, (owner, _))| context.is_active(world, owner.0)).collect();
133    actors.sort_unstable_by_key(|(instance, (owner, _))| order::<TriggerEvents>(world, context, owner.0, *instance));
134    let events = context.services.scene_events;
135    for (instance, (owner, trigger)) in actors {
136        let Some(me) = world.object_id(owner.0) else { continue };
137        for event in events {
138            if let SceneEvent::Contact { object, other, trigger: true, phase, .. } = event {
139                if *object != me { continue; }
140                match phase {
141                    ContactPhase::Enter => trigger.enter(world, context, instance, me, *other),
142                    ContactPhase::Exit => trigger.exit(world, context, instance, me, *other),
143                    ContactPhase::Stay => {},
144                }
145            }
146        }
147    }
148}

150fn animation(world: &SystemWorld<'_>, context: &mut SystemContext<'_>) {
151    let mut query = world.query::<(&Owner, &AnimationEventBindings)>();
152    let mut actors: Vec<_> = query.iter().filter(|(_, (owner, _))| context.is_active(world, owner.0)).collect();
153    actors.sort_unstable_by_key(|(instance, (owner, _))| order::<AnimationEventBindings>(world, context, owner.0, *instance));
154    let events = context.services.scene_events;
155    for (_, (owner, bindings)) in actors {
156        let Some(me) = world.object_id(owner.0) else { continue };
157        for event in events {
158            if let SceneEvent::Animation { object, name } = event {
159                if *object == me { bindings.fire(context, me, name); }
160            }
161        }
162    }
