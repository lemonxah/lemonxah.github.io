1//! Swimming for the walking controllers: once the water is deeper than the
2//! character's chest it swims instead of walks. It floats with its head just
3//! out of the water, strokes along where it looks (diving when looking down,
4//! rising with jump, sinking with crouch), runs out of breath under water,
5//! and climbs out over an edge it swims into. A [`Swimmer`] on the same
6//! object tunes it; without one the defaults apply.
7
8use glam::{Quat, Vec3};
9use serde::{Deserialize, Serialize};
10
11use crate::TypedComponent;
12
13/// How a character swims.
14#[derive(Clone, Copy, Debug, PartialEq, Serialize, Deserialize)]
15#[serde(default)]
16pub struct Swimmer {
17    /// Water this deep, as a share of the character's height, makes it swim
18    /// (it wades below that).
19    pub swim_depth: f32,
20    /// Swimming speed (m/s).
21    pub swim_speed: f32,
22    /// Sprint swimming speed (m/s).
23    pub sprint_speed: f32,
24    /// How quickly strokes reach their speed (1/s).
25    pub stroke_accel: f32,
26    /// How quickly it slows when it stops stroking (1/s).
27    pub water_drag: f32,
28    /// How high the eyes float above the surface at rest (m).
29    pub float_height: f32,
30    /// Seconds it can stay with its head under.
31    pub breath: f32,
32    /// Highest edge above the surface it climbs out over (m).
33    pub climb_height: f32,
34    /// Seconds a climb out takes.
35    pub climb_time: f32,
36}
37
38impl Default for Swimmer {
39    fn default() -> Self {
40        Self {
41            swim_depth: 0.7,
42            swim_speed: 2.0,
43            sprint_speed: 3.5,
44            stroke_accel: 3.0,
45            water_drag: 1.5,
46            float_height: 0.12,
47            breath: 30.0,
48            climb_height: 0.9,
49            climb_time: 0.6,
50        }
51    }
52}

58/// Whether to swim with the surface `depth` metres above the feet of a
59/// character `height` tall: in past `swim_depth`, out again only a little
60/// shallower so it doesn't flicker at the threshold.
61pub fn swims(s: &Swimmer, swimming: bool, depth: f32, height: f32) -> bool {
62    let enter = s.swim_depth.clamp(0.2, 1.5) * height;
63    if swimming { depth > enter - 0.15 * height } else { depth > enter }
64}
65
66/// The stroke velocity: `input` (x right, y forward, -1..1) along `look`
67/// (diving when it looks down), plus `rise` (-1 sink .. 1 rise), at
68/// `speed`.
69pub fn stroke(look: Quat, input: glam::Vec2, rise: f32, speed: f32) -> Vec3 {
70    let wish = look * Vec3::X * input.x + look * Vec3::NEG_Z * input.y + Vec3::Y * rise;
71    wish.clamp_length_max(1.0) * speed
72}
73
74/// Vertical acceleration that floats the eyes `float_height` above the
75/// surface: a damped spring on the eye height, by the character's buoyancy
76/// (none once diving takes it under).
77pub fn float_accel(s: &Swimmer, eye_y: f32, surface: f32, vel_y: f32) -> f32 {
78    let error = surface + s.float_height - eye_y;
79    // Critically damped at about 1.5 Hz: settles without bobbing for long.
80    let w = std::f32::consts::TAU * 1.5;
81    w * w * error.clamp(-1.0, 1.0) - 2.0 * w * vel_y
82}

84/// A climb out in progress: from `from` to `to` (feet, world) over the
85/// swimmer's `climb_time`.
86#[derive(Clone, Copy, Debug, PartialEq)]
87pub struct Climb {
88    /// Climb start (feet, world).
89    pub from: Vec3,
90    /// Climb end (feet, world).
91    pub to: Vec3,
92    /// Progress 0..1 through the climb.
93    pub t: f32,
94}
95
96impl Climb {
97    /// Feet position at progress `t` (0..1): up over the edge first, then in.
98    pub fn at(&self, t: f32) -> Vec3 {
99        let t = t.clamp(0.0, 1.0);
100        let up = (t / 0.6).min(1.0);
101        let over = ((t - 0.4) / 0.6).clamp(0.0, 1.0);
102        let ease = |x: f32| x * x * (3.0 - 2.0 * x);
103        Vec3::new(
104            self.from.x + (self.to.x - self.from.x) * ease(over),
105            self.from.y + (self.to.y - self.from.y) * ease(up),
106            self.from.z + (self.to.z - self.from.z) * ease(over),
107        )
108    }
109}
110
111/// Breath left after `dt` with the head under (`under`) or out: it runs out
112/// at one second per second and comes back three times as fast.
113pub fn breathe(s: &Swimmer, breath: f32, under: bool, dt: f32) -> f32 {
114    if under { (breath - dt).max(0.0) } else { (breath + 3.0 * dt).min(s.breath.max(0.0)) }
115}

117#[cfg(test)]
118mod tests {
119    use super::*;
120
121    #[test]
122    fn swims_past_chest_depth_with_some_hysteresis() {
123        let s = Swimmer::default();
124        assert!(!swims(&s, false, 1.0, 1.8), "waist deep: wades");
125        assert!(swims(&s, false, 1.4, 1.8), "chest deep: swims");
126        assert!(swims(&s, true, 1.2, 1.8), "keeps swimming just under the threshold");
127        assert!(!swims(&s, true, 0.8, 1.8), "stands up in the shallows");
128    }
129
130    #[test]
131    fn strokes_dive_where_it_looks_and_rise_on_jump() {
132        let down = Quat::from_rotation_x(-0.6);
133        let v = stroke(down, glam::Vec2::new(0.0, 1.0), 0.0, 2.0);
134        assert!(v.y < -0.5 && v.z < -1.0 && (v.length() - 2.0).abs() < 1e-4, "{v}");
135        let up = stroke(Quat::IDENTITY, glam::Vec2::ZERO, 1.0, 2.0);
136        assert!((up - Vec3::Y * 2.0).length() < 1e-4);
137    }
138
139    #[test]
140    fn floats_with_the_eyes_just_out_of_the_water() {
141        let s = Swimmer::default();
142        let (mut eye, mut v) = (-1.0f32, 0.0f32);
143        for _ in 0..600 {
144            v += float_accel(&s, eye, 0.0, v) / 120.0;
145            eye += v / 120.0;
146        }
147        assert!((eye - s.float_height).abs() < 0.01 && v.abs() < 0.01, "{eye} {v}");
148    }
149
150    #[test]
151    fn breath_runs_out_under_water_and_comes_back() {
152        let s = Swimmer { breath: 10.0, ..Default::default() };
153        let mut b = s.breath;
154        for _ in 0..12 {
155            b = breathe(&s, b, true, 1.0);
156        }
157        assert_eq!(b, 0.0);
158        b = breathe(&s, b, false, 1.0);
159        assert!((b - 3.0).abs() < 1e-5);
160        for _ in 0..10 {
161            b = breathe(&s, b, false, 1.0);
162        }
163        assert_eq!(b, s.breath);
164    }
165
166    #[test]
167    fn a_climb_goes_up_then_over() {
168        let c = Climb { from: Vec3::ZERO, to: Vec3::new(0.0, 1.0, -1.0), t: 0.0 };
169        let mid = c.at(0.4);
170        assert!(mid.y > 0.6 && mid.z.abs() < 1e-4, "up first: {mid}");
171        assert!((c.at(1.0) - c.to).length() < 1e-5);
172    }
