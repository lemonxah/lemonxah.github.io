13/// An [`Interactable`] was used.
14#[derive(Clone, Copy, Debug)]
15pub struct InteractEvent {
16    /// The interactable that was used.
17    pub object: ObjectId,
18    /// The player who used it.
19    pub by: ObjectId,
20    /// Total uses so far.
21    pub uses: u32,

24/// How an [`Interactable`] reacts to being used.
25#[derive(Clone, Copy, PartialEq, Eq, Debug, Default, Serialize, Deserialize)]
26pub enum InteractMode {
27    /// Every use runs `on_interact` (a button, a pickup, a door that only opens).
28    #[default]
29    Trigger,
30    /// Every use flips `on`: `on_on` runs when it turns on, `on_off` when it
31    /// turns off (a lamp, a lever, a gate).
32    Toggle,
33    /// Use = press (`on_press`), `request_release` = release (`on_release`):
34    /// a held lever, a pressure plate someone stands on.
35    Hold,
36}
37
38impl InteractMode {
39    /// Every mode, in declaration order.
40    pub const ALL: [InteractMode; 3] = [Self::Trigger, Self::Toggle, Self::Hold];
41
42    /// Inspector label of this mode.
43    pub fn label(self) -> &'static str {
44        match self {
45            Self::Trigger => "Trigger",
46            Self::Toggle => "Toggle",
47            Self::Hold => "Hold",
48        }
49    }
50}

52/// Something the player can use: a switch, a door, a chest. Used by
53/// [`ClickToMove`] (walk there and use it) or from code with
54/// `ctx.edit::<Interactable>(id, |i| i.request(user))`.
55///
56/// What a use does is data: the `on_*` lists call exposed functions on any
57/// component of any object (see [`crate::actions`]), with the user as the
58/// instigator. The older `toggle` reference still works next to them.
59///
60/// Networking: with a `Network Variables` component on the same object the
61/// toggle state `on` is replicated (key `"on"`), and the actions run only on
62/// the peer with authority over the object; other peers mirror `on`, and a use
63/// or release requested on another peer is sent to the authority as an RPC
64/// (the user's object id travels with it, so the actions' instigator is right).
65/// Without one, every peer runs its own uses locally as before.
66#[derive(Serialize, Deserialize)]
67#[serde(default)]
68pub struct Interactable {
69    /// The prompt shown while in range.
70    pub prompt: String,
71    /// How close the user must be.
72    pub range: f32,
73    /// How the interactable reacts to being used.
74    pub mode: InteractMode,
75    /// Toggle this object's active state on use (open a gate, switch a lamp).
76    pub toggle: ObjectRef,
77    /// Re-bake navmeshes after toggling (gates that open new paths).
78    pub rebake_navmesh: bool,
79    /// Sound played on use (empty = silent).
80    pub sound: String,
81    /// Seconds before it can be used again.
82    pub cooldown: f32,
83    /// Only usable once.
84    pub single_use: bool,
85    /// Toggle mode: the current state (saved, so a lamp stays lit).
86    pub on: bool,
87    /// Runs on every use (the press, in Hold mode).
88    pub on_interact: EventActions,
89    /// Toggle mode: runs when a use turns it on.
90    pub on_on: EventActions,
91    /// Toggle mode: runs when a use turns it off.
92    pub on_off: EventActions,
93    /// Hold mode: runs on the press.
94    pub on_press: EventActions,
95    /// Hold mode: runs on the release.
96    pub on_release: EventActions,
97    /// Total uses so far (runtime).
98    #[serde(skip)]
99    pub uses: u32,
100    /// A use was requested by this player (runtime).
101    #[serde(skip)]
102    pub requested: Option<ObjectId>,
103    /// A release was requested by this player (runtime).
104    #[serde(skip)]
105    pub release_requested: Option<ObjectId>,
106    /// Hold mode: pressed and not yet released.
107    #[serde(skip)]
108    pub held: bool,
109    /// Cooldown remaining (runtime).
110    #[serde(skip)]
111    pub cooldown_left: f32,
112}

140impl Interactable {
141    /// Ask to be used by `by` (applied on this component's next update).
142    pub fn request(&mut self, by: ObjectId) {
143        self.requested = Some(by);
144    }
145
146    /// Hold mode: `by` let go (applied on the next update).
147    pub fn request_release(&mut self, by: ObjectId) {
148        self.release_requested = Some(by);
149    }
150
151    /// Whether the interactable can be used right now (cooldown over, not
152    /// spent if single-use).
153    pub fn available(&self) -> bool {
154        self.cooldown_left <= 0.0 && !(self.single_use && self.uses > 0)
155    }
156
157    /// Whether `on` is replicated and actions are authority-only here.
158    fn networked(ctx: &EcsMoveHost<'_, '_>) -> bool {
159        ctx.connected() && ctx.world.get::<NetworkVariables>(ctx.owner).is_some()
160    }
161
162    /// Hold mode: `by` let go. Runs on the peer that has authority.
163    fn release(&mut self, ctx: &mut EcsMoveHost<'_, '_>, by: ObjectId) {
164        if !self.held {
165            return;
166        }
167        let me = ctx.self_id();
168        self.held = false;
169        let payload = [("instigator", ArgValue::Object(by.into())), ("object", ArgValue::Object(me.into()))];
170        queue_actions(&self.on_release, &FireSource { source: me, instigator: Some(by), payload: &payload }, ctx.commands);
171    }
172
173    /// `by` uses this. Runs on the peer that has authority.
174    fn use_by(&mut self, ctx: &mut EcsMoveHost<'_, '_>, by: ObjectId, networked: bool) {
175        let me = ctx.self_id();
176        if !self.available() {
177            return;
178        }
179        self.uses += 1;
180        self.cooldown_left = self.cooldown;
181        if self.toggle.is_set() {
182            let on = ctx.is_active(self.toggle);
183            ctx.set_active(self.toggle, !on);
184            if self.rebake_navmesh {
185                ctx.rebake_navmesh();
186            }
187        }
188        if !self.sound.is_empty() {
189            let at = ctx.self_transform().translation;
190            ctx.play_sound_at(self.sound.clone(), at, 1.0);
191        }
192        if self.mode == InteractMode::Toggle {
193            self.on = !self.on;
194            if networked {
195                ctx.set_net_var_self("on", self.on);
196            }
197        }
198        if self.mode == InteractMode::Hold {
199            self.held = true;
200        }
201        let payload = [
202            ("instigator", ArgValue::Object(by.into())),
203            ("object", ArgValue::Object(me.into())),
204            ("uses", ArgValue::Int(self.uses as i64)),
205            ("on", ArgValue::Bool(self.on)),
206        ];
207        let src = FireSource { source: me, instigator: Some(by), payload: &payload };
208        queue_actions(&self.on_interact, &src, ctx.commands);
209        match self.mode {
210            InteractMode::Trigger => {}
211            InteractMode::Toggle => queue_actions(if self.on { &self.on_on } else { &self.on_off }, &src, ctx.commands),
212            InteractMode::Hold => queue_actions(&self.on_press, &src, ctx.commands),
213        }
214        ctx.emit(InteractEvent { object: me, by, uses: self.uses });
215    }
216}

218/// RPC names a non-authority peer's use / release travels under (the argument
219/// is the user's object id as 16 big-endian bytes), sent to the authority.
220const RPC_USE: &str = "zestengine.interact";
221const RPC_RELEASE: &str = "zestengine.interact_release";
222
223static INTERACTABLE_ACTIONS: &[ActionDesc] = &[
224    ActionDesc::new("Interact", "Interact", &[], true),
225    ActionDesc::new("Release", "Release", &[], true),
226];
227static INTERACTABLE_PROPERTIES: &[PropertyDesc] = &[
228    PropertyDesc::new("on", ActionType::Bool),
229    PropertyDesc::new("held", ActionType::Bool),
230    PropertyDesc::new("uses", ActionType::Int),
231];
232
233impl TypedComponent for Interactable {
234    const NAME: &'static str = "Interactable";
235
236    fn actions(&self) -> &'static [ActionDesc] {
237        INTERACTABLE_ACTIONS
238    }
239    fn invoke_action(&mut self, ctx: &mut ActionCtx, name: &str, _args: &[ArgValue]) -> Result<(), String> {
240        let by = ctx.instigator.unwrap_or(ctx.self_id);
241        match name {
242            "Interact" => self.request(by),
243            "Release" => self.request_release(by),
244            _ => return Err(crate::actions::unknown_action(Self::NAME, name)),
245        }
246        Ok(())
247    }
248    fn properties(&self) -> &'static [PropertyDesc] {
249        INTERACTABLE_PROPERTIES
250    }
251    fn read_property(&self, name: &str) -> Option<ArgValue> {
252        match name {
253            "on" => Some(ArgValue::Bool(self.on)),
254            "held" => Some(ArgValue::Bool(self.held)),
255            "uses" => Some(ArgValue::Int(self.uses as i64)),
256            _ => None,
257        }
258    }
259}

261impl Interactable {
262    fn update_native(&mut self, ctx: &mut EcsMoveHost<'_, '_>) {
263        self.cooldown_left = (self.cooldown_left - ctx.dt).max(0.0);
264        let me = ctx.self_id();
265        let networked = Self::networked(ctx);
266        let authority = !networked || ctx.has_authority(me);
267        if networked
268            && !authority
269            && let Some(on) = ctx.world.get::<NetworkVariables>(ctx.owner).and_then(|vars| match vars.get("on") { Some(NetValue::Bool(value)) => Some(*value), _ => None })
270        {
271            self.on = on;
272        }
273        if networked && !authority {
274            // A use by this peer goes to the authority, which runs the actions
275            // and replicates `on`; it is not answered locally.
276            if let Some(by) = self.release_requested.take() {
277                ctx.rpc_self(RPC_RELEASE, vec![NetValue::Bytes(by.raw().to_be_bytes().to_vec())], RpcTarget::Owner);
278            }
279            if let Some(by) = self.requested.take() {
280                ctx.rpc_self(RPC_USE, vec![NetValue::Bytes(by.raw().to_be_bytes().to_vec())], RpcTarget::Owner);
281            }
282            return;
283        }
284        if let Some(by) = self.release_requested.take() {
285            self.release(ctx, by);
286        }
287        if let Some(by) = self.requested.take() {
288            self.use_by(ctx, by, networked);
289        }
290        if networked {
291            // Uses other peers asked for, in arrival order.
292            let asks: Vec<(bool, ObjectId)> = ctx
293                .rpcs_for_self()
294                .filter(|r| r.from != ctx.local_peer() && (r.name == RPC_USE || r.name == RPC_RELEASE))
295                .filter_map(|r| match r.args.first() {
296                    Some(NetValue::Bytes(b)) => Some((r.name == RPC_RELEASE, ObjectId::from_raw(u128::from_be_bytes(b.as_slice().try_into().ok()?)))),
297                    _ => None,
298                })
299                .collect();
300            for (release, by) in asks {
301                if release {
302                    self.release(ctx, by);
303                } else {
304                    self.use_by(ctx, by, networked);
305                }
306            }
307        }
308    }

312/// Click the ground to walk there over the navmesh (needs a [`NavAgent`] on
313/// the same object and a baked `NavMesh Surface` in the scene); click an
314/// [`Interactable`] to walk up to it and use it. Shows `marker` at the target.
315#[derive(Serialize, Deserialize)]
316#[serde(default)]
317pub struct ClickToMove {
318    /// Input action for clicking.
319    pub click_action: String,
320    /// Keep re-targeting while the button is held.
321    pub hold_to_move: bool,
322    /// Click-target marker shown at the world destination, including under a parent.
323    pub marker: ObjectRef,
324    /// How far a click ray may travel (m).
325    pub max_ray_distance: f32,
326    /// The clicked destination (runtime).
327    #[serde(skip)]
328    pub destination: Option<Vec3>,
329    /// The interactable being walked to (runtime).
330    #[serde(skip)]
331    pub pending_interact: Option<ObjectId>,
332    /// A Hold-mode Interactable this component pressed; released when the
333    /// click button is up (the frame after the press at the earliest).
334    #[serde(skip)]
335    pub held: Option<ObjectId>,
336    /// Seconds between destination updates while the button is held (runtime).
337    #[serde(skip)]
338    pub retarget: f32,
339    /// This instance has initialized its marker (runtime).
340    #[serde(skip)]
341    pub started: bool,
342}

344impl Default for ClickToMove {
345    fn default() -> Self {
346        Self {
347            click_action: "Click".into(),
348            hold_to_move: true,
349            marker: ObjectRef::NONE,
350            max_ray_distance: 500.0,
351            destination: None,
352            pending_interact: None,
353            held: None,
354            retarget: 0.0,
355            started: false,
356        }
357    }
358}

360impl ClickToMove {
361    /// The held Interactable to let go of now: the click button is up.
362    pub(crate) fn take_release(&mut self, button_down: bool) -> Option<ObjectId> {
363        if button_down { None } else { self.held.take() }
364    }
365
366    fn click(&mut self, ctx: &mut EcsMoveHost<'_, '_>) {
367        let Some((origin, dir)) = ctx.pointer_ray() else {
368            return;
369        };
370        let Some(hit) = ctx.raycast(origin, dir, self.max_ray_distance) else {
371            return;
372        };
373        let interactable = ctx.get::<Interactable>(hit.object).map(|i| ctx.position_of(hit.object.into()).map(|p| (p, i.range)));
374        let (dest, marker_at) = match interactable {
375            Some(Some((pos, range))) => {
376                self.pending_interact = Some(hit.object);
377                let me = ctx.self_transform().translation;
378                let back = (me - pos).with_y(0.0).normalize_or(Vec3::Z) * (range * 0.6);
379                let stand = ctx.nearest_walkable(pos + back).unwrap_or(pos + back);
380                (stand, pos)
381            }
382            _ => {
383                self.pending_interact = None;
384                let p = ctx.nearest_walkable(hit.point).unwrap_or(hit.point);
385                (p, hit.point)
386            }
387        };
388        self.destination = Some(dest);
389        ctx.set_destination(ctx.self_id(), dest);
390        if self.marker.is_set() {
391            ctx.set_world_transform_of(self.marker, Some(marker_at + Vec3::Y * 0.03), None);
392            ctx.set_active(self.marker, true);
393        }
394    }
395}

599pub(crate) fn register_systems(registry: &mut crate::ComponentRegistry) {
600    registry.register_ecs::<ClickToMove>();
601    registry.register_ecs::<Interactable>();
602    registry.register_system_if("zestengine::click_start", Stage::Start, initialize, |world|
603        world.query::<&ClickToMove>().iter().any(|(_, value)| !value.started));
604    registry.register_system_for::<ClickToMove>("zestengine::click_to_move", Stage::Gameplay, move_and_use);
605    registry.register_system_for::<Interactable>("zestengine::interactions", Stage::Gameplay, interact);
606}

608fn initialize(world: &SystemWorld<'_>, context: &mut SystemContext<'_>) {
609    let mut query = world.query::<(&Owner, &mut ClickToMove)>();
610    let mut actors: Vec<_> = query.iter().filter(|(_, (owner, value))| !value.started && context.is_active(world, owner.0)).collect();
611    actors.sort_unstable_by_key(|(instance, (owner, _))| context.instance_order::<ClickToMove>(world, owner.0, *instance));
612    for (_, (owner, value)) in actors { value.start_native(&mut EcsMoveHost::new(world, owner.0, context)); }
613}

615fn move_and_use(world: &SystemWorld<'_>, context: &mut SystemContext<'_>) {
616    let mut query = world.query::<(&Owner, &mut ClickToMove)>();
617    let mut actors: Vec<_> = query.iter().filter(|(_, (owner, _))| context.is_active(world, owner.0)).collect();
618    actors.sort_unstable_by_key(|(instance, (owner, _))| context.instance_order::<ClickToMove>(world, owner.0, *instance));
619    for (_, (owner, value)) in actors { value.update_native(&mut EcsMoveHost::new(world, owner.0, context)); }
620}

622fn interact(world: &SystemWorld<'_>, context: &mut SystemContext<'_>) {
623    let mut query = world.query::<(&Owner, &mut Interactable)>();
624    let mut actors: Vec<_> = query.iter().filter(|(_, (owner, _))| context.is_active(world, owner.0)).collect();
625    actors.sort_unstable_by_key(|(instance, (owner, _))| context.instance_order::<Interactable>(world, owner.0, *instance));
626    for (_, (owner, value)) in actors { value.update_native(&mut EcsMoveHost::new(world, owner.0, context)); }
627}

629impl TypedComponent for ClickToMove {
630    const NAME: &'static str = "Click To Move";
631}
632
633impl ClickToMove {
634    fn start_native(&mut self, ctx: &mut EcsMoveHost<'_, '_>) {
635        if self.marker.is_set() {
636            ctx.set_active(self.marker, false);
637        }
638        self.started = true;
639    }
640
641    fn update_native(&mut self, ctx: &mut EcsMoveHost<'_, '_>) {
642        if !self.started { self.start_native(ctx); }
643        if ctx.dt <= 0.0 {
644            return;
645        }
646        self.retarget -= ctx.dt;
647        if let Some(target) = self.take_release(ctx.input.down(&self.click_action)) {
648            let by = ctx.self_id();
649            ctx.edit::<Interactable>(target, move |i| i.request_release(by));
650        }
651        let clicked = !ctx.pointer_over_ui()
652            && (ctx.input.pressed(&self.click_action)
653                || (self.hold_to_move && ctx.input.down(&self.click_action) && self.retarget <= 0.0));
654        if clicked {
655            self.retarget = 0.15;
656            self.click(ctx);
657            // The sibling agent sees its native destination immediately.
658            return;
659        }
660        let me = ctx.self_transform().translation;
661        if let Some(target) = self.pending_interact {
662            let range = ctx.get::<Interactable>(target).map(|i| i.range);
663            match (range, ctx.position_of(target.into())) {
664                (Some(range), Some(pos)) if me.with_y(0.0).distance(pos.with_y(0.0)) <= range => {
665                    let by = ctx.self_id();
666                    ctx.edit::<Interactable>(target, move |i| i.request(by));
667                    if ctx.get::<Interactable>(target).is_some_and(|i| i.mode == InteractMode::Hold) {
668                        self.held = Some(target);
669                    }
670                    ctx.stop_agent(ctx.self_id());
671                    self.pending_interact = None;
672                    self.destination = None;
673                }
674                (None, _) | (_, None) => self.pending_interact = None,
675                _ => {}
676            }
677        }
678        let arrived = ctx.world.get::<NavAgent>(ctx.owner).is_none_or(|a| a.arrived());
679        if let Some(d) = self.destination
680            && (arrived || me.with_y(0.0).distance(d.with_y(0.0)) < 0.3)
681        {
682            self.destination = None;
683        }
684        if self.destination.is_none() && self.pending_interact.is_none() && self.marker.is_set() && ctx.is_active(self.marker) {
685            ctx.set_active(self.marker, false);
686        }
687    }
