18/// Share of a hand's speed through the water a stroke turns into the
19/// swimmer's speed (the other way).
20const ARM_STROKE: f32 = 0.6;
21
22/// How a VR player moves through the world.
23#[derive(Clone, Copy, PartialEq, Eq, Debug, Serialize, Deserialize)]
24pub enum VrLocomotion {
25    /// Physical room-scale movement only.
26    Disabled,
27    /// Point and teleport.
28    Teleport,
29    /// Smooth stick movement.
30    Smooth,
31    /// Both methods available.
32    Both,
33}
34
35impl VrLocomotion {
36    /// Every mode, in declaration order.
37    pub const ALL: [VrLocomotion; 4] = [Self::Disabled, Self::Teleport, Self::Smooth, Self::Both];
38    /// Inspector label of this mode.
39    pub fn label(self) -> &'static str {
40        match self {
41            Self::Disabled => "Room-scale only",
42            Self::Teleport => "Teleport",
43            Self::Smooth => "Smooth",
44            Self::Both => "Teleport + smooth",
45        }
46    }
47    fn teleport(self) -> bool {
48        matches!(self, Self::Teleport | Self::Both)
49    }
50    fn smooth(self) -> bool {
51        matches!(self, Self::Smooth | Self::Both)
52    }
53}

74/// Whether a teleport a client announces lands somewhere a body can stand: clear head room above
75/// the target and ground within reach below it. `hit(origin, dir, len)` says whether a ray meets
76/// anything solid. A server runs this against the owner's claimed target (NET-4); the owner's
77/// own arc only ever lands on a surface, so an honest teleport passes.
78pub fn teleport_landing_ok(target: Vec3, hit: impl Fn(Vec3, Vec3, f32) -> bool) -> bool {
79    target.is_finite() && !hit(target + Vec3::Y * 0.05, Vec3::Y, 1.2) && hit(target + Vec3::Y * 0.4, Vec3::NEG_Y, 1.2)
80}

82/// Where a ballistic teleport arc from `origin` along `dir` lands. `cast`
83/// is a ray query `(origin, unit dir, max distance) -> (point, normal)`. Only
84/// walkable (upward-facing) surfaces are accepted.
85pub fn teleport_arc(
86    origin: Vec3,
87    dir: Vec3,
88    speed: f32,
89    max_steps: usize,
90    mut cast: impl FnMut(Vec3, Vec3, f32) -> Option<(Vec3, Vec3)>,
91) -> Option<Vec3> {
92    const STEP: f32 = 0.05;
93    let gravity = Vec3::new(0.0, -9.81, 0.0);
94    let mut p = origin;
95    let mut v = dir.normalize_or(Vec3::NEG_Z) * speed;
96    for _ in 0..max_steps {
97        let next = p + v * STEP;
98        v += gravity * STEP;
99        let seg = next - p;
100        let len = seg.length();
101        if len > 1e-5
102            && let Some((point, normal)) = cast(p, seg / len, len)
103        {
104            return (normal.y > 0.7).then_some(point);
105        }
106        p = next;
107    }
108    None
109}

111/// The active scheme is a VR controller scheme that binds `action`. When it
112/// isn't (a desktop scheme left active, an older project) the VR Character
113/// reads the raw controller input that action defaults to.
114fn vr_bound(ctx: &EcsMoveHost<'_, '_>, action: &str) -> bool {
115    ctx.control_scheme().is_some_and(|s| s.device() == Some(crate::InputDevice::Vr) && s.action(action).is_some())
116}

118fn vr_button(ctx: &EcsMoveHost<'_, '_>, action: &str, raw: bool) -> bool {
119    if vr_bound(ctx, action) { ctx.input.down(action) } else { raw }
120}

122/// One frame of a VR Character's locomotion input as the rig read it (stick,
123/// turn, headset pose), so networked prediction can replay it without the
124/// headset.
125#[derive(Clone, Copy, Debug, Default, PartialEq)]
126#[doc(hidden)]
127pub struct VrStride {
128    /// The play space's rotation before the frame's turn.
129    pub rig: Quat,
130    pub stick: glam::Vec2,
131    /// Degrees the play space turns about the head.
132    pub turn: f32,
133    /// Head pose in stage space.
134    pub head: (Vec3, Quat),
135    /// Hands in stage space and their world velocity, while there is water
136    /// at the head (arm strokes).
137    pub hands: Option<[(Vec3, Vec3); 2]>,
138    /// Feet target of a teleport this frame.
139    pub teleport: Option<Vec3>,
140}
141
142const STRIDE_HANDS: u32 = 1;
143const STRIDE_TELEPORT: u32 = 2;
144
145impl VrStride {
146    /// Flat form carried by an input command.
147    pub fn pack(&self) -> Vec<f32> {
148        let flags = if self.hands.is_some() { STRIDE_HANDS } else { 0 } | if self.teleport.is_some() { STRIDE_TELEPORT } else { 0 };
149        let ((p, q), r) = (self.head, self.rig);
150        let mut v = vec![flags as f32, self.stick.x, self.stick.y, self.turn, p.x, p.y, p.z, q.x, q.y, q.z, q.w, r.x, r.y, r.z, r.w];
151        if let Some(hands) = self.hands {
152            for (pos, vel) in hands {
153                v.extend_from_slice(&[pos.x, pos.y, pos.z, vel.x, vel.y, vel.z]);
154            }
155        }
156        if let Some(t) = self.teleport {
157            v.extend_from_slice(&[t.x, t.y, t.z]);
158        }
159        v
160    }
161
162    pub fn unpack(v: &[f32]) -> Option<Self> {
163        fn vec3(s: &[f32]) -> Vec3 {
164            Vec3::new(s[0], s[1], s[2])
165        }
166        if v.iter().any(|x| !x.is_finite()) { return None; }
167        let (&flags, rest) = v.split_first()?;
168        let flags = flags as u32;
169        let (fixed, mut tail) = rest.split_at_checked(14)?;
170        let mut take = |n: usize| {
171            let (a, b) = tail.split_at_checked(n)?;
172            tail = b;
173            Some(a)
174        };
175        let hands = if flags & STRIDE_HANDS != 0 {
176            let h = take(12)?;
177            Some([(vec3(&h[0..]), vec3(&h[3..])), (vec3(&h[6..]), vec3(&h[9..]))])
178        } else {
179            None
180        };
181        let teleport = if flags & STRIDE_TELEPORT != 0 { Some(vec3(take(3)?)) } else { None };
182        let rot = Quat::from_xyzw(fixed[6], fixed[7], fixed[8], fixed[9]);
183        let rig = Quat::from_xyzw(fixed[10], fixed[11], fixed[12], fixed[13]);
184        (rot.length_squared() > 0.5 && rig.length_squared() > 0.5).then(|| Self {
185            rig: rig.normalize(),
186            stick: glam::Vec2::new(fixed[0], fixed[1]),
187            turn: fixed[2],
188            head: (vec3(&fixed[3..]), rot.normalize()),
189            hands,
190            teleport,
191        })
192    }
193}

392    /// Turn the play space by `delta` degrees about a world pivot (the head).
393    fn turn(host: &mut dyn MoveHost, pivot: Vec3, delta: f32) {
394        let r = Quat::from_rotation_y(delta.to_radians());
395        let world = host.world();
396        let origin = world.w_axis.truncate();
397        host.set_world_position(pivot + r * (origin - pivot));
398        let rotation = r * world.to_scale_rotation_translation().1;
399        let local = host.parent_world().map_or(rotation, |parent| parent.to_scale_rotation_translation().1.inverse() * rotation);
400        host.set_rotation(local.normalize());
401    }

403    /// Head / hand poses in the rig's local (stage) space.
404    fn stage_poses(&self, ctx: &EcsMoveHost<'_, '_>) -> [(Vec3, Quat); 3] {
405        let vr = ctx.vr_frame();
406        if vr.active {
407            let head = vr.head.unwrap_or((Vec3::new(0.0, self.eye_height, 0.0), Quat::IDENTITY));
408            let hand = |h: Option<(Vec3, Quat)>, side: f32| {
409                h.unwrap_or((head.0 + head.1 * Vec3::new(0.2 * side, -0.35, -0.3), head.1))
410            };
411            [head, hand(vr.left_hand, -1.0), hand(vr.right_hand, 1.0)]
412        } else {
413            let head_rot = Quat::from_rotation_x(self.pitch.to_radians());
414            let head = Vec3::new(0.0, self.eye_height, 0.0);
415            [
416                (head, head_rot),
417                (head + head_rot * Vec3::new(-0.22, -0.28, -0.38), head_rot),
418                (head + head_rot * Vec3::new(0.2, -0.2, -0.42), head_rot),
419            ]
420        }
421    }

423    /// Swim when the water is past the chest (see [`super::Swimmer`], tuned
424    /// by one on this object): the stick strokes where the head looks,
425    /// pulling a hand back through the water pushes the other way, the rig
426    /// floats with the head just out, and breath runs out with it under.
427    /// True while swimming (the walk and teleport are skipped).
428    fn swim(&mut self, host: &mut dyn MoveHost, stride: &VrStride, dt: f32) -> bool {
429        let s = host.swimmer();
430        let rig = host.world();
431        let (head_local, head_rot) = stride.head;
432        let head = rig.transform_point3(head_local);
433        let feet = rig.w_axis.truncate();
434        let height = (head.y - feet.y).max(1.0) / 0.92;
435        let surface = host.water_surface(head);
436        self.swimming = surface.is_some_and(|h| swims(&s, self.swimming, h - feet.y, height));
437        let Some(h) = surface.filter(|_| self.swimming) else {
438            self.swim_vel = Vec3::ZERO;
439            self.breath = self.breath.map(|b| breathe(&s, b, false, dt)).filter(|b| *b < s.breath);
440            return false;
441        };
442        self.breath = Some(breathe(&s, self.breath.unwrap_or(s.breath), head.y < h, dt));
443        let look = rig.to_scale_rotation_translation().1 * head_rot;
444        let mut target = stroke(look, stride.stick, 0.0, s.swim_speed);
445        // Arm strokes: hands under the surface push against the water.
446        for (p, vel) in stride.hands.into_iter().flatten() {
447            if rig.transform_point3(p).y < h {
448                target -= (vel - self.swim_vel) * ARM_STROKE;
449            }
450        }
451        let target = target.clamp_length_max(s.sprint_speed);
452        let rate = if target.length_squared() > 1e-4 { s.stroke_accel } else { s.water_drag };
453        self.swim_vel = self.swim_vel.lerp(target, 1.0 - (-rate * dt).exp());
454        if stride.stick.length_squared() < 0.02 {
455            self.swim_vel.y += float_accel(&s, head.y, h, self.swim_vel.y) * dt;
456        }
457        if host.has_character() {
458            host.move_character(self.swim_vel * dt);
459            return true;
460        }
461        let mut origin = feet + self.swim_vel * dt;
462        // Not through the floor.
463        let filter = RayFilter { exclude: Some(host.self_id()), ..Default::default() };
464        if let Some(floor) = host.raycast(origin + Vec3::Y * 1.0, Vec3::NEG_Y, 1.2, &filter)
465            && origin.y < floor.point.y
466        {
467            origin.y = floor.point.y;
468            self.swim_vel.y = self.swim_vel.y.max(0.0);
469        }
470        host.set_world_position(origin);
471        true
472    }

474    /// The movement part of a frame: turn, then swim or walk with the stick.
475    /// Live frames and networked prediction's replays both run this. True
476    /// while swimming.
477    fn stride(&mut self, host: &mut dyn MoveHost, stride: &VrStride, dt: f32) -> bool {
478        // The owner's facing, like a first-person view: a replay adopts it.
479        host.set_rotation(stride.rig);
480        let rig = host.world();
481        let head_world = rig.transform_point3(stride.head.0);
482        if stride.turn != 0.0 {
483            Self::turn(host, head_world, stride.turn);
484        }
485        // The body follows tracking; the camera and hands remain exact tracked poses.
486        if self.track_capsule && let Some((radius, _, _)) = host.capsule() {
487            let head = stride.head.0;
488            let height = (head.y + 0.12).clamp(radius * 2.0, 3.5_f32.max(radius * 2.0));
489            host.set_capsule(height, [head.x, height * 0.5, head.z]);
490        }
491        if self.swim(host, stride, dt) {
492            return true;
493        }
494        let f = (host.world().to_scale_rotation_translation().1 * stride.head.1) * Vec3::NEG_Z;
495        let head_yaw = Quat::from_rotation_y((-f.x).atan2(-f.z));
496        let mv = if self.locomotion.smooth() && stride.stick.length_squared() > 0.02 {
497            stride.stick.clamp_length_max(1.0)
498        } else { glam::Vec2::ZERO };
499        let dir = head_yaw * Vec3::new(mv.x, 0.0, -mv.y);
500        self.walker.walk(host, Stride {
501            wish: dir * self.move_speed,
502            accel: self.acceleration * mv.length(), air_control: self.air_control,
503            braking: self.braking, jump: None, gravity: self.gravity,
504            swim: SwimInput::default(),
505        }, dt);
506        false
507    }

509    /// Snap the play space so the head stands over `target`.
510    fn teleport_to(host: &mut dyn MoveHost, head_local: Vec3, target: Vec3) {
511        let rig = host.world();
512        let head_offset = (rig.transform_point3(head_local) - rig.w_axis.truncate()).with_y(0.0);
513        host.set_world_position(target - head_offset);
514    }

516    fn turn_input(&mut self, rx: f32, dt: f32) -> f32 {
517        let mut turn = 0.0;
518        if self.turning == VrTurning::Disabled {
519            self.turn_latched = false;
520        } else if self.turning == VrTurning::Snap && self.snap_turn > 0.0 {
521            if rx.abs() > 0.7 && !self.turn_latched {
522                self.turn_latched = true;
523                turn = -rx.signum() * self.snap_turn;
524            } else if rx.abs() < 0.3 {
525                self.turn_latched = false;
526            }
527        } else if rx.abs() > 0.15 {
528            turn = -rx * self.smooth_turn_speed * dt;
529        }
530        turn
531    }

533    fn locomotion(&mut self, ctx: &mut EcsMoveHost<'_, '_>, dt: f32) {
534        let vr = ctx.vr_frame();
535        let (mv, turn, teleport_down) = if vr.active {
536            let rx = if vr_bound(ctx, &self.turn_action) { ctx.input.axis(&self.turn_action) } else { vr.right_stick.x };
537            let turn = self.turn_input(rx, dt);
538            let mv = if vr_bound(ctx, &self.move_action) { ctx.input.axis2(&self.move_action) } else { vr.left_stick };
539            (mv, turn, vr_button(ctx, &self.teleport_action, vr.right_stick.y > 0.6))
540        } else {
541            let look = ctx.input.axis2(&self.look_action);
542            self.pitch = (self.pitch - look.y * self.look_sensitivity).clamp(-85.0, 85.0);
543            (ctx.input.axis2(&self.move_action), -look.x * self.look_sensitivity, ctx.input.down(&self.teleport_action))
544        };
545        let poses = self.stage_poses(ctx);
546        let rig = Self::rig_world(ctx);
547        let wet = ctx.water_surface(rig.transform_point3(poses[0].0)).is_some();
548        let hands = wet.then(|| [(poses[1].0, self.hand_vel[0]), (poses[2].0, self.hand_vel[1])]);
549        let mut stride = VrStride { rig: ctx.rotation(), stick: mv, turn, head: poses[0], hands, teleport: None };
550        if self.stride(ctx, &stride, dt) {
551            self.last_stride = Some(stride);
552            return;
553        }
554
555        if self.locomotion.teleport() {
556            if teleport_down {
557                self.aiming = true;
558                let rig = Self::rig_world(ctx);
559                let (from, rot) = if vr.active { poses[2] } else { poses[0] };
560                let origin = rig.transform_point3(from);
561                let dir = rig.to_scale_rotation_translation().1 * rot * Vec3::NEG_Z;
562                let filter = RayFilter { exclude: Some(ctx.self_id()), ..Default::default() };
563                self.teleport_target = teleport_arc(origin, dir, self.teleport_speed, 60, |o, d, len| {
564                    ctx.raycast_filtered(o, d, len, &filter).map(|h| (h.point, h.normal))
565                });
566                if self.teleport_marker.is_set() {
567                    match self.teleport_target {
568                        Some(p) => {
569                            ctx.set_world_transform_of(self.teleport_marker, Some(p + Vec3::Y * 0.02), None);
570                            ctx.set_active(self.teleport_marker, true);
571                        }
572                        None => ctx.set_active(self.teleport_marker, false),
573                    }
574                }
575            } else if self.aiming {
576                self.aiming = false;
577                if let Some(target) = self.teleport_target.take() {
578                    Self::teleport_to(ctx, poses[0].0, target);
579                    self.walker = CharacterMovement { floor_y: ctx.translation().y, grounded: true, ..Default::default() };
580                    // Drop the pre-teleport movement queued earlier in this pass.
581                    let me = ctx.self_id();
582                    ctx.commands.retain(|c| !matches!(c, ComponentCommand::MoveCharacter { id, .. } if *id == me));
583                    if MoveHost::has_character(ctx) { ctx.move_character(Vec3::ZERO); }
584                    stride.teleport = Some(target);
585                }
586                if self.teleport_marker.is_set() {
587                    ctx.set_active(self.teleport_marker, false);
588                }
589            }
590        }
591        self.last_stride = Some(stride);
592    }
593
594    /// Pick up the nearest free [`Grabbable`] within reach of the hand, or
595    /// (desktop fallback, `look` = head ray) the one looked at within 2.5 m.
596    fn grab(&mut self, ctx: &mut EcsMoveHost<'_, '_>, hand: usize, hand_world: Mat4, look: Option<(Vec3, Vec3)>) -> Option<Held> {
597        const LOOK_REACH: f32 = 2.5;
598        let pos = hand_world.w_axis.truncate();
599        let taken: Vec<ObjectId> = self.held.iter().flatten().map(|h| h.id).collect();
600        let mut best: Option<(f32, ObjectId)> = None;
601        for id in ctx.objects_with::<Grabbable>() {
602            if taken.contains(&id) || !ctx.is_active(id) {
603                continue;
604            }
605            let (Some(g), Some(p)) = (ctx.get::<Grabbable>(id), ctx.position_of(id.into())) else {
606                continue;
607            };
608            if g.held { continue; }
609            let score = match look {
610                Some((o, d)) => {
611                    let t = (p - o).dot(d);
612                    let miss = (o + d * t.clamp(0.0, LOOK_REACH)).distance(p);
613                    (t > 0.0 && t <= LOOK_REACH && miss <= g.radius + 0.15).then_some(t)
614                }
615                None => {
616                    let d = p.distance(pos);
617                    (d <= self.grab_radius + g.radius).then_some(d)
618                }
619            };
620            if let Some(score) = score
621                && best.is_none_or(|(b, _)| score < b)
622            {
623                best = Some((score, id));
624            }
625        }
626        let (_, id) = best?;
627        let g = ctx.get::<Grabbable>(id)?;
628        let offset = if g.snap_to_hand {
629            let [x, y, z] = g.hold_rotation.map(f32::to_radians);
630            Mat4::from_rotation_translation(Quat::from_euler(EulerRot::XYZ, x, y, z), Vec3::from(g.hold_position))
631        } else {
632            let rel = hand_world.inverse() * ctx.matrix_of(id)?;
633            if look.is_some() {
634                // Desktop: bring it into the simulated hand instead of holding it at a distance.
635                let (_, rot, _) = rel.to_scale_rotation_translation();
636                Mat4::from_rotation_translation(rot, Vec3::new(0.0, 0.0, -0.12))
637            } else {
638                rel
639            }
640        };
641        drop(g);
642        ctx.edit::<Grabbable>(id, move |g| {
643            g.held = true;
644            g.held_by = Some(if hand == 0 { VrHand::Left } else { VrHand::Right });
645        });
646        if ctx.connected() {
647            ctx.request_ownership_of(id.into());
648        }
649        Some(Held { id, offset })
650    }
651
652    fn release(&mut self, ctx: &mut EcsMoveHost<'_, '_>, hand: usize) {
653        let Some(held) = self.held[hand].take() else {
654            return;
655        };
656        let throw = ctx.get::<Grabbable>(held.id).is_some_and(|g| g.throwable);
657        if throw {
658            ctx.set_velocity(held.id, self.hand_vel[hand]);
659        }
660        ctx.edit::<Grabbable>(held.id, |g| {
661            g.held = false;
662            g.held_by = None;
663            g.trigger_down = false;
664        });
665        if ctx.connected() {
666            ctx.commands.push(ComponentCommand::ReleaseOwnership(held.id));
667        }
668    }
669}

671impl Predictable for VrCharacter {
672    fn motion(&self) -> MotionState {
673        let mut m = MotionState::default();
674        self.walker.save(&mut m);
675        if self.swimming { m.vel = self.swim_vel; m.grounded = false; }
676        m.pitch = self.pitch;
677        m.swimming = self.swimming;
678        m.breath = self.breath;
679        m
680    }
681
682    fn set_motion(&mut self, m: &MotionState, _host: &mut dyn MoveHost) {
683        self.walker.load(m);
684        self.swim_vel = if m.swimming { m.vel } else { Vec3::ZERO };
685        self.pitch = m.pitch;
686        self.swimming = m.swimming;
687        self.breath = m.breath;
688    }
689
690    fn frame_extra(&self) -> Vec<f32> {
691        self.last_stride.map(|s| s.pack()).unwrap_or_default()
692    }
693
694    fn replay(&mut self, host: &mut dyn MoveHost, _input: &InputState, extra: &[f32], dt: f32) {
695        let dt = dt.min(0.1);
696        let Some(stride) = VrStride::unpack(extra).filter(|_| dt > 0.0) else {
697            return;
698        };
699        if self.stride(host, &stride, dt) {
700            return;
701        }
702        // The owner's teleport, if the arc could have reached it and it lands on open ground.
703        if let Some(target) = stride.teleport
704            && self.locomotion.teleport()
705            && (target - host.world_position()).with_y(0.0).length() <= self.teleport_speed * 3.0 + 1.0
706            && {
707                let filter = RayFilter { exclude: Some(host.self_id()), ..Default::default() };
708                teleport_landing_ok(target, |o, d, len| host.raycast(o, d, len, &filter).is_some())
709            }
710        {
711            Self::teleport_to(host, stride.head.0, target);
712            self.walker = CharacterMovement { floor_y: host.translation().y, grounded: true, ..Default::default() };
713            if host.has_character() { host.move_character(Vec3::ZERO); }
714        }
715    }
716}

730impl VrCharacter {
731    fn start_native(&mut self, ctx: &mut EcsMoveHost<'_, '_>) {
732        if !self.spawn_tag.is_empty() {
733            if let Some(t) = ctx.spawn_point_for(&self.spawn_tag, ctx.local_peer() as usize) {
734                ctx.set_world_position(t.translation);
735                let yaw = t.rotation.to_euler(EulerRot::YXZ).0;
736                ctx.set_world_rotation(Quat::from_rotation_y(yaw));
737            }
738        }
739        if self.head.is_set() && !ctx.driven_remotely() {
740            ctx.set_active_camera(self.head);
741        }
742        if !ctx.vr_frame().active && self.desktop_fallback && !ctx.driven_remotely() {
743            ctx.set_cursor_locked(true);
744        }
745        if self.teleport_marker.is_set() {
746            ctx.set_active(self.teleport_marker, false);
747        }
748        self.walker = CharacterMovement { floor_y: ctx.translation().y, grounded: true, ..Default::default() };
749        self.started = true;
750    }
751
752    fn update_native(&mut self, ctx: &mut EcsMoveHost<'_, '_>) {
753        if !self.started {
754            self.start_native(ctx);
755        }
756        self.last_stride = None;
757        if ctx.driven_remotely() {
758            return;
759        }
760        let vr = ctx.vr_frame();
761        self.vr_active = vr.active;
762        let menu = vr.active && !self.body_menu_action.is_empty() && vr_button(ctx, &self.body_menu_action, vr.menu);
763        if menu && !self.body_menu_prev {
764            ctx.commands.push(ComponentCommand::Body(crate::BodyCommand::Menu(None)));
765        }
766        self.body_menu_prev = menu;
767        if !vr.active && !self.desktop_fallback {
768            return;
769        }
770        let dt = ctx.dt.min(0.1);
771        if dt > 0.0 && !ctx.ui_text_editing() && !ctx.body_calibrating() {
772            self.locomotion(ctx, dt);
773        }
774
775        let poses = self.stage_poses(ctx);
776        for (target, (p, r)) in [self.head, self.left_hand, self.right_hand].into_iter().zip(poses) {
777            ctx.set_local_transform(target, Some(p), Some(r), None);
778        }
779        if ctx.body_calibrating() { return; }
780
781
782        let rig = Self::rig_world(ctx);
783        let grips = if vr.active {
784            [
785                vr_button(ctx, &self.grab_left_action, vr.left_grip > 0.6),
786                vr_button(ctx, &self.grab_right_action, vr.right_grip > 0.6),
787            ]
788        } else {
789            [false, ctx.input.down(&self.grab_action)]
790        };
791        let triggers = if vr.active {
792            [
793                vr_button(ctx, &self.use_left_action, vr.left_trigger),
794                vr_button(ctx, &self.use_right_action, vr.right_trigger),
795            ]
796        } else {
797            [false, ctx.input.down(&self.fire_action)]
798        };
799        for hand in 0..2 {
800            let (p, r) = poses[hand + 1];
801            let hand_world = rig * Mat4::from_rotation_translation(r, p);
802            let pos = hand_world.w_axis.truncate();
803            if let Some(prev) = self.hand_prev[hand]
804                && dt > 0.0
805            {
806                self.hand_vel[hand] = self.hand_vel[hand].lerp((pos - prev) / dt, 0.5);
807            }
808            self.hand_prev[hand] = Some(pos);
809
810            let grip = grips[hand];
811            if grip && !self.grip_prev[hand] && self.held[hand].is_none() {
812                let look = (!vr.active).then(|| {
813                    let (p, r) = poses[0];
814                    (rig.transform_point3(p), (rig.to_scale_rotation_translation().1 * r * Vec3::NEG_Z).normalize())
815                });
816                self.held[hand] = self.grab(ctx, hand, hand_world, look);
817            } else if !grip && self.grip_prev[hand] {
818                self.release(ctx, hand);
819            }
820            self.grip_prev[hand] = grip;
821
822            if let Some(held) = self.held[hand] {
823                if ctx.entity_of(held.id).is_none() {
824                    self.held[hand] = None;
825                    continue;
826                }
827                let (_, rot, t) = (hand_world * held.offset).to_scale_rotation_translation();
828                ctx.set_world_transform_of(held.id.into(), Some(t), Some(rot));
829                ctx.set_velocity(held.id, Vec3::ZERO);
830                let trig = triggers[hand];
831                ctx.edit::<Grabbable>(held.id, move |g| g.trigger_down = trig);
832            }
833        }
834    }
835}

837/// Something a [`VrCharacter`] can pick up, including parented objects. Weapons read
838/// `held` / `trigger_down` (see `Hitscan Weapon`'s held trigger mode).
839#[derive(Serialize, Deserialize)]
840#[serde(default)]
841pub struct Grabbable {
842    /// Snap into the hand at `hold_position` / `hold_rotation` (guns, tools);
843    /// otherwise it keeps the offset it was grabbed at.
844    pub snap_to_hand: bool,
845    /// Object position in hand space when snapped.
846    pub hold_position: [f32; 3],
847    /// Object rotation in hand space when snapped (Euler XYZ, degrees).
848    pub hold_rotation: [f32; 3],
849    /// Extra reach around the object's origin.
850    pub radius: f32,
851    /// Keep the hand's velocity when released (needs a dynamic Rigid Body).
852    pub throwable: bool,
853    /// Return to the start pose after this many seconds untouched away from it (0 = never).
854    pub return_after: f32,
855    /// Held by someone (runtime).
856    #[serde(skip)]
857    pub held: bool,
858    /// The hand holding it (runtime).
859    #[serde(skip)]
860    pub held_by: Option<VrHand>,
861    /// The holder's trigger is pressed (runtime).
862    #[serde(skip)]
863    pub trigger_down: bool,
864    /// The grab point's rest pose (runtime).
865    #[serde(skip)]
866    pub home: Option<(Vec3, Quat)>,
867    /// Seconds untouched away from its home (runtime).
868    #[serde(skip)]
869    pub idle: f32,
870}
871
872impl Default for Grabbable {
873    fn default() -> Self {
874        Self {
875            snap_to_hand: false,
876            hold_position: [0.0; 3],
877            hold_rotation: [0.0; 3],
878            radius: 0.15,
879            throwable: true,
880            return_after: 0.0,
881            held: false,
882            held_by: None,
883            trigger_down: false,
884            home: None,
885            idle: 0.0,
886        }
887    }
888}

890impl TypedComponent for Grabbable {
891    const NAME: &'static str = "Grabbable";
892}
893
894impl Grabbable {
895    fn start_native(&mut self, ctx: &mut EcsMoveHost<'_, '_>) {
896        self.home = Some((ctx.translation(), ctx.rotation()));
897    }
898
899    fn update_native(&mut self, ctx: &mut EcsMoveHost<'_, '_>) {
900        let Some((pos, rot)) = self.home else {
901            self.home = Some((ctx.translation(), ctx.rotation()));
902            return;
903        };
904        if self.held || self.return_after <= 0.0 || ctx.translation().distance(pos) < 0.05 {
905            self.idle = 0.0;
906            return;
907        }
908        self.idle += ctx.dt;
909        if self.idle >= self.return_after {
910            self.idle = 0.0;
911            ctx.set_translation(pos);
912            ctx.set_rotation(rot);
913            ctx.set_velocity(ctx.self_id(), Vec3::ZERO);
914        }
915    }
916}

918/// Avatar body that follows a head: sits `neck_drop` below `head` and turns
919/// toward the head's facing. Put the torso / limbs under it. A humanoid
920/// skinned model can be driven by the engine's tracker IK instead.
921#[derive(Serialize, Deserialize)]
922#[serde(default)]
923pub struct VrAvatar {
924    /// The head object the body follows.
925    pub head: ObjectRef,
926    /// How far the neck drops below the head (m).
927    pub neck_drop: f32,
928    /// Body turn rate toward the head (1/s).
929    pub turn_rate: f32,
930    /// Optional yaw override (degrees; the engine sets it).
931    #[serde(skip)]
932    pub yaw: Option<f32>,
933}
934
935impl Default for VrAvatar {
936    fn default() -> Self {
937        Self { head: ObjectRef::NONE, neck_drop: 0.22, turn_rate: 6.0, yaw: None }
938    }
939}

945impl VrAvatar {
946    fn late_update_native(&mut self, ctx: &mut EcsMoveHost<'_, '_>) {
947        let Some(head) = ctx.transform_of(self.head) else {
948            return;
949        };
950        let f = head.forward();
951        let goal = (-f.x).atan2(-f.z);
952        let k = 1.0 - (-self.turn_rate.max(0.0) * ctx.unscaled_dt.min(0.1)).exp();
953        let yaw = match self.yaw {
954            Some(y) => {
955                let diff = (goal - y + std::f32::consts::PI * 3.0).rem_euclid(std::f32::consts::TAU) - std::f32::consts::PI;
956                y + diff * k
957            }
958            None => goal,
959        };
960        self.yaw = Some(yaw);
961        ctx.set_world_position(head.translation - Vec3::Y * self.neck_drop);
962        let world = Quat::from_rotation_y(yaw);
963        ctx.set_world_rotation(world);
964    }
965}

979fn start(world: &SystemWorld<'_>, context: &mut SystemContext<'_>) {
980    for (_, (owner, value)) in world.query::<(&Owner, &mut Grabbable)>().iter() {
981        if value.home.is_none() && context.is_active(world, owner.0) {
982            value.start_native(&mut EcsMoveHost::new(world, owner.0, context));
983        }
984    }
985    let mut query = world.query::<(&Owner, &mut VrCharacter)>();
986    let mut actors: Vec<_> = query.iter().filter(|(_, (owner, value))| !value.started && context.is_active(world, owner.0)).collect();
987    actors.sort_unstable_by_key(|(_, (owner, _))| context.order_key(world, owner.0));
988    for (_, (owner, value)) in actors { value.start_native(&mut EcsMoveHost::new(world, owner.0, context)); }
989}
990
991fn update(world: &SystemWorld<'_>, context: &mut SystemContext<'_>) {
992    let mut query = world.query::<(&Owner, &mut VrCharacter)>();
993    let mut actors: Vec<_> = query.iter().filter(|(_, (owner, _))| context.is_active(world, owner.0)).collect();
994    actors.sort_unstable_by_key(|(_, (owner, _))| context.order_key(world, owner.0));
995    for (_, (owner, value)) in actors { value.update_native(&mut EcsMoveHost::new(world, owner.0, context)); }
996}

998fn return_objects(world: &SystemWorld<'_>, context: &mut SystemContext<'_>) {
999    for (_, (owner, value)) in world.query::<(&Owner, &mut Grabbable)>().iter() {
1000        if context.is_active(world, owner.0) { value.update_native(&mut EcsMoveHost::new(world, owner.0, context)); }
1001    }
1002}

1006        if context.is_active(world, owner.0) { value.late_update_native(&mut EcsMoveHost::new(world, owner.0, context)); }
1007    }

1185    fn teleport_landing_needs_head_room_and_ground() {
1186        let floor_y = 0.0;
1187        let ceiling_y = 3.0;
1188        // A world with a floor and (optionally) a ceiling: ray hits when it crosses either plane.
1189        let world = |ceiling: bool| {
1190            move |o: Vec3, d: Vec3, len: f32| {
1191                let end = o + d * len;
1192                let crosses = |y: f32| (o.y - y) * (end.y - y) <= 0.0 && (o.y - end.y).abs() > 1e-6;
1193                crosses(floor_y) || (ceiling && crosses(ceiling_y))
1194            }
1195        };
1196        assert!(teleport_landing_ok(Vec3::new(1.0, 0.0, 2.0), world(false)), "open floor");
1197        assert!(!teleport_landing_ok(Vec3::new(1.0, 5.0, 2.0), world(false)), "floating in the air");
1198        assert!(!teleport_landing_ok(Vec3::new(1.0, 2.5, 2.0), world(true)), "a ceiling right above the target");
1199        assert!(!teleport_landing_ok(Vec3::new(f32::NAN, 0.0, 0.0), world(false)));
1200    }

1202    fn ground(o: Vec3, d: Vec3, len: f32) -> Option<(Vec3, Vec3)> {
1203        if d.y >= 0.0 || o.y < 0.0 {
1204            return None;
1205        }

1207        (t <= len).then(|| (o + d * t, Vec3::Y))
1208    }
