1//! In-game controls / remap screen. Fills a fixed set of UI rows (built by
2//! the Controls UI template) with the actions of a control scheme and lets
3//! the player rebind them: click a binding, press the new input. Changes go
4//! through `ctx.rebind_with` and are saved to the player's overrides.
5
6use serde::{Deserialize, Serialize};
7use crate::ecs::{ObjectContext, Stage};
8
9use crate::{
10    CaptureResult, ConflictPolicy, ObjectId, RemapEntry, TypedComponent, UiButton, UiText,
11};

13/// Rebind menu driven by named UI objects. With prefix `P` it looks for:
14/// `P Scheme` (button + text: cycles the scheme shown), `P Row 1..rows`
15/// (shown / hidden per action) holding `P Row n Label` (text) and
16/// `P Row n Bind 1..columns` (buttons + text), `P Prev` / `P Next` / `P Page`
17/// for paging, `P Reset` (back to the game's defaults) and `P Status`.
18#[derive(Serialize, Deserialize)]
19#[serde(default)]
20pub struct ControlsMenu {
21    /// Name prefix of the UI objects (see the component docs).
22    pub prefix: String,
23    /// Row objects in the layout.
24    pub rows: u32,
25    /// Bindings per row (1 or 2).
26    pub columns: u32,
27    /// When the new input is already used by another action.
28    pub conflicts: ConflictPolicy,
29    /// Actions not listed (camera look, push-to-talk, ...).
30    pub hidden_actions: Vec<String>,
31    /// Engine-filled state of the menu (not saved).
32    #[serde(skip)]
33    pub state: ControlsMenuState,
34}
35
36#[doc(hidden)]
37#[derive(Default)]
38pub struct ControlsMenuState {
39    scheme: Option<String>,
40    page: usize,
41    /// The entry, slot index and column being rebound.
42    listening: Option<(RemapEntry, usize, usize)>,
43    /// Frames to wait for the host to start the capture.
44    pending: u8,
45    /// Seconds clicks are ignored after a capture (its mouse release).
46    cooldown: f32,
47    message: String,
48}
49
50impl Default for ControlsMenu {
51    fn default() -> Self {
52        Self {
53            prefix: "Controls".into(),
54            rows: 8,
55            columns: 2,
56            conflicts: ConflictPolicy::Swap,
57            hidden_actions: Vec::new(),
58            state: ControlsMenuState::default(),
59        }
60    }
61}

63impl ControlsMenu {
64    fn id(&self, ctx: &ObjectContext<'_, '_>, name: &str) -> Option<ObjectId> {
65        let full = format!("{} {name}", self.prefix);
66        ctx.find_object(&full)
67    }
68
69    fn clicked(&self, ctx: &ObjectContext<'_, '_>, name: &str) -> bool {
70        self.id(ctx, name).is_some_and(|id| ctx.ui_clicked(id))
71    }
72
73    fn set_text(&mut self, ctx: &mut ObjectContext<'_, '_>, name: &str, text: String) {
74        let Some(id) = self.id(ctx, name) else { return };
75        ctx.edit::<UiText>(id, |value| { if value.text != text { value.text = text; } });
76    }
77
78    fn set_shown(&mut self, ctx: &mut ObjectContext<'_, '_>, name: &str, on: bool) {
79        let Some(id) = self.id(ctx, name) else { return };
80        let enabled = ctx.entity_of(id).and_then(|entity| ctx.world.get::<crate::ecs::Enabled>(entity).map(|value| value.0));
81        if enabled != Some(on) { ctx.set_active(id, on); }
82    }
83
84    fn set_interactable(ctx: &mut ObjectContext<'_, '_>, id: Option<ObjectId>, on: bool) {
85        if let Some(id) = id { ctx.edit::<UiButton>(id, |button| button.interactable = on); }
86    }
87
88    /// The text of binding `index` of an entry.
89    fn binding_text(ctx: &ObjectContext<'_, '_>, scheme: &str, entry: &RemapEntry, index: usize) -> String {
90        let Some(a) = ctx.input_bindings().and_then(|bindings| bindings.scheme(scheme)).and_then(|s| s.action(&entry.action)) else {
91            return String::new();
92        };
93        if entry.slot.is_analog() {
94            return if index == 0 { a.analog_label().unwrap_or_else(|| "-".into()) } else { String::new() };
95        }
96        a.slot(entry.slot).get(index).map(|s| s.display_name()).unwrap_or_else(|| "-".into())
97    }
98}

104impl ControlsMenu {
105    fn update_native(&mut self, ctx: &mut ObjectContext<'_, '_>) {
106        let Some(screen) = ctx.services.screen else { return };
107        let bindings = std::sync::Arc::clone(&screen.controls.bindings);
108        if bindings.schemes.is_empty() {
109            return;
110        }
111        let active = bindings.active_scheme().map(|s| s.name.clone()).unwrap_or_default();
112        let scheme_name = self.state.scheme.clone().filter(|n| bindings.scheme(n).is_some()).unwrap_or_else(|| active.clone());
113        let Some(scheme) = bindings.scheme(&scheme_name) else { return };
114        let entries: Vec<RemapEntry> =
115            scheme.remap_entries().into_iter().filter(|e| !self.hidden_actions.contains(&e.action)).collect();
116        let rows = self.rows.max(1) as usize;
117        let columns = self.columns.clamp(1, 2) as usize;
118        let pages = entries.len().div_ceil(rows).max(1);
119        self.state.page = self.state.page.min(pages - 1);
120        self.state.cooldown = (self.state.cooldown - ctx.unscaled_dt).max(0.0);
121
122        // A capture in flight.
123        if let Some((entry, index, _)) = self.state.listening.clone() {
124            match ctx.captured_input() {
125                Some(CaptureResult::Captured(src)) => {
126                    ctx.rebind_with(&scheme_name, &entry.action, entry.slot, index, src, self.conflicts);
127                    self.state.message = format!("{}: {}", entry.label, src.display_name());
128                    self.state.listening = None;
129                    self.state.cooldown = 0.3;
130                }
131                Some(CaptureResult::Cancelled) => {
132                    self.state.message.clear();
133                    self.state.listening = None;
134                    self.state.cooldown = 0.3;
135                }
136                None if self.state.pending > 0 => self.state.pending -= 1,
137                None if !ctx.is_capturing_input() => self.state.listening = None,
138                None => {}
139            }
140        } else if self.state.cooldown <= 0.0 {
141            for row in 0..rows {
142                let Some(entry) = entries.get(self.state.page * rows + row) else { break };
143                for col in 0..columns {
144                    if (entry.slot.is_analog() && col > 0) || !self.clicked(ctx, &format!("Row {} Bind {}", row + 1, col + 1)) {
145                        continue;
146                    }
147                    let device = scheme.device();
148                    ctx.capture_next_input(entry.slot.capture_kind(), device);
149                    let index = if entry.slot.is_analog() { 0 } else { col.min(scheme.action(&entry.action).map_or(0, |a| a.slot(entry.slot).len())) };
150                    self.state.listening = Some((entry.clone(), index, col));
151                    self.state.pending = 2;
152                    self.state.message = if entry.slot.is_analog() {
153                        format!("Move a stick for {} (Esc cancels)", entry.label)
154                    } else {
155                        format!("Press an input for {} (Esc cancels)", entry.label)
156                    };
157                }
158            }
159            if self.clicked(ctx, "Prev") {
160                self.state.page = (self.state.page + pages - 1) % pages;
161            }
162            if self.clicked(ctx, "Next") {
163                self.state.page = (self.state.page + 1) % pages;
164            }
165            if self.clicked(ctx, "Scheme") {
166                let i = bindings.scheme_index(&scheme_name).unwrap_or(0);
167                self.state.scheme = Some(bindings.schemes[(i + 1) % bindings.schemes.len()].name.clone());
168                self.state.page = 0;
169                self.state.message.clear();
170            }
171            if self.clicked(ctx, "Reset") {
172                ctx.reset_scheme_bindings(&scheme_name);
173                self.state.message = format!("{scheme_name} reset to defaults");
174            }
175        }
176
177        // Show it.
178        let listening = self.state.listening.clone();
179        let tag = if scheme_name == active { " (in use)" } else { "" };
180        self.set_text(ctx, "Scheme", format!("{scheme_name}{tag}"));
181        self.set_text(ctx, "Page", format!("{} / {}", self.state.page + 1, pages));
182        for row in 0..rows {
183            let entry = entries.get(self.state.page * rows + row);
184            self.set_shown(ctx, &format!("Row {}", row + 1), entry.is_some());
185            let Some(entry) = entry else { continue };
186            self.set_text(ctx, &format!("Row {} Label", row + 1), entry.label.clone());
187            for col in 0..2 {
188                let name = format!("Row {} Bind {}", row + 1, col + 1);
189                let usable = col < columns && !(entry.slot.is_analog() && col > 0);
190                self.set_shown(ctx, &name, usable);
191                if !usable {
192                    continue;
193                }
194                let waiting = listening.as_ref().is_some_and(|(e, _, c)| e == entry && *c == col);
195                let text = if waiting { "...".to_string() } else { Self::binding_text(ctx, &scheme_name, entry, col) };
196                self.set_text(ctx, &name, text);
197            }
198        }
199        let conflicts = ctx.binding_conflicts(&scheme_name);
200        let status = if self.state.message.is_empty() && !conflicts.is_empty() {
201            let c = &conflicts[0];
202            format!("{} is used by {}", c.source.display_name(), c.actions.join(" and "))
203        } else {
204            self.state.message.clone()
205        };
206        self.set_text(ctx, "Status", status);
207        let id = self.id(ctx, "Reset");
208        Self::set_interactable(ctx, id, listening.is_none());
209    }
210}

213pub(crate) fn register_systems(registry: &mut crate::ComponentRegistry) {
214    registry.register_ecs::<ControlsMenu>();
215    registry.register_system_for::<ControlsMenu>("zestengine::controls_menu", Stage::Gameplay,
216        |world, ctx| crate::ecs::run_ordered(world, ctx, ControlsMenu::update_native));
217}
