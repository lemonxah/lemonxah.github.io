1//! XPBD seams sampled along panel boundaries, including unequal tessellations.
2use super::*;
3#[derive(Clone,Debug)]
4pub(super) struct ParticleStitch{pub terms:Vec<(usize,f32)>,pub compliance:f32,pub lambda:f32,vertices:[u32;3],coefficients:[f32;3],start:Vec3,end:Vec3}
5impl MeshClothSim{
6    pub fn set_stitches(&mut self,stitches:&[crate::ClothStitch])->Result<(),String>{
7        let mut seams=Vec::new();let mut neighbours=std::collections::HashSet::new();
8        for seam in stitches{
9            if [seam.vertex,seam.edge[0],seam.edge[1]].iter().any(|&v|v as usize>=self.vertex_count)||!seam.weight.is_finite()||!(0.0..=1.0).contains(&seam.weight)||!seam.compliance.is_finite()||seam.compliance<0.0{return Err("Invalid cloth stitch".into());}
10            let ids=[self.owner[seam.vertex as usize],self.owner[seam.edge[0] as usize],self.owner[seam.edge[1] as usize]];
11            let coefficients=[1.0,-(1.0-seam.weight),-seam.weight];let mut terms:Vec<(usize,f32)>=Vec::new();
12            for (id,c) in ids.into_iter().zip(coefficients){if let Some((_,weight))=terms.iter_mut().find(|(i,_)|*i==id as usize){*weight+=c;}else{terms.push((id as usize,c));}}
13            terms.retain(|(_,w)|w.abs()>1e-6);
14            for &a in &ids{for &b in &ids{if a!=b{neighbours.insert((a.min(b),a.max(b)));}}}
15            if terms.len()>1{seams.push(ParticleStitch{terms,compliance:seam.compliance,lambda:0.0,vertices:[seam.vertex,seam.edge[0],seam.edge[1]],coefficients,start:Vec3::ZERO,end:Vec3::ZERO});}
16        }
17        self.sewing=seams;self.seam_neighbours=neighbours;self.fabric.exclude_seams(&self.seam_neighbours);self.rebuild_surface_contacts();Ok(())
18    }
19    pub(super) fn prepare_seams(&mut self,skinned:&[Vec3],world:Mat4,anim:&[Vec3]){
20        if self.sewing.is_empty(){return;}
21        let previous=if self.anim.len()==self.particles.len(){self.anim.clone()}else{self.animated(&self.animated_vertices,self.last_world)};
22        for seam in &mut self.sewing{
23            seam.start=seam.vertices.iter().zip(seam.coefficients).map(|(&v,c)|(self.last_world.transform_point3(self.animated_vertices[v as usize])-previous[self.owner[v as usize] as usize])*c).sum();
24            seam.end=seam.vertices.iter().zip(seam.coefficients).map(|(&v,c)|(world.transform_point3(skinned[v as usize])-anim[self.owner[v as usize] as usize])*c).sum();
25        }
26    }
27    pub(super) fn carry_seams(&mut self,delta:Mat4){
28        // The bias is a vector (the coefficients sum to zero). Teleport Keep
29        // has already transported particles and their animation targets.
30        for seam in &mut self.sewing{seam.start=delta.transform_vector3(seam.start);}
31    }
32    pub(super) fn solve_seams(&mut self,physical:bool,h:f32,fraction:f32){
33        for seam in &mut self.sewing{
34            let delta=seam.terms.iter().map(|&(i,c)|self.particles[i].pos*c).sum::<Vec3>()+seam.start.lerp(seam.end,fraction);let len=delta.length();if len<1e-8{continue;}
35            let mass=|i:usize|if physical{self.fabric.weights.get(i).copied().unwrap_or(0.0)}else{1.0-self.particles[i].pin.clamp(0.0,1.0)};
36            let mut weights=[0.0;3];for (k,&(i,_)) in seam.terms.iter().enumerate(){weights[k]=mass(i);}
37            let alpha=seam.compliance/(h*h).max(1e-12);let denominator=seam.terms.iter().zip(&weights).map(|((_,c),w)|c*c*w).sum::<f32>()+alpha;
38            if denominator<1e-10{continue;}let dl=(-len-alpha*seam.lambda)/denominator;seam.lambda+=dl;let normal=delta/len;
39            for (&(i,c),w) in seam.terms.iter().zip(weights){self.particles[i].pos+=normal*(dl*c*w);}
40        }
41    }
